class_name CoreEnding
extends Node

const CHECK_EVERY := 0.5
const COL_MERGE := Color(0.72, 0.35, 1.0)
const COL_PATCH := Color(1.0, 0.88, 0.62)
const COL_TERMINAL := Color(0.75, 0.9, 1.0)
const TERMINAL_OFFSET := Vector3(0.0, 2.2, 0.0)

var shown := "-"
var _check := 0.0
var _orig := {}
var _extra: Array[Node] = []


func _process(delta: float) -> void:
	_check -= delta
	if _check > 0.0:
		return
	_check = CHECK_EVERY
	var player := get_tree().get_first_node_in_group("players")
	var qm: QuestManager = player.get("quests") if player != null else null
	var id := Endings.current(qm)
	if id != shown:
		_apply(id)


func _exit_tree() -> void:
	_restore()


func _apply(id: String) -> void:
	_restore()
	shown = id
	if id == "":
		return
	var sun := _sun()
	var env := _env()
	_remember(sun, ["light_energy", "light_color"])
	_remember(env, ["ambient_light_energy", "ambient_light_color"])
	for glow in _glows():
		_remember(glow, ["light_energy", "omni_range"])
	match id:
		"merge":
			for glow in _glows():
				glow.light_energy = 6.0
				glow.omni_range = 24.0
			_light(Vector3(36, 5, 33), COL_MERGE, 4.0, 22.0)
			if sun != null:
				sun.light_color = sun.light_color.lerp(COL_MERGE, 0.45)
			if env != null:
				env.ambient_light_color = env.ambient_light_color.lerp(COL_MERGE, 0.5)
		"overwrite":
			for glow in _glows():
				glow.light_energy = 0.0
			_light(Vector3(32, 5, 30), COL_PATCH, 3.0, 20.0)
			_light(Vector3(42, 5, 35), COL_PATCH, 3.0, 20.0)
			if sun != null:
				sun.light_energy *= 1.2
				sun.light_color = sun.light_color.lerp(COL_PATCH, 0.35)
			if env != null:
				env.ambient_light_color = env.ambient_light_color.lerp(COL_PATCH, 0.4)
		"terminate":
			for glow in _glows():
				glow.light_energy = 0.0
			if sun != null:
				sun.light_energy *= 0.1
			if env != null:
				env.ambient_light_energy *= 0.15
			var log := get_parent().get_node_or_null("LastLog") as Node3D
			if log != null:
				_light(log.position + TERMINAL_OFFSET, COL_TERMINAL, 2.5, 7.0)


func _restore() -> void:
	for obj in _orig:
		if is_instance_valid(obj):
			for prop in _orig[obj]:
				obj.set(prop, _orig[obj][prop])
	_orig.clear()
	for n in _extra:
		if is_instance_valid(n):
			n.queue_free()
	_extra.clear()


func _remember(obj: Object, props: Array) -> void:
	if obj == null:
		return
	var saved: Dictionary = _orig.get(obj, {})
	for p in props:
		if not saved.has(p):
			saved[p] = obj.get(p)
	_orig[obj] = saved


func _light(at: Vector3, col: Color, energy: float, reach: float) -> void:
	var l := OmniLight3D.new()
	l.name = "EndingLight"
	l.light_color = col
	l.light_energy = energy
	l.omni_range = reach
	get_parent().add_child(l)
	l.position = at
	_extra.append(l)


func _glows() -> Array:
	var out: Array = []
	for n in ["NullGlowW", "NullGlowE"]:
		var g := get_parent().get_node_or_null(n) as OmniLight3D
		if g != null:
			out.append(g)
	return out


func _sun() -> DirectionalLight3D:
	return get_tree().root.find_child("DirectionalLight3D", true, false) as DirectionalLight3D


func _env() -> Environment:
	var w := get_parent().get_viewport().find_world_3d() if get_parent() != null else null
	return w.environment if w != null else null

static func is_down(tree: SceneTree, id: String) -> bool:
	var qm := _quests(tree)
	return qm != null and int(qm.deeds.get(DEED_PREFIX + id, 0)) > 0


static func find(tree: SceneTree, id: String) -> DeckSystem:
	for s in tree.get_nodes_in_group(GROUP):
		if (s as DeckSystem).system_id == id:
			return s
	return null


static func _quests(tree: SceneTree) -> QuestManager:
	var p := tree.get_first_node_in_group("players") if tree != null else null
	return p.get("quests") as QuestManager if p != null else null


func is_offline() -> bool:
	return _offline


func is_online() -> bool:
	return not _offline


func set_contested(on: bool) -> void:
	contested = on
	if on and not _offline:
		integrity = max_integrity
	under_attack = false
	_refresh()


func attackers(enemies: Array) -> int:
	var n := 0
	for e in enemies:
		if is_instance_valid(e) and (e as Node3D).global_position.distance_to(global_position) <= reach:
			n += 1
	return n


func drain(enemies: Array, delta: float) -> void:
	if _offline or not contested:
		return
	var n := attackers(enemies)
	if n > 0:
		_damage(drain_per_enemy * n * delta)


func _damage(amount: float) -> void:
	if _offline or not contested or amount <= 0.0:
		return
	integrity = maxf(integrity - amount, 0.0)
	_attack_memory = 1.5
	if integrity <= 0.0:
		_go_offline()


func show_remote(value: float, down: bool, hit: bool) -> void:
	if _offline or not contested:
		return
	integrity = clampf(value, 0.0, max_integrity)
	if hit:
		_attack_memory = 0.6
	if down and not Coop.visiting:
		integrity = 0.0
		_go_offline()


func knock_out() -> void:
	if not _offline:
		integrity = 0.0
		_go_offline()


func _go_offline() -> void:
	_offline = true
	var qm := _quests(get_tree())
	if qm != null:
		qm.deeds[DEED_PREFIX + system_id] = 1
	_refresh()
	went_offline.emit(self)


func repair() -> void:
	if not _offline:
		return
	var qm := _quests(get_tree())
	if qm != null:
		qm.deeds.erase(DEED_PREFIX + system_id)
	_offline = false
	integrity = max_integrity
	_refresh()
	QuestItems.toast(get_tree(), "%s PATCHED  //  ONLINE\n%s" % [npc_name.to_upper(), keeps], COL_OK)
	if qm != null and recovers_on_repair != "":
		qm.recover(recovers_on_repair)
	repaired.emit(self)


func fix_hint() -> String:
	return FIX_HINT % npc_name


func interact(player: Node3D) -> void:
	if _offline:
		if contested:
			QuestItems.toast(get_tree(), "%s OFFLINE\nClear the deck first: they'd only wreck it again." % npc_name.to_upper(), COL_DOWN)

		PatchPanelScript.open(self, player)
		return

func _process(delta: float) -> void:
	_t += delta
	_hit_flash = maxf(_hit_flash - delta * 4.0, 0.0)
	_attack_memory = maxf(_attack_memory - delta, 0.0)
	var was := under_attack
	under_attack = contested and not _offline and _attack_memory > 0.0
	if was != under_attack:
		_refresh()
	_animate(delta)


func _animate(delta: float) -> void:
	if _offline:
		var f := 1.0 if fposmod(_t * 7.3, 1.0) < 0.12 and fposmod(_t * 0.9, 1.0) < 0.5 else 0.0
		_core_mat.emission_energy_multiplier = 0.15 + f * 1.4
		_light.light_energy = 0.25 + f * 0.9
		_alarm_mat.emission_energy_multiplier = 3.0 if fposmod(_t, 1.0) < 0.5 else 0.3
	elif under_attack:
		var f := 0.5 + 0.5 * sin(_t * 18.0)
		_core_mat.emission_energy_multiplier = 1.2 + f * 2.0 + _hit_flash * 3.0
		_light.light_energy = 0.8 + f * 0.8
		_alarm_mat.emission_energy_multiplier = 3.0 if fposmod(_t * 3.0, 1.0) < 0.5 else 0.2
	else:
		var f := 0.5 + 0.5 * sin(_t * 2.0)
		_core_mat.emission_energy_multiplier = 1.6 + f * 0.8
		_light.light_energy = 0.7 + f * 0.25
	if _spin != null and not _offline:
		_spin.rotate_y(delta * (2.4 if under_attack else 0.8))
	if _bar_root.visible:
		var frac := clampf(integrity / maxf(max_integrity, 1.0), 0.0, 1.0)
		_bar_fill.scale.x = maxf(frac, 0.001)
		_bar_fill.position.x = -BAR_WIDTH * 0.5 * (1.0 - frac)
		_bar_mat.albedo_color = COL_OK.lerp(COL_DOWN, 1.0 - frac) if not under_attack \
				else COL_HIT.lerp(Color.WHITE, _hit_flash)
		_label.text = "%d%%" % ceili(frac * 100.0) if not _offline else "OFFLINE"


func _refresh() -> void:
	if _core_mat == null:
		return
	var col := COL_DOWN if _offline else (COL_HIT if under_attack else COL_OK)
	_core_mat.emission = col
	_core_mat.albedo_color = col.darkened(0.4)
	_light.light_color = col
	_alarm.visible = _offline or under_attack
	_sparks.emitting = _offline
	_smoke.emitting = _offline
	_bar_root.visible = contested
	_label.modulate = COL_DOWN if _offline else Color(0.85, 0.97, 1.0)

		info_title = "OFFLINE  //  PATCH ITS CODE"
		prompt_text = "PATCH CODE"
	else:
		info_title = "ONLINE"
		prompt_text = "INSPECT"
	info_description = keeps + ("\nOFFLINE: " + offline_text if _offline else "")

func _collect_spawn_points() -> void:

		return

	for p in town.find_children("*", "Portal", true, false):
		if p != _portal:
			others.append((p as Portal).global_position)
	var near_portals: Array[Vector3] = []
	for c in town._director.grid_map.get_used_cells():
		if c.y != TownArea.FLOOR_LAYER or not town._is_floor(Vector2i(c.x, c.z)):
			continue
		var at := town._cell_to_world(Vector2i(c.x, c.z))
		var d := Vector2(at.x - centre.x, at.z - centre.z).length()
		if d < spawn_min or d > spawn_max:
			continue
		var clear := true
		for o in others:
			if Vector2(at.x - o.x, at.z - o.z).length() < portal_clearance:
				clear = false
		if clear:
			_spawn_points.append(at)
		else:
			near_portals.append(at)
	if _spawn_points.is_empty():

		return

		_spawned(_spawn_one(def, rarity, [], grow), i)
	if bosses_last_wave_only and _wave < wave_sizes.size():
		return
	for boss in mini_bosses:
		_spawned(_spawn_one(boss, mini_boss_rarity, null, grow), -1)

func _unstick() -> void:
	var region := _region()
	if region == null:
		return
	var map := region.get_navigation_map()
	var cam := get_viewport().get_camera_3d()
	var seen := {}
	for e in _alive:
		var id := e.get_instance_id()
		seen[id] = true
		var ai := e.get("ai") as AIComponent
		var w: Dictionary = _stuck.get(id, {"at": e.global_position, "still": 0.0, "hops": 0})
		_stuck[id] = w
		var on := NavigationServer3D.map_get_closest_point(map, e.global_position)
		var off_mesh := Vector2(on.x - e.global_position.x, on.z - e.global_position.z).length() > OFF_MESH
		var moved := Vector2(e.global_position.x - w.at.x, e.global_position.z - w.at.z).length()
		w.at = e.global_position
		if ai == null or ai.target == null or not is_instance_valid(ai.target) or ai.state != AIComponent.State.CHASE \
				or ai._is_busy() or _content(e, ai):
			w.still = 0.0
			if moved > STUCK_MOVE:
				w.hops = 0
			if not off_mesh:
				continue
		if moved > STUCK_MOVE and not off_mesh:
			w.still = 0.0
			w.hops = 0
			continue
		w.still += STUCK_CHECK
		if w.still < STUCK_TIME and not off_mesh:
			continue
		w.still = 0.0
		var goal: Vector3 = ai.target.global_position if ai != null and ai.target != null and is_instance_valid(ai.target) \
				else _portal.global_position
		var path := NavigationServer3D.map_get_path(map, on, NavigationServer3D.map_get_closest_point(map, goal), true)
		var reaches := not path.is_empty() and path[path.size() - 1].distance_to(goal) < 3.0
		w.hops += 1
		var visible := cam != null and cam.is_position_in_frustum(e.global_position + Vector3.UP)
		if reaches and w.hops <= HOPS_BEFORE_RESPAWN:
			_hop(e, _along(path, HOP) if path.size() > 1 else on, visible)
		elif not visible and not _spawn_points.is_empty():
			_hop(e, _spawn_points[randi() % _spawn_points.size()], false)
			w.hops = 0
		elif off_mesh:
			_hop(e, on, visible)
	for id in _stuck.keys():
		if not seen.has(id):
			_stuck.erase(id)


func _content(e: Node3D, ai: AIComponent) -> bool:
	var d := e.global_position.distance_to(ai.target.global_position)
	if ai.preferred_distance > 0.0 and d <= ai.preferred_distance * 1.3:
		return true
	return d <= ai.attack_range * 1.5


static func _along(path: PackedVector3Array, metres: float) -> Vector3:
	var left := metres
	for i in range(1, path.size()):
		var seg := path[i] - path[i - 1]
		var l := seg.length()
		if l >= left:
			return path[i - 1] + seg * (left / maxf(l, 0.001))
		left -= l
	return path[path.size() - 1]


func _hop(e: Node3D, to: Vector3, seen: bool) -> void:
	var at := to + Vector3.UP * 0.1
	if not seen:
		e.global_position = at
		e.set("velocity", Vector3.ZERO)
		return
	var model := e.get("model") as Node3D
	if model != null:
		model.visible = false
	var tw := create_tween()
	tw.tween_interval(0.07)
	tw.tween_callback(func() -> void:
		if is_instance_valid(e):
			e.global_position = at
			e.set("velocity", Vector3.ZERO))
	tw.tween_interval(0.07)
	tw.tween_callback(func() -> void:
		if is_instance_valid(e) and model != null and is_instance_valid(model):
			model.visible = true)

	var npcs: Array = get_parent().find_children("*", "DialogueNPC", true, false)

		return a.global_position.distance_to(_portal.global_position) \
				< b.global_position.distance_to(_portal.global_position))
	for i in npcs.size():
		var npc := npcs[i] as DialogueNPC
		var body := npc.get_node_or_null("Body") as CollisionObject3D
		_runners.append({
			"npc": npc, "home": npc.global_transform, "delay": 0.3 + i * npc_stagger,
			"path": PackedVector3Array(), "i": 0, "gone": false, "started": false,
			"body": body, "layer": body.collision_layer if body != null else 0,
		})


func _move_runner(r: Dictionary, delta: float) -> void:
	var npc := r.npc as DialogueNPC
	if r.gone or not is_instance_valid(npc):
		return
	r.delay -= delta
	if r.delay > 0.0:
		return
	if not r.started:
		r.started = true
		_set_out(r)
	var path: PackedVector3Array = r.path
	var at := npc.global_position
	var goal := _portal.global_position
	if Vector2(goal.x - at.x, goal.z - at.z).length() <= VANISH_DISTANCE or r.i >= path.size():
		_vanish(r)
		return
	var next := path[r.i]
	var to := Vector2(next.x - at.x, next.z - at.z)
	var step := npc_speed * delta
	if to.length() <= step:
		r.i += 1
		step = to.length()
	if to.length() > 0.001:
		var dir := to.normalized()
		npc.global_position += Vector3(dir.x, 0.0, dir.y) * step
		var turner := npc.get_node_or_null("ModelAnchor") as Node3D
		if turner == null:
			turner = npc
		var yaw := atan2(dir.x, dir.y) + deg_to_rad(npc.model_forward_offset_deg)
		turner.global_rotation.y = lerp_angle(turner.global_rotation.y, yaw, 1.0 - exp(-12.0 * delta))


func _set_out(r: Dictionary) -> void:
	var npc := r.npc as DialogueNPC
	npc.set_deferred("monitoring", false)
	npc.set_process(false)
	if r.body != null:
		(r.body as CollisionObject3D).collision_layer = 0
	var goal := _portal.global_position
	var region := _region()
	if region != null:
		r.path = NavigationServer3D.map_get_path(region.get_navigation_map(), npc.global_position, goal, true)
	if (r.path as PackedVector3Array).is_empty():
		r.path = PackedVector3Array([goal])
	r.i = 0
	var ap := npc.find_child("AnimationPlayer", true, false) as AnimationPlayer
	if ap != null:
		var clip := RUN_CLIP if ap.has_animation(RUN_CLIP) else ""
		for c in ap.get_animation_list():
			if clip == "" and (String(c).to_lower().contains("jog") or String(c).to_lower().contains("run")):
				clip = c
		if clip != "":
			ap.get_animation(clip).loop_mode = Animation.LOOP_LINEAR
			ap.play(clip, 0.15)


func _vanish(r: Dictionary) -> void:

	var npc := r.npc as DialogueNPC

	t.tween_property(npc, "scale", Vector3.ONE * 0.05, 0.25).set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_IN)
	t.tween_callback(npc.hide)
	r["tween"] = t

static func current(qm: QuestManager) -> String:
	if qm == null:
		return ""
	for id in IDS:
		if qm.deed_count(NOW_DEED, id) > 0:
			return id
	for id in IDS:
		if qm.deed_count("ending", id) > 0:
			return id
	return ""


static func choose(qm: QuestManager, id: String) -> void:
	if qm == null or not id in IDS:
		return
	for other in IDS:
		qm.deeds.erase("%s:%s" % [NOW_DEED, other])
	qm.deeds["%s:%s" % [NOW_DEED, id]] = 1
	GameEvents.report("ending", id)
	Coop.kernel_changed()


static func times(qm: QuestManager, id: String) -> int:
	return qm.deed_count("ending", id) if qm != null else 0


static func runs_everywhere(qm: QuestManager) -> bool:
	return current(qm) in ["merge", "overwrite"]


static func kernel_editable(qm: QuestManager) -> bool:
	return current(qm) == "overwrite"

static func kernel_lines(qm: QuestManager) -> PackedStringArray:
	var out := PackedStringArray()

static func final_log(qm: QuestManager, player: Node) -> PackedStringArray:
	var out := PackedStringArray()
	var who := "EXECUTOR"
	if player != null and is_instance_valid(player):
		var c = player.get("class_data")
		who = str(c.display_name).to_upper() if c != null and c.get("display_name") != null else who
	out.append("> executor   %s" % who)
	out.append("> rank       %s" % Ranks.current(qm))
	var grades := PackedStringArray()
	for r in Ranks.LADDER:
		var g := qm.exam_grade(r["id"]) if r["id"] != "" else -1
		if g >= 0:
			grades.append("%s %s" % [str(r["name"]).to_lower(), ExamSpec.GRADES[clampi(g, 0, ExamSpec.GRADES.size() - 1)]])
	out.append("> exams      %s" % (", ".join(grades) if not grades.is_empty() else "none sat"))
	var bosses := ScriptBudget.act_bosses_killed(qm)
	out.append("> bosses     %d broken" % bosses.size())
	out.append("> areas      %d cleared" % ScriptBudget.cleared_areas(qm).size())
	var logs := 0
	for k in ["architect_log_a", "architect_log_b", "architect_log_c"]:
		if qm.has_recovered(k):
			logs += 1
	out.append("> logs       %d of 3 Architect logs found" % logs)
	var t := KernelScript.tally(qm)
	out.append("> kernel     merged %d, overwrote %d, terminated %d" % [t["merge"], t["overwrite"], t["terminate"]])
	var before := PackedStringArray()
	for id in IDS:
		if id != "terminate" and times(qm, id) > 0:
			before.append(str(id))
	if not before.is_empty():
		out.append("> before     you also tried: %s" % ", ".join(before))
	return out

static func _convoy(qm: QuestManager) -> String:

	tag.pixel_size = 0.005
	tag.font_size = size_px
	tag.outline_size = 8
	tag.modulate = col
	tag.position.y = height
	e.add_child(tag)


func _pick_system():
	var best = null
	var best_n := 1 << 30
	for sys in _systems:
		if not is_instance_valid(sys) or sys.is_offline() or not sys.sabotage_target:
			continue
		var n := 0
		for s in _saboteurs.values():
			if s == sys:
				n += 1
		if n < best_n:
			best_n = n
			best = sys
	return best


func _release(e: Node3D) -> void:
	if not is_instance_valid(e) or not _saboteurs.has(e):
		return
	_saboteurs.erase(e)
	_untag(e)
	var ai = e.get("ai")
	if ai != null:
		ai.objective = null
		ai.aggro_range = spawn_max * 3.0


func _untag(e: Variant) -> void:
	if is_instance_valid(e):
		var tag := (e as Node).get_node_or_null("SaboteurTag")
		if tag != null:
			tag.queue_free()


func _warn(sys) -> void:
	if sys.is_offline():
		return
	var level := int(_warned.get(sys, 0))
	var frac: float = sys.integrity / maxf(sys.max_integrity, 1.0)
	if level < 1 and sys.under_attack:
		_warned[sys] = 1
		QuestItems.toast(get_tree(), "THE NULL ARE AT THE %s\nIf it falls: %s" % [
			String(sys.npc_name).to_upper(), sys.offline_text], COL_HIT)
	elif level < 2 and frac <= 0.5:
		_warned[sys] = 2
		QuestItems.toast(get_tree(), "%s AT %d%%" % [String(sys.npc_name).to_upper(), ceili(frac * 100.0)], COL_WARN)


func _on_system_down(sys) -> void:
	if state != State.RUNNING and state != State.CLEARING:
		return
	var name_ := String(sys.npc_name).to_upper()
	_lost.append(name_)
	QuestItems.toast(get_tree(), "%s OFFLINE\n%s\nThe Null corrupted its code: patch it once the deck is clear." % [
		name_, sys.offline_text], COL_WARN)
	for e in _saboteurs.keys():
		if _saboteurs[e] != sys:
			continue
		var next = _pick_system()
		if next == null:
			_release(e)
		else:
			_saboteurs[e] = next
			e.get("ai").objective = next


func _update_systems_hud() -> void:

		if not is_instance_valid(sys) or not (sys.sabotage_target or sys.is_offline()):

			elif pct >= 100:

func _end_contest() -> void:

	_saboteurs.clear()
	for sys in _systems:
		if is_instance_valid(sys):
			sys.set_contested(false)

func _pay_out(kept_cargo: bool, lost: Array) -> void:
	var owed := not Coop.in_game or is_due()
	if owed:
		GameEvents.report(DEED, after_stop)
		GameEvents.report(ScriptBudget.CLEAR_DEED, "breach_" + after_stop)
	var credits := reward_credits + (cargo_bonus if kept_cargo else 0)
	var wallet := Coop.own_player(get_tree()).get("currency") as CurrencyComponent \
			if Coop.own_player(get_tree()) != null else null

	elif lost.is_empty():

	return null

			return w
	return null

class_name KernelDirector

const SPOT_RANGE := 14.0
const SCAN_EVERY := 0.25
const HISTORY := 5.0
const DRAIN_GAP := 8.0
const ENTRY_BARRIER := 0.2
const ENTRY_BARRIER_SECONDS := 8.0

var _player: Node3D
var _started := false
var _scan := 0.0
var _history: Array = []
var _rolled := false
var _drained_at := -100.0
var _strip: KernelStrip
var _scheduling := false
var _scheduled := ""
var _schedule_left := 0.0
var _schedule_index := -1
var _mine := false
var _in_act5 := false
var _card: PullRequestCard


func _ready() -> void:
	GameEvents.happened.connect(_on_event)
	Coop.event_register("kernel", self)
	_begin.call_deferred()


func _exit_tree() -> void:
	if GameEvents.happened.is_connected(_on_event):
		GameEvents.happened.disconnect(_on_event)
	if is_instance_valid(_strip):
		_strip.queue_free()
	if is_instance_valid(_card):
		_card.queue_free()
	_set_answering(false)


func _begin() -> void:
	for i in 120:
		_player = get_tree().get_first_node_in_group("players") as Node3D
		if _player != null and _player.is_inside_tree():
			break
		await get_tree().process_frame
	var qm := _qm()
	if _player == null:
		queue_free()
		return
	_in_act5 = KernelScript.is_zone(get_parent().scene_file_path)
	_mine = KernelScript.unlocked(qm) and (_in_act5 or Endings.runs_everywhere(qm))
	if not _mine:
		if Coop.in_game and Coop.is_host():
			_started = true
		else:
			queue_free()
		return
	_build_strip()
	var zone_line := KernelScript.null_line_of(get_parent().scene_file_path)
	if zone_line != "" and KernelScript.answer_of(qm, zone_line) == "" and not Coop.visiting:
		if not KernelScript.is_inserted(qm, zone_line):
			GameEvents.report(KernelScript.NULL_DEED, zone_line)
		await get_tree().create_timer(0.6).timeout
		if not is_instance_valid(_player) or not is_inside_tree() or KernelScript.answer_of(qm, zone_line) != "":
			return
		var ids: Array = KernelScript.NULL_LINES.keys()
		var w := Waypoints.for_scene(get_parent().scene_file_path)
		_card = PullRequestCard.open(get_tree(), zone_line, w.display_name if w != null else "",
			ids.find(zone_line) + 1, ids.size(), _lines())
		_set_answering(true)
		var reply: Array = await _card.answered
		_set_answering(false)
		answer(zone_line, str(reply[0]), str(reply[1]))
	if "barrier_on_entry" in _lines():
		var hp := _hp()
		if hp != null:
			hp.add_barrier(_max_life() * ENTRY_BARRIER, ENTRY_BARRIER_SECONDS)
			_pop("barrier(20%)", KernelScript.COL_ROOT)
	_scheduling = get_parent().scene_file_path.get_file().get_basename() == KernelScript.SCHEDULE_ZONE
	if _scheduling:
		_next_schedule()
	_started = true


func _set_answering(on: bool) -> void:
	Coop.answering = on and Coop.in_game
	var hp := _hp()
	if hp != null:
		hp.untouchable = on and Coop.in_game


func _process(delta: float) -> void:
	if not _started or not is_instance_valid(_player):
		return
	if not _mine:
		_scan -= delta
		if _scan <= 0.0:
			_scan = SCAN_EVERY
			_scan_enemies()
		return
	if _scheduling:
		_schedule_left -= delta
		if _schedule_left <= 0.0:
			_next_schedule()
	_update_strip()
	var hp := _hp()
	if hp == null or hp.current_health <= 0.0:
		return
	var now := Time.get_ticks_msec() / 1000.0
	_history.append([now, hp.current_health])
	while not _history.is_empty() and now - float(_history[0][0]) > HISTORY:
		_history.pop_front()
	var top := _max_life()
	var frac := hp.current_health / top if top > 0.0 else 1.0
	if frac < 0.3 and not _rolled and "rollback" in _lines():
		var best := hp.current_health
		for h in _history:
			best = maxf(best, float(h[1]))
		if best > hp.current_health:
			_rolled = true
			hp.heal(best - hp.current_health)
			_pop("stack.rollback(5s)", KernelScript.COL_ROOT)
	if frac < 0.5 and now - _drained_at >= DRAIN_GAP and "null_drain" in _running():
		var mana := _player.get("mana") as ManaComponent
		if mana != null and mana.current_mana > 0.0:
			_drained_at = now
			var take := minf(mana.get_max_mana() * 0.25, mana.current_mana)
			mana.spend(take)
			_pop("mana.drain(25%%)  -%d" % roundi(take), KernelScript.COL_NULL)
	_scan -= delta
	if _scan <= 0.0:
		_scan = SCAN_EVERY
		_scan_enemies()


func _scan_enemies() -> void:
	var at := _player.global_position
	var lines := _lines() if _mine else []
	var real := not Coop.in_game or Coop.is_host()
	var party := Coop.party_kernel(_player if _mine else null, not _in_act5)
	var enemy_lines: Array = party.lines
	var enemy_null: Array = party.merged.duplicate()
	if _scheduled != "" and not _scheduled in enemy_null:
		enemy_null.append(_scheduled)
	var players := get_tree().get_nodes_in_group("players")
	for e in get_tree().get_nodes_in_group(&"enemies"):
		var body := e as Node3D
		if body == null or SkillBehavior.is_dead(body):
			continue
		if _mine and not body.has_meta(&"kernel_spotted") and body.global_position.distance_to(at) <= SPOT_RANGE:
			body.set_meta(&"kernel_spotted", true)
			_spotted(body, lines)
		if not body.has_meta(&"kernel_marked") and _near_any(body, players):
			body.set_meta(&"kernel_marked", true)
			_mark(body, enemy_lines, enemy_null, real)


func _near_any(body: Node3D, players: Array) -> bool:
	for p in players:
		if is_instance_valid(p) and (p as Node3D).global_position.distance_to(body.global_position) <= SPOT_RANGE:
			return true
	return false


func _spotted(e: Node3D, lines: Array) -> void:
	if "registry_add" in lines:
		var sk := _player.get("skills") as SkillComponent
		var key: Variant = ScriptRegisters.entry_key(e)
		if sk != null and key != null:
			for s in sk.sockets:
				if s.program != null:
					s.program.state().registers[key] = true


func _mark(e: Node3D, lines: Array, running: Array, real: bool) -> void:
	if "defragment" in lines and "traits" in e:
		for t in (e.get("traits") as Array).duplicate():
			if t is FragmentLock:
				(e.get("traits") as Array).erase(t)
				(t as Node).queue_free()
	if not real:
		return
	var stats := e.get("stats") as StatComponent
	var hp := e.get("health") as HealthComponent
	if stats != null:
		if "weaken" in lines:
			_mod(stats, StatTypes.Stat.MAX_HEALTH, -0.15, "kernel_weaken")
		if "slow" in lines:
			_mod(stats, StatTypes.Stat.MOVE_SPEED, -0.25, "kernel_slow")
		if "null_armour" in running:
			_mod(stats, StatTypes.Stat.ARMOUR, 0.4, "kernel_null_armour")
		if "null_regrow" in running:
			var before := stats.get_stat(StatTypes.Stat.MAX_HEALTH)
			_mod(stats, StatTypes.Stat.MAX_HEALTH, 0.2, "kernel_null_regrow")
			if hp != null:
				hp.heal(stats.get_stat(StatTypes.Stat.MAX_HEALTH) - before)
	if hp != null and stats != null:
		hp.current_health = minf(hp.current_health, stats.get_stat(StatTypes.Stat.MAX_HEALTH))


static func _mod(stats: StatComponent, stat: int, value: float, source: String) -> void:
	var m := Modifier.new()
	m.stat = stat
	m.op = Modifier.ModOp.PERCENT_MULT
	m.value = value
	m.source_id = source
	stats.add_modifier(m)


func _on_event(kind: String, _target: String, _amount: int) -> void:
	if kind != "kill" or not _started or not is_instance_valid(_player):
		return
	var lines := _lines()
	if "heal_on_kill" in lines:
		var hp := _hp()

	if "mana_on_kill" in lines:
		var mana := _player.get("mana") as ManaComponent
		if mana != null:
			mana.restore(mana.get_max_mana() * 0.10)
	if "null_swap" in _running():
		var sk := _player.get("skills") as SkillComponent
		if sk != null:
			for s in sk.sockets:
				if s.program != null:
					s.program.state().registers.clear()
			_pop("registry.clear()", KernelScript.COL_NULL)

static func unlocked(qm: QuestManager) -> bool:
	return qm != null and qm.is_completed(UNLOCK_QUEST)


static func is_zone(scene_path: String) -> bool:
	return scene_path.get_file().get_basename() in ZONES


static func is_inserted(qm: QuestManager, null_id: String) -> bool:
	return qm != null and qm.deed_count(NULL_DEED, null_id) > 0


static func answer_of(qm: QuestManager, null_id: String) -> String:
	if qm == null:
		return ""
	for a in ANSWERS:
		if qm.deed_count(deed_of(a), null_id) > 0:
			return a
	return ""


static func deed_of(answer: String) -> String:
	return {"merge": MERGE_DEED, "overwrite": OVERWRITE_DEED, "terminate": TERMINATE_DEED}.get(answer, "")


static func answer(qm: QuestManager, lines: Array[String], null_id: String, how: String, pick := "") -> void:
	if qm == null or not NULL_LINES.has(null_id) or answer_of(qm, null_id) != "" or not how in ANSWERS:
		return
	var line := ""
	if how == "merge":
		line = str(NULL_LINES[null_id]["gift"])
	elif how == "overwrite":
		line = pick if pick in offers(lines, null_id) else ""
	if line != "" and not line in lines:
		lines.append(line)
	GameEvents.report(deed_of(how), null_id)


static func offers(lines: Array, null_id: String) -> Array:
	var all: Array = NULL_LINES.get(null_id, {}).get("offers", [])
	var fresh := all.filter(func(id: String) -> bool: return not id in lines)
	return (fresh if fresh.size() >= 2 else all).slice(0, 2)


static func null_running(qm: QuestManager) -> Array:
	if Endings.current(qm) == "overwrite":
		return []
	return NULL_LINES.keys().filter(func(id: String) -> bool: return answer_of(qm, id) == "merge")


static func tally(qm: QuestManager) -> Dictionary:
	var out := {"merge": 0, "overwrite": 0, "terminate": 0}
	for id in NULL_LINES:
		var a := answer_of(qm, id)
		if a != "":
			out[a] += 1
	return out


static func lean(qm: QuestManager) -> String:
	var t := tally(qm)
	var best := 0
	for a in ANSWERS:
		best = maxi(best, int(t[a]))
	if best == 0:
		return "terminate"
	var ids: Array = NULL_LINES.keys()
	ids.reverse()
	for id in ids:
		var a := answer_of(qm, id)
		if a != "" and int(t[a]) == best:
			return a
	return "terminate"


static func null_line_of(scene_path: String) -> String:
	var zone := scene_path.get_file().get_basename()
	for id in NULL_LINES:
		if NULL_LINES[id]["zone"] == zone:
			return id
	return ""


static func code_of(id: String) -> String:
	var d: Dictionary = LINES.get(id, NULL_LINES.get(id, {}))
	return "when %s: %s" % [d.get("trigger", "?"), d.get("code", "?")]

class PowerPuzzle extends Puzzle:
	const OPS := ["<", ">", "<=", ">=", "==", "!="]
	const LIMITS := [0, 40, 80, 120, 160, 200]
	var op := 0
	var limit := 3

	func _init() -> void:
		file = "breaker.prog"
		brief = "The Null flipped a check in the breaker: it cuts the market row's power on a normal day " \
				+ "and keeps it on through an overload. Fix it so both tests pass."

	func lines() -> Array:
		return [
			{"tokens": [["# keeps the market row lit; trips on an overload (max safe draw: 120)", "cm"]]},
			{"tokens": [["always", "kw"], [":", "op"]]},
			{"indent": 1, "tokens": [["power_on", "fn"], ["()", "op"]]},
			{"tokens": [["when ", "kw"], ["load", "fn"], ["() ", "op"], {"edit": "op", "text": OPS[op]},
					[" ", "op"], {"edit": "limit", "text": str(LIMITS[limit])}, [":", "op"]]},
			{"indent": 1, "tokens": [["cut_power", "fn"], ["()", "op"], ["      # trip the breaker", "cm"]]},
		]

	func step(id: String, dir: int) -> void:
		if id == "op":
			op = posmod(op + dir, OPS.size())
		else:
			limit = posmod(limit + dir, LIMITS.size())

	func run() -> Array:
		var out := []
		for case in [["a normal day", 40, false], ["an overload", 160, true]]:
			var cut := _compare(int(case[1]), OPS[op], LIMITS[limit])
			out.append(["%s: load %d -> power %s (should be %s)" % [case[0], case[1],
					"CUT" if cut else "ON", "CUT" if case[2] else "ON"], cut == case[2]])
		return out

	static func _compare(a: int, o: String, b: int) -> bool:
		match o:
			"<": return a < b
			">": return a > b
			"<=": return a <= b
			">=": return a >= b
			"==": return a == b
		return a != b


class CargoPuzzle extends Puzzle:
	const BOLTS := 3
	var count := 5

	func _init() -> void:
		file = "seal.prog"
		brief = "The Null changed the seal's loop: it tries to drive bolts the hold door doesn't have, " \
				+ "and the bolt drive jams. Make it drive every bolt, and only those."

	func lines() -> Array:
		return [
			{"tokens": [["# the hold door has 3 bolts: bolt 0, 1 and 2", "cm"]]},
			{"tokens": [["when ", "kw"], ["door_closed", "fn"], ["():", "op"]]},
			{"indent": 1, "tokens": [["for ", "kw"], ["i", "var"], [" in ", "kw"], ["range", "fn"], ["(", "op"],
					{"edit": "count", "text": str(count)}, ["):", "op"]]},
			{"indent": 2, "tokens": [["drive_bolt", "fn"], ["(", "op"], ["i", "var"], [")", "op"]]},
			{"indent": 1, "tokens": [["lock", "fn"], ["()", "op"]]},
		]

	func step(_id: String, dir: int) -> void:
		count = clampi(count + dir, 1, 6)

	func run() -> Array:
		var driven := mini(count, BOLTS)
		var out := [["drive_bolt(0..%d): %d of %d bolts home" % [driven - 1, driven, BOLTS], driven == BOLTS]]
		if count > BOLTS:
			out.append(["drive_bolt(%d): NO SUCH BOLT, the drive jams" % BOLTS, false])
		else:
			out.append(["no stray bolts driven", true])
		out.append(["lock(): the seal %s" % ("holds" if count == BOLTS else "won't hold"), count == BOLTS])
		return out

			"transmit": [["transmit", "fn"], ["(", "op"], ["convoy_id", "var"], [")", "op"]]}
	var order := ["transmit", "tune", "power_on"]

	func _init() -> void:
		file = "relay.prog"
		brief = "The Null shuffled the relay's start-up: it transmits before it's switched on, so the " \
				+ "signal never goes out. Put the steps in the order that works."

	func lines() -> Array:
		var out := [
			{"tokens": [["# start-up: the convoy's signal on band 7", "cm"]]},
			{"tokens": [["always", "kw"], [":", "op"]]},
		]
		for i in order.size():
			out.append({"indent": 1, "move": i, "tokens": TEXT[order[i]]})
		return out

	func move(slot: int, dir: int) -> void:
		var to := slot + dir
		if to < 0 or to >= order.size():
			return
		var a = order[slot]
		order[slot] = order[to]
		order[to] = a

	func run() -> Array:
		var on := false
		var tuned := false
		var out := []
		for s in order:
			match s:
				"power_on":
					on = true
					out.append(["power_on(): relay on", true])
				"tune":
					if on:
						tuned = true
						out.append(["tune(band=7): locked on", true])
					else:
						out.append(["tune(band=7): no power, nothing to tune", false])
				"transmit":
					if not on:
						out.append(["transmit(): the relay is off", false])
					elif not tuned:
						out.append(["transmit(): untuned, the signal is noise", false])
					else:
						out.append(["transmit(): signal out, the convoy is back on the net", true])
		return out


class IffPuzzle extends Puzzle:
	const OPS := ["==", "!="]
	const STATUSES := ["INBOUND", "DELETED", "DOCKED"]
	const RECORDS := [
		{"n": 1, "status": "DELETED", "note": "arrived 11 days ago, flagged NULL, deleted"},
		{"n": 2, "status": "INBOUND", "note": "the convoy on approach"},
	]
	var op := 1
	var status := 0

	func _init() -> void:
		file = "iff.prog"
		brief = "Checkpoint's guns challenged us and the transponder answered with the wrong record, " \
				+ "so they read us as Null. Its registry has two Hardfaults. Make it answer as ours."

	func lines() -> Array:
		return [
			{"tokens": [["# Checkpoint's registry, sent with the challenge:", "cm"]]},
			{"tokens": [["#   record 1  HARDFAULT  DELETED   arrived 11 days ago, flagged NULL", "cm"]]},
			{"tokens": [["#   record 2  HARDFAULT  INBOUND", "cm"]]},
			{"tokens": [["when ", "kw"], ["challenged", "fn"], ["():", "op"]]},
			{"indent": 1, "tokens": [["rec", "var"], [" = ", "op"], ["registry", "var"], [".", "op"],
					["find", "fn"], ["(", "op"], ["name", "var"], [" == ", "op"], ["\"HARDFAULT\"", "num"],
					[", ", "op"], ["status", "var"], [" ", "op"], {"edit": "op", "text": OPS[op]},
					[" ", "op"], {"edit": "status", "text": STATUSES[status]}, [")", "op"]]},
			{"indent": 1, "tokens": [["answer", "fn"], ["(", "op"], ["rec", "var"], [")", "op"],
					["      # the guns hold fire for a live record", "cm"]]},
		]

	func step(id: String, dir: int) -> void:
		if id == "op":
			op = posmod(op + dir, OPS.size())
		else:
			status = posmod(status + dir, STATUSES.size())

	func run() -> Array:
		var found := []
		for r in RECORDS:
			var same: bool = r["status"] == STATUSES[status]
			if same == (OPS[op] == "=="):
				found.append(r)
		var out := [["registry.find(): %d record%s match%s" % [found.size(),
				"" if found.size() == 1 else "s", "es" if found.size() == 1 else ""], found.size() == 1]]
		if found.is_empty():
			out.append(["answer(nothing): no answer, the guns read silence as Null -> FIRE", false])
		elif found.size() > 1:
			out.append(["answer(records 1 and 2): two of us at once, the guns read a copy -> FIRE", false])
		elif int(found[0]["n"]) == 1:
			out.append(["answer(record 1): HARDFAULT, %s -> FIRE" % found[0]["note"], false])
		else:
			out.append(["answer(record 2): HARDFAULT, INBOUND -> HOLD FIRE", true])
		return out

class WardenCrowdPuzzle extends Puzzle:
	const OPS := ["<", ">", "<=", ">=", "==", "!="]
	var op := 0
	var count := 3

	func _init() -> void:
		file = "crowd.prog"
		brief = "CHALLENGE: prove you read a crowd. Answer when 3 or more enemies are close, " \
				+ "and stay silent when fewer are."

	func lines() -> Array:
		return [
			{"tokens": [["# the Warden sends 2, then 3, then 5 drones at you", "cm"]]},
			{"tokens": [["when ", "kw"], ["enemies_nearby", "fn"], ["() ", "op"], {"edit": "op", "text": OPS[op]},
					[" ", "op"], {"edit": "count", "text": str(count)}, [":", "op"]]},
			{"indent": 1, "tokens": [["answer", "fn"], ["()", "op"]]},
		]

	func step(id: String, dir: int) -> void:
		if id == "op":
			op = posmod(op + dir, OPS.size())
		else:
			count = clampi(count + dir, 1, 6)

	func run() -> Array:
		var out := []
		for n in [2, 3, 5]:
			var fired := _cmp(n, OPS[op], count)
			var want: bool = n >= 3
			out.append(["%d drones close: %s (should %s)" % [n, "ANSWER" if fired else "silent",
					"answer" if want else "stay silent"], fired == want])
		return out

	static func _cmp(a: int, o: String, b: int) -> bool:
		match o:
			"<": return a < b
			">": return a > b
			"<=": return a <= b
			">=": return a >= b
			"==": return a == b
		return a != b

			{"indent": 2, "tokens": [["reply", "fn"], ["(", "op"], ["ping", "var"], ["[", "op"], ["i", "var"], ["])", "op"]]},
		]

			"reply": [["reply", "fn"], ["(", "op"], ["signature", "var"], [")", "op"]]}
	var order := ["reply", "sign", "receive"]

	func _init() -> void:
		file = "handshake.prog"
		brief = "CHALLENGE: a handshake has an order. Take the challenge in, sign it with our key, " \
				+ "then send the signature back."

	func lines() -> Array:
		var out := [{"tokens": [["when ", "kw"], ["challenged", "fn"], ["():", "op"]]}]
		for i in order.size():
			out.append({"indent": 1, "move": i, "tokens": TEXT[order[i]]})
		return out

	func move(slot: int, dir: int) -> void:
		var to := slot + dir
		if to < 0 or to >= order.size():
			return
		var a = order[slot]
		order[slot] = order[to]
		order[to] = a

	func run() -> Array:
		var got := false
		var signed := false
		var out := []
		for s in order:
			match s:
				"receive":
					got = true
					out.append(["receive(): challenge in", true])
				"sign":
					if got:
						signed = true
						out.append(["sign(): signed with our key", true])
					else:
						out.append(["sign(): nothing to sign yet", false])
				"reply":
					if signed:
						out.append(["reply(): signature sent, the Warden checks it", true])
					else:
						out.append(["reply(): an unsigned reply, read as forged", false])
		return out


class WardenIdentifyPuzzle extends Puzzle:
	const OPS := ["==", "!="]
	const STATUSES := ["DELETED", "INBOUND", "DOCKED"]
	var op := 0
	var status := 0

	func _init() -> void:
		file = "identify.prog"
		brief = "CHALLENGE: identify yourself. The registry still holds both Hardfaults. " \
				+ "Name the one that is you."

	func lines() -> Array:
		return [
			{"tokens": [["#   record 1  HARDFAULT  DELETED   flagged NULL", "cm"]]},
			{"tokens": [["#   record 2  HARDFAULT  INBOUND", "cm"]]},
			{"tokens": [["when ", "kw"], ["challenged", "fn"], ["():", "op"]]},
			{"indent": 1, "tokens": [["identify", "fn"], ["(", "op"], ["status", "var"], [" ", "op"],
					{"edit": "op", "text": OPS[op]}, [" ", "op"], {"edit": "status", "text": STATUSES[status]}, [")", "op"]]},
		]

	func step(id: String, dir: int) -> void:
		if id == "op":
			op = posmod(op + dir, OPS.size())
		else:
			status = posmod(status + dir, STATUSES.size())

	func run() -> Array:
		var records := [[1, "DELETED"], [2, "INBOUND"]]
		var found := []
		for r in records:
			if (r[1] == STATUSES[status]) == (OPS[op] == "=="):
				found.append(r[0])
		if found.size() == 1 and found[0] == 2:
			return [["identify(): record 2, HARDFAULT, INBOUND", true]]
		if found.is_empty():
			return [["identify(): no record. The Warden reads silence as Null", false]]
		if found.size() > 1:
			return [["identify(): both records at once. The Warden reads a copy", false]]
		return [["identify(): record 1, HARDFAULT, DELETED. The Warden reads Null", false]]
