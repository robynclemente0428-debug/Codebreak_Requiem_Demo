class_name BlastBehavior
extends SkillBehavior

const PLAYER_SWARM_GROUP := &"player_blast_swarm"
const ENEMY_SWARM_GROUP := &"enemy_blast_swarm"
const BASE_HIT_MASK := 5

func release(cast: SkillCast) -> void:
	var caster := cast.caster
	var swarm := _find_swarm(caster)
	if swarm == null:
		push_error("BlastBehavior: no BlastSwarm found for %s" % caster.name)
		return

	var style := SkillStyle.for_caster(caster)
	swarm.set_style(style)

	var origin := SkillDeliveries.spawn_point(caster, cast.flag("spawn_height"), cast.flag("spawn_forward"))
	var exclude_rid := RID()
	if caster is CollisionObject3D:
		exclude_rid = caster.get_rid()
	var mask := SkillTargets.projectile_mask(caster, BASE_HIT_MASK)

	var directions := SkillDeliveries.fan_directions(cast.forward(),
			int(cast.flag("max_projectiles")), cast.flag("spread_angle"))
	for direction in directions:
		swarm.spawn(origin, direction, style["primary"],
			cast.flag("speed"), cast.flag("range"), mask, exclude_rid,
			_explode.bind(cast, style, direction))


func aim_preview(caster: Node3D, skill: SkillInstance) -> Array:
	var origin := SkillDeliveries.spawn_point(caster, skill.get_flag("spawn_height"), skill.get_flag("spawn_forward"))
	var out := []
	for d in SkillDeliveries.fan_directions(SkillDeliveries.forward(caster),
			int(skill.get_flag("max_projectiles")), skill.get_flag("spread_angle")):
		out.append({"from": origin, "direction": d, "range": skill.get_flag("range"),
				"radius": skill.get_flag("radius"), "pierce": 0})
	return out


func waits_for_impact() -> bool:
	return true


func repeat_release(cast: SkillCast, from_impact: Variant, _direction: Vector3) -> void:
	var origin := SkillDeliveries.spawn_point(cast.caster, cast.flag("spawn_height"), cast.flag("spawn_forward"))
	var aim := {"from": origin, "direction": cast.forward(), "range": cast.flag("range"),
		"seek_from": from_impact if from_impact is Vector3 else origin}
	for s in cast.skill.support_behaviors:
		s.aim_repeat(cast, aim)
	for direction in SkillDeliveries.fan_directions(aim["direction"],
			int(cast.flag("max_projectiles")), cast.flag("spread_angle")):
		fire_shot(cast, aim["from"], direction, aim["range"])


func _find_swarm(caster: Node3D) -> BlastSwarm:
	var group := ENEMY_SWARM_GROUP if SkillTargets.is_enemy(caster) else PLAYER_SWARM_GROUP
	return caster.get_tree().get_first_node_in_group(group) as BlastSwarm


func _explode(center: Vector3, cast: SkillCast, style: Dictionary, direction := Vector3.ZERO) -> void:
	if not cast.is_valid():
		return
	var radius := cast.flag("radius") * float(cast.extra.get("radius_mult", 1.0))
	ImpactEffect.spawn(cast.effect_parent(), center, radius, style["secondary"], style["primary"])
	SoundManager.play_blast_impact(cast.effect_parent(), center)
	for target in SkillDeliveries.hits_in_circle(cast, center, radius):
		cast.hit(target)
	for s in cast.skill.support_behaviors:
		s.on_area_impact(cast, center, direction)


func fire_shot(cast: SkillCast, origin: Vector3, direction: Vector3, max_range: float) -> void:
	var caster := cast.caster
	if not cast.is_valid():
		return
	var swarm := _find_swarm(caster)
	if swarm == null:
		return
	var style := SkillStyle.for_caster(caster)
	var exclude_rid := RID()
	if caster is CollisionObject3D:
		exclude_rid = caster.get_rid()
	swarm.spawn(origin, direction, style["primary"], cast.flag("speed"), maxf(max_range, 0.5),
			SkillTargets.projectile_mask(caster, BASE_HIT_MASK), exclude_rid,
			_explode.bind(cast, style, direction))

class_name ClassTrait
extends Resource

static func for_class(class_id: String) -> ClassTrait:
	var path := "res://data/class_traits/%s.tres" % class_id
	return load(path) as ClassTrait if class_id != "" and ResourceLoader.exists(path) else null


@export var display_name: String = ""
@export_multiline var description: String = ""

@export_group("Program knobs")
@export var knob_cost_mult: float = 1.0
@export var knob_effect_mult: float = 1.0
@export var extra_points: float = 0.0
@export var buff_cost_mult: float = 1.0
@export var nth_rule_cost_mult: float = 1.0
@export var nth_rule_effect_mult: float = 1.0
@export var nth_rule_effect_per_n: float = 0.0
@export var hit_rule_cost_mult: float = 1.0
@export var hit_rule_effect_mult: float = 1.0
@export var per_hit_any_knob: bool = false

@export_group("Stats")
@export var armour_per_always_rule: float = 0.0
@export var armour_per_block: float = 0.0
@export var armour_per_when_fire: float = 0.0
@export var when_buff_seconds: float = 4.0
@export var damage_per_10_armour: float = 0.0
@export var flat_damage_per_10_armour: float = 0.0
@export var move_speed_per_rule: float = 0.0
@export var move_speed_per_knob_step: float = 0.0
@export var moving_damage_more: float = 0.0
@export var moving_damage_more_per_speed: float = 0.0
@export var full_life_damage_more: float = 0.0

@export_group("Movement")
@export var phase_through_enemies: bool = false


func rule_cost_mult(rule: ProgramRule) -> float:
	return (nth_rule_cost_mult if _is_nth(rule) else 1.0) * (hit_rule_cost_mult if _is_per_hit(rule) else 1.0)


func rule_effect_mult(rule: ProgramRule, include_hit := true) -> float:
	return (nth_effect_mult(rule) if _is_nth(rule) else 1.0) \
			* (hit_rule_effect_mult if include_hit and _is_per_hit(rule) else 1.0)


func nth_effect_mult(rule: ProgramRule) -> float:
	var n := 2
	for c in rule.conditions:
		if c is EveryNthCastCondition:
			n = maxi(n, (c as EveryNthCastCondition).n)
	return nth_rule_effect_mult + nth_rule_effect_per_n * (n - 2)


static func _is_per_hit(rule: ProgramRule) -> bool:
	return rule != null and rule.loop == null and rule.scope() == ProgramCondition.Scope.HIT


static func _is_nth(rule: ProgramRule) -> bool:
	if rule == null:
		return false
	for c in rule.conditions:
		if c is EveryNthCastCondition:
			return true
	return false


func adjust_knob(k: SkillKnob) -> SkillKnob:
	if is_equal_approx(knob_cost_mult, 1.0) and is_equal_approx(knob_effect_mult, 1.0):
		return k
	var c := k.duplicate() as SkillKnob
	c.cost_per_step = k.cost_per_step * knob_cost_mult
	c.step = k.step * knob_effect_mult
	return c


const RULES_SOURCE := "class_trait_rules"

func apply_rule_bonuses(stats: StatComponent, always_rules: int, total_rules: int, knob_steps: int = 0,
		blocks: int = 0) -> void:
	if stats == null:
		return
	stats.remove_by_source(RULES_SOURCE)
	if armour_per_always_rule > 0.0 and always_rules > 0:
		stats.add_modifier(_mod(StatTypes.Stat.ARMOUR, Modifier.ModOp.PERCENT_ADD,
				armour_per_always_rule * always_rules, RULES_SOURCE))
	if armour_per_block > 0.0 and blocks > 0:
		stats.add_modifier(_mod(StatTypes.Stat.ARMOUR, Modifier.ModOp.PERCENT_ADD,
				armour_per_block * blocks, RULES_SOURCE))
	if move_speed_per_rule > 0.0 and total_rules > 0:
		stats.add_modifier(_mod(StatTypes.Stat.MOVE_SPEED, Modifier.ModOp.FLAT_ADD,
				move_speed_per_rule * total_rules, RULES_SOURCE))
	if move_speed_per_knob_step > 0.0 and knob_steps > 0:
		stats.add_modifier(_mod(StatTypes.Stat.MOVE_SPEED, Modifier.ModOp.FLAT_ADD,
				move_speed_per_knob_step * knob_steps, RULES_SOURCE))


func armour_damage_increase(caster_stats: StatComponent) -> float:
	if damage_per_10_armour <= 0.0 or caster_stats == null:
		return 0.0
	return damage_per_10_armour * floorf(maxf(caster_stats.get_stat(StatTypes.Stat.ARMOUR), 0.0) / 10.0)


func armour_flat_damage(caster_stats: StatComponent) -> float:
	if flat_damage_per_10_armour <= 0.0 or caster_stats == null:
		return 0.0
	return flat_damage_per_10_armour * floorf(maxf(caster_stats.get_stat(StatTypes.Stat.ARMOUR), 0.0) / 10.0)


func on_rule_fired(caster: Node, key: String) -> void:
	if armour_per_when_fire <= 0.0 or not is_instance_valid(caster):
		return
	var stats := caster.get("stats") as StatComponent
	if stats == null:
		return
	var source := "class_trait_when_%s" % key
	stats.remove_by_source(source)
	stats.add_modifier(_mod(StatTypes.Stat.ARMOUR, Modifier.ModOp.PERCENT_ADD, armour_per_when_fire, source))
	var token: int = int(stats.get_meta(source, 0)) + 1
	stats.set_meta(source, token)
	var timer := caster.get_tree().create_timer(when_buff_seconds, false)
	timer.timeout.connect(func() -> void:
		if is_instance_valid(stats) and int(stats.get_meta(source, 0)) == token:
			stats.remove_by_source(source))


func hit_damage_mult(caster: Node, target: Node = null) -> float:
	var m := 1.0
	if (moving_damage_more > 0.0 or moving_damage_more_per_speed > 0.0) and caster is CharacterBody3D:
		var v := (caster as CharacterBody3D).velocity
		if Vector2(v.x, v.z).length() > 0.5:
			m *= 1.0 + moving_damage_more
			var stats := caster.get("stats") as StatComponent
			if moving_damage_more_per_speed > 0.0 and stats != null:
				m *= 1.0 + moving_damage_more_per_speed * maxf(stats.get_stat(StatTypes.Stat.MOVE_SPEED), 0.0)
	if full_life_damage_more > 0.0 and target != null and not target.is_in_group(&"breakables") \
			and ProgramCondition.health_fraction(target) >= 0.999:
		m *= 1.0 + full_life_damage_more
	return m


func summary() -> PackedStringArray:
	var out := PackedStringArray()
	if not is_equal_approx(knob_cost_mult, 1.0) or not is_equal_approx(knob_effect_mult, 1.0):
		out.append("knob steps cost x%s, effect x%s" % [_n(knob_cost_mult), _n(knob_effect_mult)])
	if extra_points != 0.0:
		out.append("%+d program points" % roundi(extra_points))
	if not is_equal_approx(buff_cost_mult, 1.0):
		out.append("buffs cost %d%% %s points" % [roundi(absf(1.0 - buff_cost_mult) * 100.0),
			"fewer" if buff_cost_mult < 1.0 else "more"])
	if not is_equal_approx(nth_rule_cost_mult, 1.0) or not is_equal_approx(nth_rule_effect_mult, 1.0):
		var eff := "x%s" % _n(nth_rule_effect_mult)
		if nth_rule_effect_per_n > 0.0:
			eff += " (+%s per N above 2)" % _n(nth_rule_effect_per_n)
		out.append("every-Nth-cast rules cost x%s, effect %s" % [_n(nth_rule_cost_mult), eff])
	if not is_equal_approx(hit_rule_cost_mult, 1.0) or not is_equal_approx(hit_rule_effect_mult, 1.0):
		out.append("per-hit rules cost x%s, effect x%s" % [_n(hit_rule_cost_mult), _n(hit_rule_effect_mult)])
	if per_hit_any_knob:
		out.append("per-hit rules can change any flag knob, and attack speed for your next attack")
	if armour_per_always_rule > 0.0:
		out.append("+%d%% armour per ALWAYS rule" % roundi(armour_per_always_rule * 100.0))
	if armour_per_block > 0.0:
		out.append("+%d%% armour per block in your programs" % roundi(armour_per_block * 100.0))
	if armour_per_when_fire > 0.0:
		out.append("+%d%% armour for %ds when a WHEN rule fires" % [roundi(armour_per_when_fire * 100.0), roundi(when_buff_seconds)])
	if damage_per_10_armour > 0.0:
		out.append("%s%% increased damage per 10 armour" % _n(damage_per_10_armour * 100.0))
	if flat_damage_per_10_armour > 0.0:
		out.append("adds %s - %s damage per 10 armour" % [_n(flat_damage_per_10_armour), _n(flat_damage_per_10_armour)])
	if move_speed_per_rule > 0.0:
		out.append("+%s movement speed per rule" % _n(move_speed_per_rule))
	if move_speed_per_knob_step > 0.0:
		out.append("+%s movement speed per knob step" % _n(move_speed_per_knob_step))
	if moving_damage_more > 0.0:
		out.append("%d%% more damage while moving" % roundi(moving_damage_more * 100.0))
	if moving_damage_more_per_speed > 0.0:
		out.append("%d%% more damage per 1 movement speed while moving" % roundi(moving_damage_more_per_speed * 100.0))
	if full_life_damage_more > 0.0:
		out.append("%d%% more damage against enemies at full life" % roundi(full_life_damage_more * 100.0))
	if phase_through_enemies:
		out.append("move through enemies")
	return out


static func _n(v: float) -> String:
	return str(snappedf(v, 0.01)).trim_suffix(".0")


static func _mod(stat: int, op: int, value: float, source: String) -> Modifier:
	var m := Modifier.new()
	m.stat = stat
	m.op = op
	m.value = value
	m.source_id = source
	return m

class_name HealthComponent
extends Node

@export var stats: StatComponent
@export var regen_interval: float = 0.25

@export var max_damage_reduction: float = 0.9
const ARMOUR_CONSTANT := 10.0

var current_health: float = 0.0
var barrier: float = 0.0
var guard: float = 0.0
var _barrier_until := 0
var _barrier_seconds := 0.0
var _guard_until := 0
var last_damaged_msec := -1000000
var _initialized: bool = false
var _regen_timer: float = 0.0
var _last_max: float = 0.0

signal health_changed(current: float, max: float)
signal damage_taken(amount: float)
signal died
signal barrier_changed(amount: float)

var remote_damage := Callable()
var remote_shielded := false
var untouchable := false


signal damage_shown(amount: float)


func apply_remote(value: float, show_hit := true) -> void:
	var was_alive := current_health > 0.0
	if show_hit and was_alive and value < current_health:
		damage_shown.emit(current_health - maxf(value, 0.0))
	current_health = value
	_emit_changed()
	if was_alive and value <= 0.0:
		died.emit()

func initialize() -> void:
	current_health = stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	_initialized = true
	_emit_changed()

func _process(delta: float) -> void:
	if barrier > 0.0 and Time.get_ticks_msec() >= _barrier_until:
		barrier = 0.0
		barrier_changed.emit(0.0)
	if not _initialized or current_health <= 0.0:
		return
	_regen_timer += delta
	if _regen_timer < regen_interval:
		return
	var elapsed := _regen_timer
	_regen_timer = 0.0
	var regen := stats.get_stat(StatTypes.Stat.REGENERATION)
	if regen <= 0.0 or current_health >= stats.get_stat(StatTypes.Stat.MAX_HEALTH):
		return
	heal(regen * elapsed)

func take_damage(amount: float, ignore_armour := false) -> float:
	if not _initialized:
		push_warning("take_damage called before HealthComponent.initialize()")
		return 0.0
	if current_health <= 0.0:
		return 0.0
	if remote_damage.is_valid():
		remote_damage.call(amount, ignore_armour)
		return amount if ignore_armour else mitigate(amount, stats.get_stat(StatTypes.Stat.ARMOUR))
	if untouchable:
		return 0.0
	var now := Time.get_ticks_msec()
	if guard > 0.0 and now < _guard_until:
		amount *= 1.0 - guard
	if barrier > 0.0 and now < _barrier_until:
		var absorbed := minf(barrier, amount)
		barrier -= absorbed
		amount -= absorbed
		barrier_changed.emit(barrier)
		if amount <= 0.0:
			return 0.0
	var armour: float = stats.get_stat(StatTypes.Stat.ARMOUR)
	var mitigated := amount if ignore_armour else mitigate(amount, armour)
	last_damaged_msec = now
	current_health -= mitigated
	damage_taken.emit(mitigated)
	_emit_changed()
	if current_health <= 0.0:
		died.emit()
	return mitigated

func add_barrier(amount: float, seconds: float) -> void:
	if amount <= 0.0 or current_health <= 0.0:
		return
	barrier = maxf(barrier if Time.get_ticks_msec() < _barrier_until else 0.0, amount)
	_barrier_until = Time.get_ticks_msec() + int(seconds * 1000.0)
	_barrier_seconds = seconds
	barrier_changed.emit(barrier)


func barrier_time_left() -> float:
	if barrier <= 0.0 or _barrier_seconds <= 0.0:
		return 0.0
	return clampf((_barrier_until - Time.get_ticks_msec()) / (_barrier_seconds * 1000.0), 0.0, 1.0)


func add_guard(fraction: float, seconds: float) -> void:
	if fraction <= 0.0:
		return
	var now := Time.get_ticks_msec()
	guard = maxf(guard if now < _guard_until else 0.0, clampf(fraction, 0.0, 0.9))
	_guard_until = maxi(_guard_until, now + int(seconds * 1000.0))


func mitigate(amount: float, armour: float) -> float:
	var reduction := armour / (armour + ARMOUR_CONSTANT * amount)
	reduction = minf(reduction, max_damage_reduction)
	return maxf(amount * (1.0 - reduction), 1.0)    

func heal(amount: float) -> void:
	if not _initialized or current_health <= 0.0:
		return
	var max_hp := stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	current_health = min(current_health + amount, max_hp)
	_emit_changed()

func _on_stats_changed() -> void:
	if not _initialized:
		return
	var new_max := stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	var clamped := minf(current_health, new_max)
	if is_equal_approx(clamped, current_health) and is_equal_approx(new_max, _last_max):
		return
	current_health = clamped
	_emit_changed()

func revive(fraction: float = 1.0) -> void:
	current_health = stats.get_stat(StatTypes.Stat.MAX_HEALTH) * fraction
	_regen_timer = 0.0
	_emit_changed()

func _emit_changed() -> void:
	_last_max = stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	health_changed.emit(current_health, _last_max)

func is_shielded() -> bool:
	if remote_damage.is_valid():
		return remote_shielded
	var now := Time.get_ticks_msec()
	return (barrier > 0.0 and now < _barrier_until) or (guard > 0.0 and now < _guard_until)

class_name HitFacts
extends RefCounted

var caster: Node
var by_player := false
var gem_id := ""
var gem: SkillGemDefinition
var script_set := -1
var owner_set := 0
var fired: Array = []
var rule_pairs: Array = []
var tier := ""
var has_script := false
var blocks_to_next := 0
var cast_id := ""
var root_id := ""
var root_damage_mult := 1.0
var loop_id := ""
var loop_repeat := false
var forked := false


static func from_cast(cast: SkillCast) -> HitFacts:
	if cast == null or not cast.is_valid():
		return null
	var f := HitFacts.new()
	f.caster = cast.caster
	f.by_player = cast.caster.is_in_group(&"players")
	var skill := cast.skill
	if skill != null and skill.gem != null:
		f.gem = skill.gem
		f.gem_id = String(skill.gem.id)
	var skills := cast.caster.get("skills") as SkillComponent
	f.script_set = skills.script_set if skills != null else -1
	f.owner_set = f.script_set
	if skills == null:
		var own := Coop.own_player(cast.caster.get_tree())
		var own_skills := own.get("skills") as SkillComponent if own != null else null
		f.owner_set = own_skills.script_set if own_skills != null else 0
	f.fired = ProgramSupport.fired_conditions(cast)
	var support := ProgramSupport.of(skill)
	if support != null and support.program != null:
		for rule in support.program.flat_rules():
			if rule.conditions.size() == 2:
				f.rule_pairs.append([rule.conditions[0].type_id(), rule.conditions[1].type_id()])
	var v = skill.program_validation if skill != null else null
	if v != null and v.valid:
		f.tier = v.tier.display_name.to_lower() if v.tier != null else ""
		f.has_script = skill.program != null and skill.program.has_rules()
		f.blocks_to_next = v.actions_to_next_tier if v.next_tier != null else 0
	f.cast_id = _id(cast)
	var runner := cast.extra.get(ProgramSupport.LOOP_KEY) as ProgramLoopRunner
	var root: SkillCast = runner.root if runner != null and runner.root != null else cast
	f.root_id = _id(root)
	f.root_damage_mult = root.damage_mult
	if runner != null:
		f.loop_id = _id(runner)
		_track_loop(f.loop_id, runner)
	f.loop_repeat = bool(cast.extra.get("loop_repeat", false))
	f.forked = bool(cast.extra.get("forked", false))
	return f


static var _loops := {}
const LOOPS_KEEP := 256


static var _remote := {}


static func loop_running(id: String) -> bool:
	if _remote.has(id):
		return bool(_remote[id]["running"])
	var w: WeakRef = _loops.get(id)
	var runner := w.get_ref() as ProgramLoopRunner if w != null else null
	if runner == null:
		_loops.erase(id)
		return false
	return not runner.finished


static func remote_loop(id: String, running: bool) -> void:
	if not running and not _remote.has(id):
		return
	if running and _remote.has(id) and not bool(_remote[id]["running"]):
		return
	var now := Time.get_ticks_msec()
	if _remote.size() > LOOPS_KEEP:
		for k: String in _remote.keys():
			if now - int(_remote[k]["at"]) > 60000:
				_remote.erase(k)
	_remote[id] = {"running": running, "at": now}


static func _track_loop(id: String, runner: ProgramLoopRunner) -> void:
	if _loops.has(id):
		return
	if _loops.size() > LOOPS_KEEP:
		for k: String in _loops.keys():
			if (_loops[k] as WeakRef).get_ref() == null:
				_loops.erase(k)
	_loops[id] = weakref(runner)


static func _id(o: Object) -> String:

	f.script_set = int(d.get("set", -1))

	f.fired = _short_list(d.get("fired"), 16)

class_name ManaComponent
extends Node

@export var stats: StatComponent
@export var regen_interval: float = 0.25
@export var base_regen_per_second: float = 2.0
@export var regen_delay: float = 0.0
@export var start_full: bool = true

var current_mana: float = 0.0
var _initialized: bool = false
var _regen_timer: float = 0.0
var _delay_left: float = 0.0
var _last_max: float = 0.0

signal mana_changed(current: float, max: float)
signal spend_failed(needed: float, current: float)


func initialize() -> void:
	var m := get_max_mana()
	current_mana = m if start_full else 0.0
	_initialized = true
	_emit_changed()


func get_max_mana() -> float:
	if stats == null:
		return 0.0
	return maxf(stats.get_stat(StatTypes.Stat.MANA), 0.0)

func get_regen() -> float:
	var r := base_regen_per_second
	if stats:
		r += stats.get_stat(StatTypes.Stat.MANA_REGEN)
	return maxf(r, 0.0)


func has_enough(amount: float) -> bool:
	return _initialized and current_mana >= amount


func spend(amount: float) -> bool:
	if not _initialized or amount <= 0.0:
		return _initialized
	if current_mana < amount:
		spend_failed.emit(amount, current_mana)
		return false
	current_mana -= amount
	_delay_left = regen_delay
	_emit_changed()
	return true


func restore(amount: float) -> void:
	if not _initialized:
		return
	current_mana = minf(current_mana + amount, get_max_mana())
	_emit_changed()


func restore_full() -> void:
	if _initialized:
		current_mana = get_max_mana()
		_emit_changed()


func _process(delta: float) -> void:
	if not _initialized:
		return
	if _delay_left > 0.0:
		_delay_left -= delta
		return
	_regen_timer += delta
	if _regen_timer < regen_interval:
		return
	var elapsed := _regen_timer
	_regen_timer = 0.0
	var regen := get_regen()
	if regen <= 0.0 or current_mana >= get_max_mana():
		return
	restore(regen * elapsed)


func _on_stats_changed() -> void:
	if not _initialized:
		return
	var new_max := get_max_mana()
	var clamped := minf(current_mana, new_max)
	if is_equal_approx(clamped, current_mana) and is_equal_approx(new_max, _last_max):
		return
	current_mana = clamped
	_emit_changed()


func _emit_changed() -> void:
	_last_max = get_max_mana()
	mana_changed.emit(current_mana, _last_max)

func gain_on_hit(amount: float) -> void:
	if amount > 0.0:
		restore(amount)

	var origin := SkillDeliveries.spawn_point(caster, spawn_height, spawn_forward)

	if caster.has_method("get_muzzle_point"):
		var muzzle = caster.get_muzzle_point()

			origin = muzzle
			if caster.has_method("get_aim_point"):

				to.y = 0.0
				if to.length() > 0.5:

	var caster := cast.caster
	var projectile := PROJECTILE_SCENE.instantiate() as ShootProjectile
	projectile.direction = direction
	projectile.tint = ShootProjectile.TINT_PLAYER if caster.is_in_group("players") 			else ShootProjectile.TINT_ENEMY

	projectile.global_position = origin

	projectile.hit_world.connect(_on_hit_world.bind(cast))


func _on_hit_enemy(target: Node, position: Vector3, pierced_so_far: int,
		projectile: ShootProjectile, cast: SkillCast) -> void:
	if not cast.is_valid() or not is_instance_valid(target):
		return
	var keep := maxf(1.0 - cast.flag("pierce_falloff"), 0.0)
	ShootTracers.spark(cast.effect_parent(), position, 0.3, projectile.tint if is_instance_valid(projectile) \
			else ShootProjectile.TINT_PLAYER)
	var info := cast.hit(target, pow(keep, pierced_so_far))
	if is_instance_valid(projectile):
		projectile.pierce = maxi(projectile.pierce, int(cast.flag("pierce")))
	if pierced_so_far == 0 and info.flag("fork_count") >= 1.0 and not cast.extra.get("forked", false) \
			and not target.is_in_group(&"breakables") and is_instance_valid(projectile):
		_fork(cast, projectile, position, info)


func _fork(cast: SkillCast, projectile: ShootProjectile, at: Vector3, info: HitInfo = null) -> void:
	var child := cast.spawn_child()
	child.extra = cast.extra.duplicate()
	child.extra["forked"] = true
	if info != null:
		for k in ["fork_count", "fork_angle"]:
			if info.flag_bonus.has(k):
				child.flag_bonus[k] = float(child.flag_bonus.get(k, 0.0)) + float(info.flag_bonus[k])
	child.damage_mult *= maxf(cast.flag("fork_damage"), 0.0)
	for s in cast.skill.support_behaviors:
		if s is ProgramSupport:
			(s as ProgramSupport).apply_fork_rules(child)
	var reach := maxf(projectile.max_distance - projectile.travelled(), cast.flag("range") * 0.5)
	var dirs := SkillDeliveries.fan_directions(projectile.direction,
			int(round(child.flag("fork_count"))), child.flag("fork_angle"))
	ShootTracers.spark(cast.effect_parent(), at, 0.45, Color(0.4, 0.95, 1.0))
	for d in dirs:
		_spawn_projectile(child, at, d, projectile.exclude, reach)


func _on_hit_world(position: Vector3, cast: SkillCast) -> void:
	if cast.is_valid():
		ShootTracers.spark(cast.effect_parent(), position, 0.25, Color(0.8, 0.8, 0.8))

class_name SkillBehavior
extends RefCounted

const BARRIER_SECONDS := 4.0
const GUARD_MIN_SECONDS := 0.6
const GUARD_MAX_SECONDS := 2.0


func execute(caster: Node3D, skill: SkillInstance, stats: StatComponent) -> void:
	var cast := SkillCast.new(caster, skill, stats, self)
	var delay := _play_animation(cast)
	if delay > 0.0:
		await caster.get_tree().create_timer(delay).timeout
		if not cast.is_valid() or is_dead(cast.caster):
			return
	if is_dead(caster):
		return
	for s in skill.support_behaviors:
		s.before_release(cast)
	_apply_defense(cast)
	release(cast)


func _apply_defense(cast: SkillCast) -> void:
	if cast.depth > 0 or cast.extra.get("hologram", false):
		return
	var hp := cast.caster.get("health") as HealthComponent
	if hp == null:
		return
	var barrier := cast.flag("barrier")
	if barrier > 0.0 and hp.stats != null:
		hp.add_barrier(barrier * hp.stats.get_stat(StatTypes.Stat.MAX_HEALTH), BARRIER_SECONDS)
	var guard := cast.flag("guard")
	if guard > 0.0:
		hp.add_guard(guard, clampf(recovery_time(cast), GUARD_MIN_SECONDS, GUARD_MAX_SECONDS))


static func is_dead(node: Node) -> bool:
	if not is_instance_valid(node):
		return true
	var h = node.get("health")
	return h is HealthComponent and h.current_health <= 0.0


func release(_cast: SkillCast) -> void:
	push_error("%s: release() not implemented" % get_script().resource_path)


func repeat_release(cast: SkillCast, _from_impact: Variant, _direction: Vector3) -> void:
	release(cast)


func aim_preview(_caster: Node3D, _skill: SkillInstance) -> Array:
	return []


func waits_for_impact() -> bool:
	return false


func recovery_time(cast: SkillCast) -> float:
	var recovery := maxf(cast.flag("recovery_mult"), 1.0)
	return recovery / maxf(cast.stats.get_stat(StatTypes.Stat.ATTACK_SPEED) * cast.skill.speed_mult(), 0.1)


func _play_animation(cast: SkillCast) -> float:
	var caster := cast.caster
	if not caster.has_method("play_skill_animation"):
		return 0.0
	var cooldown := recovery_time(cast)
	var key := cast.skill.peek_animation_key(cooldown + cast.flag("combo_grace"))
	var delay: float = caster.play_skill_animation(key, cooldown)
	if delay > 0.0:
		cast.skill.advance_animation()
	return delay


func hit(cast: SkillCast, target: Node, damage_mult := 1.0) -> HitInfo:
	var info := HitInfo.new()
	info.cast = cast
	info.target = target
	if not cast.is_valid() or not is_instance_valid(target):
		return info
	if cast.caster.get("is_remote") == true:
		return info
	var target_health := target.get("health") as HealthComponent
	if target_health == null:
		return info
	if target is Node3D:
		info.position = (target as Node3D).global_position
	info.damage = cast.skill.roll_damage(cast.stats) * damage_mult * cast.damage_mult
	if cast.skill.class_trait != null:
		info.damage *= cast.skill.class_trait.hit_damage_mult(cast.caster, target)
	info.is_crit = cast.skill.last_roll_was_crit

	for s in cast.skill.support_behaviors:
		s.modify_hit(cast, target, info)
	if info.cancelled:
		return info
	var traits: Array = target.get("traits") if "traits" in target else []
	var on_copy: bool = target.get("is_remote") == true and target is EnemyBase
	if not traits.is_empty() or on_copy:
		info.facts = HitFacts.from_cast(cast)
	if on_copy:
		Coop.hit_facts = info.facts
		traits = []
	for t: EnemyTrait in traits:
		t.modify_incoming(info)
	if info.cancelled:
		return info
	if info.absorbed:
		if target_health.current_health <= 0.0:
			return info
		info.landed = true
		cast.already_hit.append(target)
		for t: EnemyTrait in traits:
			if is_instance_valid(t):
				t.after_hit(info)
		for s in cast.skill.support_behaviors:
			s.on_hit(cast, target, info)
		return info
	var own: Array = cast.caster.get("traits") if "traits" in cast.caster else []
	for t: EnemyTrait in own:
		if is_instance_valid(t):
			t.on_outgoing(info)

	Coop.hit_source = Coop.Hit.PLAYER if cast.caster is Player else Coop.Hit.ENEMY
	Coop.damage_by = Coop.my_id() if cast.caster is Player and Coop.in_game else 0
	info.dealt = target_health.take_damage(info.damage)
	Coop.damage_by = 0
	Coop.hit_source = Coop.Hit.NONE
	Coop.hit_facts = null
	if info.dealt <= 0.0:
		return info
	info.landed = true
	info.killed = target_health.current_health <= 0.0
	Coop.my_hit(target, cast, info)
	cast.already_hit.append(target)
	for t: EnemyTrait in traits:
		if is_instance_valid(t):
			t.after_hit(info)

	if not target.is_in_group(&"breakables") and not on_copy:
		landing_rewards(cast, info)

	for s in cast.skill.support_behaviors:
		s.on_hit(cast, target, info)
		if info.killed:
			s.on_kill(cast, target, info)
	return info


func landing_rewards(cast: SkillCast, info: HitInfo) -> void:
	if not cast.is_valid():
		return
	_apply_on_hit_stats(cast)
	var leech := cast.flag("leech") + info.leech_bonus
	var caster_hp := cast.caster.get("health") as HealthComponent
	if leech > 0.0 and caster_hp != null:
		caster_hp.heal(info.dealt * leech)
	if caster_hp != null:
		if info.barrier_bonus > 0.0 and caster_hp.stats != null:
			caster_hp.add_barrier(info.barrier_bonus * caster_hp.stats.get_stat(StatTypes.Stat.MAX_HEALTH), BARRIER_SECONDS)
		if info.guard_bonus > 0.0:
			caster_hp.add_guard(info.guard_bonus, clampf(recovery_time(cast), GUARD_MIN_SECONDS, GUARD_MAX_SECONDS))


func _apply_on_hit_stats(cast: SkillCast) -> void:
	var caster := cast.caster
	var life_on_hit := cast.stats.get_stat(StatTypes.Stat.LIFE_ON_HIT)
	if life_on_hit > 0.0:
		var caster_health := caster.get("health") as HealthComponent
		if caster_health != null:
			caster_health.heal(life_on_hit)

	var mana_on_hit := cast.stats.get_stat(StatTypes.Stat.MANA_ON_HIT)
	if mana_on_hit > 0.0:
		var caster_mana := caster.get("mana") as ManaComponent
		if caster_mana != null:
			caster_mana.gain_on_hit(mana_on_hit)

		if index >= 0 and index < sockets.size() and sockets[index].gem != null:
			sockets[index].program = null
			sockets[index].sets[script_set] = null
			_rebuild(sockets[index])
			changed.emit()
		return null
	var result := preview_program(index, program)
	if result.valid:
		var s := sockets[index]
		s.program = program
		s.sets[script_set] = program
		_rebuild(s)
		_report_program(index)
		changed.emit()
	return result


func _report_program(index: int) -> void:
	var s := sockets[index]
	if s.program == null or s.skill == null:
		return
	var gem := s.gem.definition.skill_gem
	GameEvents.report("program_applied", gem.id)
	for t in gem.tags:
		GameEvents.report("program_applied_tag", TagTypes.name_of(TagTypes.TagGem, t))
	var conditions := {}
	var nested := false
	var branches := 0
	for rule in s.program.flat_rules():
		if not rule.is_always():
			branches += 1
		if rule.conditions.size() >= 2:
			nested = true
		for c in rule.conditions:
			conditions[c.type_id()] = true
	for type_id in conditions:
		GameEvents.report("program_condition", type_id)
	for rule in s.program.rules:
		if rule.loop != null:
			GameEvents.report("program_loop", "while" if rule.loop.mode == ProgramLoop.Mode.WHILE else "for")
	if nested:
		GameEvents.report("program_nested")
	if branches >= 2:
		GameEvents.report("program_branches")
	for routine_index in s.program.called_routines():
		GameEvents.report("program_routine", str(routine_index))
	var v := s.skill.program_validation
	if v != null and v.tier != null:
		GameEvents.report("program_efficient", v.tier.display_name.to_lower())

		skill.supports.append(item.definition.support_gem)
		support_ids.append(item.definition.support_gem.id)
	if s.program != null:
		s.program.prune_owners(support_ids)
	skill.program = s.program
	skill.routines = routines
	skill.class_trait = _class_trait()
	skill.rank = rank()
	skill.bonus_points = bonus_points()
	skill.bonus_steps = bonus_steps()

	var t := _class_trait()
	var stats := get_parent().get("stats") as StatComponent
	if t == null or stats == null:
		return
	var always_rules := 0
	var total_rules := 0
	var knob_steps := 0
	var blocks := 0
	for sock in sockets:
		if sock.skill == null or sock.program == null or sock.skill.program_validation == null \
				or not sock.skill.program_validation.valid:
			continue
		blocks += sock.skill.program_validation.actions_used
		var limits := sock.skill.get_program_limits()
		for r in sock.program.rules:
			total_rules += 1
			if r.is_always():
				always_rules += 1
			for e in r.effects:
				if ScriptRegisters.is_write(e.knob) or limits == null:
					continue
				var k := limits.get_knob(e.knob)
				if k != null and k.cost_for(e.steps) > 0.0:
					knob_steps += absi(e.steps)
	t.apply_rule_bonuses(stats, always_rules, total_rules, knob_steps, blocks)

class_name SkillDeliveries
extends RefCounted


static func forward(caster: Node3D) -> Vector3:
	var f := caster.global_transform.basis.z
	f.y = 0.0
	return f.normalized()


static func spawn_point(caster: Node3D, height: float, forward_offset: float) -> Vector3:
	return caster.global_position + Vector3.UP * height + forward(caster) * forward_offset


static func fan_directions(fwd: Vector3, count: int, step_degrees: float) -> Array[Vector3]:
	var dirs: Array[Vector3] = []
	count = maxi(count, 1)
	for i in count:
		var offset_deg := (float(i) - (count - 1) * 0.5) * step_degrees
		dirs.append(fwd.rotated(Vector3.UP, deg_to_rad(offset_deg)))
	return dirs


const NO_CAP := 0


static func hits_in_circle(cast: SkillCast, center: Vector3, radius: float, max_targets := NO_CAP,
		exclude: Array = []) -> Array:
	var targets := _candidates(cast, exclude)
	var indices := resolve_circle(center, radius, max_targets, _positions(targets))
	return _pick(targets, indices)


static func hits_in_arc(cast: SkillCast, origin: Vector3, fwd: Vector3, arc_degrees: float,
		reach: float, max_targets := NO_CAP, exclude: Array = []) -> Array:
	var targets := _candidates(cast, exclude)
	var indices := resolve_arc(origin, fwd, arc_degrees, reach, max_targets, _positions(targets))
	return _pick(targets, indices)


static func nearest_target(cast: SkillCast, point: Vector3, max_distance: float,
		exclude: Array = []) -> Node3D:
	var found := hits_in_circle(cast, point, max_distance, 1, exclude)
	return found[0] if not found.is_empty() else null


static func resolve_circle(center: Vector3, radius: float, max_targets: int,
		candidate_positions: Array) -> Array[int]:
	var found: Array = []
	for i in candidate_positions.size():
		var pos: Vector3 = candidate_positions[i]
		var dist := Vector2(pos.x - center.x, pos.z - center.z).length()
		if dist <= radius:
			found.append([dist, i])
	return _closest(found, max_targets)


static func resolve_arc(origin: Vector3, fwd: Vector3, arc_degrees: float, reach: float,
		max_targets: int, candidate_positions: Array) -> Array[int]:
	var found: Array = []
	for i in candidate_positions.size():
		var pos: Vector3 = candidate_positions[i]
		var to_target: Vector3 = pos - origin
		var dist := to_target.length()
		if dist > reach or dist < 0.001:
			continue
		var angle_deg := rad_to_deg(fwd.angle_to(to_target.normalized()))
		if angle_deg <= arc_degrees / 2.0:
			found.append([dist, i])
	return _closest(found, max_targets)


static func _candidates(cast: SkillCast, exclude: Array) -> Array:
	var result: Array = []
	for t in SkillTargets.hittable(cast.caster):
		if is_instance_valid(t) and t is Node3D and not (t in exclude):
			result.append(t)
	return result


static func _positions(targets: Array) -> Array[Vector3]:
	var positions: Array[Vector3] = []
	for t in targets:
		positions.append((t as Node3D).global_position)
	return positions


static func _pick(targets: Array, indices: Array[int]) -> Array:
	var picked: Array = []
	for i in indices:
		picked.append(targets[i])
	return picked


static func _closest(found: Array, max_targets: int) -> Array[int]:
	found.sort_custom(func(a, b): return a[0] < b[0])
	var hits: Array[int] = []
	for entry in found:
		if max_targets > 0 and hits.size() >= max_targets:
			break
		hits.append(entry[1])
	return hits

class_name SkillInstance
extends Resource

@export var gem: SkillGemDefinition
@export var supports: Array[SupportGemDefinition] = []

func _init() -> void:
	stats = StatComponent.new()
	_effective = StatComponent.new()


func rebuild() -> void:
	stats.clear_modifiers()
	stats.base_stats = {StatTypes.Stat.MANA_COST: gem.base_mana_cost}
	_modifiers.clear()
	_flags = gem.base_flags.duplicate()
	_anim_override = ""
	support_behaviors.clear()

	for m in gem.stat_modifiers:
		_add_modifier(_tagged(m, "gem_%s" % gem.id))

	for support in supports:
		if not _tags_compatible(support.allowed_tags, gem.tags):
			continue
		if support.animation_override != "":
			_anim_override = support.animation_override
		if support.behavior_script != null:
			var sb := support.behavior_script.new() as SupportBehavior
			if sb == null:
				push_error("Support %s: behavior_script must extend SupportBehavior" % support.id)
			else:
				sb.definition = support
				support_behaviors.append(sb)
		for m in support.stat_modifiers:
			_add_modifier(_tagged(m, "support_%s" % support.id))
		for key in support.flag_modifiers.keys():
			_flags[key] = _flags.get(key, 0.0) + support.flag_modifiers[key]
		if support.mana_cost_increase != 0.0:
			var cost_mod := Modifier.new()
			cost_mod.stat = StatTypes.Stat.MANA_COST
			cost_mod.op = Modifier.ModOp.PERCENT_MULT
			cost_mod.value = support.mana_cost_increase
			cost_mod.source_id = "support_%s_cost" % support.id
			stats.add_modifier(cost_mod)
	_apply_program()
	for key in flag_overrides.keys():
		_flags[key] = float(flag_overrides[key])
	print("[Skill] %s flags=%s overrides=%s" % [gem.id, _flags, flag_overrides])

func _apply_program() -> void:
	program_validation = null
	if program == null or not program.has_rules():
		return
	var limits := get_program_limits()
	program_validation = program.validate(limits)
	if not program_validation.valid:
		push_warning("[Skill] %s: program ignored: %s" % [gem.id, "; ".join(program_validation.errors)])
		return
	var run := program.resolve(limits.routines, limits) if program.has_calls() else program
	var has_conditional := run != program
	for rule in run.rules:
		if not rule.is_baked():
			has_conditional = true
			continue
		has_conditional = has_conditional or rule.uses_registers()
		for e in rule.effects:
			var knob := limits.get_knob(e.knob)
			if knob == null or e.steps == 0 or knob.kind == SkillKnob.Kind.VARIABLE or e.ramps():
				continue
			if knob.kind == SkillKnob.Kind.FLAG:
				_flags[knob.flag_key] = _flags.get(knob.flag_key, 0.0) + knob.value_for(e.steps)
			else:
				_add_modifier(knob.make_modifier(e.steps, "program"))
	var tier := program_validation.tier
	if tier != null:
		for m in tier.modifiers:
			_add_modifier(_tagged(m, "program_tier_%s" % tier.display_name))
		for key in tier.flag_bonus.keys():
			_flags[key] = _flags.get(key, 0.0) + float(tier.flag_bonus[key])
	if has_conditional:
		support_behaviors.push_front(ProgramSupport.new(run, limits))

func get_program_limits() -> SkillProgramLimits:
	var active: Array = []
	for support in supports:
		if _tags_compatible(support.allowed_tags, gem.tags):
			active.append(support)
	var limits := SkillProgramLimits.for_socket(gem, active, class_trait, rank, bonus_points, bonus_steps)
	if limits != null:
		limits.routines = routines
	return limits

	copy.source_id = source
	return copy

func build_effective_stats(caster_stats: StatComponent) -> StatComponent:
	var base: Dictionary = {}
	for stat in StatTypes.Stat.values():
		base[stat] = caster_stats.get_stat(stat)
	_effective.clear_modifiers()
	_effective.base_stats = base
	for m in _modifiers:
		_effective.add_modifier(m)
	if class_trait != null:
		var inc := class_trait.armour_damage_increase(caster_stats)
		if inc > 0.0:
			var m := Modifier.new()
			m.stat = StatTypes.Stat.SKILL_DAMAGE_MULT
			m.op = Modifier.ModOp.PERCENT_ADD
			m.value = inc
			m.source_id = "class_trait_armour_damage"
			_effective.add_modifier(m)
		var flat := class_trait.armour_flat_damage(caster_stats)
		if flat > 0.0:
			for stat in [StatTypes.Stat.MIN_ATTACK_DAMAGE, StatTypes.Stat.MAX_ATTACK_DAMAGE]:
				var f := Modifier.new()
				f.stat = stat
				f.op = Modifier.ModOp.FLAT_ADD
				f.value = flat
				f.source_id = "class_trait_armour_flat"

	return _effective

func roll_damage(effective_stats: StatComponent) -> float:
	var lo := effective_stats.get_stat(StatTypes.Stat.MIN_ATTACK_DAMAGE)
	var hi := effective_stats.get_stat(StatTypes.Stat.MAX_ATTACK_DAMAGE)
	var mult := effective_stats.get_stat(StatTypes.Stat.SKILL_DAMAGE_MULT)
	var base := randf_range(lo, hi)
	var total := base * mult

	var chance := clampf(effective_stats.get_stat(StatTypes.Stat.CRIT_CHANCE), 0.0, 1.0)
	var crit_mult := maxf(effective_stats.get_stat(StatTypes.Stat.CRIT_MULTIPLIER), 1.0)
	last_roll_was_crit = randf() < chance
	if last_roll_was_crit:
		total *= crit_mult
	return total

func count_swing(cast: SkillCast) -> int:
	if cast.extra.has(SWING_KEY):
		return int(cast.extra[SWING_KEY])
	var now := Time.get_ticks_msec()
	var base := maxf(cast.flag("recovery_mult"), 1.0) / maxf(cast.stats.get_stat(StatTypes.Stat.ATTACK_SPEED), 0.1)
	var window := base + cast.flag("combo_grace") + 0.35
	if (now - _combo_last_msec) / 1000.0 > window:
		_combo_swings = 0
	_combo_swings += 1
	_combo_last_msec = now
	cast.extra[SWING_KEY] = _combo_swings
	return _combo_swings


func get_cooldown(caster_stats: StatComponent) -> float:
	var speed := build_effective_stats(caster_stats).get_stat(StatTypes.Stat.ATTACK_SPEED) * speed_mult()
	return maxf(get_flag("recovery_mult"), 1.0) / maxf(speed, 0.1)

	var m := 1.0
	for s in support_behaviors:
		m *= s.speed_mult(self)
	return m

	var weapon: ItemInstance = equipment.get_equipped(EquipmentSlots.Slot.WEAPON)
	if weapon == null or weapon.definition == null or weapon.definition.weapon_type == null:
		return 0.0
	return weapon.definition.weapon_type.skill_mana_cost

func can_afford(mana: ManaComponent) -> bool:

func cast(caster: Node3D, caster_stats: StatComponent, mana: ManaComponent = null) -> void:

class_name StatComponent
extends Node

@export var base_stats: Dictionary = {}:
	set(value):
		base_stats = value
		_mark_dirty()

var _modifiers: Array[Modifier] = []
var _cache: Dictionary = {}
var _dirty: bool = true

signal stats_changed

func add_modifier(mod: Modifier) -> void:
	_modifiers.append(mod)
	_mark_dirty()

func remove_by_source(source_id: String) -> void:
	_modifiers = _modifiers.filter(func(m): return m.source_id != source_id)
	_mark_dirty()

func clear_modifiers() -> void:
	_modifiers.clear()
	_mark_dirty()

func remove_by_source_prefix(prefix: String) -> void:
	_modifiers = _modifiers.filter(func(m): return not m.source_id.begins_with(prefix))
	_mark_dirty()

func get_stat(stat: StatTypes.Stat) -> float:
	if _dirty:
		_recompute()
	return _cache.get(stat, 0.0)

func _mark_dirty() -> void:
	_dirty = true
	stats_changed.emit()

func _recompute() -> void:
	_cache.clear()
	var touched: Dictionary = {}

	for stat in base_stats.keys():
		touched[stat] = true

	for m in _modifiers:
		touched[m.stat] = true

	for stat in touched.keys():
		var base: float = base_stats.get(stat, 0.0)
		var flat := 0.0
		var percent_add := 0.0
		var percent_mult := 1.0
		var override_val = null
		for m in _modifiers:
			if m.stat != stat:
				continue
			match m.op:
				Modifier.ModOp.FLAT_ADD:
					flat += m.value
				Modifier.ModOp.PERCENT_ADD:
					percent_add += m.value
				Modifier.ModOp.PERCENT_MULT:
					percent_mult *= (1.0 + m.value)
				Modifier.ModOp.OVERRIDE:
					override_val = m.value
		var result: float = override_val if override_val != null \
		else (base + flat) * (1.0 + percent_add) * percent_mult
		if stat == StatTypes.Stat.MAX_HEALTH and override_val == null:
			result = maxf(result, 1.0)
		_cache[stat] = result
	_dirty = false

class_name BreakpointSupport
extends SupportBehavior

const META_PREFIX := "breakpoint_"
const LAST_TARGET_PREFIX := "breakpoint_last_"
const MARK_NODE := "BreakpointMark"
const COL_MARK := Color(1.0, 0.55, 0.15)
const COL_BURST := Color(1.0, 0.3, 0.1)


func modify_hit(cast: SkillCast, target: Node, hit: HitInfo) -> void:
	if target.is_in_group(&"breakables"):
		return
	var n := mark_count(cast.caster, target) + 1
	if n <= 1:
		return
	if n >= _hits_needed(cast, hit):
		hit.damage *= 1.0 + maxf(hit.flag("breakpoint_burst"), 0.0)
		hit.set_meta(&"breakpoint_consume", true)
	else:
		hit.damage *= 1.0 + maxf(hit.flag("breakpoint_bonus"), 0.0)
	hit.set_meta(&"breakpoint_n", n)


func on_hit(cast: SkillCast, target: Node, hit: HitInfo) -> void:
	if target.is_in_group(&"breakables"):
		return
	var owner := VariableStoreSupport.owner_of(cast.caster)
	if owner == null:
		return
	owner.set_meta(_last_key(owner), target.get_instance_id())
	var key := _key(owner)
	if hit.get_meta(&"breakpoint_consume", false):
		target.remove_meta(key)
		_remove_marker(target)
		if target is Node3D:
			ImpactEffect.spawn(cast.effect_parent(), (target as Node3D).global_position + Vector3.UP,
					1.1, COL_BURST, COL_MARK)
		return
	if hit.killed:
		target.remove_meta(key)
		_remove_marker(target)
		return
	var n: int = hit.get_meta(&"breakpoint_n", 1)
	target.set_meta(key, {"n": n, "msec": Time.get_ticks_msec(),
			"duration": maxf(cast.flag("breakpoint_duration"), 0.5)})
	_show_marker(target, n, _hits_needed(cast, hit), maxf(cast.flag("breakpoint_duration"), 0.5), key)


static func mark_count(caster: Node, target: Node) -> int:
	var owner := VariableStoreSupport.owner_of(caster)
	if owner == null or not is_instance_valid(target):
		return 0
	var key := _key(owner)
	if not target.has_meta(key):
		return 0
	var m: Dictionary = target.get_meta(key)
	if (Time.get_ticks_msec() - int(m["msec"])) / 1000.0 > float(m["duration"]):
		target.remove_meta(key)
		_remove_marker(target)
		return 0
	return int(m["n"])


static func is_marked(caster: Node, target: Node) -> bool:
	return is_instance_valid(target) and not SkillBehavior.is_dead(target) \
			and mark_count(caster, target) > 0


static func last_target(caster: Node) -> Node:
	var owner := VariableStoreSupport.owner_of(caster)
	if owner == null:
		return null
	var id: int = owner.get_meta(_last_key(owner), 0)
	var n := instance_from_id(id) if id != 0 else null
	return n as Node if is_instance_valid(n) else null


func _hits_needed(cast: SkillCast, hit: HitInfo = null) -> int:
	return maxi(int(round(hit.flag("breakpoint_hits") if hit != null else cast.flag("breakpoint_hits"))), 2)


static func _key(owner: Node) -> String:
	return META_PREFIX + str(owner.get_instance_id())


static func _last_key(owner: Node) -> String:
	return LAST_TARGET_PREFIX + str(owner.get_instance_id())


static func _show_marker(target: Node, n: int, needed: int, duration: float, key: String) -> void:
	if not target is Node3D:
		return
	var label := target.get_node_or_null(MARK_NODE) as Label3D
	if label == null:
		label = Label3D.new()
		label.name = MARK_NODE
		label.billboard = BaseMaterial3D.BILLBOARD_ENABLED
		label.no_depth_test = true
		label.fixed_size = true
		label.pixel_size = 0.0011
		label.font_size = 26
		label.outline_size = 8
		label.outline_modulate = Color(0, 0, 0, 0.9)
		label.modulate = COL_MARK
		label.position = Vector3.UP * 2.3
		target.add_child(label)
		var timer := Timer.new()
		timer.name = "Expire"
		timer.one_shot = true
		label.add_child(timer)
		timer.timeout.connect(func():
			if is_instance_valid(target):
				target.remove_meta(key)
			label.queue_free())
	(label.get_node("Expire") as Timer).start(duration)
	label.text = "BREAKPOINT %d/%d" % [n, needed]
	label.modulate = COL_BURST if n >= needed - 1 else COL_MARK


static func _remove_marker(target: Node) -> void:
	if not is_instance_valid(target):
		return
	var label := target.get_node_or_null(MARK_NODE)
	if label != null:
		label.queue_free()

class_name MulticastSupport

const SHADER := preload("res://assets/shaders/materials/game_shaders/hologram_glitch.gdshader")
const KEEP_ALIVE := 6.0

var _side := 1.0

func _summon(cast: SkillCast, side: float) -> void:
	var caster := cast.caster
	var parent := cast.effect_parent()
	if parent == null:
		return
	await caster.get_tree().process_frame
	if not cast.is_valid():
		return

	var holo := Hologram.new()
	holo.source = caster
	if SkillTargets.is_enemy(caster):
		holo.set_meta(&"hologram_of_enemy", true)
	parent.add_child(holo)
	var right := caster.global_transform.basis.x.normalized()
	holo.global_transform = caster.global_transform
	holo.global_position += right * side * cast.flag("multicast_offset")
	var mat := ShaderMaterial.new()
	mat.shader = SHADER
	holo.build_visual(caster, mat)

	var hit_in := holo.play_like(caster, cast.skill, cast.behavior.recovery_time(cast))
	SoundManager.play_skill(cast.skill.get_current_animation(), holo, hit_in)
	await caster.get_tree().create_timer(maxf(hit_in, 0.12), false).timeout
	if not is_instance_valid(holo):
		return
	if not cast.is_valid():
		holo.queue_free()
		return

	var copy := SkillCast.new(holo, cast.skill, cast.stats, cast.behavior)
	copy.flag_bonus = cast.flag_bonus.duplicate()
	copy.extra = cast.extra.duplicate()
	copy.extra["hologram"] = true
	copy.damage_mult = cast.damage_mult * cast.flag("multicast_effectiveness")
	for s in cast.skill.support_behaviors:
		s.before_release(copy)
	cast.behavior.release(copy)
	holo.vanish(maxf(cast.flag("multicast_linger"), 0.1), KEEP_ALIVE)


class Hologram extends Node3D:
	var source: Node3D
	var _mat: ShaderMaterial
	var _anim: AnimationPlayer

	func get_effect_style() -> Dictionary:
		return SkillStyle.for_caster(source) if is_instance_valid(source) else SkillStyle.default_style()

	func build_visual(from: Node3D, mat: ShaderMaterial) -> void:
		_mat = mat
		var visual := _visual_root(from)
		if visual == null:
			return
		var copy := visual.duplicate(0) as Node3D
		add_child(copy)
		copy.transform = from.global_transform.affine_inverse() * visual.global_transform
		for n in copy.find_children("*", "", true, false):
			if n is AnimationTree or n is CollisionObject3D or n is CollisionShape3D \
					or n is Light3D or n is AudioStreamPlayer3D or n is GPUParticles3D:
				n.queue_free()
			elif n is GeometryInstance3D:
				(n as GeometryInstance3D).material_override = mat
				(n as GeometryInstance3D).cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
		if copy is GeometryInstance3D:
			(copy as GeometryInstance3D).material_override = mat
		_fit_shader(copy)
		var players := copy.find_children("*", "AnimationPlayer", true, false)
		_anim = players[0] if not players.is_empty() else null

	func _fit_shader(root: Node) -> void:
		var best := AABB()
		for n in root.find_children("*", "MeshInstance3D", true, false):
			var box := (n as MeshInstance3D).get_aabb()
			if box.size.y > best.size.y:
				best = box
		if best.size.y <= 0.0001:
			return
		var k := best.size.y / 1.8
		_mat.set_shader_parameter("height_min", best.position.y)
		_mat.set_shader_parameter("height_max", best.end.y)
		_mat.set_shader_parameter("glitch_strength", 0.12 * k)
		_mat.set_shader_parameter("band_density", 12.0 / k)
		_mat.set_shader_parameter("scan_density", 60.0)

	func play_like(from: Node3D, skill: SkillInstance, recovery: float) -> float:
		var model = from.get_node_or_null("model")
		var key := skill.get_current_animation()
		if _anim == null or model == null or key == "" or not model.has_method("get_clip_length"):
			return 0.25
		var length: float = model.get_clip_length(key)
		if length <= 0.0:
			return 0.25
		var speed := clampf(length / maxf(recovery * 0.9, 0.05), model.min_action_speed, model.max_action_speed)
		var clip: String = model._clip(key)
		if not _anim.has_animation(clip):
			return 0.25
		_anim.play(clip, 0.0, speed)
		return length * float(model.hit_points.get(key, 0.5)) / speed

	func vanish(after: float, keep_alive: float) -> void:
		var t := create_tween()
		t.tween_interval(after)
		var start = _mat.get_shader_parameter("intensity") if _mat else null
		var from_v: float = float(start) if start != null else 1.6
		t.tween_method(func(v: float) -> void:
				if _mat: _mat.set_shader_parameter("intensity", v), from_v, 0.0, 0.35)
		t.tween_callback(hide)
		t.tween_interval(keep_alive)
		t.tween_callback(queue_free)

	static func _visual_root(from: Node3D) -> Node3D:
		var model = from.get_node_or_null("model")
		if model != null and model.get("_model") is Node3D:
			return model.get("_model")
		for c in from.get_children():
			if c is Node3D and not (c is CollisionObject3D or c is CollisionShape3D) \
					and not c.find_children("*", "MeshInstance3D", true, false).is_empty():
				return c
		return null

func _hop(cast: SkillCast, sub: SkillCast, from: Vector3, left: int) -> void:
	if left <= 0 or not cast.is_valid():
		return
	var target: Node3D = null
	for t in SkillDeliveries.hits_in_circle(sub, from, maxf(cast.flag("pointer_range"), 0.5), 32, sub.already_hit):
		if not t.is_in_group(&"breakables"):
			target = t
			break
	if target == null:
		return
	await cast.caster.get_tree().create_timer(HOP_DELAY, false).timeout
	if not cast.is_valid() or not is_instance_valid(target):
		return
	var at := target.global_position
	_link_fx(cast.effect_parent(), from, at)
	var radius := cast.flag("radius") * HOP_RADIUS
	ImpactEffect.spawn(cast.effect_parent(), at, radius, LINK_COLOR, HOP_ACCENT)
	for t in SkillDeliveries.hits_in_circle(sub, at, radius, SkillDeliveries.NO_CAP, sub.already_hit):
		sub.hit(t, cast.flag("pointer_hop_damage"))
	if not (target in sub.already_hit):
		sub.already_hit.append(target)
	_hop(cast, sub, at, left - 1)


static func _link_fx(parent: Node, a: Vector3, b: Vector3) -> void:
	var lift := Vector3.UP * 0.6
	var seg := (b + lift) - (a + lift)
	var length := seg.length()
	if length < 0.05:
		return
	var mesh := BoxMesh.new()
	mesh.size = Vector3(0.08, 0.08, length)
	var mat := StandardMaterial3D.new()
	mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	mat.albedo_color = LINK_COLOR
	mat.emission_enabled = true
	mat.emission = LINK_COLOR
	mat.emission_energy_multiplier = 3.0
	mesh.material = mat
	var line := MeshInstance3D.new()
	line.mesh = mesh
	line.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	parent.add_child(line)
	line.global_position = a + lift + seg * 0.5
	line.look_at(b + lift, Vector3.UP if absf(seg.normalized().y) < 0.99 else Vector3.RIGHT)
	var t := line.create_tween()
	t.tween_property(mat, "albedo_color:a", 0.0, 0.25)
	t.tween_callback(line.queue_free)

func aim_repeat(cast: SkillCast, aim: Dictionary) -> void:
	var origin: Vector3 = aim["from"]
	var seek_from: Vector3 = aim["seek_from"]
	var facing: Vector3 = aim["direction"]
	var reach := maxf(cast.flag("pointer_range"), 0.5)
	var max_turn := deg_to_rad(clampf(cast.flag("pointer_turn"), 0.0, 180.0))
	var hit_before: Array = cast.extra.get("loop_hit_so_far", [])
	var found := SkillDeliveries.hits_in_circle(cast, seek_from, reach, 32)
	var target: Node3D = null
	for pass_no in 4:
		for t in found:
			var fresh: bool = not (t in hit_before)
			var reachable: bool = facing.angle_to(_flat(t.global_position - origin, facing)) <= max_turn + 0.001
			if (pass_no == 0 and fresh and reachable) or (pass_no == 1 and fresh) \
					or (pass_no == 2 and reachable) or pass_no == 3:
				target = t
				break
		if target != null:
			break
	if target == null:
		return
	var to := target.global_position - origin
	to.y = 0.0
	var want := _flat(to, facing)
	var angle := facing.angle_to(want)
	if angle <= max_turn + 0.001:
		aim["direction"] = want
		aim["range"] = maxf(aim["range"], to.length() + 1.5)
	else:
		var side := signf(facing.cross(want).y)
		aim["direction"] = facing.rotated(Vector3.UP, max_turn * (side if side != 0.0 else 1.0))


static func _flat(v: Vector3, fallback: Vector3) -> Vector3:
	v.y = 0.0
	return v.normalized() if v.length() > 0.05 else fallback

class_name RollbackWatch
extends Node

const SUPPORT_ID := "rollback_support"
const SAMPLE_EVERY := 0.2
const KEEP_SECONDS := 8.0
const COL := Color(0.45, 0.85, 1.0)

var skills: SkillComponent
var window := 3.0
var cooldown := 20.0
var trigger := 0.3
var _history: Array = []
var _sample_t := 0.0
var _ready_at := 0


static func sync(component: SkillComponent) -> void:
	var watch := component.get_node_or_null("RollbackWatch") as RollbackWatch
	var found := false
	var w := 0.0
	var cd := INF
	var trig := 0.0
	for s in component.sockets:
		if s.skill == null or not s.supports.any(func(it: ItemInstance) -> bool:
				return it.definition.support_gem != null and it.definition.support_gem.id == SUPPORT_ID):
			continue
		found = true
		w = maxf(w, s.skill.get_flag("rollback_window"))
		cd = minf(cd, s.skill.get_flag("rollback_cooldown"))
		trig = maxf(trig, s.skill.get_flag("rollback_trigger"))
	if not found:
		if watch != null:
			watch.queue_free()
		return
	if watch == null:
		watch = RollbackWatch.new()
		watch.name = "RollbackWatch"
		watch.skills = component
		component.add_child(watch)
	watch.window = clampf(w, 0.5, KEEP_SECONDS - 1.0)
	watch.cooldown = maxf(cd, 4.0)
	watch.trigger = clampf(trig, 0.1, 0.9)


func seconds_left() -> float:
	return maxf((_ready_at - Time.get_ticks_msec()) / 1000.0, 0.0)


func _process(delta: float) -> void:
	var hp := _health()
	if hp == null or hp.current_health <= 0.0:
		_history.clear()
		return
	var now := Time.get_ticks_msec()
	_sample_t -= delta
	if _sample_t <= 0.0:
		_sample_t = SAMPLE_EVERY
		_history.append([now, hp.current_health])
		while not _history.is_empty() and now - int(_history[0][0]) > KEEP_SECONDS * 1000.0:
			_history.pop_front()
	var max_hp := hp.stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	if now < _ready_at or max_hp <= 0.0 or hp.current_health / max_hp >= trigger:
		return
	var back := _life_at(now - int(window * 1000.0))
	if back <= hp.current_health:
		return
	hp.heal(back - hp.current_health)
	_ready_at = now + int(cooldown * 1000.0)
	_history.clear()
	var numbers := get_tree().get_first_node_in_group("damage_numbers") as DamageNumbers
	var owner_3d := skills.get_parent() as Node3D
	if numbers != null and owner_3d != null:
		numbers.pop_text(owner_3d.global_position + Vector3.UP * 2.4, "ROLLBACK", COL, 1.1)


func _life_at(msec: int) -> float:
	for entry in _history:
		if int(entry[0]) >= msec:
			return float(entry[1])
	return float(_history[-1][1]) if not _history.is_empty() else 0.0


func _health() -> HealthComponent:
	return skills.get_parent().get("health") as HealthComponent if skills != null else null

class_name TryExceptSupport
extends SupportBehavior

const META_STACKS := "try_except_stacks_"
const META_MISSED := "try_except_missed_"
const COL_ERROR := Color(1.0, 0.35, 0.35)
const COL_HANDLED := Color(0.4, 1.0, 0.6)


func modify_hit(cast: SkillCast, target: Node, hit: HitInfo) -> void:
	if not _counts(cast) or target.is_in_group(&"breakables"):
		return
	var stacks := stacks_of(cast.caster, cast.skill)
	if stacks <= 0:
		return
	hit.damage *= 1.0 + hit.flag("except_stack_damage") * stacks
	cast.extra["except_handled"] = stacks


func on_hit(cast: SkillCast, target: Node, _hit: HitInfo) -> void:
	if _counts(cast) and not target.is_in_group(&"breakables"):
		cast.extra["except_hits"] = int(cast.extra.get("except_hits", 0)) + 1


func on_area_impact(cast: SkillCast, center: Vector3, _direction: Vector3) -> void:
	if not _counts(cast):
		return
	var hits := int(cast.extra.get("except_hits", 0))
	cast.extra["except_hits"] = 0
	var owner := VariableStoreSupport.owner_of(cast.caster)
	if owner == null:
		return
	var key := String(cast.skill.gem.id)
	if hits > 0:
		owner.set_meta(META_MISSED + key, false)
		var handled := int(cast.extra.get("except_handled", 0))
		if handled > 0 and stacks_of(owner, cast.skill) > 0:
			owner.set_meta(META_STACKS + key, 0)
			_pop(cast, center, "HANDLED x%d" % handled, COL_HANDLED)
		return
	owner.set_meta(META_MISSED + key, true)
	var mana := cast.caster.get("mana") as ManaComponent
	if mana != null:
		var shots := maxf(cast.flag("max_projectiles"), 1.0)
		mana.restore(cast.skill.get_mana_cost() * clampf(cast.flag("except_refund"), 0.0, 1.0) / shots)
	var cap := int(round(cast.flag("except_max_stacks")))
	var now := stacks_of(owner, cast.skill)
	if now < cap:
		owner.set_meta(META_STACKS + key, now + 1)
		_pop(cast, center, "EXCEPTION %d/%d" % [now + 1, cap], COL_ERROR)


static func _counts(cast: SkillCast) -> bool:
	return cast.depth == 0 and not cast.extra.get("hologram", false)


static func stacks_of(caster: Node, skill: SkillInstance) -> int:
	var owner := VariableStoreSupport.owner_of(caster)
	if owner == null or skill == null or skill.gem == null:
		return 0
	return int(owner.get_meta(META_STACKS + String(skill.gem.id), 0))


static func last_missed(caster: Node, skill: SkillInstance) -> bool:
	var owner := VariableStoreSupport.owner_of(caster)
	if owner == null or skill == null or skill.gem == null:
		return false
	return bool(owner.get_meta(META_MISSED + String(skill.gem.id), false))


static func on_skill(skill: SkillInstance) -> bool:
	if skill == null:
		return false
	for s in skill.support_behaviors:
		if s is TryExceptSupport:
			return true
	return false


static func _pop(cast: SkillCast, at: Vector3, text: String, color: Color) -> void:
	if cast.caster.is_in_group("players"):
		cast.caster.get_tree().call_group("damage_numbers", "pop_text", at + Vector3.UP * 1.4, text, color)

class_name VariableStoreSupport
extends SupportBehavior

const META_PREFIX := "variable_store_"
const EFFECT_PER_CHARGE := 0.25


func on_hit(cast: SkillCast, target: Node, _hit: HitInfo) -> void:
	if target.is_in_group(&"breakables"):
		return
	var owner := owner_of(cast.caster)
	if owner == null:
		return
	var cap := int(cast.flag("var_store_max"))
	var now := read(owner, cast.skill)
	if now < cap:
		owner.set_meta(_key(cast.skill), mini(now + maxi(int(_hit.flag("var_store_gain")), 1), cap))


static func owner_of(caster: Node) -> Node:
	if not is_instance_valid(caster):
		return null
	var src = caster.get("source")
	return src if src is Node and is_instance_valid(src) else caster


static func read(caster: Node, skill: SkillInstance) -> int:
	var owner := owner_of(caster)
	if owner == null or skill == null or skill.gem == null:
		return 0
	return int(owner.get_meta(_key(skill), 0))


static func spend(caster: Node, skill: SkillInstance, amount: int) -> void:
	var owner := owner_of(caster)
	if owner == null or skill == null or skill.gem == null:
		return
	owner.set_meta(_key(skill), maxi(read(owner, skill) - amount, 0))


const CAPACITY_KNOB := &"store_capacity"


static func capacity_with(steps: int, knob: SkillKnob) -> int:
	var def: SupportGemDefinition = SkillGemRegistry.get_support_gem("variable_store_support")
	var base: float = def.flag_modifiers.get("var_store_max", 10.0) if def != null else 10.0
	return int(round(base + knob.value_for(steps)))


static func rule_effect_mult(rule: ProgramRule) -> float:
	var mult := 1.0
	for c in rule.conditions:
		if c is ChargeAtLeastCondition:
			mult *= 1.0 + EFFECT_PER_CHARGE * (c as ChargeAtLeastCondition).n
	return mult


static func on_skill(skill: SkillInstance) -> bool:
	if skill == null:
		return false
	for s in skill.support_behaviors:
		if s is VariableStoreSupport:
			return true
	return false


static func _key(skill: SkillInstance) -> String:
	return META_PREFIX + String(skill.gem.id)
