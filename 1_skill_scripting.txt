class_name ProgramCondition
extends Resource

enum Scope { CAST, HIT }


func type_id() -> String:
	return ""


func scope() -> Scope:
	return Scope.CAST


func required_support() -> String:
	return ""


func on_fired(_cast: SkillCast) -> void:
	pass


func per_repeat() -> bool:
	return false


func defensive() -> bool:
	return false


func keyword() -> String:
	return ""


func modular() -> bool:
	return false


func param_text(_param: String) -> String:
	return ""


func fired_id() -> String:
	return type_id()


func allows_repeats() -> bool:
	return false


func needs_loop() -> bool:
	return false


func changes_during_cast() -> bool:
	return false


func is_true(_cast: SkillCast, _target: Node) -> bool:
	return false


func cost_multiplier() -> float:
	return 1.0


func describe() -> String:
	return type_id()


func param_names() -> PackedStringArray:
	return PackedStringArray()


func param_step(_param: String) -> float:
	return 1.0


func param_label(param: String) -> String:
	return param.to_upper()


func palette_name() -> String:
	return describe().trim_prefix("WHEN ")


func nudge_param(param: String, clicks: int) -> void:
	var current = get(param)
	if current is int:
		set(param, current + int(param_step(param)) * clicks)
	else:
		set(param, snappedf(float(current) + param_step(param) * clicks, 0.001))
	clamp_params()


func clamp_params() -> void:
	pass


func to_dict() -> Dictionary:
	var d := {"type": type_id()}
	for p in param_names():
		d[p] = get(p)
	return d


func load_params(d: Dictionary) -> void:
	for p in param_names():
		if not d.has(p):
			continue
		var current = get(p)
		if current is int:
			set(p, int(d[p]))
		elif current is float:
			set(p, float(d[p]))
		else:
			set(p, d[p])
	clamp_params()


static func health_fraction(node: Node) -> float:
	if not is_instance_valid(node):
		return 1.0
	var h := node.get("health") as HealthComponent
	if h == null or h.stats == null:
		return 1.0
	var max_hp := h.stats.get_stat(StatTypes.Stat.MAX_HEALTH)
	return h.current_health / max_hp if max_hp > 0.0 else 1.0

class_name ProgramEffect
extends Resource

@export var knob: StringName
@export var steps: int = 1
@export var op: int = 0
@export var ramp: int = -1
@export var reg: int = 0
@export var sort_by: int = 0

const CALL_KNOB := &"call"


static func make(knob_id: StringName, step_count: int) -> ProgramEffect:
	var e := ProgramEffect.new()
	e.knob = knob_id
	e.steps = step_count
	return e


static func make_write(op_text: String) -> ProgramEffect:
	var e := make(ScriptRegisters.WRITE_KNOB, 1)
	e.op = maxi(ScriptRegisters.WRITE_OPS.find(op_text), 0)
	return e


static func make_sort(order: int) -> ProgramEffect:
	var e := make_write("sort")
	e.sort_by = order
	return e


static func make_call(index: int) -> ProgramEffect:
	var e := make(CALL_KNOB, 1)
	e.op = index
	return e


func is_call() -> bool:
	return knob == CALL_KNOB


func register_name() -> String:
	return "target"


func write_op() -> String:
	return ScriptRegisters.WRITE_OPS[posmod(op, ScriptRegisters.WRITE_OPS.size())]


func describe_write() -> String:
	if write_op() == "sort":
		return "registry.sort(%s)" % ScriptRegisters.sort_text(sort_by)
	return "registry.%s(target)" % write_op()


func ramps() -> bool:
	return ramp >= 0


func ramp_register() -> String:
	return "size"


func ramp_text() -> String:
	return "registry.size" if ramps() else ""


func to_dict() -> Dictionary:
	var d := {"knob": String(knob), "steps": steps}
	if ramps():
		d["ramp"] = 0
	if ScriptRegisters.is_write(knob):
		d["set_op"] = op
		if write_op() == "sort":
			d["sort_by"] = sort_by
	if is_call():
		d = {"knob": String(CALL_KNOB), "steps": 1, "routine": op}
	return d


static func from_dict(d: Dictionary) -> ProgramEffect:
	var e := make(StringName(str(d.get("knob", ""))), int(d.get("steps", 0)))
	if e.is_call():
		e.steps = 1
		e.op = int(d.get("routine", 0))
		return e
	e.ramp = 0 if int(d.get("ramp", -1)) >= 0 else -1
	if ScriptRegisters.is_write(e.knob):
		if d.has("set_op"):
			e.op = int(d["set_op"])
			e.sort_by = int(d.get("sort_by", 0))
		else:
			var old_op := int(d.get("op", 1))
			e.op = 1 if old_op == 2 or (old_op == 0 and e.steps == 0) else 0
		e.steps = 1
	return e

class_name ProgramLoopRunner
extends RefCounted

const GAP := 0.22
const IMPACT_GAP := 0.1
const DEBUG := true


var support: ProgramSupport
var root: SkillCast
var current: SkillCast
var cap := ProgramLoop.BASE_CAP
var index := 0
var finished := false

var hits := 0
var kills := 0
var landed := false
var landed_hits := 0
var landed_kills := 0
var last_targets: Array = []
var last_impact: Variant = null
var last_direction := Vector3.ZERO

var _tickets: Array[ProgramRule] = []
var _whiles: Array[ProgramRule] = []
var _late_fors: Array[ProgramRule] = []
var _while_used := {}
var _stopped := {}
var _iter_done := false
var _tree: SceneTree
var _current_rule: ProgramRule
var _current_paid := 0.0
var _forgiven := 0
var _hit_so_far: Array = []


func _init(p_support: ProgramSupport, p_root: SkillCast, p_cap: int,
		tickets: Array[ProgramRule], whiles: Array[ProgramRule], late_fors: Array[ProgramRule] = []) -> void:
	support = p_support
	root = p_root
	cap = p_cap
	_tickets = tickets
	_whiles = whiles
	_late_fors = late_fors


func start() -> void:
	_tree = root.caster.get_tree()
	root.extra[ProgramSupport.LOOP_KEY] = self
	root.extra["loop_index"] = 0
	root.extra["loop_cap"] = cap
	_begin(root)


func repeat_mana_cost() -> float:
	return root.skill.get_mana_cost() * ProgramLoop.MANA_FRACTION * support.limits.repeat_mana_mult()


func any_target_alive() -> bool:
	for t in last_targets:
		if is_instance_valid(t) and not SkillBehavior.is_dead(t):
			return true
	return false


func record_hit(cast: SkillCast, target: Node, hit: HitInfo) -> void:
	if finished or cast != current or target.is_in_group(&"breakables"):
		return
	hits += 1
	if hit.killed:
		kills += 1
	if not (target in last_targets):
		last_targets.append(target)


func on_impact(cast: SkillCast, center: Vector3, direction: Vector3) -> void:
	if finished or cast != current or _iter_done or last_impact != null:
		return
	last_impact = center
	last_direction = direction
	_tree.create_timer(IMPACT_GAP, false).timeout.connect(_iteration_done.bind(cast))


func _begin(cast: SkillCast) -> void:
	current = cast
	_iter_done = false
	hits = 0
	kills = 0
	last_targets = []
	last_impact = null
	var wait := GAP
	if cast.behavior.waits_for_impact():
		wait = cast.flag("range") / maxf(cast.flag("speed"), 1.0) + 0.6
	_tree.create_timer(wait, false).timeout.connect(_iteration_done.bind(cast))


func _iteration_done(cast: SkillCast) -> void:
	if finished or cast != current or _iter_done:
		return
	_iter_done = true
	landed = true
	landed_hits = hits
	landed_kills = kills
	if not root.is_valid() or SkillBehavior.is_dead(root.caster):
		_log("stop: caster gone or dead")
		_finish()
		return
	var mana := root.caster.get("mana") as ManaComponent
	var cost := repeat_mana_cost()
	if DEBUG:
		var alive := 0
		for t in last_targets:
			if is_instance_valid(t) and not SkillBehavior.is_dead(t):
				alive += 1
		_log("iteration %d landed: %d hits, %d kills, %d targets (%d alive), cast #%s, mana %.1f / repeat costs %.1f" % [
			index, hits, kills, last_targets.size(), alive, str(root.extra.get(ProgramSupport.CAST_NUMBER_KEY, "?")),
			mana.current_mana if mana != null and mana.get("current_mana") != null else -1.0, cost])
	if current != root and hits == 0 and _forgiven < int(round(root.flag("except_retries"))):
		_except(mana, cost)
		return
	if index >= cap:
		_log("stop: reached the cap (%d)" % cap)
		_finish()
		return
	if mana != null and cost > 0.0 and not mana.has_enough(cost):
		_log("stop: not enough mana for a repeat")
		_finish()
		return
	var rule := _next_rule()
	if rule == null:
		_log("stop: no loop wants another repeat")
		_finish()
		return
	_log("repeat %d (rule: %s)" % [index + 1, _rule_text(rule)])
	_current_paid = 0.0
	if mana != null and cost > 0.0 and mana.spend(cost):
		_current_paid = cost
	index += 1
	if not SkillTargets.is_enemy(root.caster):
		GameEvents.report("loop_repeat", "stop" if rule.loop_stops_itself() else "for")
	_repeat(rule)


func _except(mana: ManaComponent, cost: float) -> void:
	_forgiven += 1
	if mana != null and _current_paid > 0.0:
		mana.restore(_current_paid * clampf(root.flag("except_refund"), 0.0, 1.0))
	_current_paid = 0.0
	if mana != null and cost > 0.0:
		if not mana.has_enough(cost):
			_finish()
			return
		mana.spend(cost)
		_current_paid = cost
	_repeat(_current_rule, true)


func _next_rule() -> ProgramRule:
	if current == root and not _late_fors.is_empty():
		for r in _late_fors:
			if DEBUG:
				var parts: Array[String] = []
				for c in r.conditions:
					parts.append("%s=%s" % [c.type_id(), str(c.is_true(root, null))])
				_log("judging FOR rule after landing: %s" % ", ".join(parts))
			if r.all_true(root, null):
				support.announce(root, r)
				for i in r.loop.count:
					_tickets.append(r)
		_late_fors.clear()
	for rule in _tickets + _whiles:
		if not _stopped.has(rule) and _breaks_now(rule):
			_stopped[rule] = true
	while not _tickets.is_empty():
		var r: ProgramRule = _tickets.pop_front()
		if not _stopped.has(r):
			return r
	for r in _whiles:
		if _stopped.has(r) or int(_while_used.get(r, 0)) >= r.loop.count:
			continue
		if r.all_true(current, null):
			_while_used[r] = int(_while_used.get(r, 0)) + 1
			support.announce(current, r)
			return r
	return null


func _breaks_now(rule: ProgramRule) -> bool:
	for i in rule.branches.size():
		var b := rule.branches[i]
		if not b.breaks or b.conditions.is_empty():
			continue
		var all := true
		for c in b.conditions:
			if not c.is_true(current, null):
				all = false
				break
		if all:
			_log("break: if %s" % " and ".join(b.conditions.map(func(c: ProgramCondition) -> String: return c.type_id())))
			support.announce(current, rule.branch_rule(i))
			return true
	return false


func _repeat(rule: ProgramRule, retry := false) -> void:
	_current_rule = rule
	var child := SkillCast.new(root.caster, root.skill, root.stats, root.behavior)
	child.depth = 1
	child.flag_bonus = root.flag_bonus.duplicate()
	child.damage_mult = root.damage_mult * pow(ProgramLoop.FALLOFF, index)
	child.extra = root.extra.duplicate()
	child.extra["loop_index"] = index
	child.extra["loop_repeat"] = true
	child.extra["loop_retry"] = retry
	for t in last_targets:
		if not (t in _hit_so_far):
			_hit_so_far.append(t)
	child.extra["loop_hit_so_far"] = _hit_so_far.duplicate()
	child.extra[ProgramSupport.LOOP_RULE_KEY] = support.rule_key(rule)
	support.apply_repeat_rules(child, rule)
	var from = null if retry else last_impact
	var dir := last_direction
	_begin(child)
	root.behavior.repeat_release(child, from, dir)


func _finish() -> void:
	finished = true
	support.runner_done(self)


func _log(text: String) -> void:
	if DEBUG and root != null and is_instance_valid(root.caster) and root.caster.is_in_group("players"):
		print("[Loop] %s: %s" % [root.skill.gem.id, text])


func _rule_text(rule: ProgramRule) -> String:
	var parts: Array[String] = []
	for c in rule.conditions:
		parts.append(c.type_id())
	return ("when " + " and ".join(parts)) if not parts.is_empty() else "always"

class_name ProgramRoutine
extends Resource

const MAX_ROUTINES := 3
const MAX_BLOCKS := 6
const NAME_MAX := 14
const DEFAULT_NAMES := ["routine_a", "routine_b", "routine_c"]
const OWNER := "@routines"

@export var routine_name := "routine_a"
@export var body: ProgramRule


func _init() -> void:
	body = ProgramRule.new()
	body.owner = OWNER


static func make(index: int) -> ProgramRoutine:
	var r := ProgramRoutine.new()
	r.routine_name = DEFAULT_NAMES[clampi(index, 0, DEFAULT_NAMES.size() - 1)]
	return r


func block_count() -> int:
	return body.action_count()


func is_empty() -> bool:
	return body.effects.is_empty() and body.branches.is_empty()


func has_if() -> bool:
	return not body.branches.is_empty()


func has_call() -> bool:
	for e in body.effects:
		if e.is_call():
			return true
	for b in body.branches:
		for e in b.effects:
			if e.is_call():
				return true
	return false


func problem() -> Array:
	if is_empty():
		return ["%s() is empty." % routine_name, "%s() is empty" % routine_name,
			"open routines.gd and drop knobs into %s()" % routine_name]
	if block_count() > MAX_BLOCKS:
		return ["%s() has %d blocks, a routine holds at most %d." % [routine_name, block_count(), MAX_BLOCKS],
			"%s() has %d blocks, max %d" % [routine_name, block_count(), MAX_BLOCKS],
			"remove %d block%s from %s() in routines.gd" % [block_count() - MAX_BLOCKS,
				"" if block_count() - MAX_BLOCKS == 1 else "s", routine_name]]
	if not body.conditions.is_empty() or body.loop != null:
		return ["%s() has a condition or loop of its own." % routine_name,
			"%s() has a condition or loop of its own" % routine_name,
			"a routine has no conditions: put them on the rule that calls it"]
	if has_call():
		return ["%s() calls a routine." % routine_name, "%s() calls a routine" % routine_name,
			"a routine can't call a routine: copy its lines in instead"]
	return []


static func clean_name(text: String, index: int) -> String:
	var out := ""
	for ch in text.strip_edges().to_lower().replace(" ", "_"):
		if (ch >= "a" and ch <= "z") or (ch >= "0" and ch <= "9") or ch == "_":
			out += ch
	out = out.left(NAME_MAX)
	return out if out != "" and not (out[0] >= "0" and out[0] <= "9") else DEFAULT_NAMES[clampi(index, 0, DEFAULT_NAMES.size() - 1)]


func duplicate_routine() -> ProgramRoutine:
	return ProgramRoutine.from_dict(to_dict())


func to_dict() -> Dictionary:
	return {"name": routine_name, "body": body.to_dict()}


static func from_dict(d: Dictionary) -> ProgramRoutine:
	var r := ProgramRoutine.new()
	r.routine_name = str(d.get("name", DEFAULT_NAMES[0]))
	if d.get("body") is Dictionary:
		r.body = ProgramRule.from_dict(d["body"])
	r.body.owner = OWNER
	r.body.conditions.clear()
	r.body.loop = null
	return r

class_name ProgramRule
extends Resource

const REMOVED_KNOBS: Array[StringName] = [&"max_targets", &"cleave_targets"]

@export var owner: String = ""
@export var conditions: Array[ProgramCondition] = []
@export var effects: Array[ProgramEffect] = []
@export var loop: ProgramLoop
@export var branches: Array[ProgramBranch] = []

var parent: ProgramRule
var branch_index := -1
var _own_count := 0


func is_always() -> bool:
	return conditions.is_empty()


func is_branch() -> bool:
	return parent != null


func branch_rule(i: int) -> ProgramRule:
	var b := branches[i]
	var r := ProgramRule.new()
	r.owner = owner
	r.conditions.assign(conditions + b.conditions)
	r.effects.assign(b.effects)
	r.parent = self
	r.branch_index = i
	r._own_count = b.conditions.size()
	return r


func flat() -> Array[ProgramRule]:
	var out: Array[ProgramRule] = [self]
	for i in branches.size():
		out.append(branch_rule(i))
	return out


func in_loop() -> bool:
	return parent != null and parent.loop != null


func has_break() -> bool:
	for b in branches:
		if b.breaks:
			return true
	return false


func loop_stops_itself() -> bool:
	return loop != null and (loop.mode == ProgramLoop.Mode.WHILE or has_break())


func branch() -> ProgramBranch:
	return parent.branches[branch_index] if parent != null else null


func own_conditions() -> Array[ProgramCondition]:
	if parent == null:
		return conditions
	var out: Array[ProgramCondition] = []
	out.assign(conditions.slice(conditions.size() - _own_count))
	return out


func scope() -> ProgramCondition.Scope:
	for c in conditions:
		if c.scope() == ProgramCondition.Scope.HIT:
			return ProgramCondition.Scope.HIT
	return ProgramCondition.Scope.CAST


func all_true(cast: SkillCast, target: Node) -> bool:
	if parent != null:
		if not parent.is_always():
			if parent.scope() == ProgramCondition.Scope.CAST:
				if not ProgramSupport.rule_fired(cast, parent):
					return false
			elif not parent.all_true(cast, target):
				return false
		for c in own_conditions():
			if not c.is_true(cast, target):
				return false
		return true
	for c in conditions:
		if not c.is_true(cast, target):
			return false
	return true


const MIN_COST_MULT := 0.25
const CASH_IN_MIN_COST_MULT := 0.18

func cost_multiplier() -> float:
	var best := {}
	for c in conditions:
		var t := c.type_id()
		best[t] = minf(best.get(t, 1.0), c.cost_multiplier())
	return maxf(raw_cost_multiplier_of(best), min_cost_mult())


func is_discount_capped() -> bool:
	var best := {}
	for c in conditions:
		var t := c.type_id()
		best[t] = minf(best.get(t, 1.0), c.cost_multiplier())
	return raw_cost_multiplier_of(best) < min_cost_mult()


func min_cost_mult() -> float:
	return CASH_IN_MIN_COST_MULT if spends_marks() else MIN_COST_MULT


func spends_marks() -> bool:
	var checks := false
	for c in conditions:
		var read := c as VarCondition
		checks = checks or (read != null and read.is_membership() and read.op_text() == "in")
	if not checks:
		return false
	for e in effects:
		if ScriptRegisters.is_write(e.knob) and e.write_op() == "remove":
			return true
	return false


static func raw_cost_multiplier_of(best: Dictionary) -> float:
	var m := 1.0
	for t in best:
		m *= best[t]
	return m


func repeated_condition() -> String:
	var seen := {}
	for c in conditions:
		if seen.has(c.type_id()) and not c.allows_repeats():
			return c.palette_name().to_lower().replace(" ", "_")
		seen[c.type_id()] = true
	return ""


func only_on_repeats() -> bool:
	if loop != null:
		return false
	for c in conditions:
		if c.needs_loop():
			return true
	return false


func action_count() -> int:
	var n := conditions.size() + (loop.action_count() if loop != null else 0)
	for e in effects:
		if not ScriptRegisters.is_write(e.knob):
			n += 1
	for b in branches:
		n += b.conditions.size() + (1 if b.breaks else 0)
		for e in b.effects:
			if not ScriptRegisters.is_write(e.knob):
				n += 1
	return n


func uses_registers() -> bool:
	if not branches.is_empty():
		return true
	for e in effects:
		if ScriptRegisters.is_write(e.knob) or e.ramps():
			return true
	return false


func is_baked() -> bool:
	return is_always() and loop == null


func describe(limits: SkillProgramLimits) -> String:
	var whens: Array[String] = []
	for c in conditions:
		whens.append(c.describe())
	var dos: Array[String] = []
	for e in effects:
		if e.is_call():
			dos.append("CALL routine %d" % (e.op + 1))
			continue
		if ScriptRegisters.is_write(e.knob):
			dos.append(e.describe_write())
			continue
		var k := limits.get_knob(e.knob) if limits != null else null
		dos.append(k.describe(e.steps) if k != null else "%s %+d" % [e.knob, e.steps])
	var head := "ALWAYS" if whens.is_empty() else " AND ".join(whens)
	if loop != null:
		if loop.mode == ProgramLoop.Mode.WHILE:
			head = head.replace("WHEN ", "WHILE ")
		var body := loop.describe()
		if not dos.is_empty():
			body += " with " + ", ".join(dos)
		return "%s -> %s" % [head, body]
	for b in branches:
		var ifs: Array[String] = []
		for c in b.conditions:
			ifs.append(c.describe().trim_prefix("WHEN "))
		var bdos: Array[String] = []
		for e in b.effects:
			if e.is_call():
				bdos.append("CALL routine %d" % (e.op + 1))
			elif ScriptRegisters.is_write(e.knob):
				bdos.append(e.describe_write())
			else:
				var bk := limits.get_knob(e.knob) if limits != null else null
				bdos.append(bk.describe(e.steps) if bk != null else "%s %+d" % [e.knob, e.steps])
		dos.append("IF %s -> [%s]" % [" AND ".join(ifs), ", ".join(bdos)])
	return "%s -> %s" % [head, ", ".join(dos)]


func to_dict() -> Dictionary:
	var conds: Array = []
	for c in conditions:
		conds.append(c.to_dict())
	var effs: Array = []
	for e in effects:
		effs.append(e.to_dict())
	var d := {"if": conds, "do": effs}
	if owner != "":
		d["owner"] = owner
	if loop != null:
		d["loop"] = loop.to_dict()
	if not branches.is_empty():
		var bs: Array = []
		for b in branches:
			bs.append(b.to_dict())
		d["branches"] = bs
	return d


static func from_dict(d: Dictionary) -> ProgramRule:
	var r := ProgramRule.new()
	r.owner = str(d.get("owner", ""))
	for cd in d.get("if", []):
		var c := ProgramConditions.from_dict(cd)
		if c != null:
			r.conditions.append(c)
	for ed in d.get("do", []):
		var e := ProgramEffect.from_dict(ed)
		if not e.knob in REMOVED_KNOBS:
			r.effects.append(e)
	if d.get("loop") is Dictionary:
		r.loop = ProgramLoop.from_dict(d["loop"])
	for bd in d.get("branches", []):
		if bd is Dictionary:
			r.branches.append(ProgramBranch.from_dict(bd))
	if r.loop != null and r.loop.break_on != ProgramLoop.Break.NONE:
		r.branches.append(ProgramBranch.legacy_break(r.loop.break_on == ProgramLoop.Break.KILL))
		r.loop.break_on = ProgramLoop.Break.NONE
	return r

class_name ProgramSupport
extends SupportBehavior

const CAST_NUMBER_KEY := "program_cast_number"
const REPEAT_UNDO_KEY := "program_repeat_undo"
const LOOP_KEY := "program_loop"
const FIRED_KEY := "program_fired"
const HIT_WRITES_KEY := "registry_hit_writes"
const RULES_FIRED_KEY := "program_rules_fired"
const LOOP_RULE_KEY := "program_loop_rule"

var program: SkillProgram
var limits: SkillProgramLimits
var _flat: Array[ProgramRule] = []
var _cast_count := 0
var registers: Dictionary:
	get:
		return program.state().registers
var _caster: Node3D
var _runners: Array[ProgramLoopRunner] = []
const HIT_SPEED_WINDOW := 2.5
var _hit_speed := 0.0
var _hit_speed_msec := 0
var _order: Array = []
var _order_frame := -1


func _init(p_program: SkillProgram = null, p_limits: SkillProgramLimits = null) -> void:
	program = p_program
	limits = p_limits


func before_release(cast: SkillCast) -> void:
	if cast.extra.get("hologram", false) or cast.depth > 0:
		return
	_caster = cast.caster
	_clear_stale_registers()
	program.state().sort_by = 0
	_order_frame = -1
	_hit_speed = 0.0
	if _counts_swings(cast.skill):
		cast.extra[CAST_NUMBER_KEY] = cast.skill.count_swing(cast)
	else:
		_cast_count += 1
		cast.extra[CAST_NUMBER_KEY] = _cast_count
	var undo := {"damage": 1.0, "flags": {}}
	var tickets: Array[ProgramRule] = []
	var whiles: Array[ProgramRule] = []
	var late_fors: Array[ProgramRule] = []
	_flat = program.flat_rules()
	for rule in program.rules:
		if rule.is_always() and rule.loop == null:
			_report_calls(rule, cast)
	for rule in _flat:
		if rule.scope() != ProgramCondition.Scope.CAST or rule.in_loop():
			continue
		if rule.loop != null:
			if rule.loop.mode == ProgramLoop.Mode.WHILE:
				whiles.append(rule)
			elif _is_repeat_rule(rule):
				late_fors.append(rule)
			elif rule.is_always() or rule.all_true(cast, null):
				if not rule.is_always():
					announce(cast, rule)
				for i in rule.loop.count:
					tickets.append(rule)
			continue
		if rule.is_always():
			_write_registers(rule, cast)
			_apply_effects(cast, rule, {}, true)
			continue
		if not rule.all_true(cast, null):
			continue
		_fire_cast_rule(cast, rule, undo if _is_repeat_rule(rule) else {})
	cast.extra[REPEAT_UNDO_KEY] = undo
	if tickets.is_empty() and whiles.is_empty() and late_fors.is_empty():
		return
	var runner := ProgramLoopRunner.new(self, cast, limits.loop_cap(), tickets, whiles, late_fors)
	_runners.append(runner)
	runner.start()


func apply_repeat_rules(child: SkillCast, loop_rule: ProgramRule = null) -> void:
	var undo: Dictionary = child.extra.get(REPEAT_UNDO_KEY, {})
	if not undo.is_empty():
		child.damage_mult /= maxf(float(undo["damage"]), 0.0001)
		for k in undo["flags"]:
			child.flag_bonus[k] = child.flag_bonus.get(k, 0.0) - float(undo["flags"][k])
	child.extra[REPEAT_UNDO_KEY] = {}
	for rule in _rules():
		if rule.loop != null or rule.is_always() or rule.scope() != ProgramCondition.Scope.CAST \
				or not _is_repeat_rule(rule) or rule.in_loop():
			continue
		if rule.all_true(child, null):
			_fire_cast_rule(child, rule, {})
	if loop_rule != null:
		_apply_effects(child, loop_rule, {})
		_apply_loop_branches(child, loop_rule)


func _apply_loop_branches(child: SkillCast, loop_rule: ProgramRule) -> void:
	for i in loop_rule.branches.size():
		var br := loop_rule.branch_rule(i)
		if br.effects.is_empty() or br.scope() != ProgramCondition.Scope.CAST:
			continue
		if _own_true(br, child, null):
			_fire_cast_rule(child, br, {})


static func _own_true(rule: ProgramRule, cast: SkillCast, target: Node) -> bool:
	for c in rule.own_conditions():
		if not c.is_true(cast, target):
			return false
	return true


static func _counts_swings(skill: SkillInstance) -> bool:
	for s in skill.support_behaviors:
		if s is FinishingBlowSupport or s is OverclockSupport:
			return true
	return false


func apply_fork_rules(child: SkillCast) -> void:
	for rule in _rules():
		if rule.loop != null or rule.is_always() or rule.scope() != ProgramCondition.Scope.CAST \
				or rule.in_loop():
			continue
		var about_forks := false
		for c in rule.conditions:
			if c is ForkedCondition:
				about_forks = true
		if about_forks and rule.all_true(child, null):
			_fire_cast_rule(child, rule, {})


func runner_done(runner: ProgramLoopRunner) -> void:
	_runners.erase(runner)


static func _is_repeat_rule(rule: ProgramRule) -> bool:
	for c in rule.conditions:
		if c.per_repeat():
			return true
	return false


func announce(cast: SkillCast, rule: ProgramRule) -> void:
	_mark_fired(cast, rule)
	_report_triggered(rule, cast)
	_report_calls(rule, cast)
	_trait_fired(cast, rule)
	var fired: Array = cast.extra.get(RULES_FIRED_KEY, [])
	if not rule.is_branch() and not rule.get_instance_id() in fired:
		cast.extra[RULES_FIRED_KEY] = fired + [rule.get_instance_id()]
	for c in rule.own_conditions():
		c.on_fired(cast)


static func rule_fired(cast: SkillCast, rule: ProgramRule) -> bool:
	return cast != null and rule != null and rule.get_instance_id() in cast.extra.get(RULES_FIRED_KEY, [])


func _rules() -> Array[ProgramRule]:
	if _flat.is_empty() and program != null:
		_flat = program.flat_rules()
	return _flat


func rule_key(rule: ProgramRule) -> String:
	return _rule_key(rule)


func _rule_key(rule: ProgramRule) -> String:
	if rule.is_branch():
		return "%d.%d" % [program.rules.find(rule.parent), rule.branch_index]
	return str(program.rules.find(rule))


func _fire_cast_rule(cast: SkillCast, rule: ProgramRule, record: Dictionary) -> void:
	announce(cast, rule)
	_apply_effects(cast, rule, record)


func _apply_effects(cast: SkillCast, rule: ProgramRule, record: Dictionary, only_growing := false) -> void:
	var mult := limits.rule_effect_mult(rule)
	for e in rule.effects:
		if only_growing and not e.ramps():
			continue
		var knob := limits.get_knob(e.knob)
		if knob == null:
			continue
		var v := knob.step * _steps_now(e, knob, cast) * mult
		match knob.kind:
			SkillKnob.Kind.VARIABLE:
				_queue_write(e, cast, null)
			SkillKnob.Kind.DAMAGE:
				cast.damage_mult *= 1.0 + v
				if not record.is_empty():
					record["damage"] = float(record["damage"]) * (1.0 + v)
			SkillKnob.Kind.FLAG:
				cast.flag_bonus[knob.flag_key] = cast.flag_bonus.get(knob.flag_key, 0.0) + v
				if not record.is_empty():
					record["flags"][knob.flag_key] = float(record["flags"].get(knob.flag_key, 0.0)) + v


func _write_registers(rule: ProgramRule, cast: SkillCast) -> void:
	for e in rule.effects:
		if ScriptRegisters.is_write(e.knob):
			_queue_write(e, cast, null)


func _queue_write(e: ProgramEffect, cast: SkillCast, target: Node) -> void:
	if ScriptRegisters.is_sort(e):
		_set_order(e.sort_by)
		return
	if target != null:
		ScriptRegisters.write(registers, e, target)
	elif cast != null:
		cast.extra[HIT_WRITES_KEY] = (cast.extra.get(HIT_WRITES_KEY, []) as Array) + [e]


func _steps_now(e: ProgramEffect, knob: SkillKnob, cast: SkillCast, _target: Node = null) -> float:
	if not e.ramps():
		return float(e.steps)
	var count := ScriptRegisters.size_of(registers)
	if count > 0 and not SkillTargets.is_enemy(cast.caster):
		GameEvents.report("rule_triggered", "var_growth")
	return ScriptRegisters.ramp_steps(count, e.steps, program.growth_room(e, knob))


func _clear_stale_registers() -> void:
	if not is_instance_valid(_caster):
		return
	var now := Time.get_ticks_msec()
	var st := program.state()
	if st.last_active_msec < 0:
		st.last_active_msec = now
	var active := st.last_active_msec
	var hp := _caster.get("health") as HealthComponent
	if hp != null:
		active = maxi(active, hp.last_damaged_msec)
	var am := _caster.get_tree().get_first_node_in_group("area_manager") as AreaManager if _caster.is_inside_tree() else null
	var area := am.current_area.get_instance_id() if am != null and am.current_area != null else st.area_id
	if st.area_id == 0:
		st.area_id = area
	if area != st.area_id or now - active > int(ScriptRegisters.CLEAR_AFTER * 1000.0):
		registers.clear()
		st.area_id = area

func modify_hit(cast: SkillCast, target: Node, hit: HitInfo) -> void:
	program.state().last_active_msec = Time.get_ticks_msec()
	var hit_writes: Array[ProgramEffect] = []
	for rule in _rules():
		if rule.is_always() or rule.loop != null or rule.scope() != ProgramCondition.Scope.HIT:
			continue
		if rule.in_loop():
			if cast.extra.get(LOOP_RULE_KEY, "") != _rule_key(rule.parent) or not _own_true(rule, cast, target):
				continue
		elif not rule.all_true(cast, target):
			continue
		_mark_fired(cast, rule)
		_report_triggered(rule, cast)
		_trait_fired(cast, rule)
		var mult := limits.rule_effect_mult(rule)
		var flags_key := "program_hit_flags_%s" % _rule_key(rule)
		for e in rule.effects:
			var knob := limits.get_knob(e.knob)
			if knob == null:
				continue
			if knob.kind == SkillKnob.Kind.VARIABLE:
				hit_writes.append(e)
			elif knob.kind == SkillKnob.Kind.DAMAGE:
				hit.damage *= 1.0 + knob.step * _steps_now(e, knob, cast, target) * mult
			elif SkillProgramLimits.hit_defense_knob(knob) and not limits.per_hit_any_knob():
				_hit_defense(cast, rule, knob, e, hit)
			elif SkillProgramLimits.hit_support_knob(knob) and not limits.per_hit_any_knob():
				var add := knob.step * _steps_now(e, knob, cast, target) * limits.rule_effect_mult(rule, false)
				hit.flag_bonus[knob.flag_key] = float(hit.flag_bonus.get(knob.flag_key, 0.0)) + add
			elif limits.per_hit_attack_speed(knob) and not cast.extra.has(flags_key):
				_hit_speed += knob.value_for(e.steps) * mult
				_hit_speed_msec = Time.get_ticks_msec()
			elif knob.kind == SkillKnob.Kind.FLAG and limits.per_hit_any_knob() \
					and not cast.extra.has(flags_key):
				cast.flag_bonus[knob.flag_key] = cast.flag_bonus.get(knob.flag_key, 0.0) + knob.step * _steps_now(e, knob, cast, target) * mult
		if limits.per_hit_any_knob():
			cast.extra[flags_key] = true
	_write_hit_entries(cast, target, hit_writes)


func _write_hit_entries(cast: SkillCast, target: Node, hit_writes: Array[ProgramEffect]) -> void:
	for e in cast.extra.get(HIT_WRITES_KEY, []):
		ScriptRegisters.write(registers, e, target)
	for e in hit_writes:
		if ScriptRegisters.is_sort(e):
			_set_order(e.sort_by)
		else:
			ScriptRegisters.write(registers, e, target)


func _set_order(sort_by: int) -> void:
	if program.state().sort_by != sort_by:
		program.state().sort_by = sort_by
		_order_frame = -1


func registry_order() -> Array:
	var frame := Engine.get_physics_frames()
	if frame != _order_frame:
		_order_frame = frame
		_order = ScriptRegisters.ordered(registers, program.state().sort_by, _caster)
	return _order


func position_of(target: Node) -> int:
	var at: Variant = ScriptRegisters.entry_key(target)
	return registry_order().find(at) if at != null else -1


func _hit_defense(cast: SkillCast, rule: ProgramRule, knob: SkillKnob, e: ProgramEffect, hit: HitInfo) -> void:
	var v := knob.step * _steps_now(e, knob, cast, hit.target) * limits.rule_effect_mult(rule, false)
	if v <= 0.0:
		return
	if knob.id == &"leech":
		hit.leech_bonus += v
		return
	var once := "program_hit_def_%s_%s" % [_rule_key(rule), knob.id]
	if cast.extra.has(once):
		return
	cast.extra[once] = true
	if knob.id == &"barrier":
		hit.barrier_bonus += v
	else:
		hit.guard_bonus += v


func speed_mult(_skill: SkillInstance) -> float:
	if _hit_speed == 0.0 or (Time.get_ticks_msec() - _hit_speed_msec) / 1000.0 > HIT_SPEED_WINDOW:
		return 1.0
	return maxf(1.0 + _hit_speed, 0.1)


static func _mark_fired(cast: SkillCast, rule: ProgramRule) -> void:
	var fired: Array = cast.extra.get(FIRED_KEY, [])
	for c in rule.own_conditions():
		if not c.fired_id() in fired:
			fired.append(c.fired_id())
	cast.extra[FIRED_KEY] = fired


static func fired_conditions(cast: SkillCast) -> Array:
	if cast == null:
		return []
	var out: Array = cast.extra.get(FIRED_KEY, []).duplicate()
	var runner := cast.extra.get(LOOP_KEY) as ProgramLoopRunner
	if runner != null and runner.root != null and runner.root != cast:
		for id in runner.root.extra.get(FIRED_KEY, []):
			if not id in out:
				out.append(id)
	return out


func _trait_fired(cast: SkillCast, rule: ProgramRule) -> void:
	if limits.class_trait != null and not SkillTargets.is_enemy(cast.caster):
		limits.class_trait.on_rule_fired(cast.caster, "%d_%s" % [get_instance_id(), _rule_key(rule)])


func _report_triggered(rule: ProgramRule, cast: SkillCast) -> void:
	if SkillTargets.is_enemy(cast.caster):
		return
	for c in rule.own_conditions():
		GameEvents.report("rule_triggered", c.type_id())
	if cast.skill != null and cast.skill.gem != null:
		GameEvents.report("rule_fired", "%s#%s" % [cast.skill.gem.id, _rule_key(rule)])


func _report_calls(rule: ProgramRule, cast: SkillCast) -> void:
	if SkillTargets.is_enemy(cast.caster) or cast.skill == null or cast.skill.gem == null:
		return
	var holder: Object = rule.branch() if rule.is_branch() else rule
	if holder == null or not holder.has_meta(&"calls"):
		return
	for index in holder.get_meta(&"calls"):
		GameEvents.report("routine_called", "%d|%s" % [int(index), cast.skill.gem.id])

class_name ScriptBudget
extends RefCounted

const GRADE_POINTS := [0.0, 0.5, 1.0]
const POINTS_PER_CLEAR := 1.0
const STEPS_PER_DISTINCTION := 1
const STEPS_PER_ACT_BOSS := 1
const BOSS_DEED := "act_boss"
const DISTINCTION := 2
const CLEAR_DEED := "clear_area"


static func breakdown(qm: QuestManager) -> Dictionary:
	var out := {"rank": 0.0, "grades": 0.0, "clears": 0.0, "areas": 0, "total": 0.0,
		"distinctions": 0, "bosses": 0, "steps": 0}
	if qm == null:
		return out
	out["rank"] = float(Ranks.perk(Ranks.index_of(qm), "budget"))
	for r in Ranks.LADDER:
		var g := qm.exam_grade(r["id"]) if r["id"] != "" else -1
		if g >= 0 and g < GRADE_POINTS.size():
			out["grades"] += GRADE_POINTS[g]
		if g == DISTINCTION:
			out["distinctions"] += 1
	out["bosses"] = _deeds(qm, BOSS_DEED).size()
	out["steps"] = out["distinctions"] * STEPS_PER_DISTINCTION + out["bosses"] * STEPS_PER_ACT_BOSS
	out["areas"] = cleared_areas(qm).size()
	out["clears"] = out["areas"] * POINTS_PER_CLEAR
	out["total"] = out["rank"] + out["grades"] + out["clears"]
	return out


static func points(qm: QuestManager) -> float:
	return breakdown(qm)["total"]


static func steps(qm: QuestManager) -> int:
	return breakdown(qm)["steps"]


static func cleared_areas(qm: QuestManager) -> PackedStringArray:
	return _deeds(qm, CLEAR_DEED)


static func act_bosses_killed(qm: QuestManager) -> PackedStringArray:
	return _deeds(qm, BOSS_DEED)


static func is_boss_killed(qm: QuestManager, enemy_id: String) -> bool:
	return qm != null and qm.deed_count(BOSS_DEED, enemy_id) > 0


static func _deeds(qm: QuestManager, kind: String) -> PackedStringArray:
	var out := PackedStringArray()
	if qm == null:
		return out
	var prefix := kind + ":"
	for key: String in qm.deeds:
		if key.begins_with(prefix) and int(qm.deeds[key]) > 0:
			out.append(key.substr(prefix.length()))
	return out


static func is_cleared(qm: QuestManager, area_id: String) -> bool:
	return qm != null and qm.deed_count(CLEAR_DEED, area_id) > 0

class_name ScriptRegisters
extends RefCounted

const WRITE_OPS := ["add", "remove", "sort"]
const MEMBER_WRITES := 2
const SORT_KEYS := ["age", "hp", "max_hp", "distance"]
const POSITIONS := ["[0]", "[1]", "[2]", "[-1]", "[:2]", "[:3]"]
const READ_KEYS := ["target", "size"]
const MEMBER_OPS := ["in", "not in"]
const OPS := ["<", ">", "<=", ">=", "==", "!="]
const MAX_VALUE := 99
const WRITE_KNOB := &"var"
const CLEAR_AFTER := 10.0
const RAMP_PER_STEP := 5
const RAMP_COST_SLOW := 0.3
const RAMP_COST_FAST := 0.6
const GROW_KNOBS: Array[StringName] = [&"radius", &"range_bonus", &"arc_angle", &"range", &"spread", &"speed", &"leech", &"barrier", &"guard"]

static var _knobs: Array[SkillKnob] = []


static func knobs() -> Array[SkillKnob]:
	if _knobs.is_empty():
		var k := SkillKnob.new()
		k.id = WRITE_KNOB
		k.display_name = "registry"
		k.kind = SkillKnob.Kind.VARIABLE
		k.step = 1.0
		k.min_steps = 0
		k.max_steps = 9999
		k.cost_per_step = 0.0
		k.unit = ""
		_knobs.append(k)
	return _knobs


static func is_write(knob_id: StringName) -> bool:
	return knob_id == WRITE_KNOB


static func is_sort(e: ProgramEffect) -> bool:
	return e != null and is_write(e.knob) and e.write_op() == "sort"


static func sort_text(sort_by: int) -> String:
	var n := SORT_KEYS.size() * 2
	var s := posmod(sort_by, n)
	return "key=%s%s" % [SORT_KEYS[s / 2], ", reverse=true" if s % 2 == 1 else ""]


const SORT_ENDS := {"age": ["oldest", "newest"], "hp": ["lowest-hp", "highest-hp"],
	"max_hp": ["frailest", "toughest"], "distance": ["nearest", "farthest"]}
const POSITION_WORDS := {"[0]": "first", "[1]": "2nd", "[2]": "3rd", "[-1]": "last",
	"[:2]": "first 2", "[:3]": "first 3"}


static func position_meaning(pos: int, sort_by: int) -> String:
	var s := posmod(sort_by, SORT_KEYS.size() * 2)
	var ends: Array = SORT_ENDS[SORT_KEYS[s / 2]]
	var front: String = ends[1 if s % 2 == 1 else 0]
	var back: String = ends[0 if s % 2 == 1 else 1]
	match POSITIONS[posmod(pos, POSITIONS.size())]:
		"[1]":
			return "the 2nd " + front
		"[2]":
			return "the 3rd " + front
		"[-1]":
			return "the " + back
		"[:2]":
			return "the 2 " + front
		"[:3]":
			return "the 3 " + front
	return "the " + front


static func sort_meaning(sort_by: int) -> String:
	var s := posmod(sort_by, SORT_KEYS.size() * 2)
	var rev := s % 2 == 1
	match SORT_KEYS[s / 2]:
		"hp":
			return "most life left first" if rev else "least life left first"
		"max_hp":
			return "toughest first (most maximum life)" if rev else "frailest first (least maximum life)"
		"distance":
			return "farthest from you first" if rev else "nearest to you first"
	return "newest first (the last one added)" if rev else "oldest first (the first one added)"


static func raises(e: ProgramEffect) -> bool:
	return is_write(e.knob) and e.write_op() == "add"


static func can_grow(knob: SkillKnob) -> bool:
	return knob != null and not knob.always_only \
			and (knob.kind == SkillKnob.Kind.DAMAGE or GROW_KNOBS.has(knob.id))


static func ramp_steps(count: int, value: int, room: int) -> float:
	return clampf(float(count) * value / RAMP_PER_STEP, 0.0, float(maxi(room, 0)))


static func ramp_reach(value: int, room: int) -> int:
	return ceili(float(room) * RAMP_PER_STEP / maxi(value, 1))


static func ramp_cost_share(value: int, room: int) -> float:
	if room <= 1:
		return RAMP_COST_FAST
	return lerpf(RAMP_COST_SLOW, RAMP_COST_FAST, clampf(float(value - 1) / (room - 1), 0.0, 1.0))


static func compare(value: int, op: String, n: int) -> bool:
	match op:
		"<": return value < n
		">": return value > n
		"<=": return value <= n
		">=": return value >= n
		"==": return value == n
		"!=": return value != n
	return false


static func entry_key(target: Node) -> Variant:
	if target == null or not is_instance_valid(target) or target.is_in_group(&"breakables"):
		return null
	return target.get_instance_id()


static func has(store: Dictionary, target: Node) -> bool:
	var at: Variant = entry_key(target)
	return at != null and store.has(at)


static func write(store: Dictionary, e: ProgramEffect, target: Node) -> bool:
	var at: Variant = entry_key(target)
	if at == null or e.write_op() == "sort":
		return false
	if e.write_op() == "add":
		if store.has(at) or SkillBehavior.is_dead(target):
			return false
		store[at] = true
		return true
	return store.erase(at)


static func size_of(store: Dictionary) -> int:
	prune(store)
	return store.size()


static func ordered(store: Dictionary, sort_by: int, from: Node3D) -> Array:
	prune(store)
	var ids: Array = store.keys()
	var s := posmod(sort_by, SORT_KEYS.size() * 2)
	var key: String = SORT_KEYS[s / 2]
	var rev := s % 2 == 1
	if key == "age":
		if rev:
			ids.reverse()
		return ids
	var value := {}
	for i in ids.size():
		value[ids[i]] = [_sort_value(instance_from_id(ids[i]) as Node, key, from), i]
	ids.sort_custom(func(a, b) -> bool:
		var va: Array = value[a]
		var vb: Array = value[b]
		if not is_equal_approx(float(va[0]), float(vb[0])):
			return (float(va[0]) > float(vb[0])) if rev else (float(va[0]) < float(vb[0]))
		return int(va[1]) < int(vb[1]))
	return ids


static func _sort_value(node: Node, key: String, from: Node3D) -> float:
	if node == null:
		return 0.0
	match key:
		"hp", "max_hp":
			var h := node.get("health") as HealthComponent
			if h == null:
				return 0.0
			if key == "hp":
				return h.current_health
			return h.stats.get_stat(StatTypes.Stat.MAX_HEALTH) if h.stats != null else h.current_health
		"distance":
			var n3 := node as Node3D
			return n3.global_position.distance_to(from.global_position) if n3 != null and is_instance_valid(from) else 0.0
	return 0.0


static func at_position(index: int, count: int, pos: int) -> bool:
	if index < 0 or count <= 0:
		return false
	match POSITIONS[posmod(pos, POSITIONS.size())]:
		"[0]": return index == 0
		"[1]": return index == 1
		"[2]": return index == 2
		"[-1]": return index == count - 1
		"[:2]": return index < 2
		"[:3]": return index < 3
	return false


static func is_slice(pos: int) -> bool:
	return POSITIONS[posmod(pos, POSITIONS.size())].begins_with("[:")


static func prune(store: Dictionary) -> void:
	for k in store.keys():
		var node := instance_from_id(k) as Node if k is int else null
		if node == null or not node.is_inside_tree() or SkillBehavior.is_dead(node):
			store.erase(k)

class_name SkillKnob
extends Resource

enum Kind {
	DAMAGE,
	FLAG,
	STAT,
	VARIABLE,
}

@export var id: StringName
@export var display_name: String
@export var kind: Kind = Kind.FLAG
@export var always_only := false
@export var first_cast_only := false
@export var stat: StatTypes.Stat = StatTypes.Stat.SKILL_DAMAGE_MULT
@export var flag_key: String = ""
@export var step: float = 0.05
@export var min_steps: int = -5
@export var max_steps: int = 5
@export var cost_per_step: float = 1.0
@export var unit: String = "%"
@export var defensive := false


func value_for(steps: int) -> float:
	return step * steps


const STEP_COST_GROWTH := 1.25
const BUFF_PRICE_MULT := 0.5


func cost_for(steps: int) -> float:
	return cost_per_step * steps


func is_buff(steps: int) -> bool:
	return cost_per_step * steps > 0.0


func buff_cost(n: int) -> float:
	if n <= 0:
		return 0.0
	return first_step_cost() * (pow(STEP_COST_GROWTH, n) - 1.0) / (STEP_COST_GROWTH - 1.0)


func first_step_cost() -> float:
	return absf(cost_per_step) * BUFF_PRICE_MULT


func next_step_cost(n: int) -> float:
	return first_step_cost() * pow(STEP_COST_GROWTH, maxi(n, 0))


func describe(steps: int) -> String:
	var v := value_for(steps)
	match unit:
		"%":
			return "%s %+d%%" % [display_name, roundi(v * 100.0)]
		"m":
			return "%s %+.2f m" % [display_name, v]
		"deg":
			return "%s %+d°" % [display_name, roundi(v)]
		_:
			if is_equal_approx(v, roundf(v)):
				return "%s %+d" % [display_name, roundi(v)]
			return "%s %+.1f" % [display_name, v]


func make_modifier(steps: int, source_id: String) -> Modifier:
	var m := Modifier.new()
	m.stat = StatTypes.Stat.SKILL_DAMAGE_MULT if kind == Kind.DAMAGE else stat
	m.op = Modifier.ModOp.PERCENT_MULT
	m.value = value_for(steps)
	m.source_id = source_id
	return m

class_name SkillProgram
extends Resource

const EPSILON := 0.001

@export var rules: Array[ProgramRule] = []
var registers := {}
var last_active_msec := -1
var area_id := 0
var sort_by := 0

var source: SkillProgram
var _actions_override := -1
var _origin := {}
var _call_problems: Array = []


func has_rules() -> bool:
	return not rules.is_empty()


func state() -> SkillProgram:
	return source if source != null else self


func has_calls() -> bool:
	for r in rules:
		for e in r.effects:
			if e.is_call():
				return true
		for b in r.branches:
			for e in b.effects:
				if e.is_call():
					return true
	return false


func called_routines() -> Array:
	var out: Array = []
	for r in rules:
		for list in [r.effects] + r.branches.map(func(b: ProgramBranch) -> Array: return b.effects):
			for e in list:
				if e.is_call() and not (e.op in out):
					out.append(e.op)
	return out


func resolve(routines: Array, limits: SkillProgramLimits = null) -> SkillProgram:
	var out := SkillProgram.new()
	out.source = self
	out.registers = registers
	out._actions_override = action_count()
	for r in rules:
		var nr := ProgramRule.new()
		nr.owner = r.owner
		nr.conditions.assign(r.conditions)
		nr.loop = r.loop
		out._origin[nr] = r
		var failed := out._call_problems.size()
		var calls := out._expand(r.effects, nr.effects, r, routines, limits, false)
		if out._call_problems.size() > failed:
			nr.set_meta(&"call_failed", true)
		for b in r.branches:
			var nb := ProgramBranch.new()
			nb.conditions.assign(b.conditions)
			nb.breaks = b.breaks
			out._origin[nb] = b
			failed = out._call_problems.size()
			var inner := out._expand(b.effects, nb.effects, r, routines, limits, true)
			if out._call_problems.size() > failed:
				nb.set_meta(&"call_failed", true)
			if not inner.is_empty():
				nb.set_meta(&"calls", inner.map(func(c: Array) -> int: return (c[0] as ProgramEffect).op))
			nr.branches.append(nb)
		for c in calls:
			var call: ProgramEffect = c[0]
			var routine: ProgramRoutine = c[1]
			for rb in routine.body.branches:
				var nb := ProgramBranch.new()
				nb.breaks = rb.breaks
				for cond in rb.conditions:
					var cc := (cond as ProgramCondition).duplicate() as ProgramCondition
					out._origin[cc] = call
					nb.conditions.append(cc)
				for e in rb.effects:
					var ec := (e as ProgramEffect).duplicate() as ProgramEffect
					out._origin[ec] = call
					nb.effects.append(ec)
				out._origin[nb] = call
				nr.branches.append(nb)
		if not calls.is_empty():
			nr.set_meta(&"calls", calls.map(func(c: Array) -> int: return (c[0] as ProgramEffect).op))
		out.rules.append(nr)
	return out


func _expand(lines: Array, into: Array, rule: ProgramRule, routines: Array, limits: SkillProgramLimits,
		in_if: bool) -> Array:
	var calls: Array = []
	for e in lines:
		var line := e as ProgramEffect
		if not line.is_call():
			into.append(line)
			continue
		var routine: ProgramRoutine = routines[line.op] if line.op >= 0 and line.op < routines.size() else null
		var problem: Array = []
		if limits != null and not limits.allows("call"):
			problem = ["routines aren't available at your rank.", "routines aren't available at your rank (Engineer)",
				"delete this line"]
		elif routine == null:
			problem = ["it calls a routine that doesn't exist.", "there's no routine %d" % (line.op + 1),
				"pick a routine (click its name) or delete this line"]
		elif not routine.problem().is_empty():
			problem = routine.problem()
		elif in_if and routine.has_if():
			problem = ["%s() has an if, and it's called inside an if." % routine.routine_name,
				"%s() has an if: it can't be called inside an if" % routine.routine_name,
				"call it from the rule's own lines, or take the if out of %s()" % routine.routine_name]
		if not problem.is_empty():
			_call_problems.append([rule, line, problem[0], problem[1], problem[2]])
			continue
		for re in routine.body.effects:
			var copy := (re as ProgramEffect).duplicate() as ProgramEffect
			_origin[copy] = line
			into.append(copy)
		calls.append([line, routine])
	return calls


func _remap(v: ProgramValidation) -> void:
	for issue in v.issues:
		var r: Variant = issue["rule"]
		var p: Variant = issue["part"]
		if r != null and _origin.has(r):
			issue["rule"] = _origin[r]
		if p != null and _origin.has(p):
			issue["part"] = _origin[p]


func flat_rules() -> Array[ProgramRule]:
	var out: Array[ProgramRule] = []
	for r in rules:
		out.append_array(r.flat())
	return out


func action_count() -> int:
	if _actions_override >= 0:
		return _actions_override
	var n := 0
	for r in rules:
		n += r.action_count()
	return n


func validate(limits: SkillProgramLimits) -> ProgramValidation:
	if limits != null and has_calls():
		var resolved := resolve(limits.routines, limits)
		var v := resolved._validate_plain(limits)
		resolved._remap(v)
		for p in resolved._call_problems:
			v.add_issue("%s rule: %s" % [_owner_name((p[0] as ProgramRule).owner), p[2]], p[3], p[4], p[0], p[1])
		return v
	return _validate_plain(limits)


func _validate_plain(limits: SkillProgramLimits) -> ProgramValidation:
	tag_registry_reads()
	var v := ProgramValidation.new()
	if limits == null:
		v.add_error("This skill can't be programmed.")
		return v
	v.point_budget = limits.point_budget
	v.max_actions = limits.max_actions
	v.actions_used = action_count()

	if rules.size() > limits.max_rules:
		v.add_error("Too many rules (%d / %d)." % [rules.size(), limits.max_rules],
			"delete a rule, or move its lines into another rule with the same conditions")
	if v.actions_used > limits.max_actions:
		var extra := v.actions_used - limits.max_actions
		v.add_error("Too many blocks (%d / %d)." % [v.actions_used, limits.max_actions],
			"remove %d block%s: every condition, knob line and loop counts as one" % [extra, "" if extra == 1 else "s"])

	var always_steps := {}
	var cond_up := {}
	var cond_down := {}
	var raising := {}
	var lowering := {}
	var rule_cost := {}

	var has_loop := false
	for r in rules:
		has_loop = has_loop or r.loop != null
	var per_owner := {}
	var labels := {}
	var all := flat_rules()
	for i in all.size():
		var rule := all[i]
		var label := ""
		if rule.is_branch():
			label = "%s, if %d" % [labels.get(rule.parent, "Rule"), rule.branch_index + 1]
		else:
			per_owner[rule.owner] = per_owner.get(rule.owner, 0) + 1
			label = "%s rule %d" % [_owner_name(rule.owner), per_owner[rule.owner]]
			labels[rule] = label
		if rule.owner != "" and not (rule.owner in limits.knob_owner.values()):
			if not rule.is_branch():
				v.add_issue("%s: that support isn't socketed in this skill." % label,
					"its support gem isn't socketed in this skill",
					"socket the gem again, or delete this rule", rule)
			continue
		if not rule.is_branch() and not rule.branches.is_empty():
			if not limits.allows("if"):
				v.add_issue("%s: if blocks aren't available at your rank." % label,
					"if blocks aren't available at your rank (Architect)", "delete the if block", rule, rule.branches[0])
		if rule.is_branch():
			if not limits.allows("if"):
				continue
			if rule.own_conditions().is_empty():
				v.add_issue("%s has no condition." % label, "this if has no condition yet",
					"drag a condition onto this if line (killed, hit_nothing, target_life_below...)", rule)
				continue
			if rule.branch().breaks and rule.parent.loop == null:
				v.add_issue("%s: break only works in a rule with a loop." % label,
					"break only works in a rule with a loop",
					"add a for or while loop to this rule, or delete the break", rule)
				continue
			if rule.effects.is_empty() and rule.branch().breaks:
				pass
			elif rule.effects.is_empty():
				if not rule.branch().has_meta(&"call_failed"):
					v.add_issue("%s does nothing." % label, "this if does nothing yet",
						"drag a knob from SNIPPETS under it, or delete the if", rule)
				continue
		elif rule.effects.is_empty() and rule.loop == null and rule.branches.is_empty():
			if rule.has_meta(&"call_failed"):
				continue
			v.add_issue("%s has no effects." % label, "this rule does nothing yet",
				"drag a knob from SNIPPETS under it, or delete the rule", rule)
			continue
		var repeated := rule.repeated_condition()
		if repeated != "" and not (rule.is_branch() and rule.parent.repeated_condition() == repeated):
			v.add_issue("%s: %s() is used more than once. Each condition can appear once per rule." % [label, repeated],
				"%s() is already used in this rule" % repeated,
				"delete this line: each condition can appear once per rule", rule, _second_use(rule))
			continue
		var missing: ProgramCondition = null
		for c in rule.own_conditions():
			if c.required_support() != "" and not limits.support_ids.has(c.required_support()):
				missing = c
		if missing != null:
			var gem := _owner_name(missing.required_support())
			v.add_issue("%s: %s needs the %s gem in this skill." % [label, "that block", gem],
				"needs the %s gem in this skill" % gem,
				"socket %s, or delete this line" % gem, rule, missing)
			continue
		var locked: ProgramCondition = null
		for c in rule.own_conditions():
			if not limits.allows("cond:" + c.type_id()):
				locked = c
		if locked != null:
			var lfn := locked.palette_name().to_lower().replace(" ", "_")
			v.add_issue("%s: %s() isn't available at your rank." % [label, lfn],
				"%s() isn't available at your rank" % lfn, "delete this line", rule, locked)
			continue
		if rule.loop != null and not limits.allows("loop"):
			v.add_issue("%s: loops aren't available at your rank." % label,
				"loops aren't available at your rank", "delete the loop", rule, rule.loop)
			continue
		if rule.only_on_repeats() and not has_loop and not (rule.is_branch() and rule.parent.only_on_repeats()):
			var needs: ProgramCondition = null
			for c in rule.conditions:
				if c.needs_loop():
					needs = c
					break
			var fn := needs.palette_name().to_lower().replace(" ", "_")
			v.add_issue("%s: %s() only changes loop repeats, and this skill has no loop." % [label, fn],
				"%s() only changes loop repeats, and this skill has no loop" % fn,
				"add a for or while loop to this skill, or use a condition that checks the cast itself (target_life_below() checks each hit)",
				rule, needs)
			continue
		var raw_cost := 0.0
		if rule.loop != null:
			var loop_problem := _check_loop(rule, limits)
			if not loop_problem.is_empty():
				v.add_issue("%s: %s" % [label, loop_problem[0]], String(loop_problem[0]).trim_suffix("."),
					loop_problem[1], rule, rule.loop)
				continue
			raw_cost += rule.loop.cost(limits.loop_cost_mult(), rule.has_break())
		for e in rule.effects:
			var knob := limits.get_knob(e.knob)
			if knob == null:
				v.add_issue("%s: %s can't be changed on this skill." % [label, e.knob],
					"%s can't be changed on this skill" % e.knob, "delete this line", rule, e)
				continue
			if limits.owner_of(e.knob) != rule.owner:
				v.add_issue("%s: %s belongs to another gem." % [label, knob.display_name],
					"%s belongs to the %s file" % [e.knob, _owner_name(limits.owner_of(e.knob))],
					"delete it here and add it in that gem's file", rule, e)
				continue
			if e.steps == 0:
				continue
			if e.ramps() and not ScriptRegisters.can_grow(knob):
				var why := "it's read before the cast starts" if knob.always_only or knob.kind == SkillKnob.Kind.STAT \
						else "it can't change gradually"
				v.add_issue("%s: %s can't grow with the registry." % [label, knob.display_name],
					"%s can't grow with the registry: %s" % [e.knob, why],
					"click its %s until the line is a plain value" % e.ramp_text(), rule, e)
				continue
			if e.ramps() and knob.cost_for(e.steps) < 0.0:
				v.add_issue("%s: only a buff can grow with the registry." % label,
					"only a buff can grow with the registry, and this pays points back",
					"raise its value above 0, or click its %s until the line is a plain value" % e.ramp_text(), rule, e)
				continue
			if e.ramps() and (e.steps > knob.max_steps or _grows_twice(e) or growth_room(e, knob) <= 0):
				var problem := "%s grows too fast: the most is one step's worth per %d enemies in the registry" % [e.knob, ScriptRegisters.RAMP_PER_STEP]
				var cure := "lower its value"
				if _grows_twice(e):
					problem = "only one line per knob can grow, and %s already has one" % e.knob
					cure = "delete this line, or click its %s until the line is a plain value" % e.ramp_text()
				elif growth_room(e, knob) <= 0:
					problem = "%s has no room left to grow: other lines reach its limit" % e.knob
					cure = "lower the other %s lines" % e.knob
				v.add_issue("%s: %s." % [label, problem], problem, cure, rule, e)
				continue
			if not rule.is_baked():
				if knob.kind == SkillKnob.Kind.STAT and not (rule.scope() == ProgramCondition.Scope.HIT \
						and limits.per_hit_attack_speed(knob)):
					var no_loop := " without a loop" if rule.is_always() else ""
					v.add_issue("%s: %s can only be changed in an ALWAYS rule%s." % [label, knob.display_name, no_loop],
						"%s only works in an always: rule%s" % [e.knob, no_loop],
						"move this line into an always: rule%s" % (" that has no loop" if no_loop != "" else ""), rule, e)
					continue
				if knob.always_only:
					v.add_issue("%s: %s can only be changed in an ALWAYS rule." % [label, knob.display_name],
						"%s only works in an always: rule (it's read before the cast starts)" % e.knob,
						"move this line into an always: rule without a loop", rule, e)
					continue
				if knob.first_cast_only and (rule.loop != null or rule.in_loop() or rule.only_on_repeats()):
					v.add_issue("%s: %s does nothing on loop repeats." % [label, knob.display_name],
						"%s does nothing on loop repeats: its gem only acts on the original cast" % e.knob,
						"move this line to a rule without a loop (and without target_alive, stack_overflow or loop_index)", rule, e)
					continue
				if rule.scope() == ProgramCondition.Scope.HIT and knob.kind != SkillKnob.Kind.DAMAGE \
						and knob.kind != SkillKnob.Kind.VARIABLE and not SkillProgramLimits.per_hit_knob(knob) \
						and not limits.per_hit_any_knob():
					v.add_issue("%s: per-hit conditions can't change %s." % [label, knob.display_name],
						"per-hit rules can change damage, leech, barrier, guard and support effects that happen on a hit or kill (explosions, breakpoints, tremors, charge, error damage, forks): %s is decided before the skill hits" % e.knob,
						"move this line to an always: rule, or to a rule without per-hit conditions", rule, e)
					continue
			raw_cost += line_cost(knob, e)
			var pushes := raising if e.steps > 0 else lowering
			if not pushes.has(e.knob):
				pushes[e.knob] = []
			pushes[e.knob].append([rule, e])
			if e.ramps():
				pass
			elif rule.is_baked():
				always_steps[e.knob] = always_steps.get(e.knob, 0) + e.steps
			elif e.steps > 0:
				cond_up[e.knob] = cond_up.get(e.knob, 0) + e.steps
			else:
				cond_down[e.knob] = cond_down.get(e.knob, 0) + e.steps
		if rule.is_baked():
			var rule_points := 0.0
			for e in rule.effects:
				var k := limits.get_knob(e.knob)
				if k == null or limits.owner_of(e.knob) != rule.owner:
					continue
				var c := line_cost(k, e)
				if c > 0.0:
					c *= limits.buff_cost_mult()
					v.buff_points += c
				else:
					v.drawback_points -= c
				v.points_used += c
				rule_points += c
			rule_cost[rule] = rule_points
		else:
			var net := maxf(raw_cost, 0.0) * rule.cost_multiplier() * limits.rule_cost_mult(rule) \
					* limits.buff_cost_mult()
			v.points_used += net
			v.buff_points += net
			rule_cost[rule] = net

	for knob in limits.knobs:
		var base: int = always_steps.get(knob.id, 0)
		var lo: int = base + cond_down.get(knob.id, 0)
		var hi: int = base + cond_up.get(knob.id, 0)
		v.knob_ranges[knob.id] = [lo, hi]
		if hi > knob.max_steps:
			_knob_issue(v, "%s can go up at most %d steps (this program: %d)." % [knob.display_name, knob.max_steps, hi],
				"%s goes up %d steps in total, max %d" % [knob.id, hi, knob.max_steps],
				"lower one of the %s lines (rules with conditions count as if all were active at once)" % knob.id,
				raising.get(knob.id, []))
		if lo < knob.min_steps:
			_knob_issue(v, "%s can go down at most %d steps (this program: %d)." % [knob.display_name, -knob.min_steps, -lo],
				"%s goes down %d steps in total, max %d" % [knob.id, -lo, -knob.min_steps],
				"raise one of the %s lines (rules with conditions count as if all were active at once)" % knob.id,
				lowering.get(knob.id, []))

	var cap_knob := limits.get_knob(VariableStoreSupport.CAPACITY_KNOB)
	if cap_knob != null:
		var capacity := VariableStoreSupport.capacity_with(int(always_steps.get(cap_knob.id, 0)), cap_knob)
		var seen := {}
		for rule in flat_rules():
			seen[rule.owner] = seen.get(rule.owner, 0) + 1
			for c in rule.own_conditions():
				var charge := c as ChargeAtLeastCondition
				if charge == null or charge.n <= capacity:
					continue
				var label := "%s rule %d" % [_owner_name(rule.owner), seen[rule.owner]]
				v.add_issue("%s: charge >= %d never fires, this skill holds at most %d charge." % [label, charge.n, capacity],
					"never fires: this skill holds at most %d charge" % capacity,
					"lower the number to %d or less, or raise Charge capacity in an always: rule (+%d per step)" % [
						capacity, int(cap_knob.step)], rule, charge)

	_check_registers(v)
	_check_sorting(v, limits)

	if v.points_used > limits.point_budget + EPSILON:
		var over_by := v.points_used - limits.point_budget
		v.add_error("Over budget by %.1f points: add drawbacks or remove buffs." % over_by,
			"lower a buff (+ line), or add a drawback (- line) that pays at least %.1f" % over_by)
		var priciest: ProgramRule = null
		for r in rule_cost:
			if rule_cost[r] > EPSILON and (priciest == null or rule_cost[r] > rule_cost[priciest]):
				priciest = r
		if priciest != null:
			v.add_note("costs the most: %.1f points" % rule_cost[priciest],
				"trimming this rule is the quickest way back under budget", priciest)

	if has_rules():
		v.tier = limits.tier_for(v.actions_used)
		for t in limits.tiers:
			if t == v.tier:
				break
			v.next_tier = t
		if v.next_tier != null:
			v.actions_to_next_tier = v.actions_used - v.next_tier.max_actions
	return v


func _check_registers(v: ProgramValidation) -> void:
	var seen := {}
	var adds := _adds_to_registry()
	var fix := "add registry.add(target) to a rule: always: adds every enemy the skill hits"
	for rule in flat_rules():
		seen[rule.owner] = seen.get(rule.owner, 0) + 1
		var label := "%s rule %d" % [_owner_name(rule.owner), seen[rule.owner]]
		for c in rule.own_conditions():
			var read := c as VarCondition
			if read == null:
				continue
			var constant := read.constant_answer()
			if constant != "":
				var text := "registry.size %s %d is %s true" % [read.op_text(), read.value, constant]
				v.add_issue("%s: %s." % [label, text], text, "change the operator or the size", rule, c)
			elif not adds:
				v.add_issue("%s: nothing ever adds to the registry, so it stays empty." % label,
					"nothing ever adds to the registry, so it stays empty", fix, rule, c)
		for e in rule.effects:
			if e.ramps() and e.steps != 0 and not adds:
				v.add_issue("%s: nothing ever adds to the registry, so this buff stays at nothing." % label,
					"nothing ever adds to the registry, so this buff stays at nothing", fix, rule, e)


func _check_sorting(v: ProgramValidation, limits: SkillProgramLimits) -> void:
	var seen := {}
	var adds := _adds_to_registry()
	var fix := "add registry.add(target) to a rule: always: adds every enemy the skill hits"
	var allowed := limits == null or limits.allows("sort")
	for rule in flat_rules():
		seen[rule.owner] = seen.get(rule.owner, 0) + 1
		var label := "%s rule %d" % [_owner_name(rule.owner), seen[rule.owner]]
		for e in rule.effects:
			if not ScriptRegisters.is_sort(e):
				continue
			if not allowed:
				v.add_issue("%s: sorting isn't available at your rank." % label,
					"sorting isn't available at your rank (Defragmenter)", "delete this line", rule, e)
			elif not adds:
				v.add_issue("%s: nothing ever adds to the registry, so there's nothing to sort." % label,
					"nothing ever adds to the registry, so there's nothing to sort", fix, rule, e)
		for c in rule.own_conditions():
			if c is RegistryPositionCondition and not adds:
				v.add_issue("%s: nothing ever adds to the registry, so it has no %s." % [label, (c as RegistryPositionCondition).pos_text()],
					"nothing ever adds to the registry, so it has no %s" % (c as RegistryPositionCondition).pos_text(), fix, rule, c)


func tag_registry_reads() -> void:
	var spends := false
	for rule in flat_rules():
		for e in rule.effects:
			spends = spends or (ScriptRegisters.is_write(e.knob) and e.write_op() == "remove")
	for rule in flat_rules():
		for c in rule.conditions:
			if c is VarCondition:
				(c as VarCondition).spent = spends


func _adds_to_registry() -> bool:
	for rule in flat_rules():
		for e in rule.effects:
			if ScriptRegisters.raises(e):
				return true
	return false


func keeps_a_count() -> bool:
	if not _adds_to_registry():
		return false
	for rule in flat_rules():
		for c in rule.conditions:
			if c is VarCondition or c is RegistryPositionCondition:
				return true
		for e in rule.effects:
			if e.ramps():
				return true
	return false


func line_cost(knob: SkillKnob, e: ProgramEffect) -> float:
	if e.ramps() and e.steps > 0:
		var room := maxi(growth_room(e, knob), 0)
		var full := room * _avg_buff_step_cost(knob)
		return full * ScriptRegisters.ramp_cost_share(e.steps, room)
	if not knob.is_buff(e.steps):
		return knob.cost_for(e.steps)
	return absi(e.steps) * _avg_buff_step_cost(knob)


func buff_steps_on(knob: SkillKnob) -> int:
	var n := 0
	for rule in flat_rules():
		for e in rule.effects:
			if e.knob != knob.id or ScriptRegisters.is_write(e.knob):
				continue
			if e.ramps():
				if e.steps > 0:
					n += maxi(growth_room(e, knob), 0)
			elif knob.is_buff(e.steps):
				n += absi(e.steps)
	return n


func _avg_buff_step_cost(knob: SkillKnob) -> float:
	var n := buff_steps_on(knob)
	return knob.buff_cost(n) / n if n > 0 else knob.first_step_cost()


func growth_room(e: ProgramEffect, knob: SkillKnob) -> int:
	var room := knob.max_steps
	for rule in flat_rules():
		for other in rule.effects:
			if other.knob == e.knob and not other.ramps() and other.steps > 0:
				room -= other.steps
	return room


func _grows_twice(e: ProgramEffect) -> bool:
	for rule in flat_rules():
		for other in rule.effects:
			if other == e:
				return false
			if other.knob == e.knob and other.ramps():
				return true
	return false


static func _knob_issue(v: ProgramValidation, full: String, text: String, hint: String, lines: Array) -> void:
	if lines.is_empty():
		v.add_error(full, hint)
		return
	for i in lines.size():
		v.add_issue(full, text, hint, lines[i][0], lines[i][1], i == 0)


static func _second_use(rule: ProgramRule) -> ProgramCondition:
	var seen := {}
	for c in rule.conditions:
		if seen.has(c.type_id()) and not c.allows_repeats():
			return c
		seen[c.type_id()] = true
	return null


static func _check_loop(rule: ProgramRule, limits: SkillProgramLimits) -> Array:
	var cap := limits.loop_cap()
	if rule.scope() == ProgramCondition.Scope.HIT:
		return ["loops can't use per-hit conditions.",
			"move the loop to a rule without per-hit conditions"]
	if rule.loop.count < 1 or rule.loop.count > cap:
		return ["a loop can repeat 1 to %d times." % cap,
			"click the repeat number (right-click lowers it) to bring it to %d or less" % cap]
	if rule.loop.mode == ProgramLoop.Mode.WHILE:
		if rule.conditions.is_empty():
			return ["a while loop needs a condition (use a for loop in always:).",
				"drag a condition onto this rule, or use a for loop instead"]
		for c in rule.conditions:
			if c.changes_during_cast():
				return []
		return ["%s never changes mid-cast, so the while would always run to its max: use a for loop." \
				% rule.conditions[0].palette_name().to_lower(),
			"use a for loop, or add a condition that can change mid-cast"]
	return []


func rules_of(owner_id: String) -> Array[ProgramRule]:
	var out: Array[ProgramRule] = []
	for r in rules:
		if r.owner == owner_id:
			out.append(r)
	return out


func prune_owners(socketed_support_ids: Array) -> bool:
	var before := rules.size()
	rules.assign(rules.filter(func(r: ProgramRule) -> bool:
		return r.owner == "" or r.owner in socketed_support_ids))
	return rules.size() != before


func duplicate_program() -> SkillProgram:
	return SkillProgram.from_dict(to_dict())


static func _owner_name(owner_id: String) -> String:
	if owner_id == "":
		return "Skill"
	var def: SupportGemDefinition = SkillGemRegistry.get_support_gem(owner_id)
	return def.display_name.trim_suffix(" Support") if def != null else owner_id


func describe(limits: SkillProgramLimits) -> PackedStringArray:
	var lines := PackedStringArray()
	for r in rules:
		lines.append(r.describe(limits))
	return lines


func to_dict() -> Dictionary:
	var out: Array = []
	for r in rules:
		out.append(r.to_dict())
	return {"rules": out}


static func from_dict(d: Dictionary) -> SkillProgram:
	var p := SkillProgram.new()
	for rd in d.get("rules", []):
		p.rules.append(ProgramRule.from_dict(rd))
	return p

func get_knob(knob_id: StringName) -> SkillKnob:

func owner_of(knob_id: StringName) -> String:
	return knob_owner.get(knob_id, "")

	var out: Array[SkillKnob] = []
	for k in knobs:
		if owner_of(k.id) == owner_id:
			out.append(k)
	return out


static func for_socket(gem: SkillGemDefinition, supports: Array, trait_res: ClassTrait = null, rank := 0,
		bonus_points := 0.0, bonus_steps := 0) -> SkillProgramLimits:
	var base := gem.program_limits if gem != null else null
	if base == null:
		return null
	var out := SkillProgramLimits.new()
	out.knobs = base.knobs.duplicate()
	out.point_budget = base.point_budget + bonus_points
	out.bonus_points = bonus_points
	out.max_actions = base.max_actions
	out.max_rules = base.max_rules + int(Ranks.perk(rank, "rules"))
	out.max_actions += int(Ranks.perk(rank, "blocks"))
	out.tiers = raised_tiers(base.tiers, int(Ranks.perk(rank, "tier_blocks")))
	out.rank = rank
	var defense := load(DEFENSE_KNOBS) as SkillProgramLimits
	if defense != null:
		for k in defense.knobs:
			if out.get_knob(k.id) == null:
				out.knobs.append(k)
	out.knobs.append_array(ScriptRegisters.knobs())
	out.knobs.assign(out.knobs.filter(func(k: SkillKnob) -> bool: return out.allows("knob:" + String(k.id))))
	for sup in supports:
		var def := sup as SupportGemDefinition
		if def == null or not out.allows("gem:" + def.id):
			continue
		out.support_ids.append(def.id)
		for k in def.program_knobs:
			out.knobs.append(k)
			out.knob_owner[k.id] = def.id
	if trait_res != null:
		out.class_trait = trait_res
		out.point_budget += trait_res.extra_points
		for i in out.knobs.size():
			out.knobs[i] = trait_res.adjust_knob(out.knobs[i])
	if bonus_steps > 0:
		out.bonus_steps = bonus_steps
		for i in out.knobs.size():
			out.knobs[i] = widen_knob(out.knobs[i], bonus_steps)
	return out


static func raised_tiers(tiers: Array[EfficiencyTier], extra: int) -> Array[EfficiencyTier]:
	var out: Array[EfficiencyTier] = []
	for t in tiers:
		if extra <= 0 or t == null:
			out.append(t)
			continue
		var raised := t.duplicate() as EfficiencyTier
		raised.max_actions = t.max_actions + extra
		raised.description = t.description.replace("%d blocks" % t.max_actions, "%d blocks" % raised.max_actions)
		out.append(raised)
	return out


static func widen_knob(k: SkillKnob, extra: int) -> SkillKnob:
	if extra <= 0:
		return k
	if k.cost_per_step > 0.0 and k.max_steps > 0:
		var c := k.duplicate() as SkillKnob
		c.max_steps += extra
		return c
	if k.cost_per_step < 0.0 and k.min_steps < 0:
		var c := k.duplicate() as SkillKnob
		c.min_steps -= extra
		return c
	return k


func allows(block: String) -> bool:
	return Ranks.allows(rank, block)


func rule_cost_mult(rule: ProgramRule) -> float:
	return class_trait.rule_cost_mult(rule) if class_trait else 1.0


func rule_effect_mult(rule: ProgramRule, include_hit := true) -> float:
	return (class_trait.rule_effect_mult(rule, include_hit) if class_trait else 1.0) * VariableStoreSupport.rule_effect_mult(rule)


const HIT_DEFENSE_KNOBS: Array[StringName] = [&"leech", &"barrier", &"guard"]


static func hit_defense_knob(knob: SkillKnob) -> bool:
	return knob != null and knob.kind == SkillKnob.Kind.FLAG and HIT_DEFENSE_KNOBS.has(knob.id)


const HIT_SUPPORT_FLAGS: Array[String] = ["explode_radius", "explode_damage", "breakpoint_bonus",
		"breakpoint_hits", "breakpoint_burst", "tremor_max_chain", "tremor_radius", "var_store_gain",
		"except_stack_damage", "fork_count", "fork_angle"]


static func hit_support_knob(knob: SkillKnob) -> bool:
	return knob != null and knob.kind == SkillKnob.Kind.FLAG and HIT_SUPPORT_FLAGS.has(knob.flag_key)


static func per_hit_knob(knob: SkillKnob) -> bool:
	return hit_defense_knob(knob) or hit_support_knob(knob)


func loop_cap() -> int:
	return ProgramLoop.BASE_CAP + (ForLoopSupport.CAP_BONUS if _has_for_loop() else 0)


func loop_cost_mult() -> float:
	return ForLoopSupport.COST_MULT if _has_for_loop() else 1.0


func repeat_mana_mult() -> float:
	return ForLoopSupport.MANA_MULT if _has_for_loop() else 1.0


func _has_for_loop() -> bool:
	return support_ids.has(ForLoopSupport.ID)


func buff_cost_mult() -> float:
	return class_trait.buff_cost_mult if class_trait else 1.0


func per_hit_any_knob() -> bool:
	return class_trait != null and class_trait.per_hit_any_knob


func per_hit_attack_speed(knob: SkillKnob) -> bool:
	return per_hit_any_knob() and knob != null and knob.kind == SkillKnob.Kind.STAT \
			and knob.stat == StatTypes.Stat.ATTACK_SPEED


func tier_for(actions: int) -> EfficiencyTier:
	for t in tiers:
		if actions <= t.max_actions:
			return t
	return null

	if not _drafts.has(_socket):
		var stored := _skills().stored_program(_socket)
		_drafts[_socket] = stored.duplicate_program() if stored != null else SkillProgram.new()
	return _drafts[_socket]

	var ids: Array = []

	return ids


func _is_dirty(socket: int) -> bool:
	if not _drafts.has(socket):
		return false
	var applied := _skills().stored_program(socket)
	var applied_json := JSON.stringify(applied.to_dict() if applied != null else {"rules": []})
	return JSON.stringify((_drafts[socket] as SkillProgram).to_dict()) != applied_json


func _drop_stale_routines() -> void:
	var sk := _skills()
	if not _routine_draft.is_empty() and (sk == null or sk.routines_version != _routine_version):
		_routine_draft.clear()
		_routine_rename = -1


func _routines() -> Array:
	_drop_stale_routines()
	if _routine_draft.is_empty():
		var sk := _skills()
		_routine_version = sk.routines_version if sk != null else -1
		for i in ProgramRoutine.MAX_ROUTINES:
			var have: ProgramRoutine = sk.routines[i] if sk != null and i < sk.routines.size() else null
			_routine_draft.append(have.duplicate_routine() if have != null else ProgramRoutine.make(i))
	return _routine_draft


func _routines_dirty() -> bool:
	_drop_stale_routines()
	if _routine_draft.is_empty():
		return false
	var sk := _skills()
	for i in _routine_draft.size():
		var have: ProgramRoutine = sk.routines[i] if i < sk.routines.size() else ProgramRoutine.make(i)
		if JSON.stringify((_routine_draft[i] as ProgramRoutine).to_dict()) != JSON.stringify(have.to_dict()):
			return true
	return false

func _kernel_lines() -> void:
	var qm := _kernel_qm()
	var lines: Array = Array(_skills().kernel)
	var running := KernelScript.null_running(qm)
	var editable := Endings.kernel_editable(qm)
	if editable:
		_comment_line("# kernel.gd: yours. You patched AXIOM: up to %d root lines, any you like, and the Null's never run." % Endings.MAX_ROOT_LINES)
		_comment_line("# these lines run on the zone, whatever you cast, in every dungeon")
	else:
		_comment_line("# kernel.gd: read only. The Null asks on every floor down; your answers are written here.")
		_comment_line("# these lines run on the zone, whatever you cast: %s" % (
			"in every dungeon (merged)" if Endings.runs_everywhere(qm) else "every Act V zone past the Core"))
	var any := false
	for trigger in KernelScript.TRIGGERS:
		var mine: Array = lines.filter(func(id: String) -> bool: return KernelScript.LINES.has(id) and KernelScript.LINES[id]["trigger"] == trigger)
		var theirs: Array = running.filter(func(id: String) -> bool: return KernelScript.NULL_LINES[id]["trigger"] == trigger)
		if mine.is_empty() and theirs.is_empty():
			continue
		any = true
		_blank_line(null)
		var head := _code_line(null)
		(head[1] as HBoxContainer).add_child(_tok("when ", COL_KEYWORD))
		(head[1] as HBoxContainer).add_child(_tok(trigger, COL_FN))
		(head[1] as HBoxContainer).add_child(_tok(":", COL_OP))
		for id in mine:
			var parts := _code_line(null)
			var code: HBoxContainer = parts[1]
			code.add_child(_tok("    " + str(KernelScript.LINES[id]["code"]), COL_KERNEL))
			code.add_child(_comment_tok("root: " + _kernel_source(qm, id), str(KernelScript.LINES[id]["tip"])))
			if editable:
				var del := _code_button("del", "Take this line out of kernel.gd.")
				del.pressed.connect(func() -> void:
					_skills().kernel.erase(id)
					Coop.kernel_changed()
					panel._notify("KERNEL: LINE REMOVED")
					_queue_rebuild())
				code.add_child(del)
		for id in theirs:
			var parts := _code_line(null, "N", KernelScript.COL_NULL)
			var code: HBoxContainer = parts[1]
			code.add_child(_tok("    " + str(KernelScript.NULL_LINES[id]["code"]), KernelScript.COL_NULL))
			code.add_child(_comment_tok("the Null's, merged on " + _kernel_floor(id), str(KernelScript.NULL_LINES[id]["tip"])))
	var answered := false
	for id in KernelScript.NULL_LINES:
		var a := KernelScript.answer_of(qm, id)
		if a == "" or a == "merge":
			continue
		if not answered:
			_blank_line(null)
			answered = true
		_comment_line("# %s on %s: %s" % ["overwritten" if a == "overwrite" else "terminated", _kernel_floor(id), KernelScript.code_of(id)])
	if not any and not answered:
		_blank_line(null)
		_comment_line("# empty: the Null's first pull request waits in the Stack Trace.")


func _kernel_source(qm: QuestManager, line_id: String) -> String:
	for id in KernelScript.NULL_LINES:
		var a := KernelScript.answer_of(qm, id)
		if a == "merge" and KernelScript.NULL_LINES[id]["gift"] == line_id:
			return "the Null's gift (merged on %s)" % _kernel_floor(id)
		if a == "overwrite" and line_id in KernelScript.NULL_LINES[id]["offers"]:
			return "yours, over the Null's on %s" % _kernel_floor(id)
	return "yours"


func _kernel_floor(null_id: String) -> String:
	var w := Waypoints.for_scene("res://scenes/map/dungeon/%s.tscn" % KernelScript.NULL_LINES[null_id]["zone"])
	return w.display_name if w != null else str(KernelScript.NULL_LINES[null_id]["zone"])

func _routine_of(rule: ProgramRule) -> ProgramRoutine:
	if rule == null or rule.owner != ROUTINES_FILE:
		return null
	for r in _routines():
		if (r as ProgramRoutine).body == rule:
			return r
	return null


func _routine_callers(index: int) -> Array:
	var out: Array = []
	var sk := _skills()
	var routines := _routines()
	var rt: ProgramRoutine = routines[index] if index < routines.size() else null
	for i in sk.sockets.size():
		var sock := sk.sockets[i]
		if sock.gem == null:
			continue
		var p: SkillProgram = _drafts[i] if _drafts.has(i) else sock.program
		if p == null or not (index in p.called_routines()):
			continue
		var text := _file_name(sock.gem.get_display_name())
		var limits := sk.program_limits_for(i)
		var top := 1.0
		if rt != null and not rt.is_empty() and limits != null:
			for r in p.rules:
				var calls := r.effects.filter(func(x: ProgramEffect) -> bool: return x.is_call() and x.op == index)
				if not calls.is_empty():
					top = maxf(top, float(_routine_here(rt, r, limits)["mult"]))
				for bi in r.branches.size():
					if r.branches[bi].effects.any(func(x: ProgramEffect) -> bool: return x.is_call() and x.op == index):
						top = maxf(top, float(_routine_here(rt, r.branch_rule(bi), limits)["mult"]))
		if not is_equal_approx(top, 1.0):
			text += " x%.2f" % top
		out.append(text)
	return out

func tutorial_target(id: String) -> Array[Control]:
	var out: Array[Control] = []
	match id:
		"script_palette", "palette":
			if _palette_box != null:
				out.append(_palette_box)
		"script_apply", "apply":
			if _apply_btn != null:
				out.append(_apply_btn)
		"output":
			if _output_frame != null:
				out.append(_output_frame)
		"guide":
			if guide_open():
				out.append(_guide)
		"guide_button":
			if _guide_btn != null:
				out.append(_guide_btn)
		"skill_file":
			if _skill_file_btn.has(_socket):
				out.append(_skill_file_btn[_socket])
		"kernel_file":
			if is_instance_valid(_kernel_file_btn) and _kernel_file_btn.is_visible_in_tree():
				out.append(_kernel_file_btn)
		"new_rule":
			if _new_rule_ctrl != null and is_instance_valid(_new_rule_ctrl):
				out.append(_new_rule_ctrl)
		"always_line":
			var always := _tutorial_always_rule()
			if always != null and _rule_line.has(always) and is_instance_valid(_rule_line[always]):
				out.append(_rule_line[always])
		_:
			var tab := _tab_of(id)
			if tab != "" and tab != _palette_tab:
				_palette_tab = tab
				_build_palette()
			if _snippets.has(id) and is_instance_valid(_snippets[id]):
				_reveal(_snippets[id])
				out.append(_snippets[id])
			elif id.begins_with("knob_number:"):
				var bits := id.split(":")
				var rule: ProgramRule = _tutorial_rule(bits[1]) if bits.size() == 3 else null
				if rule != null:
					for e in rule.effects:
						if String(e.knob) == bits[2] and _effect_num.has(e) and is_instance_valid(_effect_num[e]):
							out.append(_effect_num[e])
			elif id.begins_with("always_knob_number:"):
				var rule := _tutorial_always_rule()
				if rule != null:
					for e in rule.effects:
						if String(e.knob) == id.trim_prefix("always_knob_number:") and _effect_num.has(e) 								and is_instance_valid(_effect_num[e]):
							out.append(_effect_num[e])
			elif id.begins_with("cond_numbers:"):
				var rule := _tutorial_rule(id.trim_prefix("cond_numbers:"))
				if rule != null:
					for c in rule.conditions:
						if c.type_id() == id.trim_prefix("cond_numbers:") and _cond_nums.has(c):
							for n in _cond_nums[c]:
								if is_instance_valid(n):
									out.append(n)
			elif id.begins_with("rule_line:"):
				var rule := _tutorial_rule(id.trim_prefix("rule_line:"))
				if rule != null and _rule_line.has(rule) and is_instance_valid(_rule_line[rule]):
					out.append(_rule_line[rule])
	return out


func tutorial_state(condition_id: String, knob_id: StringName = &"damage",
		trade_knob: StringName = &"attack_speed") -> Dictionary:
	var rule := _tutorial_rule(condition_id)
	var always := _tutorial_always_rule()
	var limits := _limits()
	if limits != null and limits.get_knob(trade_knob) == null:
		trade_knob = knob_id
	var trade_steps := 0
	var trade_line := false
	if always != null:
		for e in always.effects:
			if e.knob == trade_knob:
				trade_steps += e.steps
				trade_line = true
	var knob_steps := 0
	var params: Array = []
	if rule != null:
		for e in rule.effects:
			if e.knob == knob_id:
				knob_steps += e.steps
		for c in rule.conditions:
			if c.type_id() == condition_id:
				for p in c.param_names():
					params.append(c.get(p))
	var v := _validation
	return {
		"knob_steps": knob_steps,
		"params": params,
		"discount": rule.cost_multiplier() if rule != null else 1.0,
		"points": v.buff_points if v != null else 0.0,
		"budget": (v.point_budget + v.drawback_points) if v != null else 0.0,
		"over": v != null and v.buff_points > v.point_budget + v.drawback_points + 0.001,
		"has_socket": _sock() != null and _limits() != null,
		"owner": _owner,
		"rule": rule != null,
		"selected": rule != null and rule == _selected,
		"effects": rule.effects.size() if rule != null else 0,
		"valid": _validation != null and _validation.valid,
		"dirty": _sock() != null and _is_dirty(_socket),
		"any_selected": _selected != null,
		"always_rule": always != null,
		"always_selected": always != null and _selected == always,
		"trade_knob": String(trade_knob),
		"trade_steps": trade_steps,
		"trade_line": trade_line,
		"drawback_points": v.drawback_points if v != null else 0.0,
	}


func _tutorial_always_rule() -> ProgramRule:
	if _sock() == null or _limits() == null:
		return null
	for rule in _draft().rules_of(""):
		if rule.is_always() and rule.loop == null:
			return rule
	return null


func _tutorial_rule(condition_id: String) -> ProgramRule:
	if _sock() == null or _limits() == null:
		return null
	for rule in _draft().rules_of(""):
		for c in rule.conditions:
			if c.type_id() == condition_id:
				return rule
	return null


func refresh() -> void:
	var sk := _skills()
	if sk == null:
		return
	if _set_hooked != sk:
		if _set_hooked != null and is_instance_valid(_set_hooked) 				and _set_hooked.script_set_changed.is_connected(_on_set_changed):
			_set_hooked.script_set_changed.disconnect(_on_set_changed)
		_set_hooked = sk
		sk.script_set_changed.connect(_on_set_changed)
	for i in _drafts.keys():
		if i >= sk.sockets.size() or sk.sockets[i].gem == null:
			_drafts.erase(i)
		else:
			(_drafts[i] as SkillProgram).prune_owners(_support_ids(sk.sockets[i]))
	if _sock() == null or _sock().gem == null:
		_socket = -1
		for i in sk.sockets.size():
			if sk.sockets[i].gem != null:
				_socket = i
				break
		_owner = ""
	if _sock() != null and _owner != "" and not (_owner in [ROUTINES_FILE, KERNEL_FILE]) and not (_owner in _support_ids(_sock())):
		_owner = ""
	_rebuild()


func _select_gem(socket: int, owner_id: String) -> void:
	var same_file := socket == _socket and owner_id == _owner
	_socket = socket
	_owner = owner_id
	_selected = null
	_rebuild()
	if not same_file:
		_editor_scroll.scroll_vertical = 0


func _file_owners() -> Array:
	var out: Array = [""]
	var s := _sock()
	if s == null:
		return out
	for item in s.supports:
		var def := item.definition.support_gem
		if not def.program_knobs.is_empty() and not (def.id in out):
			out.append(def.id)
	if _routines_unlocked():
		out.append(ROUTINES_FILE)
	if _kernel_unlocked():
		out.append(KERNEL_FILE)
	return out


func _queue_rebuild() -> void:
	if _rebuild_queued:
		return
	_rebuild_queued = true
	call_deferred("_rebuild")


func _rebuild() -> void:
	_rebuild_queued = false
	_validation = null
	_issue_line.clear()
	if _sock() != null and _limits() != null:
		_validation = _skills().preview_program(_socket, _draft(), _routines())
	_build_gem_list()
	_build_file_tabs()
	_build_rules()
	_build_palette()
	_update_status()
	_update_set_buttons()
	_update_library()

func _build_rules() -> void:
	_clear(_rules_box)
	_line_no = 0
	_rule_line.clear()
	_effect_num.clear()
	_cond_nums.clear()
	_new_rule_ctrl = null
	var s := _sock()
	if s == null:
		_title.text = "no file open"
		_comment_line("# socket a skill first")
		return
	var limits := _limits()
	var skill_file := _file_name(s.gem.get_display_name())
	if limits == null:
		_title.text = skill_file
		_comment_line("# this skill can't be scripted")
		return
	var owners := _file_owners()
	if not (_owner in owners):
		_owner = ""
	if _owner == "":
		_title.text = skill_file
	elif _owner == ROUTINES_FILE:
		_title.text = "routines.gd  (every skill)"
	elif _owner == KERNEL_FILE:
		_title.text = "kernel.gd  (yours: patched)" if Endings.kernel_editable(_kernel_qm()) \
			else "kernel.gd  (read only: the Null's pull requests)"
		_selected = null
		_line_no = 0
		_kernel_lines()
		return
	else:
		var sup: SupportGemDefinition = SkillGemRegistry.get_support_gem(_owner)
		_title.text = "%s / %s" % [skill_file.trim_suffix(".gd"), _file_name(sup.display_name) if sup else _owner]

	var shown: Array = []
	if _owner == ROUTINES_FILE:
		for rt in _routines():
			shown.append((rt as ProgramRoutine).body)
	else:
		shown.append_array(_draft().rules_of(_owner))
	if _selected != null and not (_selected in shown):
		_selected = null
	_line_no = 0
	if _owner == ROUTINES_FILE:
		_routine_lines(limits)
		return
	if _owner == "":
		_comment_line("# points are shared with every file of this skill (its other tabs)")
	elif owners.size() > 1:
		_comment_line("# points are shared with %s and its other tabs" % skill_file)
	for rule in _draft().rules_of(_owner):
		_blank_line(rule)
		_rule_lines(rule, limits)
	_blank_line(null)
	_new_rule_line(_owner)


func _build_file_tabs() -> void:
	_clear(_file_tabs)
	if _sock() == null or _limits() == null:
		_file_tabs.visible = false
		return

	var bad := _error_owners(_socket)
	for owner_id in _file_owners():
		var text := _owner_file(owner_id)

		if owner_id == "" and _is_dirty(_socket):

			for rt in _routines():
				has_error = has_error or (not (rt as ProgramRoutine).is_empty() and not (rt as ProgramRoutine).problem().is_empty())
		if has_error:
			text += " !"
		_file_tabs.add_child(_file_tab(owner_id, text, has_error))

func _routine_lines(limits: SkillProgramLimits) -> void:
	_comment_line("# shared by every skill: call name() is 1 block, and each skill pays for the lines")
	var routines := _routines()
	for i in routines.size():
		var rt: ProgramRoutine = routines[i]
		_blank_line(rt.body)
		_routine_header(i, rt)
		if rt.is_empty():
			var parts := _code_line(rt.body)
			var code: HBoxContainer = parts[1]
			code.add_child(_tok("    pass", COL_KEYWORD))
			code.add_child(_comment_tok("drop knobs, registry writes or an if here"))
			_make_line_interactive(parts[0], rt.body, {})
		for e in rt.body.effects:
			_effect_line(rt.body, e, limits, "    ")
		for bi in rt.body.branches.size():
			_branch_lines(rt.body, bi, limits)
	_blank_line(null)


func _routine_header(i: int, rt: ProgramRoutine) -> void:
	var parts := _code_line(rt.body)
	var line: Control = parts[0]
	var code: HBoxContainer = parts[1]
	code.add_child(_tok("def ", COL_KEYWORD))
	if _routine_rename == i:
		var edit := _name_edit(ProgramRoutine.NAME_MAX, "name")
		edit.visible = true
		edit.text = rt.routine_name
		edit.text_submitted.connect(func(_t: String) -> void: _finish_routine_rename(edit, i))
		edit.focus_exited.connect(func() -> void: _finish_routine_rename(edit, i))
		code.add_child(edit)
		edit.grab_focus.call_deferred()
		edit.select_all.call_deferred()
	else:
		var name_btn := _code_button(rt.routine_name, "Click to rename this routine. Calls keep working: they point at the routine, not its name.")
		name_btn.add_theme_color_override("font_color", COL_FN)
		name_btn.pressed.connect(_start_routine_rename.bind(i))
		code.add_child(name_btn)
	code.add_child(_tok("():", COL_OP))
	var problem: Array = rt.problem() if not rt.is_empty() else []
	var note := "%d/%d blocks" % [rt.block_count(), ProgramRoutine.MAX_BLOCKS]
	var echoes := NullEcho.callers_of(get_tree(), i) if is_inside_tree() else 0
	var echo_text := "%d Null Echo%s" % [echoes, "" if echoes == 1 else "es"]
	if not problem.is_empty():
		note = problem[1]
		if echoes > 0:
			note += " (%s call%s it: apply, and they crash)" % [echo_text, "s" if echoes == 1 else ""]
	else:
		var callers := _routine_callers(i)
		if echoes > 0:
			callers.append(echo_text)
		if not callers.is_empty():
			note += ", called by " + ", ".join(callers)
	var tok := _comment_tok(note)
	if not problem.is_empty():
		tok.add_theme_color_override("font_color", SkillPanel.COL_WARN)
	code.add_child(tok)
	var clear_btn := _code_button("clear", "Empty this routine (its name stays). A script that calls it won't compile until it has lines again.")
	clear_btn.mouse_filter = Control.MOUSE_FILTER_PASS
	clear_btn.pressed.connect(func() -> void:
		rt.body.effects.clear()
		rt.body.branches.clear()
		_changed())
	code.add_child(clear_btn)
	line.tooltip_text = "A routine: lines written once, run from any skill with  call %s()  (1 block in that script).\nIt has no conditions of its own: the rule that calls it decides when. Each skill that calls it pays the points for its lines, on its own knobs.\nThe numbers here are the base. A rule that makes its knobs stronger (a Gladiator's every-Nth rule...) makes these stronger too: \"called by strike.gd x1.75\", and the call line shows each value.\nAt most %d blocks, and one level of if." % [
		rt.routine_name, ProgramRoutine.MAX_BLOCKS]
	_make_line_interactive(line, rt.body, {})
	_rule_line[rt.body] = line


func _start_routine_rename(i: int) -> void:
	_routine_rename = i
	_lock_player(true)
	_queue_rebuild()


func _finish_routine_rename(edit: LineEdit, i: int) -> void:
	if _routine_rename != i:
		return
	_routine_rename = -1
	_lock_player(false)
	var routines := _routines()
	var wanted := ProgramRoutine.clean_name(edit.text, i)
	for j in routines.size():
		if j != i and (routines[j] as ProgramRoutine).routine_name == wanted:
			panel._notify("ANOTHER ROUTINE IS CALLED %s" % wanted.to_upper())
			_queue_rebuild()
			return
	(routines[i] as ProgramRoutine).routine_name = wanted
	_queue_rebuild()


func _call_line(code: HBoxContainer, parts: Array, rule: ProgramRule, e: ProgramEffect, branch: ProgramBranch) -> void:
	var routines := _routines()
	var rt: ProgramRoutine = routines[e.op] if e.op >= 0 and e.op < routines.size() else null
	code.add_child(_tok("call ", COL_KEYWORD))
	var name_tok := _num_tok(rt.routine_name if rt != null else "routine_%d" % (e.op + 1),
		"which routine (click / right-click: the next one)", func(d: int) -> void:
			e.op = posmod(e.op + d, routines.size())
			_changed())
	code.add_child(name_tok)
	_effect_num[e] = name_tok
	code.add_child(_tok("()", COL_OP))
	var calling := rule.branch_rule(rule.branches.find(branch)) if branch != null else rule
	var here := _routine_here(rt, calling, _limits()) if rt != null and not rt.is_empty() else {}
	var note := "1 block" + ("" if here.is_empty() else ", runs %d" % rt.block_count())
	var boosted: bool = not here.is_empty() and not is_equal_approx(float(here["mult"]), 1.0)
	if boosted:
		note += ", knobs %sx%.2f here (%s)" % ["up to " if here["mixed"] else "", here["mult"], here["why"]]
	var note_tok := _comment_tok(note)
	if boosted:
		note_tok.add_theme_color_override("font_color", Color(COL_BUFF, 0.85))
	code.add_child(note_tok)
	var body := "empty" if here.is_empty() else "\n    " + "\n    ".join(here["lines"])
	parts[0].tooltip_text = "Runs %s() here, under this rule's conditions:%s%s\n1 block in this script, whatever the routine holds; this skill pays the points for its lines.\nEdit it in routines.gd. Click the name to switch routine. Drag out or right-click to delete." % [
		rt.routine_name if rt != null else "a routine", body,
		("\nThis rule makes its knobs stronger (%s): the numbers in routines.gd are the base, these are what this call does." % here["why"]) if boosted else ""]
	_make_line_interactive(parts[0], rule, {"kind": "effect", "effect": e}, branch)
	_mark(parts, rule, e)


func _routine_here(rt: ProgramRoutine, calling: ProgramRule, limits: SkillProgramLimits) -> Dictionary:
	var lines := PackedStringArray()
	var top := 1.0
	var why := ""
	var mults := {}
	var groups: Array = [[rt.body.effects, calling, ""]]
	for b in rt.body.branches:
		var as_if := ProgramRule.new()
		as_if.owner = calling.owner
		as_if.conditions.assign(calling.conditions + b.conditions)
		groups.append([b.effects, as_if, "if: "])
	for g in groups:
		var r: ProgramRule = g[1]
		for e: ProgramEffect in g[0]:
			if e.is_call():
				continue
			if ScriptRegisters.is_write(e.knob):
				lines.append("%s%s" % [g[2], e.describe_write()])
				continue
			var knob := limits.get_knob(e.knob)
			if knob == null:
				lines.append("%s%s  (not a knob of this skill)" % [g[2], e.knob])
				continue
			var mult := limits.rule_effect_mult(r, not SkillProgramLimits.per_hit_knob(knob))
			mults[snappedf(mult, 0.01)] = true
			var base := _fmt_per_count(knob, e.steps) if e.ramps() else _fmt_knob(knob, e.steps)
			var now := _fmt_per_count(knob, e.steps, mult) if e.ramps() else _fmt_knob(knob, e.steps, mult)
			var text := "%s%s%s%s%s" % [g[2], e.knob, " -= " if e.steps < 0 else " += ",
				(e.ramp_text() + " * ") if e.ramps() else "", base]
			if not is_equal_approx(mult, 1.0):
				text += "  ->  %s here (x%.2f)" % [now, mult]
				if mult > top:
					top = mult
					why = _boost_source(r, limits)
			lines.append(text)
	return {"lines": lines, "mult": top, "why": why, "mixed": mults.size() > 1}


func _boost_source(rule: ProgramRule, limits: SkillProgramLimits) -> String:
	var parts := PackedStringArray()
	var t := limits.class_trait
	if t != null:
		var n := 0
		for c in rule.conditions:
			if c is EveryNthCastCondition:
				n = maxi(n, (c as EveryNthCastCondition).n)
		if n > 0 and not is_equal_approx(t.nth_effect_mult(rule), 1.0):
			parts.append("%s, every %s cast" % [t.display_name.to_lower(), _ordinal(n)])
		if rule.loop == null and rule.scope() == ProgramCondition.Scope.HIT and not is_equal_approx(t.hit_rule_effect_mult, 1.0):
			parts.append("%s, per-hit rule" % t.display_name.to_lower())
	if VariableStoreSupport.rule_effect_mult(rule) > 1.0:
		parts.append("spent charge")
	return ", ".join(parts) if not parts.is_empty() else "this rule"


static func _ordinal(n: int) -> String:
	if n % 100 in [11, 12, 13]:
		return "%dth" % n
	match n % 10:
		1:
			return "%dst" % n
		2:
			return "%dnd" % n
		3:
			return "%drd" % n
	return "%dth" % n


func _spacer(height: float) -> Control:
	var gap := Control.new()
	gap.custom_minimum_size.y = height * _k()
	gap.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return gap


func _rule_lines(rule: ProgramRule, limits: SkillProgramLimits) -> void:
	var discount := ""
	if not rule.is_always() and rule.cost_multiplier() < 1.0:
		discount = "buffs x%.2f" % rule.cost_multiplier()
		if rule.spends_marks():
			discount += " (spends its marks%s)" % (", max discount" if rule.is_discount_capped() else "")
		elif rule.is_discount_capped():
			discount += " (max discount)"
	var looping := rule.loop != null
	var is_while := looping and rule.loop.mode == ProgramLoop.Mode.WHILE

	if rule.is_always():
		var parts := _code_line(rule)
		var code: HBoxContainer = parts[1]
		code.add_child(_tok("always", COL_KEYWORD))
		code.add_child(_tok(":", COL_OP))
		code.add_child(_comment_tok(""))
		code.add_child(_delete_rule_button(rule))
		_make_line_interactive(parts[0], rule, {})
		_mark(parts, rule, null)
		_rule_line[rule] = parts[0]
	else:
		for i in rule.conditions.size():
			var c := rule.conditions[i]
			var last := i == rule.conditions.size() - 1
			var parts := _code_line(rule)
			var code: HBoxContainer = parts[1]
			var head := ("while " if is_while else "when ") if i == 0 else " and "
			if c.keyword() != "" and i == 0:
				head = c.keyword() + " "
			code.add_child(_tok(head, COL_KEYWORD))
			_condition_tokens(code, c, last)
			code.add_child(_comment_tok(_cond_note(c, discount if i == 0 else "")))
			if i == 0:
				code.add_child(_delete_rule_button(rule))
			parts[0].tooltip_text = "%s\nCOST: buffs in this rule x%.2f\nDrag out or right-click to delete." % [c.describe(), c.cost_multiplier()]
			_make_line_interactive(parts[0], rule, {"kind": "condition", "condition": c})
			_mark(parts, rule, c)
			if i == 0:
				_mark(parts, rule, null)
				_rule_line[rule] = parts[0]

	var indent := "    "
	if looping and not is_while:
		_loop_line(rule, limits)
		indent = "        "
	if rule.effects.is_empty() and not looping and rule.branches.is_empty():
		var parts := _code_line(rule)
		var code: HBoxContainer = parts[1]
		code.add_child(_tok("    pass", COL_KEYWORD))
		code.add_child(_comment_tok("drop a knob here"))
		_make_line_interactive(parts[0], rule, {})
	for e in rule.effects:
		_effect_line(rule, e, limits, indent)
	if not looping:
		for bi in rule.branches.size():
			_branch_lines(rule, bi, limits)
	else:
		if is_while:
			_loop_line(rule, limits)
		else:
			_cast_line(rule, indent)
		for bi in rule.branches.size():
			_branch_lines(rule, bi, limits, indent)


func _cond_note(c: ProgramCondition, discount: String) -> String:
	var at := c as RegistryPositionCondition
	if at == null:
		return discount
	var says := ScriptRegisters.position_meaning(at.pos, _draft_sort())
	return says if discount == "" else "%s, %s" % [says, discount]


func _draft_sort() -> int:
	for r in _draft().rules:
		for e in r.effects:
			if ScriptRegisters.is_sort(e):
				return e.sort_by
		for b in r.branches:
			for e in b.effects:
				if ScriptRegisters.is_sort(e):
					return e.sort_by
	return 0


func _condition_tokens(code: HBoxContainer, c: ProgramCondition, last: bool) -> void:
	var call_form := c.keyword() == "" and not c.modular()
	if not c.modular():
		code.add_child(_tok(_fn_name(c), COL_GEM if c.required_support() != "" else COL_FN))
	if call_form:
		code.add_child(_tok("(", COL_OP))
	var params := c.param_names()
	for j in params.size():
		var p := params[j]
		if j > 0:
			code.add_child(_tok(" " if c.modular() else ", ", COL_OP))
		var param_num := _num_tok(_fmt_param(c, p), c.param_label(p).to_lower(), func(d: int) -> void:
			c.nudge_param(p, d)
			_changed())
		code.add_child(param_num)
		if not _cond_nums.has(c):
			_cond_nums[c] = []
		_cond_nums[c].append(param_num)
	if call_form:
		code.add_child(_tok("):" if last else ")", COL_OP))
	elif last:
		code.add_child(_tok(":", COL_OP))


func _branch_lines(rule: ProgramRule, bi: int, limits: SkillProgramLimits, indent := "    ") -> void:
	var b := rule.branches[bi]
	var inner := indent + "    "
	var as_rule := rule.branch_rule(bi)
	var discount := ""
	if not b.conditions.is_empty() and as_rule.cost_multiplier() < 1.0:
		discount = "buffs x%.2f" % as_rule.cost_multiplier()
		if as_rule.spends_marks():
			discount += " (spends its marks%s)" % (", max discount" if as_rule.is_discount_capped() else "")
		elif as_rule.is_discount_capped():
			discount += " (max discount)"
	var per_hit := as_rule.scope() == ProgramCondition.Scope.HIT
	var head_tip := "if block\nIts lines apply when the rule fires AND its own conditions are true.%s%s\nTIP: fewer blocks than a second rule\nDrop a condition on this line, knobs below it.\nDrag out or right-click to delete the if." % [
		"\nLOOP: checked after each cast or repeat lands · its lines change the next repeat · break stops the loop" if rule.loop != null else "",
		"\nHIT: checked per hit · only damage, leech, barrier, guard and support effects" if per_hit else ""]
	if b.conditions.is_empty():
		var parts := _code_line(rule)
		var code: HBoxContainer = parts[1]
		code.add_child(_tok(indent + "if ", COL_KEYWORD))
		code.add_child(_tok("...:", COL_OP))
		code.add_child(_comment_tok("drop a condition here"))
		parts[0].tooltip_text = head_tip
		_make_line_interactive(parts[0], rule, {"kind": "branch"}, b)
		_mark(parts, rule, b)
	for i in b.conditions.size():
		var c := b.conditions[i]
		var parts := _code_line(rule)
		var code: HBoxContainer = parts[1]
		code.add_child(_tok(indent + ("if " if i == 0 else " and "), COL_KEYWORD))
		_condition_tokens(code, c, i == b.conditions.size() - 1)
		code.add_child(_comment_tok(_cond_note(c, discount if i == 0 else "")))
		if i == 0:
			parts[0].tooltip_text = head_tip
			_make_line_interactive(parts[0], rule, {"kind": "branch"}, b)
			_mark(parts, rule, b)
		else:
			parts[0].tooltip_text = "%s\nPart of the if block above.\nDrag out or right-click to delete." % c.describe()
			_make_line_interactive(parts[0], rule, {"kind": "condition", "condition": c}, b)
		_mark(parts, rule, c)
	if b.effects.is_empty() and not b.breaks:
		var parts := _code_line(rule)
		var code: HBoxContainer = parts[1]
		code.add_child(_tok(inner + "pass", COL_KEYWORD))
		code.add_child(_comment_tok("drop a knob here" if rule.loop == null else "drop a knob or break here"))
		_make_line_interactive(parts[0], rule, {}, b)
	for e in b.effects:
		_effect_line(rule, e, limits, inner, b)
	if b.breaks:
		_break_line(rule, b, inner)


func _effect_line(rule: ProgramRule, e: ProgramEffect, limits: SkillProgramLimits, indent := "    ",
		branch: ProgramBranch = null) -> void:
	var owner_rule := rule
	if branch != null:
		rule = owner_rule.branch_rule(owner_rule.branches.find(branch))
	var knob := limits.get_knob(e.knob)
	var cost := _draft().line_cost(knob, e) if knob else 0.0
	if cost > 0.0:
		cost *= limits.buff_cost_mult()
	var marker := "+" if cost > 0.0 else ("-" if cost < 0.0 else "")
	var parts := _code_line(owner_rule, marker, COL_BUFF if cost > 0.0 else COL_DRAW)
	var code: HBoxContainer = parts[1]
	code.add_child(_tok(indent, COL_OP))
	if e.is_call():
		_call_line(code, parts, owner_rule, e, branch)
		return
	if ScriptRegisters.is_sort(e):
		code.add_child(_tok("registry.sort(", COL_VAR))
		var key_num := _num_tok(ScriptRegisters.sort_text(e.sort_by), "order", func(d: int) -> void:
			e.sort_by = posmod(e.sort_by + d, ScriptRegisters.SORT_KEYS.size() * 2)
			_changed())
		code.add_child(key_num)
		_effect_num[e] = key_num
		code.add_child(_tok(")", COL_VAR))
		code.add_child(_comment_tok(ScriptRegisters.sort_meaning(e.sort_by)))
		parts[0].tooltip_text = "Puts the registry in order: %s. Then when target is registry[0]: is the first one in that order, registry[-1] the last.
The order holds from this rule's cast on; every cast starts back at oldest first, so a sort in a WHEN only holds while the WHEN does. Free: no points or blocks.
Click the key to change it (each key, then reversed). Drag out or right-click to delete." % ScriptRegisters.sort_meaning(e.sort_by)
		_make_line_interactive(parts[0], owner_rule, {"kind": "effect", "effect": e}, branch)
		_mark(parts, owner_rule, e)
		return
	if ScriptRegisters.is_write(e.knob):
		var op_num := _num_tok("registry." + e.write_op(), "operation", func(d: int) -> void:
			e.op = posmod(e.op + d, ScriptRegisters.MEMBER_WRITES)
			_changed())
		code.add_child(op_num)
		_effect_num[e] = op_num
		code.add_child(_tok("(target)", COL_OP))
		code.add_child(_comment_tok("repeats only" if rule.loop != null or rule.only_on_repeats() else ""))
		var when := "the enemy being hit" if rule.scope() == ProgramCondition.Scope.HIT \
				else "every enemy this cast hits, as each hit lands"
		var what := "Adds %s to the registry." if e.write_op() == "add" else "Takes %s out of the registry."
		parts[0].tooltip_text = (what % when) + " The registry is a set of enemies: each one is in it or not. It keeps them between casts; an enemy that dies leaves it. Free: no points or blocks.\nClick add / remove to swap. Drag out or right-click to delete."
		_make_line_interactive(parts[0], owner_rule, {"kind": "effect", "effect": e}, branch)
		_mark(parts, owner_rule, e)
		return
	if knob == null:
		code.add_child(_tok(String(e.knob), COL_VAR))
		code.add_child(_tok(" -= " if e.steps < 0 else " += ", COL_OP))
		code.add_child(_tok(str(absi(e.steps)) + " steps", COL_NUM))
		code.add_child(_comment_tok("not a knob of this skill"))
		parts[0].tooltip_text = "This skill has no %s knob: a script on it that calls this routine won't compile.\nDrag out or right-click to delete." % e.knob
		_make_line_interactive(parts[0], owner_rule, {"kind": "effect", "effect": e}, branch)
		_mark(parts, owner_rule, e)
		return
	code.add_child(_tok(String(e.knob), COL_VAR))
	code.add_child(_tok(" -= " if e.steps < 0 else " += ", COL_OP))
	var effect_mult := limits.rule_effect_mult(rule, not SkillProgramLimits.per_hit_knob(knob))
	var grow_tok := func() -> Control:
		return _num_tok(e.ramp_text(), "what this grows with (it stops at the knob's limit)",
			func(d: int) -> void:
				e.ramp = _next_grow_key(e.ramp, d)
				_changed())
	if e.ramps():
		code.add_child(grow_tok.call())
		code.add_child(_tok(" * ", COL_OP))
	var value := str(absi(e.steps))
	if knob:
		value = _fmt_per_count(knob, e.steps, effect_mult) if e.ramps() else _fmt_knob(knob, e.steps, effect_mult)
	var num := _num_tok(value, knob.display_name.to_lower() if knob else String(e.knob), func(d: int) -> void:
		e.steps += d
		_changed())
	code.add_child(num)
	_effect_num[e] = num
	var note := ""
	if cost > 0.0:
		note = "costs %.1f" % cost
		if e.ramps() and knob:
			var room := _draft().growth_room(e, knob)
			note += ", max +%s at %s = %d" % [_fmt_knob(knob, room, effect_mult), e.ramp_text(), ScriptRegisters.ramp_reach(e.steps, room)]
	elif cost < 0.0:
		note = "pays %.1f" % -cost
	if rule.loop != null or rule.only_on_repeats():
		note = ("%s, repeats only" % note) if note != "" else "repeats only"
	var charge_mult := VariableStoreSupport.rule_effect_mult(rule)
	if charge_mult > 1.0:
		var x := "x%.2f charge" % charge_mult
		note = ("%s, %s" % [note, x]) if note != "" else x
	code.add_child(_comment_tok(note))
	if knob and e.ramps():
		parts[0].tooltip_text = "%s grows with registry.size\nOne step per enemy in the registry, up to its limit (less your other %s lines).\nCOST: a higher value grows faster and costs more\nNOTE: needs a registry.add(target) line to grow\nDrag out or right-click to delete." % [
			knob.display_name, e.knob]
	elif knob:
		parts[0].tooltip_text = "%s: %s\n%s\n%s%s%s\nDrag out or right-click to delete." % [
			knob.display_name, knob.describe(e.steps), _step_price_note(knob), _knob_usage_note(knob),
			(_stat_note(knob, limits) if knob.kind == SkillKnob.Kind.STAT else "") + _knob_limits_note(knob),
			"\nLOOP: changes the repeats only" if rule.loop != null or rule.in_loop() or rule.only_on_repeats() else ""]
	_make_line_interactive(parts[0], owner_rule, {"kind": "effect", "effect": e}, branch)
	_mark(parts, owner_rule, e)


func _knob_usage_note(knob: SkillKnob) -> String:
	var range_now: Array = _validation.knob_ranges.get(knob.id, [0, 0]) if _validation != null else [0, 0]
	var lo: int = range_now[0]
	var hi: int = range_now[1]
	var have := "%+d steps" % lo if lo == hi else "%+d to %+d steps (every when rule on)" % [lo, hi]
	if lo == 0 and hi == 0:
		have = "0 steps"
	return "LIMIT: %s used, %d to %+d allowed" % [have, knob.min_steps, knob.max_steps]


func _step_price_note(knob: SkillKnob) -> String:
	var n := _draft().buff_steps_on(knob)
	var pay := "drawback steps pay %.1f points" % absf(knob.cost_per_step)
	if n <= 0:
		return "COST: first buff step %.1f points · %s" % [knob.next_step_cost(0), pay]
	return "COST: next buff step %.1f points (x%s the last) · %s" % [
		knob.next_step_cost(n), str(SkillKnob.STEP_COST_GROWTH), pay]


func _can_grow(knob: SkillKnob, limits: SkillProgramLimits) -> bool:
	return limits.allows("knob:" + String(ScriptRegisters.WRITE_KNOB)) and ScriptRegisters.can_grow(knob)


static func _next_grow_key(ramp: int, d: int) -> int:
	return -1 if (ramp >= 0) != (posmod(d, 2) == 0) else 0


func _knob_limits_note(knob: SkillKnob) -> String:
	if knob.always_only and knob.kind != SkillKnob.Kind.STAT:
		return "\nNOTE: always: rules only (read before the cast starts)"
	if knob.first_cast_only:
		return "\nNOTE: changes the original cast only, not loop repeats"
	return ""


func _stat_note(knob: SkillKnob, limits: SkillProgramLimits) -> String:
	if limits.per_hit_attack_speed(knob):
		return "\nNOTE: always: rules, or (Assassin) per-hit rules: a hit speeds up your next attack"
	return "\nNOTE: always: rules only"


func _loop_cost(rule: ProgramRule, limits: SkillProgramLimits) -> float:
	return rule.loop.cost(limits.loop_cost_mult()) * rule.cost_multiplier() * limits.rule_cost_mult(rule) \
		* limits.buff_cost_mult()


func _loop_line(rule: ProgramRule, limits: SkillProgramLimits) -> void:
	var lp := rule.loop
	var cap := limits.loop_cap()
	var parts := _code_line(rule, "+", COL_BUFF)
	var code: HBoxContainer = parts[1]
	var step := func(d: int) -> void:
		lp.count = clampi(lp.count + d, 1, cap)
		_changed()
	var cost := _loop_cost(rule, limits)
	if lp.mode == ProgramLoop.Mode.FOR:
		code.add_child(_tok("    for ", COL_KEYWORD))
		code.add_child(_tok("i", COL_VAR))
		code.add_child(_tok(" in ", COL_KEYWORD))
		code.add_child(_tok("range", COL_FN))
		code.add_child(_tok("(", COL_OP))
		code.add_child(_num_tok(str(lp.count), "repeats (1 to %d)" % cap, step))
		code.add_child(_tok("):", COL_OP))
		code.add_child(_comment_tok("%d repeat%s, costs %.1f" % [lp.count, "" if lp.count == 1 else "s", cost]))
	else:
		code.add_child(_tok("    cast", COL_FN))
		code.add_child(_tok("(", COL_OP))
		code.add_child(_tok("max=", COL_OP))
		code.add_child(_num_tok(str(lp.count), "most repeats (1 to %d)" % cap, step))
		code.add_child(_tok(")", COL_OP))
		code.add_child(_comment_tok("repeats while true, costs %.1f" % cost))
	parts[0].tooltip_text = _loop_tip(lp.mode, limits) + "\nClick the number to change the repeats.\nDrag out or right-click to delete the loop."
	_make_line_interactive(parts[0], rule, {"kind": "loop"})
	_mark(parts, rule, rule.loop)


func _cast_line(rule: ProgramRule, indent: String) -> void:
	var parts := _code_line(rule)
	var code: HBoxContainer = parts[1]
	code.add_child(_tok(indent + "cast", COL_FN))
	code.add_child(_tok("()", COL_OP))
	code.add_child(_comment_tok("the skill again"))
	_make_line_interactive(parts[0], rule, {})


func _break_line(rule: ProgramRule, b: ProgramBranch, indent: String) -> void:
	var parts := _code_line(rule, "-", COL_DRAW)
	var code: HBoxContainer = parts[1]
	code.add_child(_tok(indent + "break", COL_KEYWORD))
	code.add_child(_comment_tok("stops the loop. loop costs x%.2f" % ProgramLoop.BREAK_COST_MULT))
	parts[0].tooltip_text = "break\nStops the loop when this if is true.\nLOOP: checked after each cast or repeat lands\nCOST: the loop costs x%.2f\nDrag out or right-click to delete." % ProgramLoop.BREAK_COST_MULT
	_make_line_interactive(parts[0], rule, {"kind": "break"}, b)
	_mark(parts, rule, b)


func _loop_tip(mode: ProgramLoop.Mode, limits: SkillProgramLimits) -> String:
	var head := "for range(N)\nRepeats the skill N times after it lands." if mode == ProgramLoop.Mode.FOR \
		else "while ...: cast(max=N)\nRepeats while the rule's conditions stay true, up to N.\nNOTE: needs a condition that changes mid-cast"
	var per := ProgramLoop.FOR_COST if mode == ProgramLoop.Mode.FOR else ProgramLoop.WHILE_COST
	return "%s\nCOST: %.1f points per %s · each repeat uses %d%% of the skill's mana\nLOOP: each repeat deals x%.2f damage · at most %d per cast · the rule's lines only change the repeats\nTIP: add  if hit_nothing:  break  to stop early" % [
		head, per * limits.loop_cost_mult(), "repeat" if mode == ProgramLoop.Mode.FOR else "max repeat",
		roundi(ProgramLoop.MANA_FRACTION * limits.repeat_mana_mult() * 100.0), ProgramLoop.FALLOFF, limits.loop_cap()]

func _make_line_interactive(line: Control, rule: ProgramRule, info: Dictionary, branch: ProgramBranch = null) -> void:
	var drag_func := Callable()
	if branch != null:
		info["branch"] = branch
	if info.has("kind"):
		info["src"] = "rule"
		info["rule"] = rule
		drag_func = func(_pos: Vector2) -> Variant:
			line.set_drag_preview(_drag_preview(line))
			return info
	line.set_drag_forwarding(drag_func, _can_drop_on_rule.bind(rule), _drop_on_rule.bind(rule, info))
	line.gui_input.connect(func(ev: InputEvent) -> void:
		if not (ev is InputEventMouseButton):
			return
		if ev.button_index == MOUSE_BUTTON_LEFT and not ev.pressed and not get_viewport().gui_is_dragging():
			if _selected != rule or _owner != rule.owner:
				_selected = rule
				_owner = rule.owner
				_queue_rebuild()
		elif ev.button_index == MOUSE_BUTTON_RIGHT and ev.pressed and info.has("kind"):
			_remove_block(info)
			line.accept_event())

func _num_tok(text: String, what: String, on_step: Callable) -> Control:
	var box := PanelContainer.new()
	box.mouse_filter = Control.MOUSE_FILTER_PASS
	box.mouse_default_cursor_shape = Control.CURSOR_POINTING_HAND
	box.tooltip_text = "%s\nclick: up   right-click: down   wheel: up / down" % what
	var sb := StyleBoxFlat.new()
	sb.bg_color = Color(COL_NUM, 0.10)
	sb.border_color = Color(COL_NUM, 0.55)
	sb.border_width_bottom = 1
	sb.content_margin_left = 2
	sb.content_margin_right = 2
	box.add_theme_stylebox_override("panel", sb)
	box.add_child(_tok(text, COL_NUM))
	box.gui_input.connect(func(ev: InputEvent) -> void:
		if not (ev is InputEventMouseButton):
			return
		var step := 0
		if ev.button_index == MOUSE_BUTTON_LEFT and not ev.pressed and not get_viewport().gui_is_dragging():
			step = 1
		elif ev.button_index == MOUSE_BUTTON_RIGHT and ev.pressed:
			step = -1
		elif ev.button_index == MOUSE_BUTTON_WHEEL_UP and ev.pressed:
			step = 1
		elif ev.button_index == MOUSE_BUTTON_WHEEL_DOWN and ev.pressed:
			step = -1
		elif ev.button_index == MOUSE_BUTTON_LEFT and ev.pressed:
			box.accept_event()
			return
		if step != 0:
			on_step.call(step)
			box.accept_event())
	return box

func _build_palette() -> void:
	_clear(_palette_box)
	_snippets.clear()
	_pal_into = null
	var limits := _limits()
	if limits == null:
		return
	if _owner == KERNEL_FILE:
		_kernel_palette()
		return
	_update_tab_buttons()
	var defense := _palette_tab == "defense"
	var sel: ProgramRule = _selected if _selected != null and is_instance_valid(_selected) else null

	var knobs := limits.knobs_for(_knob_owner(_owner)).filter(func(k: SkillKnob) -> bool:
		return k.defensive == defense and k.kind != SkillKnob.Kind.VARIABLE)
	_section("knobs", "KNOBS  %s" % _owner_file(_owner), knobs.size())
	if knobs.is_empty():
		_pal_add(_comment_tok("nothing to change" if not defense else "no defensive knobs here"))
	var any_rule := knobs.filter(func(k: SkillKnob) -> bool: return not _always_only(k, limits))
	var only_always := knobs.filter(func(k: SkillKnob) -> bool: return _always_only(k, limits))
	for group in [["any rule", any_rule, "any rule\nThese knobs work in always: and when rules, and in loops.\nHIT: in a per-hit rule only damage, leech, barrier, guard and support knobs work"],
			["always: only", only_always, "always: only\nRead before the cast starts, so they only work in an always: rule without a loop.\n    always:\n        attack_speed += 5%"]]:
		if (group[1] as Array).is_empty():
			continue
		if not any_rule.is_empty() and not only_always.is_empty():
			_pal_add(_comment_tok(group[0], group[2]))
		for knob: SkillKnob in group[1]:
			var tip := "%s\n+1 step: %s\n%s\n%s" % [
				knob.display_name, knob.describe(1), _step_price_note(knob), _knob_usage_note(knob)]
			if knob.kind == SkillKnob.Kind.STAT:
				tip += _stat_note(knob, limits)
			tip += _knob_limits_note(knob)
			var knob_snippet := _snippet(String(knob.id), COL_VAR,
				{"src": "palette", "kind": "effect", "knob": knob.id}, tip,
				knob.describe(1).trim_prefix(knob.display_name).strip_edges() + "/step",
				_knob_misfit(knob, sel, limits))
			_snippets["knob:" + String(knob.id)] = knob_snippet
			_pal_add(knob_snippet)

	var per_cast: Array[String] = []
	var per_hit: Array[String] = []
	var loop_conds: Array[String] = []
	var by_gem := {}
	for type_id in ProgramConditions.TYPES.keys():
		var sample := ProgramConditions.create(type_id)
		if sample.defensive() != defense or not limits.allows("cond:" + type_id):
			continue
		if sample.modular():
			continue
		var gem_id := sample.required_support()
		if gem_id != "":
			if limits.support_ids.has(gem_id):
				if not by_gem.has(gem_id):
					by_gem[gem_id] = []
				by_gem[gem_id].append(type_id)
			continue
		if sample.needs_loop():
			loop_conds.append(type_id)
		elif sample.scope() == ProgramCondition.Scope.HIT:
			per_hit.append(type_id)
		else:
			per_cast.append(type_id)
	var gem_count := 0
	for g in by_gem:
		gem_count += (by_gem[g] as Array).size()
	if per_cast.size() + per_hit.size() + gem_count > 0:
		_section("conditions", "CONDITIONS", per_cast.size() + per_hit.size() + gem_count)
		for group in [["per cast", per_cast, "per cast\nChecked once, when you use the skill. If true, the whole cast gets the rule's lines.\n    when enemies_nearby(3, 4.0m):\n        radius += 1.0m"],
				["per hit", per_hit, "per hit\nChecked on each enemy the skill hits. Only the hits where it's true get the rule's lines.\nHIT: only damage, leech, barrier, guard and support effects\n    when target_life_below(30%):\n        damage += 20%"]]:
			if (group[1] as Array).is_empty():
				continue
			_pal_add(_comment_tok(group[0], group[2]))
			for type_id in group[1]:
				var cond_snippet := _condition_snippet(type_id, ProgramConditions.create(type_id), COL_FN, "")
				_snippets["cond:" + type_id] = cond_snippet
				_pal_add(cond_snippet)
		for gem_id in limits.support_ids:
			if not by_gem.has(gem_id):
				continue
			var def: SupportGemDefinition = SkillGemRegistry.get_support_gem(gem_id)
			var gem_name := def.display_name.trim_suffix(" Support") if def != null else gem_id
			_pal_add(_gem_header(gem_name))
			for type_id in by_gem[gem_id]:
				var gem_snippet := _condition_snippet(type_id, ProgramConditions.create(type_id), COL_GEM, gem_name)
				_snippets["cond:" + type_id] = gem_snippet
				_pal_add(gem_snippet)

	var loops := not defense and limits.allows("loop")
	var ifs := limits.allows("if")
	if loops or ifs:
		_section("loops", "LOOPS & IF" if loops else "IF", (3 + loop_conds.size() if loops else 0) + (1 if ifs else 0))
		if loops:
			_pal_add(_snippet("for range()", COL_KEYWORD, {"src": "palette", "kind": "loop", "mode": ProgramLoop.Mode.FOR},
				_loop_tip(ProgramLoop.Mode.FOR, limits) + "\nDrag into a rule, or click to add.",
				"%.1f pt/repeat" % (ProgramLoop.FOR_COST * limits.loop_cost_mult())))
			_pal_add(_snippet("while", COL_KEYWORD, {"src": "palette", "kind": "loop", "mode": ProgramLoop.Mode.WHILE},
				_loop_tip(ProgramLoop.Mode.WHILE, limits) + "\nDrag into a rule with a condition, or click to add.",
				"%.1f pt/repeat" % (ProgramLoop.WHILE_COST * limits.loop_cost_mult()),
				"" if sel == null or not sel.is_always() else "the selected rule has no condition: a while needs one"))
		if ifs:
			_pal_add(_snippet("if", COL_KEYWORD, {"src": "palette", "kind": "branch"},
				"if block\nLines that apply only when the rule fires AND the if is true.\n    when every_nth_cast(3):\n        if enemies_nearby(3, 4.0m):\n            radius += 1.0m\nTIP: fewer blocks than a second rule\nLOOP: checked after each repeat lands · can hold break\nDrop it on a rule, then a condition on its if line, then knobs below.",
				"nested"))
		if loops:
			_pal_add(_snippet("break", COL_KEYWORD, {"src": "palette", "kind": "break"},
				"break\nStops a loop early when its if is true.\n    if hit_nothing:\n        break\nCOST: the loop costs x%.2f\nDrop it on an if line in a loop rule (on the rule itself: starts a new if)." % ProgramLoop.BREAK_COST_MULT,
				"loop x%.2f" % ProgramLoop.BREAK_COST_MULT,
				"" if sel == null or sel.loop != null else "the selected rule has no loop"))
			if not loop_conds.is_empty():
				_pal_add(_comment_tok("about repeats", "about repeats\nThese read how the last cast or repeat went, so they only work in a loop rule's if.\n    for range(3):\n        if hit_nothing:\n            break\nLOOP: checked after each cast or repeat lands"))
				for type_id in loop_conds:
					var lc := _condition_snippet(type_id, ProgramConditions.create(type_id), COL_FN, "")
					_snippets["cond:" + type_id] = lc
					_pal_add(lc)

	if limits.allows("knob:" + String(ScriptRegisters.WRITE_KNOB)):
		_section("registry", "REGISTRY", -1)
		_add_variable_snippets()

	if limits.allows("call") and _owner != ROUTINES_FILE:
		var rts := _routines()
		_section("routines", "ROUTINES", rts.size())
		for i in rts.size():
			var rt: ProgramRoutine = rts[i]
			var what := "empty: write it in routines.gd" if rt.is_empty() else rt.body.describe(limits).trim_prefix("ALWAYS -> ")
			var call_snippet := _snippet("call %s()" % rt.routine_name, COL_KEYWORD, {"src": "palette", "kind": "call", "routine": i},
				"Runs %s() where you drop it: %s\n1 block, whatever it holds; this skill pays for its lines.\nDrop it in a rule of the skill's own file (or its if), or click to add." % [rt.routine_name, what],
				"1 block")
			_snippets["call:%d" % i] = call_snippet
			_pal_add(call_snippet)
	_pal_into = null


func _pal_add(c: Control) -> void:
	(_pal_into if _pal_into != null and is_instance_valid(_pal_into) else _palette_box).add_child(c)


func _section(key: String, title: String, count: int) -> void:
	_pal_into = null
	var folds: Dictionary = Settings.get_value("layout", "snippet_folds")
	var folded := bool(folds.get(key, false)) if folds != null else false
	if _palette_box.get_child_count() > 0:
		var gap := Control.new()
		gap.custom_minimum_size.y = 6.0 * _k()
		gap.mouse_filter = Control.MOUSE_FILTER_IGNORE
		_palette_box.add_child(gap)
	var head := Button.new()
	head.flat = true
	head.focus_mode = Control.FOCUS_NONE
	head.alignment = HORIZONTAL_ALIGNMENT_LEFT
	head.clip_text = true
	head.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS
	head.custom_minimum_size.x = 1.0
	head.mouse_default_cursor_shape = Control.CURSOR_POINTING_HAND
	if panel._font:
		head.add_theme_font_override("font", panel._font)
	head.add_theme_font_size_override("font_size", roundi(_sz(12) * panel.font_scale))
	for state in ["font_color", "font_hover_color", "font_pressed_color"]:
		head.add_theme_color_override(state, panel.accent if state == "font_color" else Color.WHITE)
	var line := StyleBoxFlat.new()
	line.bg_color = Color(panel.accent, 0.08)
	line.border_color = Color(panel.accent, 0.45)
	line.border_width_bottom = 1
	line.content_margin_left = 4
	for st in ["normal", "hover", "pressed"]:
		head.add_theme_stylebox_override(st, line)
	var label := "%s%s" % [title, ("  (%d)" % count) if count >= 0 else ""]
	head.text = ("> " if folded else "v ") + label
	head.tooltip_text = "Click to fold or unfold this section."
	var body := VBoxContainer.new()
	body.add_theme_constant_override("separation", 2)
	body.visible = not folded
	body.set_meta(&"section_head", head)
	body.set_meta(&"section_key", key)
	body.set_meta(&"section_label", label)
	head.pressed.connect(func() -> void: _fold(body, body.visible))
	_palette_box.add_child(head)
	_palette_box.add_child(body)
	_pal_into = body


func _fold(body: Control, fold: bool) -> void:
	body.visible = not fold
	var head := body.get_meta(&"section_head") as Button
	if head != null:
		head.text = ("> " if fold else "v ") + String(body.get_meta(&"section_label", ""))
	var folds: Dictionary = (Settings.get_value("layout", "snippet_folds") as Dictionary).duplicate() \
			if Settings.get_value("layout", "snippet_folds") is Dictionary else {}
	folds[String(body.get_meta(&"section_key", ""))] = fold
	Settings.set_value("layout", "snippet_folds", folds)


func _reveal(snippet: Control) -> void:
	var n := snippet.get_parent()
	while n != null and n != _palette_box:
		if n.has_meta(&"section_key") and not (n as Control).visible:
			_fold(n as Control, false)
		n = n.get_parent()


static func _always_only(knob: SkillKnob, limits: SkillProgramLimits) -> bool:
	return knob.always_only or (knob.kind == SkillKnob.Kind.STAT and not limits.per_hit_attack_speed(knob))


func _knob_misfit(knob: SkillKnob, sel: ProgramRule, limits: SkillProgramLimits) -> String:
	if sel == null:
		return ""
	if _always_only(knob, limits) and (not sel.is_always() or sel.loop != null):
		return "only works in an always: rule without a loop, and the selected rule isn't one"
	if sel.scope() == ProgramCondition.Scope.HIT and knob.kind != SkillKnob.Kind.DAMAGE \
			and not SkillProgramLimits.per_hit_knob(knob) and not limits.per_hit_any_knob() \
			and not limits.per_hit_attack_speed(knob):
		return "the selected rule is per hit: it can only change damage, leech, barrier, guard and support effects"
	return ""


func _add_variable_snippets() -> void:

		["registry.add(target)", {"src": "palette", "kind": "set_write", "op": 0},
			"Adds enemies to the registry. In always: or a WHEN, every enemy the cast hits; in a per-hit rule, the enemy being hit.\nDrop on a registry line to make it add." + free],
		["registry.remove(target)", {"src": "palette", "kind": "set_write", "op": 1},
			"Takes enemies out of the registry, the same way add puts them in.\nDrop on a registry line to make it remove." + free],
	])
	var member_tip := "%s\nChecked per hit, against the enemy being hit, so this rule's buffs only land on those hits (per-hit rules can change damage, leech, barrier, guard and support effects that happen on that hit or kill: explosions, breakpoints, tremors, charge, error damage, forks. Barrier and guard come once per cast when a hit lands). A hit sees the registry from before it landed.\n%s Drop on a registry check to change it."
	var in_cost := "Buffs in this rule cost x%.2f if your script spends marks with registry.remove(target): each mark pays out once. Without a remove a mark lasts all fight, so they cost x%.2f." % [
		VarCondition.COST_SPENT, VarCondition.COST_KEPT]
	_add_piece_row([

func _snippet(text: String, col: Color, data: Dictionary, tip: String, side := "", misfit := "") -> Control:
	var block := PanelContainer.new()
	block.mouse_filter = Control.MOUSE_FILTER_STOP
	block.mouse_default_cursor_shape = Control.CURSOR_DRAG
	var sb := StyleBoxFlat.new()
	sb.bg_color = Color(col, 0.07)
	sb.border_color = Color(col, 0.6)
	sb.border_width_left = 2
	if col == COL_GEM:
		sb.set_border_width_all(1)
		sb.border_width_left = 3
		sb.bg_color = Color(col, 0.12)
	sb.content_margin_left = 6
	sb.content_margin_top = 1
	sb.content_margin_bottom = 1
	block.add_theme_stylebox_override("panel", sb)
	var l := _tok(text, col)
	_shrinkable(l)
	var unlocked := NewBlocks.block_of(data)
	var is_new := NewBlocks.is_new(get_tree(), unlocked, text)
	if side != "" and not is_new:
		var row := HBoxContainer.new()
		row.mouse_filter = Control.MOUSE_FILTER_IGNORE
		row.add_theme_constant_override("separation", roundi(4 * _k()))
		row.add_child(l)
		var sl := _tok(side, COL_COMMENT)
		sl.add_theme_font_size_override("font_size", roundi(_sz(9) * panel.font_scale))
		sl.size_flags_horizontal = Control.SIZE_SHRINK_END
		sb.content_margin_right = 4
		row.add_child(sl)
		block.add_child(row)
	else:
		block.add_child(l)
	block.set_meta(&"snippet", text)
	if misfit != "":
		block.modulate.a = 0.42
		var nl := tip.find("\n")
		tip = (tip + "\nNOTE: " + misfit) if nl < 0 else tip.substr(0, nl) + "\nNOTE: " + misfit + tip.substr(nl)
	block.tooltip_text = tip
	var seen := func() -> void: pass
	if is_new:
		seen = _add_new_badge(block, l, sb, unlocked, text)
	block.set_drag_forwarding(
		func(_pos: Vector2) -> Variant:
			seen.call()
			block.set_drag_preview(_drag_preview(block))
			return data,
		_can_drop_on_palette, _drop_on_palette)
	block.gui_input.connect(func(ev: InputEvent) -> void:
		if ev is InputEventMouseButton and not ev.pressed and ev.button_index == MOUSE_BUTTON_LEFT \
				and not get_viewport().gui_is_dragging():
			seen.call()
			_add_from_palette(data, _selected))
	return block


func _add_new_badge(block: Control, label: Label, sb: StyleBoxFlat, unlocked: String, text: String) -> Callable:
	var k := _k()
	var badge := Label.new()
	badge.text = "NEW"
	badge.mouse_filter = Control.MOUSE_FILTER_IGNORE
	badge.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	badge.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	badge.add_theme_font_size_override("font_size", roundi(_sz(9) * panel.font_scale))
	badge.add_theme_color_override("font_color", Color(0.08, 0.06, 0.02))
	var pill := StyleBoxFlat.new()
	pill.bg_color = COL_NEW
	pill.set_corner_radius_all(roundi(3 * k))
	pill.content_margin_left = 3 * k
	pill.content_margin_right = 3 * k
	badge.add_theme_stylebox_override("normal", pill)
	var w := 30.0 * k
	var old_margin := sb.content_margin_right
	sb.content_margin_right = old_margin + w + 4.0 * k
	badge.anchor_left = 1.0
	badge.anchor_right = 1.0
	badge.anchor_top = 0.5
	badge.anchor_bottom = 0.5
	badge.offset_left = 4.0 * k
	badge.offset_right = 4.0 * k + w
	badge.offset_top = -7.0 * k
	badge.offset_bottom = 7.0 * k
	label.add_child(badge)
	var done := [false]
	var seen := func() -> void:
		if done[0]:
			return
		done[0] = true
		NewBlocks.mark_seen(get_tree(), unlocked, text)
		if is_instance_valid(badge):
			badge.queue_free()
		sb.content_margin_right = old_margin
	var hovering := [false]
	block.mouse_exited.connect(func() -> void: hovering[0] = false)
	block.mouse_entered.connect(func() -> void:
		hovering[0] = true
		var wait := float(ProjectSettings.get_setting("gui/timers/tooltip_delay_sec", 0.5)) + 0.6
		get_tree().create_timer(wait).timeout.connect(func() -> void:
			if hovering[0] and is_instance_valid(block) and block.is_visible_in_tree():
				seen.call()))
	return seen

func _add_from_palette(data: Dictionary, rule: ProgramRule, line: Dictionary = {}) -> void:
	if _sock() == null or _limits() == null:
		return
	if rule == null and _owner == ROUTINES_FILE:
		rule = _selected if _routine_of(_selected) != null else (_routines()[0] as ProgramRoutine).body
	if _routine_of(rule) != null:
		var why := _routine_refusal(data, line, _box(rule, line) != rule)
		if why != "":
			panel._notify(why)
			return
	if data["kind"] == "call" and (rule.owner if rule != null else _owner) != "":
		panel._notify("CALL ROUTINES FROM THE SKILL'S OWN FILE")
		return
	if data["kind"] in ["set_write", "member", "size", "compare_op", "sort", "position"]:
		_add_counter_piece(data, rule, line)
		return
	if data["kind"] == "break" and (rule == null or rule.loop == null):
		panel._notify("ADD A FOR OR WHILE LOOP FIRST")
		return
	if data["kind"] == "effect" and not _knob_fits(data["knob"], rule.owner if rule != null else _owner):
		return
	if rule == null:
		rule = _new_rule()
	var box: Object = _box(rule, line)
	if data["kind"] == "branch":
		rule.branches.append(ProgramBranch.new())
	elif data["kind"] == "condition":
		var c := ProgramConditions.create(data["type"])
		if c != null:
			box.conditions.append(c)
	elif data["kind"] == "loop":
		if rule.loop == null:
			rule.loop = ProgramLoop.make(data["mode"])
		elif rule.loop.mode == data["mode"]:
			rule.loop.count = mini(rule.loop.count + 1, _limits().loop_cap())
		else:
			rule.loop.mode = data["mode"]
	elif data["kind"] == "break":
		_add_break(rule, box as ProgramBranch)
	elif data["kind"] == "call":
		box.effects.append(ProgramEffect.make_call(int(data["routine"])))
	else:
		var existing: ProgramEffect = null
		for e in box.effects:
			if e.knob == data["knob"]:
				existing = e
		if existing != null:
			existing.steps += 1
		else:
			box.effects.append(ProgramEffect.make(data["knob"], 1))
	_selected = rule
	_changed()


func _add_break(rule: ProgramRule, branch: ProgramBranch) -> void:
	if branch == null:
		branch = ProgramBranch.new()
		rule.branches.append(branch)
	branch.breaks = true


static func _routine_refusal(data: Dictionary, line: Dictionary, in_if: bool) -> String:
	match data.get("kind"):
		"call":
			return "A ROUTINE CAN'T CALL A ROUTINE"
		"loop", "break":
			return "A ROUTINE CAN'T LOOP: PUT THE LOOP ON THE RULE THAT CALLS IT"
		"condition", "member", "compare_op", "position":
			if not in_if:
				return "A ROUTINE HAS NO CONDITIONS: PUT THEM ON THE RULE THAT CALLS IT, OR IN AN IF"
		"size":
			if not in_if and line.get("effect") == null:
				return "A ROUTINE HAS NO CONDITIONS: PUT THEM ON THE RULE THAT CALLS IT, OR IN AN IF"
	return ""


static func _box(rule: ProgramRule, line: Dictionary) -> Object:
	var b: ProgramBranch = line.get("branch") as ProgramBranch
	return b if b != null and rule != null and b in rule.branches else rule


func _add_counter_piece(data: Dictionary, rule: ProgramRule, line: Dictionary) -> void:
	var limits := _limits()
	var read: VarCondition = line.get("condition") as VarCondition
	var effect: ProgramEffect = line.get("effect") as ProgramEffect
	var write: ProgramEffect = effect if effect != null and ScriptRegisters.is_write(effect.knob) else null
	var branch: ProgramBranch = _box(rule, line) as ProgramBranch
	match data["kind"]:
		"set_write":
			if write != null:
				write.op = data["op"]
			else:
				rule = _add_write(rule, ScriptRegisters.WRITE_OPS[data["op"]], branch)
		"member":
			if read != null:
				read.key = 0
				read.op = data["op"]
			else:
				if rule == null:
					rule = _new_rule()
				_box(rule, line).conditions.append(VarCondition.make("target", ScriptRegisters.MEMBER_OPS[data["op"]]))
		"size":
			if read != null:
				read.key = 1
				read.op = ScriptRegisters.OPS.find(">=")
			elif effect != null and write == null:
				var knob := limits.get_knob(effect.knob)
				if not _can_grow(knob, limits):
					panel._notify("%s CAN'T GROW WITH THE REGISTRY" % String(effect.knob).to_upper())
					return
				effect.ramp = 0
			else:
				if rule == null:
					rule = _new_rule()
				_box(rule, line).conditions.append(VarCondition.make("size", ">=", 3))
		"compare_op":
			if read != null:
				read.key = 1
				read.op = data["op"]
			else:
				if rule == null:
					rule = _new_rule()
				_box(rule, line).conditions.append(VarCondition.make("size", ScriptRegisters.OPS[data["op"]], 3))
		"sort":
			if write != null and ScriptRegisters.is_sort(write):
				write.sort_by = data["order"]
			else:
				if not _knob_fits(ScriptRegisters.WRITE_KNOB, rule.owner if rule != null else _owner):
					return
				if rule == null:
					rule = _new_rule()
				(branch if branch != null else rule).effects.append(ProgramEffect.make_sort(data["order"]))
		"position":
			var at: RegistryPositionCondition = line.get("condition") as RegistryPositionCondition
			if at != null:
				at.pos = data["pos"]
			else:
				if rule == null:
					rule = _new_rule()
				var c := RegistryPositionCondition.new()
				c.pos = data["pos"]
				_box(rule, line).conditions.append(c)
	if rule == null:
		return
	_selected = rule
	_changed()


func _add_write(rule: ProgramRule, op_text: String, branch: ProgramBranch = null) -> ProgramRule:
	if not _knob_fits(ScriptRegisters.WRITE_KNOB, rule.owner if rule != null else _owner):
		return null
	if rule == null:
		rule = _new_rule()
	(branch if branch != null else rule).effects.append(ProgramEffect.make_write(op_text))
	return rule


func _knob_fits(knob_id: StringName, owner_id: String) -> bool:
	var limits := _limits()
	if limits == null:
		return false
	for k in limits.knobs_for(_knob_owner(owner_id)):
		if (k as SkillKnob).id == knob_id:
			return true
	panel._notify("%s ISN'T A KNOB OF %s" % [String(knob_id).to_upper(), _owner_file(owner_id).to_upper()])
	return false


func _detach(info: Dictionary) -> void:
	var from: ProgramRule = info["rule"]
	var box: Object = _box(from, info)
	match info["kind"]:
		"condition":
			box.conditions.erase(info["condition"])
		"effect":
			box.effects.erase(info["effect"])
		"branch":
			from.branches.erase(info.get("branch"))
		"loop":
			from.loop = null
		"break":
			if box is ProgramBranch:
				(box as ProgramBranch).breaks = false
	var b := box as ProgramBranch
	if b != null and b.conditions.is_empty() and b.effects.is_empty() and not b.breaks:
		from.branches.erase(b)
	if from.conditions.is_empty() and from.effects.is_empty() and from.loop == null and from.branches.is_empty():
		_draft().rules.erase(from)


func _move_block(info: Dictionary, to: ProgramRule, line: Dictionary = {}) -> void:
	var src: ProgramRule = info["rule"]
	var moved_loop: ProgramLoop = src.loop
	if info["kind"] == "break" and (to == null or to.loop == null):
		panel._notify("ADD A FOR OR WHILE LOOP FIRST")
		return
	var dest_owner := to.owner if to != null else _owner
	if _routine_of(to) != null:
		var into_if := line.get("branch") != null
		if info["kind"] in ["loop", "break"] or (info["kind"] == "condition" and not into_if) \
				or (info["kind"] == "effect" and (info["effect"] as ProgramEffect).is_call()):
			panel._notify("A ROUTINE ONLY TAKES KNOBS, REGISTRY WRITES AND IF BLOCKS")
			return
	if info["kind"] == "effect" and (info["effect"] as ProgramEffect).is_call():
		if dest_owner != "":
			panel._notify("CALL ROUTINES FROM THE SKILL'S OWN FILE")
			return
	elif info["kind"] == "effect" and not _knob_fits((info["effect"] as ProgramEffect).knob, dest_owner):
		return
	var moved_branch: ProgramBranch = info.get("branch") as ProgramBranch
	var dest_branch: ProgramBranch = line.get("branch") as ProgramBranch
	if info["kind"] == "branch" and dest_branch == moved_branch:
		return
	_detach(info)
	if to == null:
		to = _new_rule()
	var box: Object = dest_branch if dest_branch != null and dest_branch in to.branches else to
	match info["kind"]:
		"condition":
			box.conditions.append(info["condition"])
		"effect":
			box.effects.append(info["effect"])
		"branch":
			to.branches.append(moved_branch)
		"loop":
			to.loop = moved_loop
		"break":
			_add_break(to, box as ProgramBranch)
	_selected = to
	_changed()


func _remove_block(info: Dictionary) -> void:
	_detach(info)
	_changed()


func _changed() -> void:
	_queue_rebuild()

func _update_tip(draft: SkillProgram, v: ProgramValidation, limits: SkillProgramLimits) -> void:
	if _tip_row == null:
		return
	if not ScriptTips.enabled():
		_hide_tip()
		return
	if _tip_hold:
		_tip_hold = false
		return
	var next := ScriptTips.next_tip(draft, v, limits)
	if not next.is_empty():
		_show_tip(next)


func _show_tip(t: Dictionary) -> void:
	if _tip_row == null:
		return
	_tip_shown = t
	ScriptTips.mark_seen(t["id"])
	_tip_label.text = "# tip: " + String(t["text"])
	var page := 0
	for i in ScriptGuide.PAGES.size():
		if ScriptGuide.PAGES[i][0] == t["page"]:
			page = i + 1
	_tip_page_btn.text = "guide() p.%d" % page if page > 0 else "guide()"
	_tip_row.visible = true


func _hide_tip() -> void:
	_tip_shown = {}
	if _tip_row != null:
		_tip_row.visible = false


func current_tip() -> Dictionary:
	return _tip_shown


func _update_status() -> void:
	var s := _sock()
	var limits := _limits()
	var enabled := s != null and limits != null
	for b in [_apply_btn, _revert_btn, _clear_btn]:
		b.disabled = not enabled
	_state = ""
	if not enabled:
		_console.text = _c("// nothing to compile", COL_COMMENT)
		_errors_console.text = _c("// nothing to compile", COL_COMMENT)
		_set_output_alert(0)
		return
	if _owner == KERNEL_FILE:
		_kernel_status()
		return

	var draft := _draft()
	var v := _validation if _validation != null else _skills().preview_program(_socket, draft, _routines())
	_update_tip(draft, v, limits)
	var lines: Array[String] = []

	var paid := v.drawback_points + v.point_budget
	var over := v.buff_points > paid + 0.001
	var cap := maxf(maxf(paid, v.buff_points), 1.0)
	var filled := clampi(roundi(16.0 * v.buff_points / cap), 0, 16)
	var bar := "#".repeat(filled) + ".".repeat(16 - filled)
	lines.append("%s points  [lb]%s[rb]  %s" % [_c(">", COL_GUTTER), _c(bar, SkillPanel.COL_WARN if over else COL_OK),
		_c("%.1f / %.1f" % [v.buff_points, paid], SkillPanel.COL_WARN if over else SkillPanel.COL_TEXT)])
	if limits.bonus_steps > 0:
		lines.append("%s knobs   %s %s" % [_c(">", COL_GUTTER), _c("+%d step%s" % [limits.bonus_steps, "" if limits.bonus_steps == 1 else "s"], COL_OK),
			_c("// every knob goes further: " + _step_sources(), COL_COMMENT)])
	if limits.bonus_points > 0.0:
		lines.append("%s bonus   %s %s" % [_c(">", COL_GUTTER), _c("+%s points" % String.num(limits.bonus_points, 1), COL_OK),
			_c("// " + _bonus_sources(), COL_COMMENT)])
	lines.append("%s blocks  %d/%d   rules %d/%d" % [_c(">", COL_GUTTER), v.actions_used, v.max_actions,
		draft.rules.size(), limits.max_rules])
	if limits.class_trait != null:
		lines.append("%s trait   %s %s" % [_c(">", COL_GUTTER), _c(limits.class_trait.display_name.to_lower(), COL_KEYWORD),
			_c("// " + ", ".join(limits.class_trait.summary()), COL_COMMENT)])

	var has_loop := false
	for rule in draft.rules:
		has_loop = has_loop or rule.loop != null
	if has_loop:
		lines.append("%s loops   %s %s" % [_c(">", COL_GUTTER), _c("max %d repeats" % limits.loop_cap(), COL_KEYWORD),
			_c("// each repeat x%.2f damage, %d%% mana" % [ProgramLoop.FALLOFF,
				roundi(ProgramLoop.MANA_FRACTION * limits.repeat_mana_mult() * 100.0)], COL_COMMENT)])

	if not draft.has_rules():
		lines.append(_c("// add a rule to start. fewer blocks earn an efficiency bonus", COL_COMMENT))
	elif v.tier != null:
		var t := "%s tier    %s %s" % [_c(">", COL_GUTTER), _c(v.tier.display_name.to_lower(), COL_KEYWORD),
			_c("// " + v.tier.description, COL_COMMENT)]
		if v.next_tier != null:
			t += _c("  (%d fewer -> %s)" % [v.actions_to_next_tier, v.next_tier.display_name.to_lower()], COL_COMMENT)
		lines.append(t)
	elif v.next_tier != null:
		lines.append("%s tier    none %s" % [_c(">", COL_GUTTER),
			_c("// %d fewer blocks -> %s" % [v.actions_to_next_tier, v.next_tier.display_name.to_lower()], COL_COMMENT)])

	var dirty := _is_dirty(_socket) or _routines_dirty()
	if _routines_dirty():
		lines.append(_c("// routines.gd changed: apply() saves it for every skill that calls it", COL_PENDING))
	var err_lines: Array[String] = []
	if not v.valid:
		_state = "invalid"
		var count := v.error_count()
		lines.append(_c("x %d error%s: see ERRORS, fix %s to apply()" % [count, "" if count == 1 else "s",
			"it" if count == 1 else "them"], SkillPanel.COL_WARN))
		err_lines.append(_c("x %d error%s: fix %s to apply()" % [count, "" if count == 1 else "s",
			"it" if count == 1 else "them"], SkillPanel.COL_WARN))
		for i in v.issues.size():
			var issue: Dictionary = v.issues[i]
			if issue["kind"] != "error" or issue["dup"]:
				continue
			err_lines.append(_c("  error ", SkillPanel.COL_WARN) + _c(_where(issue, i), COL_FN)
				+ _c(_esc(issue["text"]), SkillPanel.COL_TEXT))
			if issue["hint"] != "":
				err_lines.append(_c("        fix: ", COL_GUTTER) + _c(_esc(issue["hint"]), COL_OK))
	else:
		err_lines.append(_c("// no errors", COL_COMMENT))
	if not v.valid:
		pass
	elif dirty:
		_state = "pending"
		lines.append(_c("ok: ", COL_PENDING) + _c("compiles. apply() to run it", Color(COL_PENDING, 0.85)))
	elif draft.has_rules():
		_state = "applied"
		lines.append(_c("ok: ", COL_OK) + _c("running", Color(COL_OK, 0.85)))
	_console.text = "\n".join(lines)
	_errors_console.text = "\n".join(err_lines)
	_apply_btn.disabled = not dirty or not v.valid
	_revert_btn.disabled = not dirty
	_set_output_alert(0 if v.valid else v.error_count())
	_apply_btn.tooltip_text = "Run this program on the skill." if v.valid \
		else "Fix the errors in OUTPUT first (red lines in the code)."


func _kernel_status() -> void:
	var qm := _kernel_qm()
	var lines: Array = Array(_skills().kernel)
	var t := KernelScript.tally(qm)
	var answered := int(t["merge"]) + int(t["overwrite"]) + int(t["terminate"])
	var out: Array[String] = []
	out.append("%s requests  %s   %s  %s  %s" % [_c(">", COL_GUTTER),
		_c("%d/%d answered" % [answered, KernelScript.NULL_LINES.size()], SkillPanel.COL_TEXT),
		_c("merge %d" % int(t["merge"]), KernelScript.COL_NULL), _c("overwrite %d" % int(t["overwrite"]), COL_KERNEL),
		_c("terminate %d" % int(t["terminate"]), KernelScript.COL_TERMINATE)])
	out.append("%s running   %s %s" % [_c(">", COL_GUTTER),
		_c("%d root line%s, %d of the Null's" % [lines.size(), "" if lines.size() == 1 else "s", KernelScript.null_running(qm).size()], SkillPanel.COL_TEXT),
		_c("// in every %s" % ("dungeon" if Endings.runs_everywhere(qm) else "Act V zone past the Core"), COL_COMMENT)])
	var ending := Endings.current(qm)
	if ending != "":
		out.append("%s ending    %s %s" % [_c(">", COL_GUTTER), _c(str(Endings.TITLES[ending]).to_lower(), Endings.COLOURS[ending]),
			_c("// the Last Log can show the others", COL_COMMENT)])
	else:
		out.append("%s leans     %s %s" % [_c(">", COL_GUTTER), _c(KernelScript.lean(qm) if answered > 0 else "nothing yet", COL_KEYWORD),
			_c("// the answer you gave most (a tie: the deepest). The Last Log reads it back", COL_COMMENT)])
	var editable := Endings.kernel_editable(qm)
	out.append(_c("ok: ", COL_OK) + _c("yours to edit: up to %d root lines, they apply at once" % Endings.MAX_ROOT_LINES if editable
		else "read only. answer the Null on each floor down", Color(COL_OK, 0.85)))
	_console.text = "\n".join(out)
	_errors_console.text = _c("// no errors", COL_COMMENT)
	_apply_btn.disabled = true
	_revert_btn.disabled = true
	_clear_btn.disabled = not editable or lines.is_empty()
	_apply_btn.tooltip_text = "kernel.gd's lines apply as soon as they're added." if editable \
		else "kernel.gd is a record: answer the Null's pull requests on each floor down."
	_set_output_alert(0)


func _mark(parts: Array, rule: ProgramRule, part: Object) -> void:
	if _validation == null:
		return
	var texts: Array[String] = []
	var hints: Array[String] = []
	var is_error := false
	for i in _validation.issues.size():
		var issue: Dictionary = _validation.issues[i]
		if issue["rule"] != rule or issue["part"] != part:
			continue
		if not _issue_line.has(i):
			_issue_line[i] = _line_no
		is_error = is_error or issue["kind"] == "error"
		texts.append(issue["text"])
		if issue["hint"] != "" and not (issue["hint"] in hints):
			hints.append(issue["hint"])
	if texts.is_empty():
		return
	var col := SkillPanel.COL_WARN if is_error else COL_PENDING
	var line: PanelContainer = parts[0]
	var sb := line.get_theme_stylebox("panel") as StyleBoxFlat
	if sb != null:
		sb.bg_color = Color(col, 0.14)
		sb.border_color = col
		sb.border_width_left = 3
	var mark := line.get_child(0).get_child(1) as Label
	mark.text = "!"
	mark.add_theme_color_override("font_color", col)
	var note := "  # %s: %s" % ["error" if is_error else "note", "; ".join(texts)]
	var code: HBoxContainer = parts[1]
	var replaced := false
	for child in code.get_children():
		if child is Label and (child as Label).get_theme_color("font_color") == COL_COMMENT:
			(child as Label).text = note
			(child as Label).add_theme_color_override("font_color", col)
			replaced = true
			break
	if not replaced:
		var tok := _tok(note, col)
		_shrinkable(tok)
		code.add_child(tok)
	var head := "%s: %s" % ["ERROR" if is_error else "NOTE", "\n".join(texts)]
	for h in hints:
		head += "\nfix: " + h
	line.tooltip_text = head + ("\n\n" + line.tooltip_text if line.tooltip_text != "" else "")


func _where(issue: Dictionary, index: int) -> String:
	var rule: ProgramRule = issue["rule"]
	if rule == null:
		return ""
	if _issue_line.has(index):
		return "line %d: " % _issue_line[index]
	return "in %s: " % _owner_file(rule.owner)


func _owner_file(owner_id: String) -> String:
	if owner_id == ROUTINES_FILE:
		return "routines.gd"
	if owner_id == KERNEL_FILE:
		return KernelScript.FILE_NAME
	if owner_id == "":
		return _file_name(_sock().gem.get_display_name()) if _sock() != null else "skill"
	var def: SupportGemDefinition = SkillGemRegistry.get_support_gem(owner_id)
	return _file_name(def.display_name) if def != null else owner_id + ".gd"


func _error_owners(socket: int) -> Dictionary:
	var v: ProgramValidation = null
	if socket == _socket and _validation != null:
		v = _validation
	else:
		var program: SkillProgram = _drafts[socket] if _drafts.has(socket) else _skills().stored_program(socket)
		if program == null or _skills().program_limits_for(socket) == null:
			return {}
		v = _skills().preview_program(socket, program, _routines())
	var out := {}
	for issue in v.issues:
		if issue["kind"] == "error":
			var rule: ProgramRule = issue["rule"]
			out[rule.owner if rule != null else ""] = true
	return out


func _set_output_alert(errors: int) -> void:
	if _output_frame == null:
		return
	var border := Color(SkillPanel.COL_WARN, 0.9) if errors > 0 else Color(panel.accent, 0.3)
	_output_frame.add_theme_stylebox_override("panel", _box_style(Color(0, 0, 0, 0.5), border))
	if errors > _last_errors:
		_out_tab = "errors"
	elif errors == 0:
		_out_tab = "output"
	_last_errors = errors
	_update_out_tabs(errors)


func _apply() -> void:
	if _routines_dirty():
		_skills().set_routines(_routines())
		_routine_draft.clear()
		if not _is_dirty(_socket):
			panel._notify("ROUTINES SAVED")
			_queue_rebuild()
			return
	var draft := _draft()
	if not draft.has_rules():
		_skills().set_program(_socket, null)
		_drafts.erase(_socket)
		panel._notify("PROGRAM REMOVED")
		return
	var result := _skills().set_program(_socket, draft.duplicate_program())
	if result.valid:
		_drafts.erase(_socket)
		panel._notify("PROGRAM APPLIED")
		if ScriptTips.enabled() and not ScriptTips.seen("applied"):
			_show_tip(ScriptTips.tip("applied"))
			_tip_hold = true
	else:
		panel._notify(result.errors[0].to_upper() if not result.errors.is_empty() else "INVALID PROGRAM")
	_queue_rebuild()


func _revert() -> void:
	_drafts.erase(_socket)
	_routine_draft.clear()
	_routine_rename = -1
	_selected = null
	_queue_rebuild()


func _clear_gem() -> void:
	if _owner == KERNEL_FILE:
		if Endings.kernel_editable(_kernel_qm()):
			_skills().kernel.clear()
			Coop.kernel_changed()
			panel._notify("KERNEL: LINES REMOVED")
			_queue_rebuild()
		return
	if _owner == ROUTINES_FILE:
		for rt in _routines():
			(rt as ProgramRoutine).body.effects.clear()
			(rt as ProgramRoutine).body.branches.clear()
		_selected = null
		_changed()
		return
	var draft := _draft()
	for r in draft.rules_of(_owner):
		draft.rules.erase(r)
	_selected = null
	_changed()


static func _file_name(display_name: String) -> String:
	var n := display_name.trim_suffix(" Support").trim_suffix(" Gem").strip_edges().to_lower()
	return n.replace(" ", "_") + ".gd"


static func _fn_name(c: ProgramCondition) -> String:
	return c.palette_name().to_lower().replace(" ", "_")


static func _fmt_param(c: ProgramCondition, p: String) -> String:
	if c.param_text(p) != "":
		return c.param_text(p)
	var v = c.get(p)
	if p == "fraction":
		return "%d%%" % roundi(float(v) * 100.0)
	if p == "radius":
		return "%.1fm" % float(v)
	return str(v)


static func _fmt_knob(knob: SkillKnob, steps: int, mult: float = 1.0) -> String:
	var v := absf(knob.value_for(steps) * mult)
	match knob.unit:
		"%":
			return "%d%%" % roundi(v * 100.0)
		"m":
			return "%.2fm" % v
		"deg":
			return "%d°" % roundi(v)
	if is_equal_approx(v, roundf(v)):
		return str(roundi(v))
	return "%.1f" % v


static func _fmt_per_count(knob: SkillKnob, steps: int, mult: float = 1.0) -> String:
	var v := absf(knob.value_for(steps) * mult) / ScriptRegisters.RAMP_PER_STEP
	match knob.unit:
		"%":
			var percent := v * 100.0
			return "%d%%" % roundi(percent) if is_equal_approx(percent, roundf(percent)) else "%.1f%%" % percent
		"m":
			return "%.2fm" % v
		"deg":
			return "%.1f°" % v
	return str(roundi(v)) if is_equal_approx(v, roundf(v)) else "%.1f" % v


static func _c(text: String, col: Color) -> String:
	return "[color=#%s]%s[/color]" % [col.to_html(false), text]


static func _esc(text: String) -> String:
	return text.replace("[", "[lb]")


func _box_style(bg: Color, border: Color) -> StyleBoxFlat:
	var sb := StyleBoxFlat.new()
	sb.bg_color = bg
	sb.border_color = border
	sb.set_border_width_all(1)
	sb.content_margin_left = 4
	sb.content_margin_right = 6
	sb.content_margin_top = 4
	sb.content_margin_bottom = 4
	return sb


func _shrinkable(l: Label) -> void:
	l.clip_text = true
	l.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS
	l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	l.custom_minimum_size.x = 1.0


func _k() -> float:
	return float(panel.script_font_size) / CODE_SIZE


func _sz(base: int) -> int:
	return roundi(base * _k())


func _clear(node: Node) -> void:
	for child in node.get_children():
		node.remove_child(child)
		child.queue_free()

func _bonus_sources() -> String:
	var owner_node := _skills().get_parent()
	var b := ScriptBudget.breakdown(owner_node.get("quests") as QuestManager)
	var parts: Array[String] = []
	if b["rank"] > 0.0:
		parts.append("rank +%s" % String.num(b["rank"], 1))
	if b["grades"] > 0.0:
		parts.append("exam grades +%s" % String.num(b["grades"], 1))
	if b["areas"] > 0:
		parts.append("%d area%s cleared +%s" % [b["areas"], "" if b["areas"] == 1 else "s", String.num(b["clears"], 1)])
	return ", ".join(parts)

func _step_sources() -> String:
	var b := ScriptBudget.breakdown(_skills().get_parent().get("quests") as QuestManager)
	var parts: Array[String] = []
	if b["distinctions"] > 0:
		parts.append("%d DISTINCTION%s +%d" % [b["distinctions"], "" if b["distinctions"] == 1 else "s",
				b["distinctions"] * ScriptBudget.STEPS_PER_DISTINCTION])
	if b["bosses"] > 0:
		parts.append("%d act boss%s +%d" % [b["bosses"], "" if b["bosses"] == 1 else "es",
				b["bosses"] * ScriptBudget.STEPS_PER_ACT_BOSS])
	return ", ".join(parts)
