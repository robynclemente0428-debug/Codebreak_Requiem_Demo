extends Node

const STRIP_TOP := 96.0
const GAP := 10.0
const HEADLINE_MIN := 230.0
const FEED_MIN := 215.0
const BASE_HEIGHT := 1080.0

const OPACITY := 0.85

const LOW := 0
const NORMAL := 10
const HIGH := 20
const URGENT := 30

var _strip: Array[Dictionary] = []
var _strip_bottom := -1.0
var _strip_aside := false
var _current: Dictionary = {}
var _queue: Array[Dictionary] = []
var _clock := 0.0
var _free_at := 0.0
const HANDOFF := 0.15
const PLACE_AFTER := 101
const FEED_AFTER := PLACE_AFTER + 1
var _tail_bottom := -1.0


func _ready() -> void:
	process_mode = Node.PROCESS_MODE_ALWAYS
	process_priority = PLACE_AFTER - 1


func _process(delta: float) -> void:
	if not get_tree().paused:
		_clock += delta
		if not _current.is_empty() and (_clock >= _current["until"] or not _alive(_current)):
			_release()
		_queue = _queue.filter(func(h: Dictionary) -> bool:
			return _alive(h) and (h["wait"] < 0.0 or _clock - h["asked"] <= h["wait"]))
		if _current.is_empty() and not _queue.is_empty() and _clock >= _free_at:
			_start(_queue.pop_front())
	_layout_strip(delta)


func screen_size() -> Vector2:
	return get_viewport().get_visible_rect().size


func fit(px: float) -> float:
	return roundf(px * minf(1.0, screen_size().y / BASE_HEIGHT))


func column_bottom() -> float:
	return screen_size().y - BOTTOM_RESERVE


const BOTTOM_RESERVE := 270.0


func _gap() -> float:
	return maxf(4.0, fit(GAP))


func add_to_strip(node: Control, order: int) -> void:
	for s in _strip:
		if s["node"] == node:
			s["order"] = order
			return
	_strip.append({"node": node, "order": order})
	_strip.sort_custom(func(a: Dictionary, b: Dictionary) -> bool: return a["order"] < b["order"])


func strip_bottom() -> float:
	return _strip_bottom


func _layout_strip(delta: float) -> void:
	_strip = _strip.filter(func(s: Dictionary) -> bool: return is_instance_valid(s["node"]))
	var gap := _gap()
	var y := fit(STRIP_TOP)
	var any := false
	for s in _strip:
		var node: Control = s["node"]
		if not node.is_inside_tree() or not _shown(node):
			continue
		if not is_equal_approx(node.global_position.y, y):
			node.global_position.y = y
		y += maxf(node.size.y, node.get_combined_minimum_size().y) + gap
		any = true
	var bottom := y - gap if any else -1.0
	_strip_aside = any and not _current.is_empty() \
			and maxf(fit(HEADLINE_MIN), bottom + gap * 2.0) + float(_current["height"]) > column_bottom()
	_strip_bottom = -1.0 if _strip_aside else bottom
	for s in _strip:
		var node: Control = s["node"]
		if _strip_aside:
			node.modulate.a = 0.0
		elif node.modulate.a < 1.0:
			node.modulate.a = minf(1.0, node.modulate.a + delta * 5.0)


static func _shown(node: CanvasItem) -> bool:
	if not node.is_visible_in_tree():
		return false
	var p := node.get_parent()
	while p != null:
		if p is CanvasLayer and not (p as CanvasLayer).visible:
			return false
		p = p.get_parent()
	return true


func headline(key: StringName, priority: int, seconds: float, show: Callable,
		hide := Callable(), wait := 8.0, keep := false) -> void:
	var h := {"key": key, "priority": priority, "seconds": seconds, "show": show, "hide": hide,
			"wait": wait, "keep": keep, "asked": _clock, "until": 0.0, "height": DEFAULT_HEIGHT}
	if not _current.is_empty() and _current["key"] == key:
		_start(h)
		return
	_queue = _queue.filter(func(q: Dictionary) -> bool: return q["key"] != key)
	if _current.is_empty() and _clock >= _free_at:
		_start(h)
		return
	if not _current.is_empty() and priority > int(_current["priority"]):
		_interrupt()
		_start(h)
		return
	var i := 0
	while i < _queue.size() and int(_queue[i]["priority"]) >= priority:
		i += 1
	_queue.insert(i, h)


const DEFAULT_HEIGHT := 60.0


func headline_done(key: StringName) -> void:
	if not _current.is_empty() and _current["key"] == key:
		_release()
	else:
		_queue = _queue.filter(func(q: Dictionary) -> bool: return q["key"] != key)


func set_headline_height(key: StringName, height: float) -> void:
	if not _current.is_empty() and _current["key"] == key \
			and not is_equal_approx(float(_current["height"]), height):
		_current["height"] = height
		_layout_strip(0.0)


func is_showing(key: StringName) -> bool:
	return not _current.is_empty() and _current["key"] == key


func headline_top() -> float:
	var y := fit(HEADLINE_MIN)
	return maxf(y, _strip_bottom + _gap() * 2.0) if _strip_bottom >= 0.0 else y


func headline_bottom() -> float:
	return -1.0 if _current.is_empty() else headline_top() + float(_current["height"])


func _release() -> void:
	_tail_bottom = headline_bottom()
	_current = {}
	_free_at = _clock + HANDOFF


func _start(h: Dictionary) -> void:
	_current = h
	h["until"] = _clock + float(h["seconds"])
	var show: Callable = h["show"]
	if show.is_valid():
		var got = show.call()
		if (got is float or got is int) and float(got) > 0.0:
			h["height"] = float(got)
	_layout_strip(0.0)


func _interrupt() -> void:
	var was := _current
	_current = {}
	var hide: Callable = was["hide"]
	if hide.is_valid():
		hide.call()
	if was["keep"] and _alive(was):
		was["asked"] = _clock
		_queue.push_front(was)


static func _alive(h: Dictionary) -> bool:
	var show: Callable = h["show"]
	return show.is_valid() and (show.get_object() == null or is_instance_valid(show.get_object()))


static func see_through(node: Node) -> void:
	var items: Array = node.find_children("*", "CanvasItem", true, false)
	if node is CanvasItem:
		items.append(node)
	for item in items:
		(item as CanvasItem).self_modulate.a = OPACITY


func feed_top() -> float:
	var gap := _gap()
	var y := fit(FEED_MIN)
	if _strip_bottom >= 0.0:
		y = maxf(y, _strip_bottom + gap * 2.0)
	var hb := headline_bottom()
	if hb < 0.0 and _clock < _free_at:
		hb = _tail_bottom
	if hb >= 0.0:
		y = maxf(y, hb + gap * 1.5)
	return y


func feed_room() -> float:
	return maxf(0.0, column_bottom() - feed_top())

func host() -> Error:
	var peer := ENetMultiplayerPeer.new()

		return err

@rpc("any_peer", "call_remote", "reliable")
func _register(version: int, info: Dictionary) -> void:
	if not is_host():
		return
	var id := multiplayer.get_remote_sender_id()
	var why := ""
	if version != PROTOCOL:
		why = "That game is on a different version."
	elif in_game:
		why = "That game has already started."
	elif roster.size() >= MAX_PLAYERS:
		why = "That game is full."
	elif str(info.get("class_id", "")) == "" or not info.get("loadout") is Dictionary:
		why = "That game couldn't read your character."
	if why != "":
		_refused.rpc_id(id, why)
		get_tree().create_timer(0.5).timeout.connect(func() -> void:
			if multiplayer.multiplayer_peer is ENetMultiplayerPeer and id in multiplayer.get_peers():
				(multiplayer.multiplayer_peer as ENetMultiplayerPeer).disconnect_peer(id))
		return
	roster[id] = {
		"name": str(info.get("name", "Player")).left(24), "class_id": str(info["class_id"]),
		"rank": str(info.get("rank", Ranks.LADDER[0]["name"])).left(16), "loadout": info["loadout"]}
	_set_roster.rpc(roster)
	_apply_roster()

func _tick_discovery(delta: float) -> void:
	if _announcer != null:
		_announce_timer -= delta
		if _announce_timer <= 0.0:
			_announce_timer = ANNOUNCE_EVERY
			_announce()
	if _listener == null:
		return
	var now := Time.get_ticks_msec() / 1000.0
	var before := _found_signature()
	while _listener.get_available_packet_count() > 0:
		var bytes := _listener.get_packet()
		var from := _listener.get_packet_ip()
		var info = JSON.parse_string(bytes.get_string_from_utf8())
		if not info is Dictionary or str(info.get("game", "")) != GAME_TAG or found.size() >= 32:
			continue
		var id := str(info.get("id", "")).left(48)
		if from.begins_with("127.") and found.has(id):
			from = found[id].address
		found[id] = {
			"name": str(info.get("name", "Host")).left(24), "class": str(info.get("class", "")).left(24),
			"address": from, "port": int(info.get("port", PORT)), "seen": now,
			"players": int(info.get("players", 1)), "max": int(info.get("max", MAX_PLAYERS)),
			"version": int(info.get("version", 0))}
	for id: String in found.keys():
		if now - found[id].seen > HOST_TIMEOUT:
			found.erase(id)
	if _found_signature() != before:
		hosts_changed.emit()


func _found_signature() -> String:
	var out := ""
	for id: String in found:
		out += "%s|%s|%d;" % [id, found[id].address, found[id].players]
	return out


func _announce() -> void:
	var me: Dictionary = roster.get(1, {})
	var packet := JSON.stringify({
		"game": GAME_TAG, "version": PROTOCOL, "id": _host_id, "port": PORT, "max": MAX_PLAYERS,
		"players": roster.size(), "name": me.get("name", "Host"),
		"class": str(me.get("class_id", "")).capitalize(),
	}).to_utf8_buffer()
	var targets := PackedStringArray(["255.255.255.255", "127.0.0.1"])
	for a in lan_addresses():
		targets.append(a.substr(0, a.rfind(".") + 1) + "255")
	for t in targets:
		_announcer.set_dest_address(t, DISCOVERY_PORT)
		_announcer.put_packet(packet)


func lan_addresses() -> PackedStringArray:
	var out := PackedStringArray()
	for a in IP.get_local_addresses():
		if a.count(".") == 3 and not a.begins_with("127.") and not a.begins_with("169.254."):
			out.append(a)
	return out

func party_kernel(mine: Node, beyond_act5 := false) -> Dictionary:
	var lines: Array = []
	var merged: Array = []
	if mine != null and is_instance_valid(mine):
		lines.append_array(Array(mine.skills.kernel))
		merged.append_array(KernelScript.null_running(mine.get("quests")))
	if in_game:
		for id: int in _players:
			var p = _players[id]
			if not is_instance_valid(p) or not (p as Player).visible:
				continue
			var loadout: Dictionary = (roster.get(id, {}) as Dictionary).get("loadout", {})
			if beyond_act5 and not bool(loadout.get("kernel_everywhere", false)):
				continue
			lines.append_array(loadout.get("kernel", []))
			merged.append_array(loadout.get("kernel_merged", []))
	var out := {"lines": [], "merged": []}
	for id in lines:
		if KernelScript.LINES.has(str(id)) and not str(id) in out.lines:
			out.lines.append(str(id))
	for id in merged:
		if KernelScript.NULL_LINES.has(str(id)) and not str(id) in out.merged:
			out.merged.append(str(id))
	return out


func party_answers() -> Array:

func loadout_of(mine: Player) -> Dictionary:
	var out := mine.loadout_save()
	out.erase("inventory")
	out["script_routines"] = mine.skills.routines_to_list()
	out["kernel"] = Array(mine.skills.kernel)
	out["kernel_merged"] = KernelScript.null_running(mine.quests)
	out["kernel_everywhere"] = Endings.runs_everywhere(mine.quests)
	out["kernel_answers"] = KernelScript.NULL_LINES.keys().map(func(id: String) -> String:
		return KernelScript.answer_of(mine.quests, id))
	out["limits"] = {"rank": mine.skills.rank(), "points": mine.skills.bonus_points(),
			"steps": mine.skills.bonus_steps()}
	out["class_id"] = String(mine.class_data.id) if mine.class_data != null else ""
	return out


@rpc("any_peer", "call_remote", "reliable")
func _p_loadout(loadout: Dictionary) -> void:
	var id := multiplayer.get_remote_sender_id()
	if not roster.has(id) or var_to_bytes(loadout).size() > MAX_LOADOUT_BYTES:
		return
	var info: Dictionary = roster[id]
	if str(loadout.get("class_id", "")) != str(info.get("class_id", "")):
		return
	if (info.get("loadout", {}) as Dictionary).hash() == loadout.hash() and _players.has(id):
		return
	info["loadout"] = loadout
	if not in_game or _am == null or not is_instance_valid(_am):
		return
	var old := _players.get(id) as Player
	var goal: Dictionary = _player_goal.get(id, {})
	var life := -1.0
	if old != null and is_instance_valid(old):
		life = old.health.current_health
		old.remove_from_group("players")
		old.queue_free()
	_players.erase(id)
	_spawn_remote_player(id)
	var p := _players.get(id) as Player
	if p == null:
		return
	if not goal.is_empty():
		_player_goal[id] = goal
		p.global_position = goal.pos
	if life >= 0.0:
		p.health.apply_remote(minf(life, p.health.current_health), false)


func _spawn_remote_player(id: int) -> void:
	var info: Dictionary = roster[id]
	var data: Dictionary = (info.get("loadout", {}) as Dictionary).duplicate(true)
	data["class_id"] = info.get("class_id", "")
	if not ResourceLoader.exists("res://data/classes/%s.tres" % data["class_id"]):
		push_warning("Coop: player %d has an unknown class '%s'" % [id, data["class_id"]])
		return
	var p := (load(PLAYER_SCENE) as PackedScene).instantiate() as Player
	p.name = "RemotePlayer%d" % id
	p.is_remote = true
	p.coop_data = data
	_am.player.add_sibling(p)
	p.global_position = _am.player.global_position
	var rig: Node3D = p.camera_rig
	rig.process_mode = Node.PROCESS_MODE_DISABLED
	rig.visible = false
	rig.camera.current = false
	for child in ["EnemyHover", "PickupController"]:
		p.get_node(child).process_mode = Node.PROCESS_MODE_DISABLED
	if GameEvents.happened.is_connected(p.quests._on_event):
		GameEvents.happened.disconnect(p.quests._on_event)
	p.health.set_process(false)
	p.health.remote_damage = func(amount: float, ignore_armour: bool) -> void:
		if is_host() and hit_source == Hit.ENEMY:
			_p_hit.rpc_id(id, amount, ignore_armour)
	_players[id] = p
	_player_goal[id] = {"area": 0, "pos": p.global_position, "rot": 0.0, "vel": Vector3.ZERO, "walking": false}
	_add_ring(p, player_colour(id))
	_show_remote(p, false)


func _add_ring(p: Player, colour: Color) -> void:
	var mat := StandardMaterial3D.new()
	mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	mat.albedo_color = Color(colour, 0.85)
	var mesh := TorusMesh.new()
	mesh.inner_radius = 0.52
	mesh.outer_radius = 0.68
	mesh.rings = 32
	mesh.ring_segments = 4
	mesh.material = mat
	var ring := MeshInstance3D.new()
	ring.name = "CoopRing"
	ring.mesh = mesh
	ring.scale.y = 0.1
	ring.position.y = 0.06
	ring.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	p.add_child(ring)

func _host_travel(scene_path: String, spawn_id: StringName, dungeon_level: int, instance_id: int,
		asker: int) -> void:
	if travelling or Time.get_ticks_msec() - _last_travel_msec < 1500:
		return
	var busy := who_is_answering()
	if busy != "":
		var text := "WAITING FOR %s\nThey're answering the Null's pull request." % busy.to_upper()
		if asker == my_id():
			QuestItems.toast(get_tree(), text, Color(1.0, 0.82, 0.30))
		elif asker in multiplayer.get_peers():
			_travel_held.rpc_id(asker, text)
		return
	if not scene_path.begins_with("res://") and not scene_path.begins_with("uid://"):
		return
	if not ResourceLoader.exists(scene_path):
		return
	_last_travel_msec = Time.get_ticks_msec()
	var inst_id := 0
	var rng_seed := 0
	var killed: Array = []
	if dungeon_level > 0:
		var inst := WorldState.prepare_dungeon(scene_path, dungeon_level, instance_id)
		inst_id = inst.id
		rng_seed = inst.rng_seed
		killed = inst.killed.keys()
	_travel.rpc(scene_path, spawn_id, clampi(dungeon_level, 0, 999), inst_id, rng_seed, killed, asker)

	if text.length() < 200:
		QuestItems.toast(get_tree(), text, Color(1.0, 0.82, 0.30))


@rpc("authority", "call_local", "reliable")
func _travel(scene_path: String, spawn_id: StringName, dungeon_level: int, inst_id: int,
		rng_seed: int, killed: Array, asker: int) -> void:
	_where = {"path": scene_path, "spawn": spawn_id, "level": dungeon_level, "inst": inst_id, "seed": rng_seed}
	_out_of_step = 0
	note("travel to %s (spawn %s, level %d, instance %d)" % [scene_path.get_file(), spawn_id, dungeon_level, inst_id])
	if _am == null or not is_instance_valid(_am):
		note("  ...before the game finished loading here: the area check will catch up")
		return
	_last_travel_msec = Time.get_ticks_msec()
	while _am.is_busy():
		await get_tree().process_frame
		if _am == null or not is_instance_valid(_am):
			return
	if dungeon_level > 0:
		if is_host():
			WorldState.prepare_dungeon(scene_path, dungeon_level, inst_id)
		else:
			WorldState.adopt(inst_id, scene_path, dungeon_level, rng_seed, killed)
	_set_visiting(scene_path, asker)
	travelling = true
	await _am.travel_now(scene_path, spawn_id)
	travelling = false


func _set_visiting(dest: String, asker: int) -> void:
	var was := visiting
	visiting = not _within_reach(dest, asker)
	if visiting != was:
		note("%s in %s" % ["VISITING (nothing unlocks)" if visiting else "no longer visiting", dest.get_file()])
	if not visiting:
		_told_visiting = false


func _within_reach(dest: String, asker: int) -> bool:
	var to := Waypoints.normalize(dest)
	var w := Waypoints.for_scene(to)
	if w != null and Waypoints.is_discovered(w):
		return true
	if to == Waypoints.normalize(SaveManager.last_town()):
		return true
	if _am != null and is_instance_valid(_am) and to == Waypoints.normalize(_am.start_area):
		return true
	if visiting:
		return false
	if asker == my_id():
		return true
	if w != null and Waypoints.is_convoy_stop(w) and Waypoints.route_open(w):
		return true
	var area: Node = _am.current_area if _am != null and is_instance_valid(_am) else null
	if area != null and is_instance_valid(area):
		for n in area.find_children("*", "Portal", true, false):
			var portal := n as Portal
			if portal != null and portal.visible and not portal.is_sealed() \
					and Waypoints.normalize(portal.target_area) == to:
				return true
	return false

const RESYNC_AFTER := 3
const RESYNC_TRIES := 3
var _where: Dictionary = {}
var _out_of_step := 0
var _resync_tries := 0
var _resync_path := ""
var _send_where := 0.0


@rpc("authority", "call_remote", "unreliable_ordered")
func _h_where(scene_path: String, spawn_id: StringName, dungeon_level: int, inst_id: int, rng_seed: int) -> void:
	if not in_game or _am == null or not is_instance_valid(_am):
		return
	if _am.is_busy() or travelling:
		_out_of_step = 0
		return
	var here := Waypoints.normalize(_am.current_scene_path()) if _am.current_area != null else ""
	if here == Waypoints.normalize(scene_path):
		_out_of_step = 0
		return
	_out_of_step += 1
	if _out_of_step < RESYNC_AFTER:
		return
	_out_of_step = 0
	if _resync_path != scene_path:
		_resync_path = scene_path
		_resync_tries = 0
	_resync_tries += 1
	if _resync_tries > RESYNC_TRIES:
		if _resync_tries == RESYNC_TRIES + 1:
			note("GIVING UP on %s: it doesn't load here" % scene_path)
			QuestItems.toast(get_tree(), "CAN'T LOAD THE HOST'S AREA\nBoth players need the same version of the game.",
					Color(1.0, 0.4, 0.4))
		return
	note("out of step: here '%s', the party is in '%s' -> going there (try %d)" % [here.get_file(), scene_path.get_file(), _resync_tries])
	if not ResourceLoader.exists(scene_path):
		note("  ...that scene doesn't exist in this build")
		return
	_where = {"path": scene_path, "spawn": spawn_id, "level": dungeon_level, "inst": inst_id, "seed": rng_seed}
	if dungeon_level > 0:
		WorldState.adopt(inst_id, scene_path, dungeon_level, rng_seed, [])
	_set_visiting(scene_path, 0)
	travelling = true
	await _am.travel_now(scene_path, spawn_id)
	travelling = false


func note(text: String) -> void:
	var line := "%s [%s] %s" % [Time.get_time_string_from_system(), "host" if is_host() else "guest", text]
	print("[Coop] ", line)
	var f := FileAccess.open("user://coop_log.txt", FileAccess.READ_WRITE if FileAccess.file_exists("user://coop_log.txt") else FileAccess.WRITE)
	if f != null:
		if f.get_length() > 400000:
			f.resize(0)
		f.seek_end()
		f.store_line(line)
		f.close()

func enemy_spawned(enemy: Node3D, def: EnemyDefinition, rarity: String, level: int, mods: Array) -> void:
	if not active or not in_game or not enemy is EnemyBase:
		return
	var e := enemy as EnemyBase
	if is_host():
		var id := _next_enemy
		_next_enemy += 1
		var mod_ids: Array = []
		for m in mods:
			mod_ids.append(m.id)
		e.set_meta(NET_ID, id)
		e.set_meta(NET_ENTRY, {
			"id": id, "def": def.resource_path, "def_id": def.id, "rarity": rarity, "level": level,
			"mods": mod_ids, "spawn": e.global_position})
		_enemies[id] = e
		e.health.health_changed.connect(func(current: float, _max: float) -> void:
			if active:
				_e_health.rpc(id, current, damage_by))
		e.ai.attack_requested.connect(func(_target: Node3D) -> void:
			if active:
				_e_attack.rpc(id))
		if _area_live:
			_announce_enemy.call_deferred(id)
		return
	e.ai.halt()
	e.is_remote = true
	_switch_off_logic(e)
	e.health.set_process(false)
	e.health.remote_damage = func(amount: float, ignore_armour: bool) -> void:
		if hit_source == Hit.PLAYER and e.has_meta(NET_ID):
			_hit_seq += 1
			_sent_seq = _hit_seq
			_sent_for = int(e.get_meta(NET_ID))
			var facts := hit_facts.to_dict() if hit_facts != null else {}
			if hit_facts != null and hit_facts.loop_id != "":
				_my_loops[hit_facts.loop_id] = true
			_e_hit.rpc_id(1, _sent_for, amount, ignore_armour, _hit_seq, facts)
	if not _adopting.is_empty():
		_enemies[int(_adopting.id)] = e
		e.set_meta(NET_ID, int(_adopting.id))
	else:
		_unmatched.append({"enemy": e, "def": def.id, "pos": e.global_position, "at": Time.get_ticks_msec()})


const COPY_KEEPS: Array[String] = ["enemy_silhouette.gd", "glitch_death.gd", "glitch_burst.gd", "lock_blast.gd"]
const ENEMY_LOGIC := "res://systems/enemies/"


func _is_enemy_logic(n: Node) -> bool:
	var s := n.get_script() as Script
	if s == null or (n is EnemyTrait and (n as EnemyTrait).coop_aware()):
		return false
	var path := s.resource_path
	return path.begins_with(ENEMY_LOGIC) and not path.get_file() in COPY_KEEPS


func _switch_off_logic(e: EnemyBase) -> void:
	for n in e.find_children("*", "", true, false):
		if _is_enemy_logic(n):
			n.process_mode = Node.PROCESS_MODE_DISABLED


func _on_node_added(n: Node) -> void:
	if not in_game or is_host() or not _is_enemy_logic(n):
		return
	var p := n.get_parent()
	while p != null and not p is EnemyBase:
		p = p.get_parent()
	if p != null and (p as EnemyBase).is_remote:
		n.process_mode = Node.PROCESS_MODE_DISABLED


func _entry_of(e: EnemyBase) -> Dictionary:
	var entry: Dictionary = (e.get_meta(NET_ENTRY) as Dictionary).duplicate()
	var area := _am.current_area if _am != null else null
	entry["parent"] = str(area.get_path_to(e.get_parent())) if area != null and area.is_ancestor_of(e) else ""
	entry["pos"] = e.global_position
	entry["rot"] = e.global_rotation.y
	entry["hp"] = e.health.current_health
	entry["area"] = _area_path
	entry["boost"] = e.get_meta(NET_BOOST, [])
	return entry


func _announce_enemy(id: int) -> void:
	var e := _enemies.get(id) as EnemyBase
	if e != null and is_instance_valid(e) and e.is_inside_tree() and e.health.current_health > 0.0:
		_e_spawned.rpc(_entry_of(e))


@rpc("any_peer", "call_remote", "reliable")
func _e_ready(scene_path: String) -> void:
	if not is_host():
		return
	var id := multiplayer.get_remote_sender_id()
	_waiting[id] = scene_path
	if _area_live and scene_path == _area_path:
		_send_enemy_list(id)


func _send_enemy_list(peer: int) -> void:
	_waiting.erase(peer)
	_tags_sent.clear()
	var entries: Array = []
	for id: int in _enemies.keys():
		var e = _enemies[id]
		if not is_instance_valid(e) or not (e as Node).is_inside_tree() or e.health.current_health <= 0.0:
			_enemies.erase(id)
			continue
		entries.append(_entry_of(e))
	_e_list.rpc_id(peer, _area_path, entries)
	for id: int in _loot_data.keys():
		if not is_instance_valid(_loot.get(id)):
			_loot_data.erase(id)
			_loot.erase(id)
	_l_list.rpc_id(peer, _area_path, _loot_data.values())


@rpc("authority", "call_remote", "reliable")
func _e_list(scene_path: String, entries: Array) -> void:
	if scene_path != _area_path or not _area_live:
		return
	for entry in entries:
		if entry is Dictionary:
			_adopt_or_spawn(entry)
	for u in _unmatched:
		if is_instance_valid(u.enemy):
			u.enemy.queue_free()
	_unmatched.clear()
	_list_done = true


@rpc("authority", "call_remote", "reliable")
func _e_spawned(entry: Dictionary) -> void:
	if _list_done and str(entry.get("area", "")) == _area_path:
		_adopt_or_spawn(entry)


func _adopt_or_spawn(entry: Dictionary) -> void:
	var id := int(entry.get("id", 0))
	var spawn: Vector3 = entry.get("spawn", Vector3.ZERO)
	var pos: Vector3 = entry.get("pos", spawn)
	if id <= 0 or _enemies.has(id) or not pos.is_finite():
		return
	var best := -1
	var best_d := MATCH_RADIUS
	for i in _unmatched.size():
		var u: Dictionary = _unmatched[i]
		if not is_instance_valid(u.enemy) or u.def != str(entry.get("def_id", "")):
			continue
		var d: float = (u.pos as Vector3).distance_to(spawn)
		if d < best_d:
			best_d = d
			best = i
	var e: EnemyBase = null
	if best >= 0:
		e = _unmatched[best].enemy
		_unmatched.remove_at(best)
		e.set_meta(NET_ID, id)
		_enemies[id] = e
	else:
		e = _spawn_from(entry)
		var boosts = entry.get("boost", [])
		if e != null and boosts is Array and boosts.size() <= 8:
			for b in boosts:
				if b is Array and b.size() == 6:
					EnemySpawner.empower(e, float(b[0]), float(b[1]), float(b[2]), str(b[3]), float(b[4]), float(b[5]))
	if e == null:
		return
	if e.global_position.distance_to(pos) > 0.1:
		e.global_position = pos
		e.global_rotation.y = float(entry.get("rot", 0.0))
	var hp := float(entry.get("hp", -1.0))
	if hp >= 0.0 and not is_equal_approx(hp, e.health.current_health):
		e.health.apply_remote(hp, false)


func _spawn_from(entry: Dictionary) -> EnemyBase:
	var path := str(entry.get("def", ""))
	if _am == null or _am.current_area == null or not path.begins_with("res://") or not ResourceLoader.exists(path):
		return null
	var def := load(path) as EnemyDefinition
	if def == null:
		return null
	var mods: Array = []
	for mod_id in entry.get("mods", []):
		for m in ModifierRegistry.all_modifiers:
			if m.id == str(mod_id):
				mods.append(m)
				break
	var area := _am.current_area
	var parent := area.get_node_or_null(str(entry.get("parent", ""))) if str(entry.get("parent", "")) != "" else null
	if parent == null:
		parent = area.get_node_or_null("Enemies")
	if parent == null:
		parent = area
	_adopting = entry
	var e := EnemySpawner.spawn_enemy(def, str(entry.get("rarity", "normal")), parent,
			entry.get("pos", Vector3.ZERO), clampi(int(entry.get("level", 1)), 1, 999), mods) as EnemyBase
	_adopting = {}
	var numbers := get_tree().get_first_node_in_group("damage_numbers") as DamageNumbers
	if e != null and numbers != null:
		numbers.track(e.health)
	return e


@rpc("any_peer", "call_remote", "reliable")
func _e_hit(id: int, amount: float, ignore_armour: bool, seq: int, facts: Dictionary) -> void:
	var e := _enemies.get(id) as EnemyBase
	var from := multiplayer.get_remote_sender_id()
	if not is_host() or e == null or not is_instance_valid(e) or not (amount > 0.0 and amount < MAX_DAMAGE) \
			or not roster.has(from) or var_to_bytes(facts).size() > 4096:
		return
	var info := HitInfo.new()
	info.target = e
	info.position = e.global_position
	info.damage = amount
	if not facts.is_empty():
		info.facts = HitFacts.from_dict(facts, _players.get(from))
		if info.facts.loop_id != "":
			HitFacts.remote_loop(info.facts.loop_id, true)
	var was_alive := e.health.current_health > 0.0
	var traits: Array = e.traits.filter(func(t) -> bool: return is_instance_valid(t))
	hit_source = Hit.ENEMY
	damage_by = from
	for t: EnemyTrait in traits:
		t.modify_incoming(info)
	damage_by = 0
	hit_source = Hit.NONE
	if info.cancelled:
		return
	if info.absorbed:
		if was_alive:
			info.landed = true
			_after_hit(traits, info)
		return
	damage_by = from
	info.dealt = e.health.take_damage(info.damage, ignore_armour)
	damage_by = 0
	if info.dealt <= 0.0:
		return
	info.landed = true
	info.killed = e.health.current_health <= 0.0
	_after_hit(traits, info)
	_e_landed.rpc_id(from, seq, info.dealt, was_alive and info.killed)


func _after_hit(traits: Array, info: HitInfo) -> void:
	hit_source = Hit.ENEMY
	for t: EnemyTrait in traits:
		if is_instance_valid(t):
			t.after_hit(info)
	hit_source = Hit.NONE

var _my_hits: Dictionary = {}
var _hit_seq := 0
var _sent_seq := 0
var _sent_for := 0
const HIT_KEEP_MSEC := 8000


func my_hit(target: Node, cast: SkillCast, info: HitInfo) -> void:
	if not in_game or is_host() or not target.has_meta(NET_ID):
		return
	var seq := _sent_seq
	_sent_seq = 0
	if seq == 0 or _sent_for != int(target.get_meta(NET_ID)):
		return
	var now := Time.get_ticks_msec()
	if _my_hits.size() > 128:
		for s: int in _my_hits.keys():
			if now - int(_my_hits[s][3]) > HIT_KEEP_MSEC or not is_instance_valid(_my_hits[s][1]):
				_my_hits.erase(s)
	_my_hits[seq] = [cast, target, info, now]


@rpc("authority", "call_remote", "reliable")
func _e_landed(seq: int, dealt: float, killed: bool) -> void:
	var rec = _my_hits.get(seq)
	_my_hits.erase(seq)
	if not rec is Array or not is_finite(dealt):
		return
	var cast := rec[0] as SkillCast
	var info := rec[2] as HitInfo
	if cast == null or info == null or not cast.is_valid():
		return
	info.dealt = clampf(dealt, 0.0, MAX_DAMAGE)
	var behavior := cast.behavior if cast.behavior != null else cast.skill.get_behavior()
	if behavior != null:
		behavior.landing_rewards(cast, info)
	if not killed or not is_instance_valid(rec[1]):
		return
	info.killed = true
	for s in cast.skill.support_behaviors:
		s.on_kill(cast, rec[1], info)

func _populator() -> DungeonPopulator:
	var area: Node = _am.current_area if _am != null and is_instance_valid(_am) else null
	if area == null or not is_instance_valid(area):
		return null
	for n in area.find_children("*", "DungeonPopulator", true, false):
		return n as DungeonPopulator
	return null


func _make_loot(entry: Dictionary) -> void:
	var id := int(entry.get("id", 0))
	var pos = entry.get("pos")
	var populator := _populator()
	if id <= 0 or _loot.has(id) or not pos is Vector3 or not (pos as Vector3).is_finite() \
			or not entry.get("item") is Dictionary or populator == null:
		return
	var item := ItemInstance.from_dict(entry["item"])
	if item == null:
		return
	var pickup := populator.place_item(item, pos)
	pickup.set_meta(LOOT_ID, id)
	_loot[id] = pickup

func _process(delta: float) -> void:
	_tick_discovery(delta)
	if in_game and _loadout_send_in > 0.0 and _am != null and is_instance_valid(_am):
		_loadout_send_in -= delta
		if _loadout_send_in <= 0.0 and is_instance_valid(_am.player):
			var loadout := _live_loadout()
			if roster.has(my_id()):
				roster[my_id()]["loadout"] = loadout
			_p_loadout.rpc(loadout)
	if in_game and not is_host():
		_check_my_loops(delta)
		_check_my_refs(delta)
	if is_host() and in_game and _area_live:
		_send_tags(delta)
	if not in_game or _list_done == false or _unmatched.is_empty():
		return
	var now := Time.get_ticks_msec()
	for i in range(_unmatched.size() - 1, -1, -1):
		var u: Dictionary = _unmatched[i]
		if not is_instance_valid(u.enemy):
			_unmatched.remove_at(i)
		elif now - int(u.at) > UNMATCHED_GRACE * 1000.0:
			u.enemy.queue_free()
			_unmatched.remove_at(i)


func _physics_process(delta: float) -> void:
	if not in_game or _am == null or not is_instance_valid(_am):
		return
	var ease_in := 1.0 - exp(-delta * 15.0)
	var here := _area_path.hash() if _area_live else -1

	for id: int in _player_goal:
		var p := _players.get(id) as Player
		if p == null or not is_instance_valid(p) or not p.is_inside_tree():
			continue
		var goal: Dictionary = _player_goal[id]
		var same_area: bool = goal.area == here and goal.area != 0
		if same_area != p.visible:
			_show_remote(p, same_area)
			p.global_position = goal.pos
		if not same_area or p.health.current_health <= 0.0:
			continue
		if p.global_position.distance_to(goal.pos) > SNAP_DISTANCE:
			p.global_position = goal.pos
		else:
			p.global_position = p.global_position.lerp(goal.pos, ease_in)
		p.global_rotation.y = lerp_angle(p.global_rotation.y, goal.rot, ease_in)
		p.velocity = goal.vel
		if p.walking != goal.walking:
			p.walking = goal.walking
			p.walking_changed.emit(p.walking)

	for id: int in _enemy_goal.keys():
		var e := _enemies.get(id) as EnemyBase
		if e == null or not is_instance_valid(e) or not e.is_inside_tree() or e.health.current_health <= 0.0:
			_enemy_goal.erase(id)
			continue
		var goal: Dictionary = _enemy_goal[id]
		var gap: Vector3 = goal.pos - e.global_position
		if gap.length() > SNAP_DISTANCE:
			e.global_position = goal.pos
		else:
			e.velocity = gap * 12.0
			e.move_and_slide()
		e.global_rotation.y = lerp_angle(e.global_rotation.y, goal.rot, ease_in)
		if gap.length() < 0.03 and absf(angle_difference(e.global_rotation.y, goal.rot)) < 0.02:
			_enemy_goal.erase(id)

	_send_player -= delta
	if _send_player <= 0.0:
		_send_player = SEND_PLAYER_EVERY
		var mine := _am.player
		if is_instance_valid(mine):
			_p_state.rpc(here, mine.global_position, mine.global_rotation.y, mine.velocity, mine.walking,
					mine.health.is_shielded(), answering)
	if not is_host() or not _area_live:
		return
	_send_where -= delta
	if _send_where <= 0.0 and not _where.is_empty() and not travelling and not _am.is_busy():
		_send_where = 1.0
		_h_where.rpc(str(_where.path), StringName(_where.spawn), int(_where.level), int(_where.inst), int(_where.seed))
	_send_enemies -= delta
	if _send_enemies > 0.0:
		return
	_send_enemies = SEND_ENEMIES_EVERY
	var ids := PackedInt32Array()
	var positions := PackedVector3Array()
	var rotations := PackedFloat32Array()
	var states := PackedByteArray()
	for id: int in _enemies.keys():
		var e = _enemies[id]
		if not is_instance_valid(e) or e.health.current_health <= 0.0:
			_enemies.erase(id)
			_enemy_sent.erase(id)
			continue
		if not (e as Node).is_inside_tree():
			continue
		if e.process_mode == Node.PROCESS_MODE_DISABLED:
			continue
		var now: Array = [e.global_position, e.global_rotation.y, int(e.ai.state)]
		var last: Array = _enemy_sent.get(id, [])
		if not last.is_empty() and (last[0] as Vector3).distance_to(now[0]) < 0.02 \
				and absf(float(last[1]) - float(now[1])) < 0.01 and last[2] == now[2]:
			continue
		_enemy_sent[id] = now
		ids.append(id)
		positions.append(now[0])
		rotations.append(now[1])
		states.append(now[2])
	for start in range(0, ids.size(), ENEMIES_PER_PACKET):
		var end := mini(start + ENEMIES_PER_PACKET, ids.size())
		_e_state.rpc(ids.slice(start, end), positions.slice(start, end), rotations.slice(start, end),
				states.slice(start, end))

	var inventory: Array = []

func save_slot(slot: int, data: Dictionary) -> bool:
	var tmp := _path(slot) + '.tmp'
	var f := FileAccess.open(tmp, FileAccess.WRITE)
	if f == null:
		push_error('Could not write save: %s' % error_string(FileAccess.get_open_error()))
		return false
	f.store_string(JSON.stringify(data, '\t'))
	f.close()
	if DirAccess.rename_absolute(tmp, _path(slot)) != OK:
		push_error('Could not finalize save for slot %d' % slot)
		return false
	return true


func load_slot(slot: int) -> Dictionary:
	if not slot_exists(slot):
		return {}
	var f := FileAccess.open(_path(slot), FileAccess.READ)
	if f == null:
		return {}
	var parsed = JSON.parse_string(f.get_as_text())
	f.close()
	if typeof(parsed) != TYPE_DICTIONARY:
		push_warning('Slot %d is corrupted' % slot)
		return {}
	return _migrate(parsed)


func _migrate(data: Dictionary) -> Dictionary:
	return data


func delete_slot(slot: int) -> void:
	if slot in party:
		return
	if slot_exists(slot):
		DirAccess.remove_absolute(_path(slot))


func swap_slots(a: int, b: int) -> bool:
	if a == b or a < 0 or b < 0 or a >= MAX_SLOTS or b >= MAX_SLOTS:
		return false
	if a in party or b in party:
		return false
	var has_a := slot_exists(a)
	var has_b := slot_exists(b)
	if not has_a and not has_b:
		return false
	var tmp := SAVE_DIR + 'slot_swap.tmp'
	if FileAccess.file_exists(tmp):
		DirAccess.remove_absolute(tmp)
	if has_a and DirAccess.rename_absolute(_path(a), tmp) != OK:
		push_error('Could not move slot %d' % a)
		return false
	if has_b and DirAccess.rename_absolute(_path(b), _path(a)) != OK:
		push_error('Could not move slot %d' % b)
		if has_a:
			DirAccess.rename_absolute(tmp, _path(a))
		return false
	if has_a and DirAccess.rename_absolute(tmp, _path(b)) != OK:
		push_error('Could not finish moving slot %d (it is in %s)' % [a, tmp])
		return false
	return true


func begin_session(count: int = 1) -> void:
	players_needed = clampi(count, 1, MAX_PLAYERS)
	party.clear()
	_loaded.clear()
	creating_slot = -1


func add_to_party(slot: int) -> bool:
	if slot in party or party.size() >= players_needed:
		return false
	var data := load_slot(slot)
	if data.is_empty():
		return false
	party.append(slot)
	_loaded[slot] = data
	return true


func remove_last_from_party() -> void:
	if party.is_empty():
		return
	_loaded.erase(party.pop_back())


func party_full() -> bool:
	return party.size() == players_needed


func party_seat_of(slot: int) -> int:
	return party.find(slot)


func get_data(slot: int) -> Dictionary:
	return _loaded.get(slot, {})


func last_town() -> String:
	if party.is_empty():
		return ""
	var path := str(get_data(party[0]).get(LAST_TOWN_KEY, ""))
	if path.is_empty() or not ResourceLoader.exists(path):
		return ""
	return path


func start_game() -> void:
	get_tree().change_scene_to_file(GAME_SCENE)


func save_party() -> void:
	for node in get_tree().get_nodes_in_group('players'):
		if node.has_method('write_save'):
			node.write_save()
	for slot in party:
		save_slot(slot, _loaded[slot])


func end_session() -> void:
	party.clear()
	_loaded.clear()
	creating_slot = -1

func _size_window() -> void:
	var win := get_window()
	if DisplayServer.window_get_mode() != DisplayServer.WINDOW_MODE_WINDOWED:
		return
	var usable := DisplayServer.screen_get_usable_rect(win.current_screen)
	var want := parse_resolution(str(get_value("display", "resolution")))
	var deco := win.get_size_with_decorations() - win.size
	deco = Vector2i(maxi(deco.x, 0), maxi(deco.y, 0))
	var size := Vector2i(mini(want.x, usable.size.x - deco.x), mini(want.y, usable.size.y - deco.y))
	win.size = size
	var corner := usable.position + (usable.size - (size + deco)) / 2
	win.position = corner + Vector2i(deco.x / 2, deco.y - deco.x / 2)

func _apply_cursor() -> void:
	var themed: bool = get_value("display", "cursor") == "themed"
	var size := clampi(int(get_value("display", "cursor_size")), 16, 128)
	for shape in CURSORS:
		if not themed:
			Input.set_custom_mouse_cursor(null, shape)
			continue
		var tex := load(CURSOR_DIR + CURSORS[shape][0]) as Texture2D
		if tex == null:
			continue
		var img := tex.get_image()
		if img.is_compressed():
			img.decompress()
		img.resize(size, size, Image.INTERPOLATE_LANCZOS)
		var hotspot: Vector2 = CURSORS[shape][1] * size / 64.0
		Input.set_custom_mouse_cursor(ImageTexture.create_from_image(img), shape, hotspot)


func _apply_audio() -> void:
	for key in AUDIO_BUSES:
		var bus := AudioServer.get_bus_index(AUDIO_BUSES[key])
		if bus < 0:
			continue
		var v := clampf(float(get_value("audio", key)), 0.0, 1.0)
		AudioServer.set_bus_volume_db(bus, linear_to_db(maxf(v, 0.0001)))
		var muted := v <= 0.001
		if key == "master" and not _focused and get_value("audio", "mute_in_background"):
			muted = true
		AudioServer.set_bus_mute(bus, muted)

func _load() -> void:
	var cfg := ConfigFile.new()
	if cfg.load(SAVE_PATH) != OK:
		return
	for section in DEFAULTS:
		for key in DEFAULTS[section]:
			if cfg.has_section_key(section, key):
				var v = cfg.get_value(section, key)
				if typeof(v) == typeof(DEFAULTS[section][key]) \
						or (DEFAULTS[section][key] is float and v is int):
					if not _values.has(section):
						_values[section] = {}
					_values[section][key] = v


func _queue_save() -> void:
	if _save_queued:
		return
	_save_queued = true
	get_tree().create_timer(0.5, true).timeout.connect(_save)


func _save() -> void:
	_save_queued = false
	var cfg := ConfigFile.new()
	for section in DEFAULTS:
		for key in DEFAULTS[section]:
			cfg.set_value(section, key, get_value(section, key))
	cfg.save(SAVE_PATH)

func play_music(path: String) -> void:
	if path == _music_path:
		return
	_music_path = path
	var old := _music[_music_on]
	_music_on = 1 - _music_on
	var new := _music[_music_on]
	new.stream = _load(path)
	new.volume_db = -40.0
	if new.stream:
		new.play()
	var t := create_tween().set_parallel(true)
	t.tween_property(new, "volume_db", MUSIC_DB, FADE_SEC)
	t.tween_property(old, "volume_db", -80.0, FADE_SEC)
	t.chain().tween_callback(old.stop)


func _on_node_added(node: Node) -> void:
	if node is HealthComponent:
		var hc := node as HealthComponent
		if not hc.damage_taken.is_connected(_on_damage_taken):
			hc.damage_taken.connect(_on_damage_taken.bind(hc))
			hc.damage_shown.connect(_on_damage_taken.bind(hc))
		return
	if node is BaseButton and not node.has_meta("no_click_sound"):
		var b := node as BaseButton
		if not b.pressed.is_connected(_on_button_pressed):
			b.pressed.connect(_on_button_pressed.bind(b))


func _on_button_pressed(b: BaseButton) -> void:
	if not is_instance_valid(b) or b.has_meta("no_click_sound"):
		return
	var text := ""
	if b is Button:
		text = (b as Button).text.to_upper()
	var path := CLICK
	if START_WORDS.any(func(w: String) -> bool: return w in text):
		path = CLICK_START
	elif EXIT_WORDS.any(func(w: String) -> bool: return w in text) or text.strip_edges() == "X":
		path = CLICK_EXIT
	play_ui(path)


func play_ui(path: String, volume_db := CLICK_DB) -> void:
	_ui.stream = _load(path)
	_ui.volume_db = volume_db
	_ui.play()


func play_skill(key: String, source: Node3D, hit_in: float) -> void:
	if not SKILL_SOUNDS.has(key) or not is_instance_valid(source):
		return
	var entry: Array = SKILL_SOUNDS[key]
	var lead: float = entry[1]
	var wait := 0.0 if lead < 0.0 else maxf(hit_in - lead, 0.0)
	if wait > 0.0:
		await get_tree().create_timer(wait, false).timeout
		if not is_instance_valid(source):
			return
	if _is_own_player(source):
		_play_flat(entry[0], entry[2])
	else:
		_play_at(source, entry[0], entry[2])


func play_gunshot(shooter: Node3D) -> void:
	if not is_instance_valid(shooter):
		return
	var gun := DEFAULT_GUN
	var eq = shooter.get("equipment")
	if eq != null and eq.has_method("get_equipped"):
		var weapon = eq.get_equipped(EquipmentSlots.Slot.WEAPON)
		if weapon != null and weapon.definition.weapon_type != null \
				and GUN_SOUNDS.has(weapon.definition.weapon_type.id):
			gun = weapon.definition.weapon_type.id
	var entry: Array = GUN_SOUNDS[gun]
	var pitch := randf_range(1.0 - GUN_PITCH_JITTER, 1.0 + GUN_PITCH_JITTER)
	if _is_own_player(shooter):
		_play_flat(entry[0], entry[1], pitch)
	else:
		_play_at(shooter, entry[0], entry[1], pitch)


func play_blast_impact(world: Node, at: Vector3) -> void:
	play_at_point(world, at, BLAST_IMPACT, BLAST_IMPACT_DB, randf_range(0.94, 1.06))


func play_at_point(world: Node, at: Vector3, path: String, volume_db: float, pitch := 1.0) -> void:
	if _impact_voices >= MAX_IMPACT_VOICES or world == null or not is_instance_valid(world) \
			or not world.is_inside_tree():
		return
	var p3 := AudioStreamPlayer3D.new()
	p3.bus = _bus("Effects")
	p3.stream = _load(path)
	if p3.stream == null:
		p3.free()
		return
	p3.volume_db = volume_db
	p3.pitch_scale = pitch
	p3.unit_size = 8.0
	p3.max_distance = 40.0
	world.add_child(p3)
	p3.global_position = at
	_impact_voices += 1
	var done := func() -> void:
		if is_instance_valid(p3) and not p3.is_queued_for_deletion():
			_impact_voices -= 1
			p3.queue_free()
	p3.finished.connect(done)
	p3.tree_exiting.connect(func() -> void:
		if not p3.is_queued_for_deletion():
			_impact_voices -= 1)
	p3.play()


func _on_damage_taken(amount: float, hc: HealthComponent) -> void:
	if amount <= 0.0 or not is_instance_valid(hc):
		return
	var target := hc.get_parent() as Node3D
	if target == null or target.is_in_group(&"breakables"):
		return
	var now := Time.get_ticks_msec()
	var key := target.get_instance_id()
	if now - int(_last_hurt.get(key, -100000)) < int(HURT_COOLDOWN * 1000.0):
		return
	_last_hurt[key] = now
	if _is_own_player(target):
		_play_flat(TAKE_DAMAGE, PLAYER_HURT_DB, randf_range(0.95, 1.05), TAKE_DAMAGE_LENGTH)
	else:
		_play_at(target, TAKE_DAMAGE, ENEMY_HURT_DB, randf_range(1.05, 1.25), TAKE_DAMAGE_LENGTH)


func _is_own_player(node: Node) -> bool:
	return node.is_in_group("players") and node.get("is_remote") != true


func _play_flat(path: String, volume_db: float, pitch := 1.0, max_len := 0.0) -> void:
	var p := _player("Effects")
	p.stream = _load(path)
	p.volume_db = volume_db
	p.pitch_scale = pitch
	p.finished.connect(p.queue_free)
	p.play()
	if max_len > 0.0:
		get_tree().create_timer(max_len, false).timeout.connect(p.queue_free)


func _play_at(source: Node3D, path: String, volume_db: float, pitch := 1.0, max_len := 0.0) -> void:
	if _enemy_voices >= MAX_ENEMY_VOICES or not is_instance_valid(source):
		return
	var p3 := AudioStreamPlayer3D.new()
	p3.bus = _bus("Effects")
	p3.stream = _load(path)
	p3.volume_db = volume_db
	p3.pitch_scale = pitch
	p3.unit_size = 6.0
	p3.max_distance = 30.0
	source.add_child(p3)
	_enemy_voices += 1
	p3.finished.connect(_end_voice.bind(p3))
	p3.tree_exiting.connect(_voice_removed.bind(p3))
	if max_len > 0.0:
		get_tree().create_timer(max_len, false).timeout.connect(_end_voice.bind(p3))
	p3.play()


func _end_voice(p3) -> void:
	if is_instance_valid(p3) and not p3.is_queued_for_deletion():
		_enemy_voices -= 1
		p3.queue_free()


func _voice_removed(p3) -> void:
	if is_instance_valid(p3) and not p3.is_queued_for_deletion():
		_enemy_voices -= 1


func _update_footsteps() -> void:
	var walking := false
	if not get_tree().paused:
		var p := get_tree().get_first_node_in_group("players")
		if p != null and p.get("walking") == true and p is CharacterBody3D and p.is_on_floor():
			var h = p.get("health")
			walking = h == null or h.current_health > 0.0
	if walking:
		if not _steps.playing:
			_steps.play()
		_steps.stream_paused = false
	elif _steps.playing:
		_steps.stream_paused = true


func _player(bus_name: String) -> AudioStreamPlayer:
	var p := AudioStreamPlayer.new()
	p.bus = _bus(bus_name)
	add_child(p)
	return p


func _bus(bus_name: String) -> StringName:
	return StringName(bus_name) if AudioServer.get_bus_index(bus_name) >= 0 else &"Master"


func _load(path: String) -> AudioStream:
	if not _cache.has(path):
		_cache[path] = load(path) if ResourceLoader.exists(path) else null
		if _cache[path] == null:
			push_warning("SoundManager: missing sound %s" % path)
	return _cache[path]

				"world": hovered, "world_height": LOOT_ARROW_HEIGHT}
		return {"text": "Left-click to pick it up",
			"sub": "It goes into your backpack. Its stats are in the card next to the mouse: hover gear there later and hold CTRL to compare it with what you wear.",
			"world": hovered, "world_height": LOOT_ARROW_HEIGHT}
	if Input.is_action_pressed(PickupController.PICKUP_ACTION):
		return {"text": "Keep holding [%s] and point at the drop" % key,
			"sub": "Its name lights up and its stats show when the mouse is on it.",
			"world": drop, "world_height": LOOT_ARROW_HEIGHT}
	var labels := get_tree().get_first_node_in_group("loot_labels")
	var names_shown: bool = labels != null and bool(labels.get("always_show"))
	return {"text": "Loot dropped: hold [%s] and point at it" % key,
		"sub": ("Names of drops are on the ground already. " if names_shown
				else "Holding [%s] shows the names of the drops around you. " % key)
			+ "Gear, gems and items for your build come from enemies and chests.",
		"world": drop, "world_height": LOOT_ARROW_HEIGHT}


func _nearest_drop(player: Node3D) -> LootPickup:
	var best: LootPickup = null
	var best_d := LOOT_HINT_RANGE
	for n in get_tree().get_nodes_in_group(LootPickup.GROUP):
		var p := n as LootPickup
		if p == null or not p.is_inside_tree() or p.is_queued_for_deletion():
			continue
		var d := player.global_position.distance_to(p.global_position)
		if d < best_d:
			best_d = d
			best = p
	return best


static func _num(v: float) -> String:
	return str(roundi(v)) if is_equal_approx(v, roundf(v)) else "%.1f" % v

func _get_terminal() -> Node3D:
	if _terminal != null and is_instance_valid(_terminal) and _terminal.is_inside_tree():
		return _terminal
	_terminal = null
	if _time < _terminal_search_wait:
		return null
	_terminal_search_wait = _time + 1.0
	var found := get_tree().root.find_children("*", "RouteTerminal", true, false)
	if not found.is_empty():
		_terminal = found[0] as Node3D
	return _terminal

func _hint_for(obj: Objective, progress: int) -> Dictionary:
	var inv := get_tree().get_first_node_in_group("inventory_panel") as InventoryPanel
	var sp := get_tree().get_first_node_in_group("skill_panel") as SkillPanel
	var inv_open := inv != null and inv.is_open()
	var sp_open := sp != null and sp.is_open()
	var inv_key := OS.get_keycode_string(inv.toggle_key) if inv != null else "I"
	var sp_key := OS.get_keycode_string(sp.toggle_key) if sp != null else "K"
	var sp_state: Dictionary = sp.tutorial_state() if sp != null else {}

	match obj.kind:
		"walk":
			return {"text": "Move with [%s] [%s] [%s] [%s]   %d/%d" % [
					GemText.key_label(&"forward"), GemText.key_label(&"left"),
					GemText.key_label(&"backward"), GemText.key_label(&"right"),
					progress, obj.required],
				"sub": "Try each direction. Your character turns to face the mouse."}

		"open_panel":
			match obj.target:
				"map":
					var mm := get_tree().get_first_node_in_group("minimap")
					var key := OS.get_keycode_string(int(mm.get("overlay_key"))) if mm != null else "TAB"
					return {"text": "Press [%s] to open the full map" % key,
						"sub": "It shows the whole area. Press [%s] again to close it." % key}
				"inventory":
					return {"text": "Press [%s] to open your backpack" % inv_key,
						"sub": "Your backpack holds your gear, skill gems and loot."}
				"skills":
					return {"text": "Press [%s] to open your skills" % sp_key,
						"sub": "Skill gems go into sockets here. Each socketed gem is a skill you can use."}

		"equip_slot":
			if not inv_open:
				return {"text": "Press [%s] to open your backpack" % inv_key}
			return {"text": "Right-click your weapon to equip it",
				"sub": "Double-click works too. Hold CTRL over an item to compare it with what you wear.",
				"targets": inv.tutorial_target("weapon_item") + inv.tutorial_target("weapon_slot")}

		"socket_skill":
			if sp_open:
				if sp_state.get("tab") != "sockets":
					return {"text": "Open the SOCKETS tab", "targets": sp.tutorial_target("tab_sockets")}
				if sp_state.get("picker", false):
					return {"text": "Pick your skill gem",
						"sub": "Only gems that fit this socket are listed.",
						"targets": sp.tutorial_target("picker")}
				return {"text": "Click an empty skill socket [+]",
					"sub": "A socketed skill gem becomes a skill. The small cells next to it are for support gems.",
					"targets": sp.tutorial_target("empty_socket")}
			if inv_open:
				return {"text": "Right-click the skill gem to socket it",
					"sub": "It goes into the first free skill socket.",
					"targets": inv.tutorial_target("skill_gem_item")}
			return {"text": "Press [%s] to open your skills" % sp_key}

		"bind_skill":
			var key := GemText.key_label(StringName(obj.target))
			if not sp_open:
				return {"text": "Press [%s] to open your skills" % sp_key}
			if sp_state.get("tab") != "sockets":
				return {"text": "Open the SOCKETS tab", "targets": sp.tutorial_target("tab_sockets")}
			if sp_state.get("pending") != StringName(obj.target):
				return {"text": "Click the [%s] hotkey slot" % key,
					"sub": "The row at the top is your hotkeys. Pick one, then pick the skill to put on it.",
					"targets": sp.tutorial_target("hotkey_" + obj.target)}
			return {"text": "Now click your skill to put it on [%s]" % key,
				"sub": "Right-click a hotkey slot later to clear it.",
				"targets": sp.tutorial_target("filled_socket")}

		"cast":
			if inv_open or sp_open:
				return {"text": "Close the menus [ESC], then use your skill"}
			var key := GemText.key_label(StringName(obj.target)) if obj.target != "" else "your skill key"
			return {"text": "Press [%s] to use your skill   %d/%d" % [key, progress, obj.required],
				"sub": "Each use costs mana (the blue bar). It refills over time."}

		"program_condition":
			var warn := _skill_missing_hint(obj, false, inv, sp, inv_open, sp_open, sp_key, sp_state)
			if not warn.is_empty():
				return warn
			return _script_hint(obj, sp, sp_open, sp_key, sp_state)

		"rule_triggered":
			var warn := _skill_missing_hint(obj, true, inv, sp, inv_open, sp_open, sp_key, sp_state)
			if not warn.is_empty():
				return warn
			return _fight_hint(obj, progress, inv_open or sp_open)

		"program_applied":
			var warn := _skill_missing_hint(obj, false, inv, sp, inv_open, sp_open, sp_key, sp_state)
			if not warn.is_empty():
				return warn
			if not sp_open:
				return {"text": "Press [%s], then open the SCRIPTING tab" % sp_key}
			if sp_state.get("tab") != "scripting":
				return {"text": "Open the SCRIPTING tab", "targets": sp.tutorial_target("tab_script")}
			return {"text": "Click a snippet to add a rule, then press apply()",
				"targets": sp.tutorial_target("script_palette") + sp.tutorial_target("script_apply")}

	return {"text": obj.description}


func _script_hint(obj: Objective, sp: SkillPanel, sp_open: bool, sp_key: String, sp_state: Dictionary) -> Dictionary:
	var cond := obj.target
	var fn := cond + "()"
	if not sp_open:
		return {"text": "Press [%s] to open your skills" % sp_key,
			"sub": "Scripts are written in the skill menu. You'll make your skill hit harder when enemies crowd you."}
	if sp_state.get("tab") != "scripting":
		return {"text": "Open the SCRIPTING tab",
			"sub": "A script is a few small rules that change how a skill behaves.",
			"targets": sp.tutorial_target("tab_script")}
	if not _lesson_seen(LESSON_GUIDE):
		_mark_lesson_seen(LESSON_GUIDE)
		sp.open_script_guide("start")
	if sp.script_guide_open():
		return {"text": "Read the SCRIPTING GUIDE, or press [ESC] to skip it",
			"sub": "Skipping is fine: tips show in OUTPUT as you build. guide() at the top of the editor opens the guide any time.",
			"targets": sp.tutorial_target("script:guide")}
	var st: Dictionary = sp.tutorial_script_state(cond)
	if not st.get("has_socket", false):
		return {"text": "Socket a skill gem first",
			"sub": "Go back to the SOCKETS tab: only a socketed skill can be scripted."}
	if st.get("owner", "") != "":
		return {"text": "Click your skill's file at the top of the list",
			"sub": "The list on the left works like files: your skill, and under it any support gems.",
			"targets": sp.tutorial_target("script:skill_file")}
	if not st.get("rule", false):
		return {"text": "Click  %s  in SNIPPETS on the right" % fn,
			"sub": "It starts a new rule:  when %s:  Whatever you put under it only happens while 3+ enemies are within 4 m of you." % fn,
			"targets": sp.tutorial_target("script:cond:" + cond)}
	if st.get("effects", 0) == 0:
		if not st.get("selected", false):
			return {"text": "Click your  when %s  line to select the rule" % fn,
				"sub": "Snippets you click go into the selected rule. It's marked with a bar on its left.",
				"targets": sp.tutorial_target("script:rule_line:" + cond)}
		return {"text": "Now click  damage  in SNIPPETS, under knobs",
			"sub": "Knobs are what a rule changes. This adds  damage += ...  under your rule: more damage, but only while enemies are close.",
			"targets": sp.tutorial_target("script:knob:damage")}

	var points := "%.1f of %.1f points used" % [st.get("points", 0.0), st.get("budget", 0.0)]
	var params: Array = st.get("params", [])
	if _lesson_start_params.is_empty() or _lesson_start_params.size() != params.size():
		_lesson_start_params = params.duplicate()
	if not _lesson_done.has("raised_knob"):
		if int(st.get("knob_steps", 0)) >= 2:
			_lesson_done["raised_knob"] = true
		else:
			return {"text": "Click the orange number on the damage line to raise it",
				"sub": "Numbers are buttons: click raises, right-click lowers, the wheel does both. Each step adds damage and costs more points. Now: %s." % points,
				"targets": sp.tutorial_target("script:knob_number:%s:damage" % cond)}
	if not _lesson_done.has("changed_condition"):
		if params != _lesson_start_params:
			_lesson_done["changed_condition"] = true
		else:
			var why := "Needing more enemies makes the rule rarer, so its buffs get cheaper: now buffs x%.2f. A wider radius does the opposite." \
				% float(st.get("discount", 1.0))
			if st.get("over", false):
				why = "You're over budget (%s). Needing more enemies makes buffs cheaper, which can bring you back under: now buffs x%.2f." \
					% [points, float(st.get("discount", 1.0))]
			return {"text": "Now click the enemy count in  when %s" % fn,
				"sub": why,
				"targets": sp.tutorial_target("script:cond_numbers:" + cond)}
	var trade: String = st.get("trade_knob", "attack_speed")
	var trade_name := trade.replace("_", " ")
	if not _lesson_done.has("lowered_knob"):
		if float(st.get("drawback_points", 0.0)) > 0.0:
			_lesson_done["lowered_knob"] = true
			_points_when_lowered = float(st.get("points", 0.0))
		elif not st.get("always_rule", false):
			if st.get("any_selected", false):
				return {"text": "Click the  drop a snippet here  line to start a new rule",
					"sub": "Knobs can cost points, or give them back. Next you'll trade a little %s for more damage." % trade_name,
					"targets": sp.tutorial_target("script:new_rule")}
			return {"text": "Click  %s  in SNIPPETS" % trade,
				"sub": "A rule with no condition is  always:  it's on all the time. %s can only be changed in an always: rule." % trade_name.capitalize(),
				"targets": sp.tutorial_target("script:knob:" + trade)}
		elif not st.get("trade_line", false) and not st.get("always_selected", false):
			return {"text": "Click your  always:  line to select it",
				"sub": "Then  %s  in SNIPPETS goes into that rule." % trade,
				"targets": sp.tutorial_target("script:always_line")}
		elif not st.get("trade_line", false):
			return {"text": "Click  %s  in SNIPPETS" % trade,
				"sub": "It goes into your always: rule. %s can only be changed there." % trade_name.capitalize(),
				"targets": sp.tutorial_target("script:knob:" + trade)}
		else:
			var at := int(st.get("trade_steps", 0))
			var left := "It's at 0 now: one more right-click." if at == 0 					else "%d more right-clicks." % (at + 1)
			return {"text": "Right-click the  %s  number until it goes below 0" % trade,
				"sub": "%s Lowering a knob is a drawback: your skill gets a little slower, and you get points back that ANY rule can spend. Now: %s." % [left, points],
				"targets": sp.tutorial_target("script:always_knob_number:" + trade)}
	if not _lesson_done.has("spent_points"):
		if float(st.get("points", 0.0)) > _points_when_lowered + 0.01:
			_lesson_done["spent_points"] = true
		else:
			return {"text": "Spend them: click the damage number in  when %s  again" % fn,
				"sub": "You gave up some %s for more damage when you're crowded. Any knob trades the same way: lower what you need least, raise what you need most. Now: %s." % [trade_name, points],
				"targets": sp.tutorial_target("script:knob_number:%s:damage" % cond)}
	if not st.get("valid", false):
		var fix := "Hover the red line in the code to see how to fix it."
		if st.get("over", false):
			fix = "Over budget (%s): right-click the damage number to lower it, or raise the enemy count for a bigger discount." % points
		return {"text": "Your script has an error: read OUTPUT", "sub": fix,
			"targets": sp.tutorial_target("script:output")}
	if st.get("dirty", false):
		return {"text": "Press  apply()  to run your script",
			"sub": "Until then it's only a draft. Your rule uses %s, with buffs at x%.2f because it doesn't always fire." \
				% [points, float(st.get("discount", 1.0))],
			"targets": sp.tutorial_target("script:apply") + sp.tutorial_target("script:rule_line:" + cond)}
	return {"text": "Press  apply()  to run your script", "targets": sp.tutorial_target("script:apply")}

func _support_hint(qm: QuestManager) -> Dictionary:
	if _lesson_seen(LESSON_SUPPORTS) or not qm.is_completed(QUEST_BASICS) or not qm.pending_choices.is_empty():
		return {}
	var player := _get_player()
	var skills: SkillComponent = player.get("skills")
	var inv_comp: InventoryComponent = player.get("inventory")
	if skills == null or inv_comp == null:
		return {}
	var inv := get_tree().get_first_node_in_group("inventory_panel") as InventoryPanel
	var sp := get_tree().get_first_node_in_group("skill_panel") as SkillPanel
	var inv_open := inv != null and inv.is_open()
	var sp_open := sp != null and sp.is_open()
	var sp_key := OS.get_keycode_string(sp.toggle_key) if sp != null else "K"
	var sp_state: Dictionary = sp.tutorial_state() if sp != null else {}

	var socketed: ItemInstance = null
	var has_skill := false
	for s in skills.sockets:
		has_skill = has_skill or s.gem != null
		if socketed == null and not s.supports.is_empty():
			socketed = s.supports[0]
	var in_backpack := false
	for it in inv_comp.items:
		if it != null and it.definition.support_gem != null:
			in_backpack = true
			break

	if socketed != null:
		if not _support_lesson_shown:
			_mark_lesson_seen(LESSON_SUPPORTS)
			return {}
		if not sp_open:
			_mark_lesson_seen(LESSON_SUPPORTS)
			return {}
		if sp_state.get("tab") == "scripting":
			_mark_lesson_seen(LESSON_SUPPORTS)
			_start_script_lesson(qm)
			return {}
		return {"text": "%s now changes your skill" % socketed.definition.support_gem.display_name,
			"sub": "Hover it to read what it does. Right-click it to take it out again. Next, your first script: open the SCRIPTING tab.",
			"targets": sp.tutorial_target("filled_support") + sp.tutorial_target("tab_script")}

	if not in_backpack:
		return {}
	_support_lesson_shown = true
	var why := "Support gems go in the small cells next to a skill and change how that skill works."
	if not has_skill:
		return {"text": "Socket your skill gem first", "sub": why}
	if sp_open:
		if sp_state.get("tab") != "sockets":
			return {"text": "Open the SOCKETS tab", "sub": why, "targets": sp.tutorial_target("tab_sockets")}
		if sp_state.get("picker", false):
			return {"text": "Pick your support gem",
				"sub": "Only supports that work with this skill are listed.",
				"targets": sp.tutorial_target("picker")}
		return {"text": "Click an empty support cell next to your skill", "sub": why,
			"targets": sp.tutorial_target("empty_support")}
	if inv_open:
		var gem := inv.tutorial_target("support_gem_item")
		if not gem.is_empty():
			return {"text": "Right-click the support gem to socket it",
				"sub": "It goes next to the first skill it works with.", "targets": gem}
	return {"text": "You got a support gem: press [%s] to socket it" % sp_key, "sub": why}

	var lesson := qm.get_def(QUEST_SCRIPT)
	if lesson == null or not qm.can_start(lesson):
		return
	qm.start(lesson)
	QuestItems.toast(get_tree(), "WARDEN BRACK: Before you face anything real, teach your skill to think.\nYour first script: make it fire when enemies close in.")

	if _waypoint_map_seen:
		_mark_lesson_seen(LESSON_WAYPOINTS)
		return {}
	if not _in_waypoint_dungeon():
		return {}
	var w := Waypoints.for_scene(_current_scene_path())
	return {"text": "Waypoint unlocked: %s" % w.display_name,
		"sub": "Every area you reach is added to your waypoint map. Press [%s] to open it." % key}


func _in_waypoint_dungeon() -> bool:
	if _in_town():
		return false
	var w := Waypoints.for_scene(_current_scene_path())
	return w != null and w.kind != Waypoint.Kind.TOWN and Waypoints.is_discovered(w)


func _current_scene_path() -> String:
	var am := get_tree().get_first_node_in_group("area_manager") as AreaManager
	return am.current_scene_path() if am != null else ""


func _lesson_seen(id: String) -> bool:
	if SaveManager.party.is_empty():
		return false
	return id in SaveManager.get_data(SaveManager.party[0]).get(SAVE_KEY_DONE, [])


func _mark_lesson_seen(id: String) -> void:
	for slot in SaveManager.party:
		var data: Dictionary = SaveManager.get_data(slot)
		var done: Array = data.get(SAVE_KEY_DONE, [])
		if not id in done:
			done.append(id)
		data[SAVE_KEY_DONE] = done


func _skill_missing_hint(obj: Objective, need_hotkey: bool, inv: InventoryPanel, sp: SkillPanel,
		inv_open: bool, sp_open: bool, sp_key: String, sp_state: Dictionary) -> Dictionary:
	var player := _get_player()
	var skills: SkillComponent = player.get("skills") if player != null else null
	if skills == null:
		return {}

	var socketed := -1
	for i in skills.sockets.size():
		var s := skills.sockets[i]
		if s.gem == null:
			continue
		if socketed < 0:
			socketed = i
		if s.program != null and _program_uses(s.program, obj.target):
			socketed = i
			break

	if socketed < 0:
		var why := "Scripts change how a skill behaves, so you need a skill gem in a socket first."
		if sp_open:
			if sp_state.get("tab") != "sockets":
				return {"text": "No skill equipped: open the SOCKETS tab", "sub": why,
					"targets": sp.tutorial_target("tab_sockets")}
			if sp_state.get("picker", false):
				return {"text": "Pick your skill gem", "sub": why, "targets": sp.tutorial_target("picker")}
			return {"text": "No skill equipped: click an empty skill socket [+]", "sub": why,
				"targets": sp.tutorial_target("empty_socket")}
		if inv_open:
			var gem := inv.tutorial_target("skill_gem_item")
			if not gem.is_empty():
				return {"text": "No skill equipped: right-click the skill gem to socket it", "sub": why,
					"targets": gem}
		return {"text": "No skill equipped: press [%s] and socket a skill gem" % sp_key, "sub": why}

	if not need_hotkey or skills.action_of(socketed) != &"":
		return {}
	var why_key := "Your rule only fires when you use the skill, and a skill needs a hotkey to be used."
	if not sp_open:
		return {"text": "Your skill isn't on a hotkey: press [%s]" % sp_key, "sub": why_key}
	if sp_state.get("tab") != "sockets":
		return {"text": "Open the SOCKETS tab", "sub": why_key, "targets": sp.tutorial_target("tab_sockets")}
	if sp_state.get("pending", &"") == &"":
		var slots: Array[Control] = []
		for action in SkillComponent.ACTIONS:
			slots.append_array(sp.tutorial_target("hotkey_" + String(action)))
		return {"text": "Click a hotkey slot for your skill", "sub": why_key, "targets": slots}
	return {"text": "Now click your skill to put it on [%s]" % GemText.key_label(sp_state.get("pending")),
		"sub": why_key, "targets": sp.tutorial_target("filled_socket")}


func _program_uses(program: SkillProgram, condition_id: String) -> bool:
	for rule in program.rules:
		for c in rule.conditions:
			if c.type_id() == condition_id:
				return true
	return false


func _fight_hint(obj: Objective, progress: int, menus_open: bool) -> Dictionary:
	var condition_text := _rule_condition_text(obj.target)
	if menus_open:
		return {"text": "Close the menus [ESC] and find a fight",
			"sub": "Your rule only fires in combat, when %s." % condition_text}
	if _in_town():
		return {"text": "Take a portal to a dungeon",
			"sub": "Your rule needs enemies. It fires when %s, and the town is safe." % condition_text,
			"world": _nearest_dungeon_portal()}
	return {"text": "Use your skill with %s   %d/%d" % [condition_text, progress, obj.required],
		"sub": "Let a group close in, then attack: each use while the rule is true counts."}


func _rule_condition_text(condition_id: String) -> String:
	var player := _get_player()
	var skills: SkillComponent = player.get("skills") if player != null else null
	if skills != null:
		for s in skills.sockets:
			if s.program == null:
				continue
			for rule in s.program.rules:
				for c in rule.conditions:
					if c.type_id() == condition_id:
						return c.describe().trim_prefix("WHEN ")
	return "enough enemies are close"


func _travelling() -> bool:
	var am := get_tree().get_first_node_in_group("area_manager") as AreaManager
	return am != null and am.is_busy()


func _in_town() -> bool:
	var am := get_tree().get_first_node_in_group("area_manager")
	return am != null and am.get("current_area") is TownArea


func _nearest_dungeon_portal() -> Node3D:
	var player := _get_player() as Node3D
	if player == null:
		return null
	var best: Node3D = null
	var best_d := INF
	for p in NpcNameplates.portals():
		if not is_instance_valid(p) or not p.is_inside_tree():
			continue
		var w: Waypoint = Waypoints.for_scene(p.target_area) if p.target_area != "" else null
		if p.dungeon_level <= 0 and (w == null or w.kind == Waypoint.Kind.TOWN):
			continue
		var d := player.global_position.distance_to(p.global_position)
		if d < best_d:
			best_d = d
			best = p
	return best


func _get_player() -> Node:
	if _player == null or not is_instance_valid(_player) or not _player.is_inside_tree():
		_player = get_tree().get_first_node_in_group("players")
	return _player


func _get_guide() -> Node3D:
	if _guide != null and is_instance_valid(_guide) and _guide.is_inside_tree():
		return _guide
	_guide = null
	if _time < _guide_search_wait:
		return null
	_guide_search_wait = _time + 1.0
	for n in get_tree().root.find_children("*", "Area3D", true, false):
		if n.get("npc_id") == GUIDE_NPC_ID:
			_guide = n as Node3D
			break
	return _guide


func _get_examiner() -> Node3D:
	if _examiner != null and is_instance_valid(_examiner) and _examiner.is_inside_tree():
		return _examiner
	_examiner = null
	if _time < _examiner_search_wait:
		return null
	_examiner_search_wait = _time + 1.0
	for n in get_tree().root.find_children("*", "Area3D", true, false):
		if n.get("npc_id") == EXAMINER_NPC_ID:
			_examiner = n as Node3D
			break
	return _examiner


func _on_canvas_draw() -> void:
	var pulse := 0.5 + 0.5 * sin(_time * 5.0)
	var line := Color(COL_HINT, 0.55 + 0.45 * pulse)
	for t in _targets:
		if not is_instance_valid(t) or not t.is_visible_in_tree():
			continue
		var r := _screen_rect(t).grow(4.0 + 3.0 * pulse)
		_canvas.draw_rect(r, Color(COL_HINT, 0.10 * pulse), true)
		_canvas.draw_rect(r, line, false, 3.0)


func _on_world_canvas_draw() -> void:
	var line := Color(COL_HINT, 0.55 + 0.45 * (0.5 + 0.5 * sin(_time * 5.0)))
	if _world_target != null and is_instance_valid(_world_target):
		_draw_world_arrow(_world_target.global_position + Vector3.UP * _world_height, line,
				NpcNameplates.top_of(_world_target))


func _screen_rect(c: Control) -> Rect2:
	var xf := c.get_global_transform_with_canvas()
	return Rect2(xf.origin, c.size * xf.get_scale())


func _draw_world_arrow(pos: Vector3, col: Color, above_y: float = NAN) -> void:
	var pr: Dictionary = NpcNameplates.project(pos)
	if pr.is_empty():
		return
	var screen := _world_canvas.get_viewport_rect().size
	var behind: bool = pr["behind"]
	var p: Vector2 = pr["pos"]
	var inner := Rect2(Vector2(EDGE_MARGIN, EDGE_MARGIN), screen - Vector2(EDGE_MARGIN, EDGE_MARGIN) * 2.0)

	if not behind and not is_nan(above_y):
		p.y = minf(p.y, above_y - 18.0)
	if not behind and inner.has_point(p):
		var tip := p + Vector2(0.0, sin(_time * 4.0) * 6.0)
		_world_canvas.draw_colored_polygon(PackedVector2Array([
			tip, tip + Vector2(-14.0, -24.0), tip + Vector2(14.0, -24.0)]), col)
		return

	var centre := screen * 0.5
	var dir := p - centre
	if behind:
		dir = -dir
	if dir.length() < 1.0:
		dir = Vector2.DOWN
	dir = dir.normalized()
	var half := inner.size * 0.5
	var reach := minf(half.x / maxf(absf(dir.x), 0.001), half.y / maxf(absf(dir.y), 0.001))
	var tip := centre + dir * reach
	var side := dir.orthogonal()
	_world_canvas.draw_colored_polygon(PackedVector2Array([
		tip, tip - dir * 26.0 + side * 15.0, tip - dir * 26.0 - side * 15.0]), col)
