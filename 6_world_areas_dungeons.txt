class_name AreaManager
extends Node

@export var area_container: Node3D
@export var player: Player
@export_file("*.tscn") var start_area: String
@export var start_spawn: StringName = &"default"
@export var fade_rect: ColorRect
@export var fade_time: float = 0.25
@export var arrival_cooldown_ms: int = 600

var current_area: GameArea
var _current_path := ""
var _busy := false
var _ignore_until_ms := 0
var _remember_spot := false


const WAYPOINT_PANEL := preload("res://scenes/ui/player_ui/waypoints/waypoint_panel.gd")
const NPC_INFO_PANEL := preload("res://scenes/ui/player_ui/npc_info/npc_info_panel.gd")
const STIM_HUD := preload("res://scenes/ui/player_ui/stims/stim_hud.gd")


func _ready() -> void:
	add_to_group("area_manager")
	_add_waypoint_panel()
	await get_tree().process_frame
	Coop.game_ready(self)
	var town := Coop.start_town if Coop.in_game else SaveManager.last_town()
	if town.is_empty():
		await _change_area(start_area, start_spawn, false)
	else:
		await _change_area(town, &"default", false)


func is_busy() -> bool:
	return _busy


func current_scene_path() -> String:
	return _current_path


func travel(scene_path: String, spawn_id: StringName, dungeon_level: int = 0, new_instance := false) -> void:
	_ignore_until_ms = 0
	_remember_spot = true
	await go_to(scene_path, spawn_id, dungeon_level,
			WorldState.ALWAYS_NEW if new_instance else WorldState.REJOIN_OR_NEW)
	_remember_spot = false


func _add_waypoint_panel() -> void:
	var layer := CanvasLayer.new()
	layer.name = "WaypointLayer"
	layer.layer = 21
	var info := NPC_INFO_PANEL.new()
	info.name = "NpcInfoPanel"
	layer.add_child(info)
	var party: Control = preload("res://scenes/ui/player_ui/coop/party_menu.gd").new()
	party.name = "PartyMenu"
	layer.add_child(party)
	var panel := WAYPOINT_PANEL.new()
	panel.name = "WaypointPanel"
	layer.add_child(panel)
	var host: Node = self
	var vp := get_viewport()
	while vp is SubViewport and vp.get_parent() != null:
		host = vp.get_parent()
		vp = host.get_viewport()
	host.add_child.call_deferred(layer)
	var stim_layer := CanvasLayer.new()
	stim_layer.name = "StimLayer"
	stim_layer.layer = 19
	var lock_tags: Control = preload("res://scenes/ui/player_ui/lock_tags/lock_tags.gd").new()
	lock_tags.name = "LockTags"
	stim_layer.add_child(lock_tags)
	var stims := STIM_HUD.new()
	stims.name = "StimHud"
	stim_layer.add_child(stims)
	var threats := ThreatPanel.new()
	threats.name = "ThreatPanel"
	stim_layer.add_child(threats)
	var lock_banner: Control = preload("res://scenes/ui/player_ui/lock_banner/lock_banner.gd").new()
	lock_banner.name = "LockBanner"
	stim_layer.add_child(lock_banner)
	host.add_child.call_deferred(stim_layer)


func can_go() -> bool:
	return not _busy and Time.get_ticks_msec() >= _ignore_until_ms


func go_to(scene_path: String, spawn_id: StringName = &"default",
		dungeon_level: int = 0, instance_id: int = 0) -> void:
	if _busy or scene_path.is_empty():
		return
	if Time.get_ticks_msec() < _ignore_until_ms:
		return
	if Coop.in_game:
		Coop.request_travel(scene_path, spawn_id, dungeon_level, instance_id)
		return
	if dungeon_level > 0:
		WorldState.prepare_dungeon(scene_path, dungeon_level, instance_id)
	await _change_area(scene_path, spawn_id, true)


func travel_now(scene_path: String, spawn_id: StringName) -> void:
	if _busy or scene_path.is_empty():
		return
	await _change_area(scene_path, spawn_id, true)


func _change_area(scene_path: String, spawn_id: StringName, use_fade: bool, revive := false) -> void:
	_busy = true
	player.set_physics_process(false)
	player.velocity = Vector3.ZERO

	if use_fade:
		await _fade_to(1.0)
	get_tree().call_group("blast_swarms", "clear")
	var leaving = current_area.get("instance") if current_area else null
	if leaving is DungeonInstance:
		leaving.has_return_spot = _remember_spot and not revive
		if leaving.has_return_spot:
			leaving.return_spot = player.global_position
	Coop.area_changing()
	if current_area:
		current_area.queue_free()
		await current_area.tree_exited
		current_area = null

	var packed := load(scene_path) as PackedScene
	var area := packed.instantiate() as GameArea if packed else null
	if area == null:
		push_error("AreaManager: '%s' is missing or its root doesn't extend GameArea" % scene_path)
		if Coop.in_game:
			Coop.note("AREA FAILED TO LOAD: %s (exists %s)" % [scene_path, ResourceLoader.exists(scene_path)])
		await _finish(use_fade)
		return

	area_container.add_child(area)
	current_area = area
	_current_path = scene_path
	GraphicsQuality.apply(get_tree())

	await get_tree().physics_frame
	await get_tree().physics_frame

	var spot := area.get_spawn_position(spawn_id)
	var inst = area.get("instance")
	if inst is DungeonInstance and inst.has_return_spot and not revive:
		spot = inst.return_spot
		inst.has_return_spot = false
	player.global_position = spot + Vector3.UP * 0.05
	player.velocity = Vector3.ZERO
	player.reset_physics_interpolation()
	if revive:
		player.revive()

	if area.has_method("on_player_arrived"):
		area.on_player_arrived(player)
	if not area is TownArea:
		GameEvents.report("enter_area", area.scene_file_path.get_file().get_basename())
	Waypoints.arrived(scene_path)
	GlitchDeath.prewarm(area, spot)
	EffectWarmup.run(area, spot)
	Coop.area_loaded(scene_path)

	await _wait_for_navigation(area, use_fade)
	await _finish(use_fade)


const NAV_WAIT_MAX := 10.0
const LOADING_FONT := "res://scenes/ui/font/Courier_Prime/CourierPrime-Bold.ttf"
var _loading_label: Label


func _wait_for_navigation(area: GameArea, use_fade: bool) -> void:
	if area.get("navigation_is_ready") != false:
		return
	if use_fade:
		_show_loading(true)
	var waited := 0.0
	while is_instance_valid(area) and area.get("navigation_is_ready") == false and waited < NAV_WAIT_MAX:
		await get_tree().process_frame
		waited += get_process_delta_time()
	if waited >= NAV_WAIT_MAX:
		push_warning("AreaManager: navmesh not ready after %.0f s, showing the area anyway" % NAV_WAIT_MAX)
	_show_loading(false)


func _show_loading(on: bool) -> void:
	if fade_rect == null:
		return
	if _loading_label == null:
		_loading_label = Label.new()
		_loading_label.text = "LOADING"
		_loading_label.mouse_filter = Control.MOUSE_FILTER_IGNORE
		_loading_label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		_loading_label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
		_loading_label.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
		_loading_label.add_theme_font_size_override("font_size", 28)
		_loading_label.add_theme_color_override("font_color", Color("#00E5FF"))
		if ResourceLoader.exists(LOADING_FONT):
			_loading_label.add_theme_font_override("font", load(LOADING_FONT))
		fade_rect.add_child(_loading_label)
	_loading_label.visible = on
	if on:
		_loading_label.modulate.a = 1.0
		var tween := _loading_label.create_tween().set_loops()
		tween.tween_property(_loading_label, "modulate:a", 0.35, 0.6)
		tween.tween_property(_loading_label, "modulate:a", 1.0, 0.6)
		_loading_label.set_meta("tween", tween)
	elif _loading_label.has_meta("tween"):
		(_loading_label.get_meta("tween") as Tween).kill()
		_loading_label.remove_meta("tween")


func _finish(use_fade: bool) -> void:
	await get_tree().physics_frame
	if use_fade:
		await _fade_to(0.0)
	player.set_physics_process(true)
	_ignore_until_ms = Time.get_ticks_msec() + arrival_cooldown_ms
	_busy = false


func _fade_to(alpha: float) -> void:
	if fade_rect == null or fade_time <= 0.0:
		return
	var tween := create_tween()
	tween.tween_property(fade_rect, "modulate:a", alpha, fade_time)
	await tween.finished

func respawn(reload_area := false) -> void:
	if _busy or current_area == null:
		return
	if Coop.in_game:
		reload_area = false
	var id: StringName = &"default"
	var v = current_area.get("respawn_spawn_id")
	if v != null and v != &"":
		id = v

	if reload_area:
		var inst = current_area.get("instance")
		if inst is DungeonInstance:
			WorldState.prepare_rejoin(inst)
		await _change_area(_current_path, id, true, true)
		return

	_busy = true
	player.set_physics_process(false)
	await _fade_to(1.0)
	get_tree().call_group("blast_swarms", "clear")
	player.global_position = current_area.get_spawn_position(id) + Vector3.UP * 0.05
	player.velocity = Vector3.ZERO
	player.reset_physics_interpolation()
	player.revive()
	await _finish(true)

extends GameArea

signal navigation_ready
var navigation_is_ready := false

@onready var director: DungeonDirector = $DungeonDirector
@onready var grid_map: GridMap = $NavigationRegion3D/GridMap
@onready var nav_region: NavigationRegion3D = $NavigationRegion3D
@onready var populator: DungeonPopulator = $DungeonPopulator
@export var dungeon_level: int = 1
@export var respawn_spawn_id: StringName = &"default"
@export var portal_scene: PackedScene

const FLOOR_LAYER := 0

var _arrival_cells: Dictionary = {}
var _arrival_points: Dictionary = {}
var instance: DungeonInstance

const DUNGEON_LIGHT := 0.08


func _init() -> void:
	light_scale = DUNGEON_LIGHT


func _ready() -> void:
	instance = WorldState.begin_visit(scene_file_path, dungeon_level)
	dungeon_level = instance.level
	director.use_random_seed = false
	director.fixed_seed = instance.rng_seed
	seed(instance.rng_seed)
	if not director.generate_from_layout():
		push_error("Dungeon generation failed")
		return
	print("[Dungeon] generated, rooms: %d (instance %d, level %d, %d killed)" % [
		director.rooms.size(), instance.id, instance.level, instance.killed.size()])
	_place_portals()
	populator.instance = instance
	_spawn_enemies()
	_scatter_null_decals()
	randomize()
	var sleeper := EnemySleep.new()
	sleeper.name = "EnemySleep"
	sleeper.enemies_root = $Enemies
	add_child(sleeper)
	_bind_minimap()
	var kernel := KernelDirector.new()
	kernel.name = "KernelDirector"
	add_child(kernel)
	await _bake_navigation()


func _exit_tree() -> void:
	WorldState.end_visit(instance)

func _bind_minimap() -> void:
	var mm := get_tree().get_first_node_in_group("minimap") as Minimap
	if mm == null:
		push_warning("Dungeon: no Minimap in the scene tree")
		return
	mm.bind_dungeon(director, grid_map, $Enemies, {"explored": instance.explored})

func _spawn_enemies() -> void:
	var skip: Array = []
	for r: RoomData in director.rooms:
		if r.room_type == "start" or r.room_type == "portal" or r.room_type == "boss":
			skip.append(r)
	populator.populate(dungeon_level, skip, _is_floor, _grid_to_world)
	populator.spawn_marked(director.layout.enemies, dungeon_level, _is_floor, _grid_to_world)
	var echoes := get_node_or_null("NullEchoes") as NullEchoDirector
	if echoes != null:
		populator.place_echoes(echoes, _is_floor, _grid_to_world)
	populator.place_ward_beacon(_is_floor, _grid_to_world)
	populator.place_salvage(scene_file_path, _is_floor, _grid_to_world)
	populator.place_queued_crates(scene_file_path, _is_floor, _grid_to_world)
	populator.place_props(_is_floor, _grid_to_world)

func _scatter_null_decals() -> void:
	var cells: Array[Vector2i] = []
	for c in grid_map.get_used_cells():
		if c.y == FLOOR_LAYER and _is_floor(c):
			cells.append(Vector2i(c.x, c.z))
	var rooms: Array[Rect2i] = []
	for r: RoomData in director.rooms:
		rooms.append(r.rect)
	NullDecals.scatter(self, cells, _grid_to_world, NullDecals.count_for_level(dungeon_level, cells.size()),
			instance.rng_seed ^ 0x5EED, director.dungeon_theme.cell_size.x, rooms, dungeon_level)


const NAV_SOURCE_GROUP := &"navmesh_source"
const REBAKE_DELAY := 0.6
const NAV_FOOTPRINT := 0.6

var _rebake_pending := false
var _tile_geometry: NavigationMeshSourceGeometryData3D


func _bake_navigation() -> void:
	var nm := nav_region.navigation_mesh
	nm.geometry_source_geometry_mode = NavigationMesh.SOURCE_GEOMETRY_GROUPS_WITH_CHILDREN
	nm.geometry_source_group_name = NAV_SOURCE_GROUP
	nm.geometry_parsed_geometry_type = NavigationMesh.PARSED_GEOMETRY_STATIC_COLLIDERS
	grid_map.add_to_group(NAV_SOURCE_GROUP)
	var props := get_node_or_null("Props")
	if props != null:
		for p in props.get_children():
			if p is LootProp:
				_add_nav_obstacle(p as LootProp)
				if not (p as LootProp).opened.is_connected(_on_prop_opened):
					(p as LootProp).opened.connect(_on_prop_opened)
	await get_tree().physics_frame
	await get_tree().physics_frame
	var data := NavigationMeshSourceGeometryData3D.new()
	NavigationServer3D.parse_source_geometry_data(nav_region.navigation_mesh, data, nav_region)
	_tile_geometry = NavigationMeshSourceGeometryData3D.new()
	_tile_geometry.set_vertices(data.get_vertices())
	_tile_geometry.set_indices(data.get_indices())
	_bake_async(data)
	while _baking_async:
		await get_tree().process_frame
	await get_tree().physics_frame
	await get_tree().physics_frame
	print("[Dungeon] navmesh ready, polygons: %d" % nav_region.navigation_mesh.get_polygon_count())
	navigation_is_ready = true
	navigation_ready.emit()


func _add_nav_obstacle(p: LootProp) -> void:
	if p.has_node("NavObstacle"):
		return
	var ob := NavigationObstacle3D.new()
	ob.name = "NavObstacle"
	ob.avoidance_enabled = false
	ob.affect_navigation_mesh = true
	ob.height = p.collision_height
	var r := p.collision_radius * NAV_FOOTPRINT
	ob.vertices = PackedVector3Array([Vector3(-r, 0, -r), Vector3(r, 0, -r), Vector3(r, 0, r), Vector3(-r, 0, r)])
	p.add_child(ob)
	ob.add_to_group(NAV_SOURCE_GROUP)


func _on_prop_opened(prop: LootProp) -> void:
	if prop.still_blocks():
		return
	if _rebake_pending:
		return
	_rebake_pending = true
	await get_tree().create_timer(REBAKE_DELAY, false).timeout
	while nav_region.is_baking() or _baking_async:
		await get_tree().process_frame
	_rebake_pending = false
	if is_inside_tree():
		_rebake()


var _baking_async := false


func _rebake() -> void:
	if _tile_geometry == null:
		nav_region.bake_navigation_mesh(true)
		return
	var data := NavigationMeshSourceGeometryData3D.new()
	data.set_vertices(_tile_geometry.get_vertices())
	data.set_indices(_tile_geometry.get_indices())
	var to_region := nav_region.global_transform.affine_inverse()
	var props := get_node_or_null("Props")
	if props != null:
		for p in props.get_children():
			var lp := p as LootProp
			if lp == null or not lp.still_blocks() or lp.is_queued_for_deletion():
				continue
			var ob := lp.get_node_or_null("NavObstacle") as NavigationObstacle3D
			if ob == null:
				continue
			var pts := PackedVector3Array()
			var base := to_region * ob.global_position
			for v in ob.vertices:
				var w := to_region * (ob.global_transform * v)
				pts.append(Vector3(w.x, 0.0, w.z))
			data.add_projected_obstruction(pts, base.y, ob.height, false)
	_bake_async(data)


func _bake_async(data: NavigationMeshSourceGeometryData3D) -> void:
	var nm := nav_region.navigation_mesh.duplicate() as NavigationMesh
	nm.clear_polygons()
	nm.set_vertices(PackedVector3Array())
	_baking_async = true
	NavigationServer3D.bake_from_source_geometry_data_async(nm, data, _on_rebaked.bind(nm))


func _on_rebaked(nm: NavigationMesh) -> void:
	_baking_async = false
	if is_inside_tree() and nm.get_polygon_count() > 0:
		nav_region.navigation_mesh = nm

func get_spawn_position(spawn_id: StringName = &"default") -> Vector3:
	var layout := director.layout
	if layout == null or layout.spawn_cell == DungeonLayout.NO_CELL:
		push_error("get_spawn_position: dungeon has no valid spawn (generation failed?)")
		return global_position
	if _arrival_points.has(spawn_id):
		return _arrival_points[spawn_id]
	if layout.spawns.has(spawn_id):
		return _grid_to_world(layout.spawns[spawn_id])
	if _arrival_cells.has(spawn_id):
		return _grid_to_world(_nearby_floor(_arrival_cells[spawn_id]))
	return _grid_to_world(layout.spawn_cell)


func _room_nearest(cell: Vector2i) -> RoomData:
	var best: RoomData = null
	var best_d := 1 << 60
	for r: RoomData in director.rooms:
		var d := (r.center() - cell).length_squared()
		if d < best_d:
			best_d = d
			best = r
	return best	

func _is_floor(cell: Vector3i) -> bool:
	var item := grid_map.get_cell_item(cell)
	if item == GridMap.INVALID_CELL_ITEM:
		return false
	var t := director.dungeon_theme
	var lib := t.mesh_library
	return item == lib.find_item_by_name(t.floor_name) or item == lib.find_item_by_name(t.room_floor_name)
	
func _nearby_floor(center: Vector2i, min_dist: int = 2, max_dist: int = 5) -> Vector2i:
	var toward := Vector2i.DOWN
	var best_d := 1 << 60
	for r: RoomData in director.rooms:
		var off := r.center() - center
		var d := off.length_squared()
		if d > 0 and d < best_d:
			best_d = d
			if absi(off.x) > absi(off.y):
				toward = Vector2i(signi(off.x), 0)
			else:
				toward = Vector2i(0, signi(off.y))

	var dirs: Array[Vector2i] = [toward, Vector2i(toward.y, toward.x),
			Vector2i(-toward.y, -toward.x), -toward]

	for d in range(min_dist, max_dist + 1):
		for dir in dirs:
			var c: Vector2i = center + dir * d
			if _is_floor(Vector3i(c.x, FLOOR_LAYER, c.y)):
				return c
		for dx in range(-d, d + 1):
			for dz in range(-d, d + 1):
				if maxi(absi(dx), absi(dz)) != d:
					continue
				var c := center + Vector2i(dx, dz)
				if _is_floor(Vector3i(c.x, FLOOR_LAYER, c.y)):
					return c
	push_warning("No floor found near %s, using it as-is" % [center])
	return center

func _place_portal(cell: Vector2i, target: String, spawn: StringName, level: int = 0, yaw := 0.0,
		sealed_by_key := "", open_after_quest := "", routing: Dictionary = {}) -> void:
	var p := portal_scene.instantiate() as Portal
	p.set_routing(routing)
	p.target_area = target
	p.target_spawn = spawn
	p.dungeon_level = level
	p.sealed_by_key = sealed_by_key
	p.open_after_quest = open_after_quest
	add_child(p)
	p.global_position = _grid_to_world(cell)
	p.global_basis = grid_map.global_basis.orthonormalized() * PortalMarker.portal_basis(yaw)


func _exit_point(cell: Vector2i, yaw: float, distance: float) -> Variant:
	var c := grid_map.map_to_local(Vector3i(cell.x, FLOOR_LAYER, cell.y))
	var local := c + PortalMarker.exit_offset(yaw, distance)
	var at := grid_map.local_to_map(local)
	if not _is_floor(Vector3i(at.x, FLOOR_LAYER, at.z)):
		return null
	return grid_map.to_global(Vector3(local.x, c.y + director.dungeon_theme.floor_half_thickness, local.z))

func _grid_to_world(grid_pos: Vector2i) -> Vector3:
	var cell := Vector3i(grid_pos.x, FLOOR_LAYER, grid_pos.y)
	assert(_is_floor(cell), "No floor under spawn cell %s" % [grid_pos])
	var c := grid_map.map_to_local(cell)
	return grid_map.to_global(Vector3(c.x, c.y + director.dungeon_theme.floor_half_thickness, c.z))

func _place_portals() -> void:
	for p in director.layout.portals:
		var yaw: float = p.get("yaw", 0.0)
		_place_portal(p.cell, p.target_area, p.target_spawn, int(p.get("dungeon_level", 0)), yaw,
				str(p.get("sealed_by_key", "")), str(p.get("open_after_quest", "")), p)
		if p.arrival_id != &"":
			_arrival_cells[p.arrival_id] = p.cell
			var dist := float(p.get("exit_distance", 1.6))
			var point = _exit_point(p.cell, yaw, dist)
			if point == null:
				point = _exit_point(p.cell, yaw + PI, dist)
			if point != null:
				_arrival_points[p.arrival_id] = point
			else:
				push_warning("Portal to %s: no floor %.1f m in front of it, arrivals use the nearest floor" % [p.target_area, float(p.get("exit_distance", 1.6))])

func generate_map(a: Vector2i, b: Vector2i, max_retries: int = 5) -> bool:
	if use_random_seed:
		rng.randomize()
	else:
		rng.seed = fixed_seed
	
	for attempt in max_retries:
		grid = GridData.new(grid_width, grid_height)
		rooms.clear()
		
		var path = PathGenerator.generate(grid, a, b, rng)
		CorridorCarver.carve(grid, path, corridor_width)
		rooms += RoomGenerator.place_along_path(grid, path, rng, min_room_size, max_room_size, room_interval, room_min_gap)
		rooms += BranchGenerator.generate(grid, path, rng, num_branch_rooms, min_room_size, max_room_size, rooms, room_min_gap)
		
		if ConnectivityChecker.validate(grid, a, b):
			_paint()
		
		push_warning("Dungeon generation attempt %d failed connectivity check, retrying..." % (attempt + 1))
	
	push_error("Dungeon generation failed after %d attempts." % max_retries)
	return false

func generate_from_layout(max_retries: int = 5) -> bool:
	if layout_root == null:
		push_error("DungeonDirector: layout_root not assigned")
		return false
	layout = DungeonLayoutReader.read(layout_root, grid_map, grid_width, grid_height)
	print("Layout root: %s, markers found: %d, spawn: %s" % [
	layout_root.get_path(), layout.rooms.size(), layout.spawn_cell])
	if layout.spawn_cell == DungeonLayout.NO_CELL:
		push_error("Layout needs one SpawnMarker")
		return false

	if use_random_seed:
		rng.randomize()
	else:
		rng.seed = fixed_seed

	for attempt in max_retries:
		grid = GridData.new(grid_width, grid_height)
		rooms.clear()
		rooms += layout.rooms
		var extra := _random_rooms(layout.rooms)
		rooms += extra

		var cells: Array = _connect_rooms(layout.rooms + extra)
		if not cells.is_empty():
			if filler_rooms:
				rooms += RoomGenerator.place_along_path(grid, cells, rng, min_room_size,
						max_room_size, room_interval, room_min_gap, rooms)
			if num_branch_rooms > 0:
				rooms += BranchGenerator.generate(grid, cells, rng, num_branch_rooms,
						min_room_size, max_room_size, rooms, room_min_gap)
		for r: RoomData in rooms:
			RoomGenerator.carve_room(grid, r.rect)

		var targets: Array[Vector2i] = []
		for r: RoomData in rooms:
			targets.append(r.center())
		for s in layout.spawns.values():
			targets.append(s)
		for en in layout.enemies:
			targets.append(en["cell"])
		if ConnectivityChecker.validate_all(grid, layout.spawn_cell, targets):
			return _paint()
		push_warning("Layout attempt %d failed connectivity, retrying..." % (attempt + 1))

	push_error("Layout generation failed after %d attempts" % max_retries)
	return false

func _random_rooms(existing: Array) -> Array:
	var out: Array = []
	if random_rooms <= 0:
		return out
	var lo := maxi(min_room_size, 3)
	var hi := maxi(max_room_size, lo)
	var taken: Array = existing.duplicate()
	var route := _marker_route(existing) if random_room_spread > 0 else []
	var tries := 0
	while out.size() < random_rooms and tries < random_rooms * 60:
		tries += 1
		var size := Vector2i(rng.randi_range(lo, hi), rng.randi_range(lo, hi))
		var pos := Vector2i(rng.randi_range(2, grid_width - size.x - 2),
				rng.randi_range(2, grid_height - size.y - 2))
		if not route.is_empty():
			var seg: Array = route[rng.randi() % route.size()]
			var on := Vector2(seg[0]).lerp(Vector2(seg[1]), rng.randf())
			var off := Vector2(rng.randf_range(-1.0, 1.0), rng.randf_range(-1.0, 1.0)) * random_room_spread
			var c := Vector2i((on + off).round()) - size / 2
			pos = Vector2i(clampi(c.x, 2, grid_width - size.x - 2), clampi(c.y, 2, grid_height - size.y - 2))
		var rect := Rect2i(pos, size)
		if RoomGenerator.room_overlaps(rect, taken, maxi(room_min_gap, 2)):
			continue
		var room := RoomData.new(rect, "generic")
		out.append(room)
		taken.append(room)
	if out.size() < random_rooms:
		push_warning("DungeonDirector: only %d of %d random rooms fit (grid %dx%d)"
				% [out.size(), random_rooms, grid_width, grid_height])
	return out


func _marker_route(list: Array) -> Array:
	var segs: Array = []
	if list.size() < 2:
		return segs
	var in_tree: Array[int] = [0]
	var pending: Array[int] = []
	for i in range(1, list.size()):
		pending.append(i)
	while not pending.is_empty():
		var best := Vector2i(-1, -1)
		var best_d := 1 << 60
		for i in in_tree:
			for j in pending:
				var d := _room_dist(list[i], list[j])
				if d < best_d:
					best_d = d
					best = Vector2i(i, j)
		segs.append([list[best.x].center(), list[best.y].center()])
		in_tree.append(best.y)
		pending.erase(best.y)
	return segs


func _connect_rooms(list: Array) -> Array:
	var cells: Array = []
	if list.size() < 2:
		return cells
	var in_tree: Array[int] = [0]
	var pending: Array[int] = []
	for i in range(1, list.size()):
		pending.append(i)

	while not pending.is_empty():
		var best := Vector2i(-1, -1)
		var best_d := 1 << 60
		for i in in_tree:
			for j in pending:
				var d := _room_dist(list[i], list[j])
				if d < best_d:
					best_d = d
					best = Vector2i(i, j)
		cells += _link(list[best.x], list[best.y])
		in_tree.append(best.y)
		pending.erase(best.y)

	for k in loop_connections:
		var a := rng.randi() % list.size()
		var b := rng.randi() % list.size()
		if a != b:
			cells += _link(list[a], list[b])
	return cells

func _room_dist(a: RoomData, b: RoomData) -> int:
	var d := a.center() - b.center()
	return absi(d.x) + absi(d.y)

func _link(a: RoomData, b: RoomData) -> Array:
	a.add_connection(b)
	b.add_connection(a)
	if a.rect.grow(1).intersects(b.rect):
		return []
	var path: Array = _l_path(a.center(), b.center()) if straight_corridors \
			else PathGenerator.generate(grid, a.center(), b.center(), rng)
	CorridorCarver.carve(grid, path, corridor_width)
	return path

func _l_path(a: Vector2i, b: Vector2i) -> Array:
	var path: Array = []
	var p := a
	while p.x != b.x:
		p.x += signi(b.x - p.x)
		path.append(p)
	while p.y != b.y:
		p.y += signi(b.y - p.y)
		path.append(p)
	return path

func preview_layout() -> void:
	if generate_from_layout():
		print("Preview: %d rooms" % rooms.size())

func clear_preview() -> void:
	grid_map.clear()
	

func _paint() -> bool:
	grid_map.mesh_library = dungeon_theme.mesh_library
	grid_map.cell_size = dungeon_theme.cell_size

	var lib := dungeon_theme.mesh_library
	var floor_id := lib.find_item_by_name(dungeon_theme.floor_name)
	var room_id := lib.find_item_by_name(dungeon_theme.room_floor_name)
	var wall_id := lib.find_item_by_name(dungeon_theme.wall_name)
	if floor_id == -1 or room_id == -1 or wall_id == -1:
		push_error("Theme library is missing an item named Floor / Room_Floor / Wall")
		return false

	var painter := GridMapPainter.new(grid_map, floor_id, room_id, wall_id)
	painter.paint(grid, dungeon_theme.wall_height)
	return true

const DROP_CHANCE := {"normal": 0.10, "magic": 0.35, "rare": 1.0, "champion": 1.0}

    "normal": [1, 4],
    "magic": [4, 10],

func populate(level: int, skip_rooms: Array, is_floor: Callable, to_world: Callable) -> void:
	_level = level
	_is_floor = is_floor
	_to_world = to_world

	var dungeon_theme := director.dungeon_theme
	if dungeon_theme.enemy_pool.is_empty():
		push_warning("Theme has no enemy_pool, no enemies spawned")
		return

	var rng := director.rng
	if threats.is_empty():
		if instance != null:
			threats = instance.threats
		else:
			var area := owner if owner != null else get_parent()
			threats = DungeonThreats.roll_for(area.scene_file_path, rng.seed) if area != null else PackedStringArray()
	_threat_rng.seed = rng.seed + 7177
	var swarm := DungeonThreats.SWARM_COUNT if "swarm" in threats else 1
	var used: Dictionary = {}
	var spawned := 0
	for room: RoomData in director.rooms:
		if room in skip_rooms:
			continue
		var is_branch := room.room_type == "branch"
		var target := rng.randi_range(
			dungeon_theme.enemies_per_room_min, dungeon_theme.enemies_per_room_max)
		if is_branch:
			target = maxi(1, target / 2)
		target *= swarm
		var weights: Dictionary = dungeon_theme.branch_rarity_weights if is_branch \
			else dungeon_theme.room_rarity_weights

		var placed := 0
		var attempts := 0
		while placed < target and attempts < target * 10:
			attempts += 1
			var cell := Vector2i(
				rng.randi_range(room.rect.position.x, room.rect.end.x - 1),
				rng.randi_range(room.rect.position.y, room.rect.end.y - 1))
			if used.has(cell) or not _is_floor.call(Vector3i(cell.x, 0, cell.y)):
				continue
			used[cell] = true
			var rarity := _roll_rarity(weights, rng)
			var def := EnemyDefinition.pick(dungeon_theme.enemy_pool, rarity, rng)
			if def == null:
				continue
			_spawn_threat(def, rarity, cell)
			placed += 1
			spawned += 1

		if is_branch:
			_drop_item(_to_world.call(room.center()), "rare")
	spawned += _populate_corridors(rng, used)
	print("[Populator] spawned %d enemies" % spawned)


func _populate_corridors(rng: RandomNumberGenerator, used: Dictionary) -> int:
	if corridor_tiles_per_group <= 0:
		return 0
	var pool := director.dungeon_theme.enemy_pool
	var in_room := {}
	var safe: Array[Rect2i] = []
	for room: RoomData in director.rooms:
		if room.room_type == "start" or room.room_type == "portal":
			safe.append(room.rect.grow(corridor_safe_distance))
		for x in range(room.rect.position.x, room.rect.end.x):
			for y in range(room.rect.position.y, room.rect.end.y):
				in_room[Vector2i(x, y)] = true
	var cells: Array[Vector2i] = []
	for x in director.grid_width:
		for y in director.grid_height:
			var c := Vector2i(x, y)
			if in_room.has(c) or used.has(c) or not _is_floor.call(Vector3i(x, 0, y)):
				continue
			if safe.any(func(r: Rect2i) -> bool: return r.has_point(c)):
				continue
			cells.append(c)
	var groups := cells.size() / corridor_tiles_per_group
	var spawned := 0
	for g in groups:
		if cells.is_empty():
			break
		var anchor: Vector2i = cells[rng.randi() % cells.size()]
		var size := rng.randi_range(corridor_group_min, corridor_group_max)
		if "swarm" in threats:
			size *= DungeonThreats.SWARM_COUNT
		var def := EnemyDefinition.pick(pool, "normal", rng)
		if def == null:
			break
		for i in size:
			var spot := anchor
			for attempt in 8:
				var c := anchor + Vector2i(rng.randi_range(-1, 1), rng.randi_range(-1, 1))
				if not used.has(c) and _is_floor.call(Vector3i(c.x, 0, c.y)):
					spot = c
					break
			if used.has(spot):
				continue
			used[spot] = true
			_spawn_threat(def, _roll_rarity(corridor_rarity_weights, rng), spot)
			spawned += 1
	return spawned

func _eligible_loot() -> Array[ItemDefinition]:
	if _loot_cache_level == _level:
		return _loot_cache
	var theme := director.dungeon_theme
	var out: Array[ItemDefinition] = []
	if theme.auto_loot:
		out.append_array(ItemCatalog.for_level(_level, loot_level_window,
				theme.loot_include_gems, PackedStringArray(theme.loot_folders)))
	for def in theme.loot_pool:
		if def not in out and def.item_level <= _level and def.item_level >= _level - loot_level_window:
			out.append(def)
	var rank := Ranks.of_player(get_tree())
	out.assign(out.filter(func(d: ItemDefinition) -> bool: return not ShopNPC._locked(d, rank)))
	_loot_cache = out
	_loot_cache_level = _level
	print("[Populator] loot pool at level %d: %d items" % [_level, out.size()])
	return out

func _spawn_enemy(def: EnemyDefinition, rarity: String, cell: Vector2i,
		forced_mods: Variant = null, add_to_random: bool = false) -> Node3D:
	var enemy := EnemySpawner.spawn_enemy(
		def, rarity, enemies_root, _to_world.call(cell) + SPAWN_LIFT, _level,
		forced_mods, add_to_random)
	if not _track(enemy):
		return null
	var numbers : DamageNumbers = get_tree().get_first_node_in_group("damage_numbers") as DamageNumbers
	if numbers:
		numbers.track(enemy.health)
	print("[Populator] %s (%s)" % [enemy.display_name, rarity])
	enemy.set_meta(&"enemy_id", def.id)
	enemy.set_meta(&"rarity", rarity)
	enemy.set_meta(&"base_name", def.display_name)
	enemy.set_meta(&"act_boss", def.act_boss)
	enemy.health.died.connect(_on_enemy_died.bind(enemy, rarity, def.id))
	EnemySleep.allow(enemy)
	return enemy

func _spawn_threat(def: EnemyDefinition, rarity: String, cell: Vector2i) -> Node3D:
	return _spawn_enemy(def, rarity, cell, DungeonThreats.roll_traits(threats, rarity, _threat_rng), true)

func _track(enemy: Node3D) -> bool:
	var index := _enemy_index
	_enemy_index += 1
	if instance == null:
		return true
	if instance.killed.has(index):
		enemy.get_parent().remove_child(enemy)
		enemy.queue_free()
		return false
	var inst := instance
	enemy.health.died.connect(func() -> void:
		inst.killed[index] = true
		_check_cleared(inst))
	return true

		return area != null and area.is_ancestor_of(n) and n.has_method("wave_event_done") \
				and not n.wave_event_done())

func _check_cleared(inst: DungeonInstance) -> void:
	if inst.cleared or inst.killed.size() < _enemy_index:
		return
	var pending := _pending_wave_events()
	if not pending.is_empty():
		if not _told_waves_left:
			_told_waves_left = true
			var names: Array[String] = []
			for n in pending:
				names.append(n.wave_event_label() if n.has_method("wave_event_label") else "a wave event")
			QuestItems.toast(get_tree(), "AREA NOT CLEARED YET\nStill to go: %s." % ", ".join(names),
					Color(1.0, 0.82, 0.30))
		return
	inst.cleared = true
	var area_id := _area_id()
	if area_id == "":
		return
	var player := get_tree().get_first_node_in_group("players")
	var qm := player.get("quests") as QuestManager if player != null else null
	var first := qm != null and not ScriptBudget.is_cleared(qm, area_id)
	GameEvents.report(ScriptBudget.CLEAR_DEED, area_id)
	var area := owner if owner != null else get_parent()
	var w := Waypoints.for_scene(area.scene_file_path) if area != null else null
	var area_name := w.display_name if w != null else area_id.capitalize()
	var reward := ""
	if first:
		var pts := ScriptBudget.POINTS_PER_CLEAR
		reward = "+%s SCRIPT POINT%s ON EVERY SKILL" % [
				str(int(pts)) if is_equal_approx(pts, roundf(pts)) else "%.1f" % pts, "" if pts <= 1.0 else "S"]
	elif qm != null:
		reward = "ALREADY CLEARED BEFORE: NO NEW POINT"
	AreaClearedBanner.announce(get_tree(), area_name, reward)

			for y in director.grid_height:
				var c := Vector2i(x, y)
				if not in_room.has(c) and is_floor.call(Vector3i(x, 0, y)) \
						and _against_wall(c, is_floor) and _leaves_gap(c, is_floor):
					corridor.append(c)
		_scatter(corridor, corridor.size() / corridor_cells_per_prop, holder, to_world, rng, counter, taken)
	var placed: int = counter[1]
	var chests := _place_chests(holder, to_world, rng, counter, taken, is_floor)
	print("[Populator] placed %d props, %d chests" % [placed, chests])


func _place_chests(holder: Node3D, to_world: Callable, rng: RandomNumberGenerator,
		counter: Array, taken: Dictionary, is_floor: Callable) -> int:
	if not chests_enabled or chests_max <= 0:
		return 0
	if chest_scene == null:
		chest_scene = load(DEFAULT_CHEST)
	var t := clampf((_level - 1) / 9.0, 0.0, 1.0)
	var want := roundi(lerpf(chests_min, chests_max, t))
	var rooms: Array = []
	for room: RoomData in director.rooms:
		if room.room_type != "start" and room.room_type != "portal":
			rooms.append(room)
	var gm := director.grid_map
	var away := Vector3(0.0, 0.0, -1.0)
	var cam := get_viewport().get_camera_3d()
	if cam != null:
		away = Vector3(-cam.global_basis.z.x, 0.0, -cam.global_basis.z.z).normalized()
	var made := 0
	while made < want and not rooms.is_empty():
		var room: RoomData = rooms.pop_at(rng.randi() % rooms.size())
		var spots: Array = []
		for x in range(room.rect.position.x, room.rect.end.x):
			for y in range(room.rect.position.y, room.rect.end.y):
				var c := Vector2i(x, y)
				if taken.has(c) or not is_floor.call(Vector3i(x, 0, y)):
					continue
				for d in [Vector2i.LEFT, Vector2i.RIGHT, Vector2i.UP, Vector2i.DOWN]:
					if (gm.global_basis * Vector3(d.x, 0.0, d.y)).dot(away) < 0.1:
						continue
					var wall: Vector2i = c + d
					var front: Vector2i = c - d
					var front2: Vector2i = c - d * 2
					if not is_floor.call(Vector3i(wall.x, 0, wall.y)) \
							and is_floor.call(Vector3i(front.x, 0, front.y)) and not taken.has(front) \
							and is_floor.call(Vector3i(front2.x, 0, front2.y)):
						spots.append([c, d])
						break
		if spots.is_empty():
			continue
		var pick: Array = spots[rng.randi() % spots.size()]
		var cell: Vector2i = pick[0]
		var wall_dir: Vector2i = pick[1]
		for dx in range(-1, 2):
			for dy in range(-1, 2):
				taken[cell + Vector2i(dx, dy)] = true
		var chest := _place_prop(cell, holder, to_world, rng, counter, chest_scene)
		var into := (gm.global_basis * Vector3(-wall_dir.x, 0.0, -wall_dir.y)).normalized()
		chest.global_rotation = Vector3(0.0, atan2(into.x, into.z), 0.0)
		chest.global_position -= into * gm.cell_size.x * 0.2
		made += 1
	return made


func _scatter(spots: Array[Vector2i], want: int, holder: Node3D, to_world: Callable,
		rng: RandomNumberGenerator, counter: Array, taken: Dictionary) -> void:
	var made := 0
	while made < want and not spots.is_empty():
		var cell: Vector2i = spots.pop_at(rng.randi() % spots.size())
		if taken.has(cell):
			continue
		var pile: Array[Vector2i] = [cell]
		if rng.randf() < prop_cluster_chance:
			var extra := rng.randi_range(1, 2)
			for d in [Vector2i.LEFT, Vector2i.RIGHT, Vector2i.UP, Vector2i.DOWN]:
				var n: Vector2i = cell + d
				if pile.size() > extra:
					break
				if spots.has(n) and not taken.has(n):
					pile.append(n)
					spots.erase(n)
		for c in pile:
			taken[c] = true
			_place_prop(c, holder, to_world, rng, counter)
			made += 1


func _place_prop(cell: Vector2i, holder: Node3D, to_world: Callable,
		rng: RandomNumberGenerator, counter: Array, scene: PackedScene = null) -> LootProp:
	if scene == null:
		scene = prop_scenes[0]
		if prop_scenes.size() > 1 and rng.randf() < rare_prop_chance:
			scene = prop_scenes[1]
	var index: int = counter[0]
	var prop := scene.instantiate() as LootProp
	prop.index = index
	holder.add_child(prop)
	prop.global_position = to_world.call(cell)
	prop.rotation.y = rng.randf() * TAU
	_props[index] = prop
	prop.opened.connect(func(_p) -> void: Coop.prop_opened(index))
	if instance != null:
		if instance.opened.has(index):
			prop.set_already_open()
		else:
			var inst := instance
			prop.opened.connect(func(_p) -> void: inst.opened[index] = true)
	counter[0] = index + 1
	counter[1] += 1
	return prop


func _leaves_gap(c: Vector2i, is_floor: Callable) -> bool:
	for d in [Vector2i.LEFT, Vector2i.UP]:
		var a: Vector2i = c + d
		var b: Vector2i = c - d
		if not is_floor.call(Vector3i(a.x, 0, a.y)) and not is_floor.call(Vector3i(b.x, 0, b.y)):
			return false
	return true


func _against_wall(c: Vector2i, is_floor: Callable) -> bool:
	for d in [Vector2i.LEFT, Vector2i.RIGHT, Vector2i.UP, Vector2i.DOWN]:
		var n: Vector2i = c + d
		if not is_floor.call(Vector3i(n.x, 0, n.y)):
			return true
	return false

func _roll_rarity(weights: Dictionary, rng: RandomNumberGenerator) -> String:
	var total := 0
	for w in weights.values():
		total += int(w)
	var roll := rng.randi_range(0, total - 1)
	for r in weights.keys():
		roll -= int(weights[r])
		if roll < 0:
			return r
	return "normal"

func spawn_marked(entries: Array, level: int, is_floor: Callable, to_world: Callable) -> void:
	_level = level
	_is_floor = is_floor
	_to_world = to_world
	for e in entries:
		var def: EnemyDefinition = e["definition"]

			continue
		var cell: Vector2i = e["cell"]

			continue
		var mods: Array = e.get("modifiers", []).duplicate()
		var replace: bool = e.get("replace_random", true)

			for extra in DungeonThreats.roll_traits(no_swarm, e["rarity"], _threat_rng):
				if not mods.has(extra):
					mods.append(extra)
		var enemy := _spawn_enemy(def, e["rarity"], cell, mods, not replace)
		var key: String = e.get("key", "")
		if key != "" and not QuestItems.owned(get_tree(), key):
			if enemy != null:
				enemy.set_meta(&"carries_key", key)
				enemy.health.died.connect(func() -> void:
					_drop_key.call_deferred(key, enemy.global_position))
			else:
				_drop_key.call_deferred(key, to_world.call(cell))

func place_queued_crates(scene_path: String, is_floor: Callable, to_world: Callable) -> void:
	if scene_path.get_file().get_basename() != "act_3_sweep":
		return
	var qm := YardCrates.qm_of(get_tree())
	var ids := YardCrates.queued(qm)
	if ids.is_empty():
		return
	var rooms: Array[RoomData] = []
	for room: RoomData in director.rooms:
		if not (room.room_type in ["start", "portal", "boss"]):
			rooms.append(room)
	if rooms.is_empty():
		return
	var used := {}
	var taken := {_beacon_cell: true, _salvage_cell: true}
	for id in ids:
		var pick := absi(hash(id)) % rooms.size()
		for step in rooms.size():
			if not used.has((pick + step) % rooms.size()):
				pick = (pick + step) % rooms.size()
				break
		used[pick] = true
		var c := rooms[pick].center()
		var placed := false
		for r in range(0, 6):
			for dx in range(-r, r + 1):
				for dy in range(-r, r + 1):
					var t := c + Vector2i(dx, dy)
					if placed or maxi(absi(dx), absi(dy)) != r or _near_taken(t, taken):
						continue
					if is_floor.call(Vector3i(t.x, 0, t.y)):
						QueuedCrate.spawn(get_parent(), to_world.call(t), id)

						placed = true

func place_echoes(director_node: NullEchoDirector, is_floor: Callable, to_world: Callable) -> void:
	var def := director_node.echo_definition()
	if def == null:
		return
	var start := director.layout.spawn_cell if director.layout != null else Vector2i.ZERO
	var rooms: Array[RoomData] = []
	var far: RoomData = null
	for room: RoomData in director.rooms:
		if room.room_type in ["start", "portal", "boss"]:
			continue
		rooms.append(room)
		if far == null or Vector2(room.center() - start).length() > Vector2(far.center() - start).length():
			far = room
	var taken := {}
	for room in rooms:
		var count := director_node.per_room + (1 if director_node.champion and room == far else 0)
		for k in count:
			var cell := _free_cell_near(room.center(), is_floor, taken)
			if cell == Vector2i(-99999, -99999):
				break
			taken[cell] = true
			var champ := director_node.champion and room == far and k == count - 1
			var echo := _spawn_enemy(def, "champion" if champ else "normal", cell, [], false)
			if echo == null:
				continue
			for t in echo.get("traits"):
				if t is NullEcho:
					(t as NullEcho).copies = director_node.copies
					(t as NullEcho).calls_all = champ


func _free_cell_near(c: Vector2i, is_floor: Callable, taken: Dictionary) -> Vector2i:
	for r in range(0, 7):
		for dx in range(-r, r + 1):
			for dy in range(-r, r + 1):
				var t := c + Vector2i(dx, dy)
				if maxi(absi(dx), absi(dy)) != r or _near_taken(t, taken):
					continue
				if is_floor.call(Vector3i(t.x, 0, t.y)):
					return t
	return Vector2i(-99999, -99999)


static func _near_taken(cell: Vector2i, taken: Dictionary) -> bool:
	for t in taken:
		if maxi(absi(cell.x - (t as Vector2i).x), absi(cell.y - (t as Vector2i).y)) < 2:
			return true
	return false


func place_ward_beacon(is_floor: Callable, to_world: Callable) -> void:
	if ward_beacon_id == "":
		return
	var start := director.layout.spawn_cell if director.layout != null else Vector2i.ZERO
	var best: RoomData = null
	for room: RoomData in director.rooms:
		if room.room_type in ["start", "portal", "boss"]:
			continue
		if best == null or Vector2(room.center() - start).length() > Vector2(best.center() - start).length():
			best = room
	if best == null:
		push_warning("No room for the ward beacon %s" % ward_beacon_id)
		return
	var c := best.center()
	var cell := c
	var found := false
	for r in range(0, 6):
		for dx in range(-r, r + 1):
			for dy in range(-r, r + 1):
				var t := c + Vector2i(dx, dy)
				if not found and maxi(absi(dx), absi(dy)) == r and is_floor.call(Vector3i(t.x, 0, t.y)):
					cell = t
					found = true
		if found:
			break
	if not found:
		push_warning("No floor for the ward beacon %s" % ward_beacon_id)
		return
	_beacon_cell = cell
	var beacon := WardBeacon.spawn(get_parent(), to_world.call(cell), ward_beacon_id)
	beacon.wave_event_finished.connect(func() -> void:
		if instance != null:
			_check_cleared(instance))

class_name Portal
extends Area3D

@export_file("*.tscn") var target_area: String
@export var target_spawn: StringName = &"default"
@export var dungeon_level: int = 0
@export var instance_id: int = 0
@export var sealed_by_key: String = ""
@export var open_after_quest: String = ""
@export var solid_while_sealed := false
@export var shown_after_quest: String = ""
@export var route_after_quest: String = ""
@export_file("*.tscn") var routed_area: String = ""
@export var routed_spawn: StringName = &"default"
@export var route_when_done := false
@export var routed_level: int = 0
@export var clears_stop: String = ""

const SEALED_A := Color(1.0, 0.12, 0.15)
const SEALED_B := Color(1.0, 0.55, 0.45)

var _was_sealed := false
var _seal_check := 0.0
var _turned_back := 0.0
var _quest_check := 0.0
var _home_area := ""
var _home_spawn: StringName = &""
var _home_level := 0
var _barrier: StaticBody3D


func is_sealed() -> bool:
	if sealed_by_key == "":
		return false
	var player := get_tree().get_first_node_in_group("players")
	var qm = player.get("quests") if player != null else null
	if qm == null:
		return false
	return not (qm.has_recovered(sealed_by_key) or (open_after_quest != "" and qm.is_completed(open_after_quest)))


static func quest_started(tree: SceneTree, quest_id: String) -> bool:
	var player := tree.get_first_node_in_group("players") if tree != null else null
	var qm = player.get("quests") if player != null else null
	return qm != null and (qm.is_active(quest_id) or qm.is_completed(quest_id))


static func quest_done(tree: SceneTree, quest_id: String) -> bool:
	var player := tree.get_first_node_in_group("players") if tree != null else null
	var qm = player.get("quests") if player != null else null
	return qm != null and qm.is_completed(quest_id)


func set_routing(d: Dictionary) -> void:
	shown_after_quest = str(d.get("shown_after_quest", ""))
	route_after_quest = str(d.get("route_after_quest", ""))
	routed_area = str(d.get("routed_area", ""))
	routed_spawn = StringName(str(d.get("routed_spawn", "default")))
	route_when_done = bool(d.get("route_when_done", false))
	routed_level = int(d.get("routed_level", 0))
	clears_stop = str(d.get("clears_stop", ""))
	if is_inside_tree():
		_update_routing()


func _update_routing() -> void:
	if shown_after_quest != "":
		var shown := quest_started(get_tree(), shown_after_quest)
		visible = shown
		set_deferred("monitoring", shown)
	if route_after_quest != "" and routed_area != "":
		if _home_area == "":
			_home_area = target_area
			_home_spawn = target_spawn
			_home_level = dungeon_level
		var routed := quest_done(get_tree(), route_after_quest) if route_when_done \
				else quest_started(get_tree(), route_after_quest)
		target_area = routed_area if routed else _home_area
		target_spawn = routed_spawn if routed else _home_spawn
		dungeon_level = routed_level if routed else _home_level


func _process(delta: float) -> void:
	_turned_back = maxf(_turned_back - delta, 0.0)
	if shown_after_quest != "" or route_after_quest != "":
		_quest_check -= delta
		if _quest_check <= 0.0:
			_quest_check = 0.5
			_update_routing()
	if sealed_by_key == "":
		return
	_seal_check -= delta
	if _seal_check > 0.0:
		return
	_seal_check = 0.5
	var sealed := is_sealed()
	if sealed != _was_sealed:
		_was_sealed = sealed
		_show_seal(sealed)
		if _barrier != null:
			_barrier.process_mode = Node.PROCESS_MODE_INHERIT if sealed else Node.PROCESS_MODE_DISABLED


func _show_seal(sealed: bool) -> void:
	var mesh := get_node_or_null("MeshInstance3D") as MeshInstance3D
	var mat := mesh.material_override as ShaderMaterial if mesh != null else null
	if mat == null:
		return
	if not has_meta(&"open_a"):
		mat = mat.duplicate()
		mesh.material_override = mat
		set_meta(&"open_a", mat.get_shader_parameter("color_a"))
		set_meta(&"open_b", mat.get_shader_parameter("color_b"))
	mat.set_shader_parameter("color_a", SEALED_A if sealed else get_meta(&"open_a"))
	mat.set_shader_parameter("color_b", SEALED_B if sealed else get_meta(&"open_b"))


func _ready() -> void:
	collision_layer = 5
	collision_mask = 3
	monitoring = true
	body_entered.connect(_on_body_entered)
	_update_routing()
	if solid_while_sealed:
		_build_barrier()


func _build_barrier() -> void:
	_barrier = StaticBody3D.new()
	_barrier.collision_layer = 1
	_barrier.collision_mask = 0
	var shape := CollisionShape3D.new()
	var box := BoxShape3D.new()
	box.size = Vector3(0.3, 2.3, 2.0)
	shape.shape = box
	shape.position.y = 0.67
	_barrier.add_child(shape)
	_barrier.process_mode = Node.PROCESS_MODE_DISABLED
	add_child(_barrier)


func _on_body_entered(body: Node3D) -> void:
	if not body is Player or not visible:
		return
	if (body as Player).is_remote:
		return
	if is_sealed():
		if _turned_back <= 0.0:
			_turned_back = 2.5
			QuestItems.toast(get_tree(), "SEALED: %s\n%s" % [QuestItems.sealed_text(sealed_by_key),
					QuestItems.hint_of(sealed_by_key)], SEALED_A)
		return
	var manager := get_tree().get_first_node_in_group("area_manager") as AreaManager
	if manager == null:
		push_error("Portal: no AreaManager found in the scene")
		return
	if clears_stop != "" and manager.can_go():
		GameEvents.report(Waypoints.FINISH_DEED, clears_stop)
		for w in Waypoints.clear_stop(clears_stop):
			QuestItems.toast(get_tree(), "ROUTE UNLOCKED: %s" % w.display_name.to_upper(),
					Color(0.25, 1.0, 0.55))
		Coop.stop_cleared(clears_stop)
	manager.go_to.call_deferred(target_area, target_spawn, dungeon_level, instance_id)

class_name Waypoints
extends RefCounted

const ACT_FOLDER := "res://data/waypoints/"
const SAVE_KEY := "waypoints"
const SAVE_KEY_CLEARED := "route_cleared"
const FINISH_DEED := "stop_finished"

static var _acts: Array[WaypointAct] = []
static var _visited: Dictionary = {}
static var _cleared: Dictionary = {}
static var _loaded_for := -2


const DOCKED_QUEST := "q04c_the_last_mile"


static func acts() -> Array[WaypointAct]:
	if _acts.is_empty():
		var dir := DirAccess.open(ACT_FOLDER)
		if dir == null:
			push_warning("Waypoints: no folder %s" % ACT_FOLDER)
			return _acts
		for file in dir.get_files():
			var f := file.trim_suffix(".remap")
			if not f.ends_with(".tres"):
				continue
			var act := load(ACT_FOLDER.path_join(f)) as WaypointAct
			if act != null:
				_acts.append(act)
		_acts.sort_custom(func(a: WaypointAct, b: WaypointAct) -> bool:
			return a.act_number < b.act_number)
	return _acts


static func for_scene(scene_path: String) -> Waypoint:
	for act in acts():
		var w := act.find_by_scene(scene_path)
		if w != null:
			return w
	return null


static func top_level(act: WaypointAct) -> int:
	var best := 0
	if act != null:
		for w in act.waypoints:
			best = maxi(best, w.dungeon_level)
	return best


static func act_of_scene(scene_path: String) -> WaypointAct:
	for act in acts():
		if act.find_by_scene(scene_path) != null:
			return act
	return null


static func is_discovered(w: Waypoint) -> bool:
	_ensure_loaded()
	return w != null and _visited.has(w.id)


static func arrived(scene_path: String) -> bool:
	_ensure_loaded()
	var w := for_scene(scene_path)
	if w == null or _visited.has(w.id):
		return false
	if Coop.visiting:
		return false
	_visited[w.id] = true
	_write_to_saves()
	print("[Waypoints] discovered %s" % w.display_name)
	return true


static func _ensure_loaded() -> void:
	var slot: int = SaveManager.party[0] if not SaveManager.party.is_empty() else -1
	if slot == _loaded_for:
		return
	_loaded_for = slot
	_visited.clear()
	_cleared.clear()
	if slot < 0:
		return
	for id in SaveManager.get_data(slot).get(SAVE_KEY, []):
		_visited[str(id)] = true
	for id in SaveManager.get_data(slot).get(SAVE_KEY_CLEARED, []):
		_cleared[str(id)] = true


static func _write_to_saves() -> void:
	var ids: Array = _visited.keys()
	var cleared: Array = _cleared.keys()
	for slot in SaveManager.party:
		SaveManager.get_data(slot)[SAVE_KEY] = ids.duplicate()
		SaveManager.get_data(slot)[SAVE_KEY_CLEARED] = cleared.duplicate()


static func is_convoy_stop(w: Waypoint) -> bool:
	if w == null or w.kind == Waypoint.Kind.TOWN or w.on_foot:
		return false
	var act := act_of_scene(w.scene)
	return act != null and act.convoy_route


static func is_cleared(w: Waypoint) -> bool:
	_ensure_loaded()
	return w != null and _cleared.has(w.id)


static func route_open(w: Waypoint) -> bool:
	if not is_convoy_stop(w):
		return false
	if not route_key_ok(w):
		return false
	if w.route_after == "" or is_cleared(w):
		return true
	var act := act_of_scene(w.scene)
	return is_cleared(act.find(w.route_after))


static func route_key_ok(w: Waypoint) -> bool:
	if w == null or (w.route_needs_key == "" and w.route_needs_quests.is_empty()):
		return true
	var tree := Engine.get_main_loop() as SceneTree
	var player := tree.get_first_node_in_group("players") if tree != null else null
	var qm = player.get("quests") if player != null else null
	if qm == null or (w.route_needs_key != "" and not qm.has_recovered(w.route_needs_key)):
		return false
	for id in w.route_needs_quests:
		if not qm.is_completed(id):
			return false
	return true


static func clear_stop(id: String) -> Array[Waypoint]:
	_ensure_loaded()
	var opened: Array[Waypoint] = []
	if id == "" or _cleared.has(id) or Coop.visiting:
		return opened
	_cleared[id] = true
	_write_to_saves()
	for act in acts():
		for w in act.waypoints:
			if w.route_after == id:
				opened.append(w)
	return opened


static func route_blocked_reason(w: Waypoint, tree: SceneTree) -> String:
	if not route_key_ok(w):
		return w.route_needs_text if w.route_needs_text != "" 				else "LOCKED: NEEDS %s" % QuestItems.name_of(w.route_needs_key)
	if not route_open(w):
		var act := act_of_scene(w.scene)
		var before := act.find(w.route_after) if act != null else null
		return "LOCKED: CLEAR %s FIRST" % (before.display_name.to_upper() if before else "THE STOP BEFORE")
	var am := tree.get_first_node_in_group("area_manager") as AreaManager
	if am == null:
		return "NO AREA MANAGER"
	if am.is_busy():
		return "TRAVELLING..."
	return breach_reason(tree)


static func is_docked(qm: QuestManager) -> bool:
	return qm != null and qm.is_completed(DOCKED_QUEST)


static func breach_reason(tree: SceneTree) -> String:
	for b in tree.get_nodes_in_group("hull_breach"):
		if b.has_method("blocks_route") and b.blocks_route():
			if b.has_method("route_block_text"):
				return b.route_block_text()
			return "HULL BREACH: CLEAR THE DECK FIRST" if b.is_running() \
					else "THE NULL ARE BOARDING: HOLD THE DECK FIRST"
	return ""


static func sail(w: Waypoint, tree: SceneTree, new_instance := false) -> bool:
	if route_blocked_reason(w, tree) != "":
		return false
	var am := tree.get_first_node_in_group("area_manager") as AreaManager
	am.go_to(w.scene, w.spawn_id, w.dungeon_level,
			WorldState.ALWAYS_NEW if new_instance else WorldState.REJOIN_OR_NEW)
	return true


static func blocked_reason(w: Waypoint, tree: SceneTree, new_instance := false) -> String:
	if not is_discovered(w):
		return "UNDISCOVERED"
	if is_convoy_stop(w):
		return "USE THE ROUTE TERMINAL OR THE DISTRICT GATE"
	var am := tree.get_first_node_in_group("area_manager") as AreaManager
	if am == null:
		return "NO AREA MANAGER"
	if am.is_busy():
		return "TRAVELLING..."
	if normalize(am.current_scene_path()) == normalize(w.scene) and not (new_instance and is_instanced(w)):
		return "YOU ARE HERE"
	for p in tree.get_nodes_in_group("players"):
		var h = p.get("health")
		if h != null and h.current_health <= 0.0:
			return "YOU ARE DEAD"
	return ""


static func travel(w: Waypoint, tree: SceneTree, new_instance := false) -> bool:
	if blocked_reason(w, tree, new_instance) != "":
		return false
	var am := tree.get_first_node_in_group("area_manager") as AreaManager
	am.travel(w.scene, w.spawn_id, w.dungeon_level, new_instance and is_instanced(w))
	return true


static func is_instanced(w: Waypoint) -> bool:
	return w != null and w.dungeon_level > 0


static func normalize(path: String) -> String:
	if path.begins_with("uid://"):
		var id := ResourceUID.text_to_id(path)
		if ResourceUID.has_id(id):
			return ResourceUID.get_id_path(id)
	return path


================================================================================
# FILE: world_state.gd

extends Node

signal instance_created(instance: DungeonInstance)
signal instance_closed(instance: DungeonInstance)

const EMPTY_TIMEOUT_SEC := 600.0
const MAX_INSTANCES := 5
const CHECK_EVERY_SEC := 1.0

const REJOIN_OR_NEW := 0
const ALWAYS_NEW := -1

var instances: Dictionary = {}
var _next_id := 1
var _pending: DungeonInstance
var _timer := 0.0


func _ready() -> void:
	process_mode = Node.PROCESS_MODE_ALWAYS


func _process(delta: float) -> void:
	_timer += delta
	if _timer < CHECK_EVERY_SEC:
		return
	_timer = 0.0
	close_expired()


func prepare_dungeon(scene_path: String, level: int, instance_id: int = REJOIN_OR_NEW) -> DungeonInstance:
	scene_path = Waypoints.normalize(scene_path)
	var inst: DungeonInstance = null
	if instance_id > 0:
		inst = instances.get(instance_id)
	elif instance_id == REJOIN_OR_NEW:
		inst = find_open(scene_path, level)
	if inst == null:
		inst = _create(scene_path, level)
	_pending = inst
	return inst


func adopt(id: int, scene_path: String, level: int, rng_seed: int, killed: Array) -> DungeonInstance:
	scene_path = Waypoints.normalize(scene_path)
	var inst: DungeonInstance = instances.get(id)
	if inst == null or inst.scene_path != scene_path or inst.rng_seed != rng_seed:
		inst = DungeonInstance.new()
		inst.id = id
		inst.scene_path = scene_path
		inst.level = level
		inst.rng_seed = rng_seed
		inst.threats = DungeonThreats.roll_for(scene_path, rng_seed)
		inst.created_msec = Time.get_ticks_msec()
		inst.empty_since_msec = inst.created_msec
		instances[id] = inst
		_next_id = maxi(_next_id, id + 1)
	for index in killed:
		inst.killed[int(index)] = true
	_pending = inst
	return inst


func prepare_rejoin(inst: DungeonInstance) -> void:
	if inst != null and instances.has(inst.id):
		_pending = inst


func find_open(scene_path: String, level: int) -> DungeonInstance:
	scene_path = Waypoints.normalize(scene_path)
	var best: DungeonInstance = null
	for inst: DungeonInstance in instances.values():
		if inst.scene_path == scene_path and inst.level == level and (best == null or inst.id > best.id):
			best = inst
	return best


func begin_visit(scene_path: String, default_level: int = 1) -> DungeonInstance:
	scene_path = Waypoints.normalize(scene_path)
	var inst := _pending
	_pending = null
	if inst == null:
		inst = _create(scene_path, default_level)
	inst.occupants += 1
	return inst


func end_visit(inst: DungeonInstance) -> void:
	if inst == null:
		return
	inst.occupants = maxi(inst.occupants - 1, 0)
	if inst.occupants == 0:
		inst.empty_since_msec = Time.get_ticks_msec()


func active_instances() -> Array[DungeonInstance]:
	var out: Array[DungeonInstance] = []
	for inst in instances.values():
		out.append(inst)
	return out


func close_expired() -> void:
	for id in instances.keys():
		var inst: DungeonInstance = instances[id]
		if inst.occupants == 0 and inst != _pending and inst.seconds_empty() >= EMPTY_TIMEOUT_SEC:
			_close(inst)


func _create(scene_path: String, level: int) -> DungeonInstance:
	_make_room()
	var inst := DungeonInstance.new()
	inst.id = _next_id
	_next_id += 1
	inst.scene_path = scene_path
	inst.level = level
	inst.rng_seed = randi()
	inst.threats = DungeonThreats.roll_for(scene_path, inst.rng_seed)
	inst.created_msec = Time.get_ticks_msec()
	inst.empty_since_msec = inst.created_msec
	instances[inst.id] = inst
	instance_created.emit(inst)
	return inst


func _make_room() -> void:
	while instances.size() >= MAX_INSTANCES:
		var oldest: DungeonInstance = null
		for inst in instances.values():
			if inst.occupants == 0 and (oldest == null or inst.empty_since_msec < oldest.empty_since_msec):
				oldest = inst
		if oldest == null:
			return
		_close(oldest)


func _close(inst: DungeonInstance) -> void:
	instances.erase(inst.id)
	instance_closed.emit(inst)
