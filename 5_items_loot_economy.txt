class_name EquipmentComponent
extends Node

@export var stat_component: StatComponent

var equipped: Dictionary = {}

signal equipment_changed

var sandbox_locked := false


func equip(item: ItemInstance, slot: int = -1) -> bool:
	if slot < 0 or not item.definition.is_stim():
		slot = slot_for(item)
	if slot < 0:
		return false
	if not can_equip(item):
		return false
	if equipped.has(slot):
		unequip(slot)
	equipped[slot] = item
	for m in item.build_modifiers():
		stat_component.add_modifier(m)
	equipment_changed.emit()
	GameEvents.report("equip", item.definition.id)
	GameEvents.report("equip_slot", str(slot))
	return true


func slot_for(item: ItemInstance) -> int:
	if item == null:
		return -1
	if item.definition.is_stim():
		if item.stim_roll.is_empty():
			StimRoller.recompute(item)
		for s in EquipmentSlots.STIM_SLOTS:
			if not equipped.has(s):
				return s
		return EquipmentSlots.STIM_SLOTS[0]
	return item.definition.get_slot()


var last_block_reason: String = ""


func can_equip(item: ItemInstance) -> bool:
	last_block_reason = ""
	if sandbox_locked:
		last_block_reason = "The exam uses its own loadout."
		return false
	var slot: int = slot_for(item)
	if slot == EquipmentSlots.Slot.OFF_HAND \
			and _is_two_handed(get_equipped(EquipmentSlots.Slot.WEAPON)):
		last_block_reason = "Can't use an off-hand with a two-handed weapon."
		return false
	if slot == EquipmentSlots.Slot.WEAPON and _is_two_handed(item) \
			and get_equipped(EquipmentSlots.Slot.OFF_HAND) != null:
		last_block_reason = "Unequip your off-hand first."
		return false
	return true


func _is_two_handed(item: ItemInstance) -> bool:
	return item != null and item.definition.weapon_type != null \
		and item.definition.weapon_type.two_handed


func unequip(slot: int) -> bool:
	last_block_reason = ""
	if sandbox_locked:
		last_block_reason = "The exam uses its own loadout."
		return false
	if not equipped.has(slot):
		return false
	var item: ItemInstance = equipped[slot]
	stat_component.remove_by_source_prefix("item_%s_" % item.definition.id)
	equipped.erase(slot)
	equipment_changed.emit()
	return true


func get_equipped(slot: int) -> ItemInstance:
	return equipped.get(slot)


func swap_to(items: Dictionary) -> void:
	var locked := sandbox_locked
	sandbox_locked = false
	for slot in equipped.keys():
		unequip(slot)
	for slot in items:
		equip(items[slot], slot)
	sandbox_locked = locked

func _enter_tree() -> void:
	add_to_group("inventory_panel")


func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_STOP
	_font = font
	if _font == null and ResourceLoader.exists(DEFAULT_FONT):
		_font = load(DEFAULT_FONT)
	_layout()
	_build_ui()
	_built = true
	visible = false
	_apply_bound_data()


func _layout() -> void:
	anchor_left = 1.0
	anchor_right = 1.0
	anchor_top = 0.0
	anchor_bottom = 1.0
	offset_top = 32.0
	offset_bottom = -32.0
	_set_offsets(false)


func _set_offsets(opened: bool) -> void:
	offset_left = -(OPEN_X + panel_width) if opened else CLOSED_GAP
	offset_right = -OPEN_X if opened else CLOSED_GAP + panel_width


func bind(inventory: InventoryComponent, equipment: EquipmentComponent) -> void:
	if _inv and _inv.changed.is_connected(_on_data_changed):
		_inv.changed.disconnect(_on_data_changed)
	if _equip and _equip.equipment_changed.is_connected(_on_data_changed):
		_equip.equipment_changed.disconnect(_on_data_changed)
	_inv = inventory
	_equip = equipment
	if _inv:
		_inv.changed.connect(_on_data_changed)
	if _equip:
		_equip.equipment_changed.connect(_on_data_changed)
	if _built:
		_apply_bound_data()


func toggle() -> void:
	if _open:
		close()
	else:
		open()

func is_open() -> bool:
	return _open

func open() -> void:
	if _open or not _built:
		return
	_open = true
	open_changed.emit(true)
	GameEvents.report("open_panel", "inventory")
	var mode := Input.mouse_mode
	if mode == Input.MOUSE_MODE_CAPTURED or mode == Input.MOUSE_MODE_HIDDEN \
			or mode == Input.MOUSE_MODE_CONFINED_HIDDEN:
		_prev_mouse_mode = mode
		_changed_mouse_mode = true
		Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
	if _inv == null or _equip == null:
		push_warning("InventoryPanel: not bound. Is bind() being called from the player?")
	_show_tab(false)
	if not visible:
		_set_offsets(false)
		modulate.a = 0.0
		visible = true
	_play_slide(true)


func close() -> void:
	if not _open:
		return
	_open = false
	open_changed.emit(false)
	_hover_slot = null
	_update_tooltip()
	if _changed_mouse_mode:
		Input.mouse_mode = _prev_mouse_mode
		_changed_mouse_mode = false
	_play_slide(false)


func _unhandled_input(event: InputEvent) -> void:
	if not (event is InputEventKey and event.pressed and not event.echo):
		return
	if event.keycode == toggle_key:
		toggle()
		get_viewport().set_input_as_handled()
	elif event.keycode == KEY_ESCAPE and _open:
		close()
		get_viewport().set_input_as_handled()


func _play_slide(opening: bool) -> void:
	if _tween:
		_tween.kill()
	_tween = create_tween().set_parallel(true)
	var left := -(OPEN_X + panel_width) if opening else CLOSED_GAP
	var right := -OPEN_X if opening else CLOSED_GAP + panel_width
	_tween.tween_property(self, "offset_left", left, 0.22) \
		.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
	_tween.tween_property(self, "offset_right", right, 0.22) \
		.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
	_tween.tween_property(self, "modulate:a", 1.0 if opening else 0.0, 0.18)
	if not opening:
		_tween.finished.connect(_on_slide_finished)


func _on_slide_finished() -> void:
	if not _open:
		visible = false


func _on_data_changed() -> void:
	if _built and not _suppress_refresh:
		_refresh()

func _refresh_quest_items() -> void:
	if _quest_row == null:
		return
	if _quests == null or not is_instance_valid(_quests):
		var p := get_tree().get_first_node_in_group("players")
		_quests = p.get("quests") as QuestManager if p != null else null
		if _quests != null:
			_quests.item_recovered.connect(func(_id: String) -> void: _refresh_quest_items())
			_quests.quest_completed.connect(func(_s: QuestState) -> void: _refresh_quest_items())
	var ids := QuestItems.carried(_quests)
	var shown := PackedStringArray()
	for c in _quest_row.get_children():
		shown.append((c as _QuestItemIcon).item_id)
	if shown != ids:
		for c in _quest_row.get_children():
			c.queue_free()
		_hover_quest = null
		for id in ids:
			var icon := _QuestItemIcon.new()
			icon.item_id = id
			icon.custom_minimum_size = Vector2(slot_size, slot_size)
			icon.mouse_entered.connect(func() -> void:
				_hover_quest = icon
				_update_tooltip())
			icon.mouse_exited.connect(func() -> void:
				if _hover_quest == icon:
					_hover_quest = null
				_update_tooltip())
			_quest_row.add_child(icon)
	_quest_none.visible = ids.is_empty()
	_tab_quest.text = "QUEST ITEMS (%d)" % ids.size() if not ids.is_empty() else "QUEST ITEMS"
	if _quest_tab:
		_count_label.text = "%d %s" % [ids.size(), "ITEM" if ids.size() == 1 else "ITEMS"]

		var d := _label(desc, 15, COL_TEXT)

		var keep := _label("Kept until %s is turned in." % users[0].title.capitalize(), 13, COL_DIM)
		keep.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		box.add_child(keep)

func _make_slot() -> ItemSlot:
	var s := ItemSlot.new()
	s.custom_minimum_size = Vector2(slot_size, slot_size)
	s.accent = accent
	s.font = _font if _font else ThemeDB.fallback_font
	s.rarity_colors = RARITY_COLORS
	s.hovered.connect(_on_slot_hovered)
	s.unhovered.connect(_on_slot_unhovered)
	s.clicked.connect(_on_slot_clicked)
	return s


func _rebuild_grid(count: int) -> void:
	for child in _grid.get_children():
		_grid.remove_child(child)
		child.queue_free()
	_grid_slots.clear()
	for i in count:
		var s := _make_slot()
		s.kind = "bag"
		_grid.add_child(s)
		_grid_slots.append(s)


func _apply_bound_data() -> void:
	var cap := _inv.capacity if _inv else DEFAULT_CAPACITY
	if cap != _grid_slots.size():
		_rebuild_grid(cap)
	_refresh()


func _refresh() -> void:
	if not _built:
		return
	if _selected != null and not _owns(_selected):
		_selected = null

	for def in EQUIP_SLOTS + STIM_ROW:
		var s: ItemSlot = _eq_slots[def["slot"]]
		s.item = _equip.get_equipped(def["slot"]) if _equip else null

	var items: Array[ItemInstance] = []
	if _inv:
		items = _inv.items
	for i in _grid_slots.size():
		_grid_slots[i].item = items[i] if i < items.size() else null
	_count_label.text = "%d / %d" % [items.size(), _grid_slots.size()]

	_refresh_quest_items()
	_sync_selection()
	_update_tooltip()
	_update_buttons()
	_refresh_credits()


func _sync_selection() -> void:
	for s in _grid_slots:
		s.selected = s.item != null and s.item == _selected
	for slot_id in _eq_slots:
		var s: ItemSlot = _eq_slots[slot_id]
		s.selected = s.item != null and s.item == _selected


func _update_buttons() -> void:
	var has := _selected != null
	_equip_btn.disabled = not has
	_equip_btn.text = "UNEQUIP" if (has and _is_equipped(_selected)) else "EQUIP"
	_discard_btn.disabled = not (has and _inv != null and _inv.has_item(_selected))

func _update_tooltip() -> void:
	if _tip == null:
		return
	if _open and _hover_quest != null and is_instance_valid(_hover_quest):
		_tip_anchor = _hover_quest
		_tip.clear()
		_fill_quest_detail(_tip.box, _hover_quest.item_id)
		_tip.visible = true
		_tip_cmp.visible = false
		_place_tips()
		return
	var ctrl := Input.is_key_pressed(KEY_CTRL)
	var slot: ItemSlot = null
	if _hover_slot != null and is_instance_valid(_hover_slot) and _hover_slot.item != null:
		slot = _hover_slot
	elif ctrl and _selected != null:
		slot = _slot_showing(_selected)
	if slot == null or not _open:
		_tip.visible = false
		_tip_cmp.visible = false
		_tip_anchor = null
		return
	var item := slot.item
	_tip_anchor = slot
	_tip.clear()
	fill_item_detail(_tip.box, item)

	var comparable := _can_compare(item)
	var current: ItemInstance = null
	if comparable:
		var target: int = _equip.slot_for(item)
		current = _equip.get_equipped(target) if target >= 0 else null
		if ctrl:
			_add_compare_lines(_tip.box, current, item)
		else:
			_tip.box.add_child(_rule())
			var h := _label("HOLD CTRL TO COMPARE", 13, accent)
			h.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
			_tip.box.add_child(h)
	_tip.visible = true

	if comparable and ctrl and current != null:
		_tip_cmp.clear()
		var head := _label("CURRENTLY EQUIPPED", 13, accent)
		_tip_cmp.box.add_child(head)
		fill_item_detail(_tip_cmp.box, current)
		_tip_cmp.visible = true
	else:
		_tip_cmp.visible = false
	_place_tips()


func can_compare(item: ItemInstance) -> bool:
	return _can_compare(item)


func equipped_for(item: ItemInstance) -> ItemInstance:
	if _equip == null or item == null:
		return null
	var target: int = _equip.slot_for(item)
	return _equip.get_equipped(target) if target >= 0 else null


func add_compare_lines(box: VBoxContainer, current: ItemInstance, item: ItemInstance) -> void:
	_add_compare_lines(box, current, item)


func _can_compare(item: ItemInstance) -> bool:
	return _equip != null and item != null and not _is_equipped(item) \
		and not item.definition.is_gem() and not item.definition.is_stim() \
		and _equip.slot_for(item) >= 0


func _add_compare_lines(box: VBoxContainer, current: ItemInstance, item: ItemInstance) -> void:
	box.add_child(_rule())
	box.add_child(_label("IF EQUIPPED" if current == null else "IF SWAPPED", 14, accent))
	if not _equip.can_equip(item):
		var why := _label(_equip.last_block_reason.to_upper(), 13, COL_WARN)
		why.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		why.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		box.add_child(why)
	var changes := stat_changes(current, item)
	if changes.is_empty():
		box.add_child(_label("NO STAT CHANGES", 14, COL_DIM))
		return
	for c in changes:
		var stat: int = c[0]
		var d: float = c[1]
		var better := (d > 0.0) != (stat in LOWER_IS_BETTER)
		var row := HBoxContainer.new()
		row.mouse_filter = Control.MOUSE_FILTER_IGNORE
		var name_l := _label(_compare_label(stat), 15, COL_TEXT)
		name_l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		row.add_child(name_l)
		row.add_child(_label(_delta_text(stat, d), 15, COL_BETTER if better else COL_WORSE))
		box.add_child(row)


func stat_changes(current: ItemInstance, item: ItemInstance) -> Array:
	var src: StatComponent = _equip.stat_component if _equip else null
	if src == null:
		return []
	var before := _stat_copy(src)
	var after := _stat_copy(src)
	if current != null:
		after.remove_by_source_prefix("item_%s_" % current.definition.id)
	for m in item.build_modifiers():
		after.add_modifier(m)
	AttributeEffects.apply(before)
	AttributeEffects.apply(after)
	var out: Array = []
	for stat in StatTypes.Stat.values():
		var d := after.get_stat(stat) - before.get_stat(stat)
		if absf(d) > 0.0005:
			out.append([stat, d])
	before.free()
	after.free()
	return out


func _stat_copy(src: StatComponent) -> StatComponent:
	var c := StatComponent.new()
	c.base_stats = src.base_stats.duplicate()
	for m in src._modifiers:
		c.add_modifier(m)
	return c


func _compare_label(stat: int) -> String:
	match stat:
		StatTypes.Stat.MIN_ATTACK_DAMAGE:
			return "Min Attack Damage"
		StatTypes.Stat.MAX_ATTACK_DAMAGE:
			return "Max Attack Damage"
	return _stat_label(stat)


func _delta_text(stat: int, d: float) -> String:
	match stat:
		StatTypes.Stat.CRIT_CHANCE, StatTypes.Stat.SKILL_DAMAGE_MULT:
			return "%+.1f%%" % (d * 100.0)
		StatTypes.Stat.CRIT_MULTIPLIER, StatTypes.Stat.ATTACK_SPEED:
			return "%+.2f" % d
		StatTypes.Stat.MOVE_SPEED, StatTypes.Stat.LIFE_ON_HIT, StatTypes.Stat.REGENERATION, \
		StatTypes.Stat.MANA_REGEN, StatTypes.Stat.ATTACK_RANGE:
			return "%+.1f" % d
	if absf(d - roundf(d)) < 0.05:
		return "%+d" % roundi(d)
	return "%+.1f" % d


func _slot_showing(item: ItemInstance) -> ItemSlot:
	for s in _grid_slots:
		if s.item == item:
			return s
	for slot_id in _eq_slots:
		if _eq_slots[slot_id].item == item:
			return _eq_slots[slot_id]
	return null


func _place_tips() -> void:
	if _tip_anchor == null or not is_instance_valid(_tip_anchor) or not _tip.visible:
		return
	var r := _tip_anchor.get_global_rect()
	var screen := get_viewport_rect().size
	var gap := 12.0
	_tip.reset_size()
	var left := minf(get_global_rect().position.x, r.position.x)
	var pos := Vector2(left - _tip.size.x - gap, r.position.y)
	if pos.x < 8.0:
		pos.x = r.end.x + gap
	pos.y = clampf(pos.y, 8.0, maxf(8.0, screen.y - _tip.size.y - 8.0))
	_tip.global_position = pos.round()
	if _tip_cmp.visible:
		_tip_cmp.reset_size()
		var p2 := Vector2(pos.x - _tip_cmp.size.x - gap, pos.y)
		if p2.x < 8.0:
			p2.x = pos.x + _tip.size.x + gap
		p2.y = clampf(p2.y, 8.0, maxf(8.0, screen.y - _tip_cmp.size.y - 8.0))
		_tip_cmp.global_position = p2.round()


func _process(_delta: float) -> void:
	if _tip != null and _tip.visible:
		_place_tips()

func fill_item_detail(box: VBoxContainer, item: ItemInstance) -> void:
	var def := item.definition
	var rarity_col: Color = RARITY_COLORS.get(item.rarity, COL_TEXT)

	var name_l := _label(item.get_display_name().to_upper(), 22, rarity_col)
	name_l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	name_l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	box.add_child(name_l)
	box.add_child(_label("%s  //  %s" % [_type_name(def), item.rarity.to_upper()], 14, COL_DIM))
	if Input.is_key_pressed(KEY_ALT):
		box.add_child(_label("ITEM LEVEL  %d" % item.item_level, 14, accent))
	var desc: String = def.get_description()
	if desc != "":
		var d := _label(desc, 14, COL_DIM)
		d.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		d.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		box.add_child(d)
	if _is_equipped(item):
		box.add_child(_label("[ EQUIPPED ]", 14, accent))
	else:
		var slot: int = _equip.slot_for(item) if _equip else def.get_slot()
		var current: ItemInstance = _equip.get_equipped(slot) if (_equip and slot >= 0) else null
		if current != null:
			box.add_child(_label("REPLACES  " + current.get_display_name().to_upper(), 13, COL_DIM))

	if def.stim != null:
		box.add_child(_rule())
		for entry in StimRoller.lines(item):
			var sl := _label(entry[0], 16 if entry[1] else 17, COL_AFFIX if entry[1] else COL_TEXT)
			sl.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
			sl.size_flags_horizontal = Control.SIZE_EXPAND_FILL
			box.add_child(sl)
		return
	var base_lines := _base_lines(def.base_stats)
	if not base_lines.is_empty() or not item.rolled_affixes.is_empty():
		box.add_child(_rule())
	for text in base_lines:
		box.add_child(_label(text, 17, COL_TEXT))
	if not base_lines.is_empty() and not item.rolled_affixes.is_empty():
		box.add_child(_rule())
	var show_ids := Input.is_key_pressed(KEY_ALT)
	for entry in item.rolled_affixes:
		if show_ids:
			var a: AffixDefinition = entry["affix"]
			box.add_child(_label(a.id, 11, Color(COL_DIM, 0.6)))
		var l := _label(_affix_text(entry), 16, COL_AFFIX)
		l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		box.add_child(l)

func _type_name(def: ItemDefinition) -> String:
	if def.weapon_type != null:
		return def.weapon_type.display_name.to_upper()
	if def.equipment_type != null:
		return def.equipment_type.display_name.to_upper()
	if def.skill_gem != null:
		return "SKILL GEM"
	if def.support_gem != null:
		return "SUPPORT GEM"
	if def.stim != null:
		return "HEALTH STIM" if def.stim.kind == StimDefinition.Kind.HEALTH else "MANA STIM"
	return "ITEM"


func _stat_label(stat: int) -> String:
	if STAT_LABELS.has(stat):
		return STAT_LABELS[stat]
	return str(StatTypes.Stat.find_key(stat)).capitalize()


func _base_lines(base: Dictionary) -> Array[String]:
	var out: Array[String] = []
	var mn: int = StatTypes.Stat.MIN_ATTACK_DAMAGE
	var mx: int = StatTypes.Stat.MAX_ATTACK_DAMAGE
	var paired := base.has(mn) and base.has(mx)
	if paired:
		out.append("Attack Damage  %d - %d" % [roundi(base[mn]), roundi(base[mx])])
	for stat in base.keys():
		if paired and (stat == mn or stat == mx):
			continue
		out.append("%s  %s" % [_stat_label(stat), _flat_text(stat, base[stat])])
	return out


func _flat_text(stat: int, v: float) -> String:
	match stat:
		StatTypes.Stat.CRIT_CHANCE, StatTypes.Stat.SKILL_DAMAGE_MULT:
			return "%+.1f%%" % (v * 100.0)
		StatTypes.Stat.CRIT_MULTIPLIER:
			return "%+.2f" % v
		StatTypes.Stat.ATTACK_SPEED, StatTypes.Stat.MOVE_SPEED, \
		StatTypes.Stat.LIFE_ON_HIT, StatTypes.Stat.REGENERATION:
			return "%+.1f" % v
	return "%+d" % roundi(v)


func _pct(v: float) -> String:
	return ("%.1f" % (v * 100.0)).trim_suffix(".0")


func _range_name(a: AffixDefinition) -> String:
	var n1 := _stat_label(a.min_stat)
	var n2 := _stat_label(a.max_stat)
	return n1 if n1 == n2 else "%s and %s" % [n1, n2]


func _affix_text(entry: Dictionary) -> String:
	var a: AffixDefinition = entry["affix"]
	var v: float = float(entry.get("rolled_value", 0.0))
	match a.shape:
		AffixDefinition.AffixShape.DAMAGE_RANGE:
			return "Adds %d - %d %s" % [roundi(entry["rolled_min"]), roundi(entry["rolled_max"]), _range_name(a)]
		AffixDefinition.AffixShape.PERCENT_RANGE:
			return "%s%% %s %s" % [_pct(absf(v)), "increased" if v >= 0.0 else "reduced", _range_name(a)]
		AffixDefinition.AffixShape.PERCENT_MULT_RANGE:
			return "%s%% %s %s" % [_pct(absf(v)), "more" if v >= 0.0 else "less", _range_name(a)]
	var label := _stat_label(a.stat)
	match a.op:
		Modifier.ModOp.FLAT_ADD:
			return "%s %s" % [_flat_text(a.stat, v), label]
		Modifier.ModOp.PERCENT_ADD:
			return "%s%% %s %s" % [_pct(absf(v)), "increased" if v >= 0.0 else "reduced", label]
		Modifier.ModOp.PERCENT_MULT:
			return "%s%% %s %s" % [_pct(absf(v)), "more" if v >= 0.0 else "less", label]
	return "%s: %s" % [label, str(v)]

func _primary_action() -> void:
	if _selected == null or _inv == null or _equip == null:
		return
	if _is_equipped(_selected):
		_unequip_selected()
	else:
		_equip_selected()


func _equip_selected() -> void:
	var item := _selected
	if item.definition.is_gem():
		var panel := get_tree().get_first_node_in_group("skill_panel") as SkillPanel
		if panel == null:
			_notify("SKILLS MENU NOT FOUND")
			return
		var error: String = panel.quick_socket(item)
		if error != "":
			_notify(error)
		return
	var slot: int = _equip.slot_for(item)
	if slot < 0:
		_notify("CANNOT EQUIP THIS ITEM")
		return
	var previous: ItemInstance = _equip.get_equipped(slot)

	_suppress_refresh = true
	_inv.remove_item(item)
	var ok: bool = _equip.equip(item, slot)
	if ok:
		if previous != null:
			_inv.add_item(previous)
	else:
		_inv.add_item(item)
		_notify(_equip.last_block_reason.to_upper() if _equip.last_block_reason != "" else "CANNOT EQUIP THIS ITEM")
	_suppress_refresh = false
	_refresh()


func _unequip_selected() -> void:
	if _inv.is_full():
		_notify("BACKPACK FULL")
		return
	var item := _selected
	var slot := _slot_of(item)
	if slot < 0:
		return
	_suppress_refresh = true
	if _equip.unequip(slot):
		_inv.add_item(item)
	else:
		_notify(_equip.last_block_reason.to_upper() if _equip.last_block_reason != "" else "CANNOT UNEQUIP THIS ITEM")
	_suppress_refresh = false
	_refresh()


func _discard_selected() -> void:
	if _selected == null or _inv == null or not _inv.has_item(_selected):
		return
	var qm: QuestManager = _inv.get_parent().get("quests")
	var keep := qm.keep_reason(_selected) if qm != null else ""
	if keep != "":
		_notify(keep)
		return
	var item := _selected
	_selected = null
	_inv.remove_item(item)
	_refresh()


func _on_sort_pressed() -> void:
	if _inv:
		_inv.sort_items()


func _is_equipped(item: ItemInstance) -> bool:
	return _equip != null and item != null and _equip.equipped.values().has(item)


func _slot_of(item: ItemInstance) -> int:
	if _equip == null:
		return -1
	for slot in _equip.equipped:
		if _equip.equipped[slot] == item:
			return slot
	return -1


func _owns(item: ItemInstance) -> bool:
	return (_inv != null and _inv.has_item(item)) or _is_equipped(item)

	_msg_tween.tween_property(_msg_label, "modulate:a", 0.0, 0.5)

class ItemSlot extends Control:
	signal hovered(slot)
	signal unhovered
	signal clicked(slot, button, double)

	var item: ItemInstance:
		set(value):
			item = value
			queue_redraw()
	var selected := false:
		set(value):
			selected = value
			queue_redraw()
	var disabled := false:
		set(value):
			disabled = value
			queue_redraw()
	var disabled_label := ""
	var kind := "bag"
	var slot_id := -1
	var placeholder := ""
	var placeholder_kind := ""
	var accent := Color.WHITE
	var font: Font
	var rarity_colors := {}
	var _hover := false

	func _init() -> void:
		mouse_filter = Control.MOUSE_FILTER_STOP
		mouse_entered.connect(func() -> void:
			_hover = true
			queue_redraw()
			hovered.emit(self))
		mouse_exited.connect(func() -> void:
			_hover = false
			queue_redraw()
			unhovered.emit())

	func _gui_input(event: InputEvent) -> void:
		if event is InputEventMouseButton and event.pressed:
			if event.button_index == MOUSE_BUTTON_LEFT or event.button_index == MOUSE_BUTTON_RIGHT:
				clicked.emit(self, event.button_index, event.double_click)
				accept_event()

	func _draw() -> void:
		var w := size.x
		var h := size.y
		var c := 10.0
		var poly := PackedVector2Array([
			Vector2(0, 0), Vector2(w - c, 0), Vector2(w, c),
			Vector2(w, h), Vector2(c, h), Vector2(0, h - c)])
		var col := accent
		if item != null:
			col = rarity_colors.get(item.rarity, accent)
		var off := disabled and item != null
		if off:
			col = Color(0.42, 0.43, 0.46)
		draw_colored_polygon(poly, Color(col, 0.16 if item != null else 0.05))

		var outline := PackedVector2Array(poly)
		outline.append(poly[0])
		var line_alpha := 0.9 if item != null else 0.25
		if off:
			line_alpha = 0.5
		elif _hover or selected:
			draw_polyline(outline, Color(col, 0.25), 7.0, true)
			line_alpha = 1.0
		draw_polyline(outline, Color(col, line_alpha), 3.0 if selected and not off else 2.0, true)

		if item != null:
			var def := item.definition

extends Node

const DEBUG := false
const MIN_TIER_MULT := 0.01

var all_affixes: Array[AffixDefinition] = []
var _affix_by_id: Dictionary = {}

func _ready() -> void:
	_load_affixes("res://data/affixes/")
	print("Loaded %d affixes" % [all_affixes.size()])

func _load_affixes(path: String) -> void:
	var dir := DirAccess.open(path)
	if dir == null:
		push_warning("Cannot open %s" % path)
		return
	for sub in dir.get_directories():
		_load_affixes(path.path_join(sub))
	for file_name in dir.get_files():
		if file_name.ends_with(".tres") or file_name.ends_with(".tres.remap"):
			var res = load(path.path_join(file_name.trim_suffix(".remap")))
			if res is AffixDefinition:
				all_affixes.append(res)
				_affix_by_id[res.id] = res

func generate_item(def: ItemDefinition, drop_level: int, rarity: String) -> ItemInstance:
	if def.skill_gem != null:
		rarity = "skill_gem"
	elif def.support_gem != null:
		rarity = "support_gem"
	var item_level := maxi(drop_level, def.item_level)
	var instance := ItemInstance.new()
	instance.definition = def
	instance.rarity = rarity
	instance.item_level = item_level
	if def.stim != null:
		StimRoller.roll(instance)
		return instance
	
	var affix_count := _roll_affix_count(rarity)
	var combined_tags := _get_affix_tags(def)
	if DEBUG: print("Combined tags for %s: %s" % [def.id, combined_tags])
	var eligible = all_affixes.filter(func(a):
		return a.min_item_level <= item_level and _tags_overlap(a.tags, combined_tags)
	)
	if DEBUG: print("Item level %d: %d eligible affixes (of %d)" % [item_level, eligible.size(), all_affixes.size()])
	var prefixes_used := 0
	var suffixes_used := 0
	var used_ids: Dictionary = {}
	var used_exclusivity_keys: Dictionary = {}
	
	for i in affix_count:
		var pool = eligible.filter(func(a):
			if a.id in used_ids: return false
			if a.affix_type == "prefix" and prefixes_used >= 3: return false
			if a.affix_type == "suffix" and suffixes_used >= 3: return false
			for key in _exclusivity_keys(a):
				if key in used_exclusivity_keys:
					return false
			return true
		)
		if pool.is_empty(): break
		var chosen = _weighted_pick(pool)
		used_ids[chosen.id] = true
		for key in _exclusivity_keys(chosen):
			used_exclusivity_keys[key] = true
			if DEBUG: print("Used exclusivity keys: %s" % used_exclusivity_keys)
		if chosen.affix_type == "prefix": prefixes_used += 1
		else: suffixes_used += 1
		
		if chosen.shape == AffixDefinition.AffixShape.DAMAGE_RANGE:
			var roll_t := randf()
			var rolled_min = lerp(chosen.added_min_low, chosen.added_min_high, roll_t)
			var rolled_max = lerp(chosen.added_max_low, chosen.added_max_high, roll_t)
			instance.rolled_affixes.append({
				"affix": chosen,
				"rolled_min": rolled_min,
				"rolled_max": rolled_max,
			})
		elif chosen.shape == AffixDefinition.AffixShape.PERCENT_RANGE:
			var rolled_percent = randf_range(chosen.min_value, chosen.max_value)
			instance.rolled_affixes.append({"affix": chosen, "rolled_value": rolled_percent})
		else:
			var rolled_value = randf_range(chosen.min_value, chosen.max_value)
			instance.rolled_affixes.append({
				"affix": chosen,
				"rolled_value": rolled_value
			})
		
	return instance
	

func _roll_affix_count(rarity: String) -> int:
	match rarity:
		"magic":    return randi_range(1, 2)
		"rare":     return randi_range(3, 6)
		"champion": return randi_range(5, 6)
		_:          return 0

func _exclusivity_keys(affix: AffixDefinition) -> Array[String]:
	var keys: Array[String] = []
	var op := int(affix.get_effective_op())
	for group in affix.exclusivity_tags:
		keys.append("%d_%d" % [group, op])
	return keys

func _get_affix_tags(def: ItemDefinition) -> Array[TagTypes.ItemTag]:
	var tags: Array[TagTypes.ItemTag] = []
	tags.append_array(def.allowed_affix_tags)
	if def.weapon_type != null:
		for t in def.weapon_type.tags:
			if t not in tags:
				tags.append(t)
	if def.equipment_type != null:
		for t in def.equipment_type.tags:
			if t not in tags:
				tags.append(t)
	return tags

func _tags_overlap(a: Array, b: Array) -> bool:
	for t in a:
		if t in b:
			return true
	return false
	

func _weighted_pick(pool: Array[AffixDefinition]) -> AffixDefinition:
	var best_level: Dictionary = {}
	for a in pool:
		for key in _exclusivity_keys(a):
			best_level[key] = maxi(int(best_level.get(key, 0)), a.min_item_level)

	var weights: Array[float] = []
	var total := 0.0
	for a in pool:
		var behind := 0
		for key in _exclusivity_keys(a):
			behind = maxi(behind, int(best_level[key]) - a.min_item_level)
		var w: float = a.weight * maxf(a.get_tier_multiplier(behind), MIN_TIER_MULT)
		weights.append(w)
		total += w

	var roll := randf() * total
	for i in pool.size():
		roll -= weights[i]
		if roll < 0.0:
			return pool[i]
	return pool[pool.size() - 1]

func get_affix(id: String) -> AffixDefinition:
	return _affix_by_id.get(id)

class_name ItemInstance
extends Resource

@export var definition: ItemDefinition
@export var rolled_affixes: Array[Dictionary] = []
@export var rarity: String = "normal"
@export var item_level: int = 1
@export var copied := false
@export var stim_roll: Dictionary = {}
@export var stim_mods: Array[String] = []
@export var charges: int = -1

func get_display_name() -> String:
	if definition.stim != null:
		return StimRoller.display_name(self)
	var prefix := ""
	var suffix := ""
	for entry in rolled_affixes:
		var affix: AffixDefinition = entry.affix
		if affix.affix_type == "prefix" and prefix == "":
			prefix = affix.display_name
		elif affix.affix_type == "suffix" and suffix == "":
			suffix = affix.display_name
	var parts: Array[String] = []
	if copied:
		parts.append("COPY OF")
	if prefix != "":
		parts.append(prefix)
	parts.append(definition.display_name)
	if suffix != "":
		parts.append(suffix)
	return " ".join(parts)

func build_modifiers() -> Array[Modifier]:
	var mods: Array[Modifier] = []
	var sid := "item_%s_" % definition.id

	for stat in definition.base_stats.keys():
		mods.append(_make_mod(stat, Modifier.ModOp.FLAT_ADD, definition.base_stats[stat], sid + "base"))

	for entry in rolled_affixes:
		var affix: AffixDefinition = entry.affix
		match affix.shape:
			AffixDefinition.AffixShape.DAMAGE_RANGE:
				mods.append(_make_mod(affix.min_stat, Modifier.ModOp.FLAT_ADD, entry.rolled_min, sid + affix.id))
				mods.append(_make_mod(affix.max_stat, Modifier.ModOp.FLAT_ADD, entry.rolled_max, sid + affix.id))
			AffixDefinition.AffixShape.PERCENT_RANGE:
				mods.append(_make_mod(affix.min_stat, Modifier.ModOp.PERCENT_ADD, entry.rolled_value, sid + affix.id))
				mods.append(_make_mod(affix.max_stat, Modifier.ModOp.PERCENT_ADD, entry.rolled_value, sid + affix.id))
			AffixDefinition.AffixShape.PERCENT_MULT_RANGE:
				mods.append(_make_mod(affix.min_stat, Modifier.ModOp.PERCENT_MULT, entry.rolled_value, sid + affix.id))
				mods.append(_make_mod(affix.max_stat, Modifier.ModOp.PERCENT_MULT, entry.rolled_value, sid + affix.id))
			_:
				mods.append(_make_mod(affix.stat, affix.op, entry.rolled_value, sid + affix.id))
	if definition.weapon_type != null and definition.weapon_type.base_range != 0.0:
		mods.append(_make_mod(StatTypes.Stat.ATTACK_RANGE, Modifier.ModOp.FLAT_ADD,
		definition.weapon_type.base_range, sid + "range"))
	return mods

func _make_mod(stat: StatTypes.Stat, op: Modifier.ModOp, value: float, source: String) -> Modifier:
	var m := Modifier.new()
	m.stat = stat
	m.op = op
	m.value = value
	m.source_id = source
	return m

static func make_gem(def: ItemDefinition) -> ItemInstance:
	if def == null:
		push_error("make_gem: the item definition is null, check the file path and that it is an ItemDefinition")
		return null
	var item := ItemInstance.new()
	item.definition = def
	item.rarity = "skill_gem" if def.skill_gem != null else "support_gem"
	return item

func to_dict() -> Dictionary:
	var affixes: Array = []
	for e in rolled_affixes:
		var a := {'id': e.affix.id}
		if e.has('rolled_value'):
			a['value'] = e.rolled_value
		if e.has('rolled_min'):
			a['min'] = e.rolled_min
			a['max'] = e.rolled_max
		affixes.append(a)
	var d := {'item': definition.id, 'rarity': rarity, 'level': item_level, 'affixes': affixes}
	if copied:
		d['copied'] = true
	if definition.stim != null:
		d['stim'] = {'base_amount': stim_roll.get('base_amount', 0.0), 'mods': stim_mods.duplicate(),
				'charges': charges}
	return d


static func from_dict(d: Dictionary) -> ItemInstance:
	var def := ItemDatabase.get_item(str(d.get('item', '')))
	if def == null:
		return null
	var inst := ItemInstance.new()
	inst.definition = def
	inst.rarity = str(d.get('rarity', 'normal'))
	inst.item_level = int(d.get('level', 1))
	inst.copied = bool(d.get('copied', false))
	for a in d.get('affixes', []):
		var affix := ItemGenerator.get_affix(str(a.get('id', '')))
		if affix == null:
			continue
		var entry := {'affix': affix}
		if a.has('value'):
			entry['rolled_value'] = float(a['value'])
		if a.has('min'):
			entry['rolled_min'] = float(a['min'])
			entry['rolled_max'] = float(a['max'])
		inst.rolled_affixes.append(entry)
	if def.stim != null:
		var s: Dictionary = d.get('stim', {})
		if s.has('base_amount') and float(s['base_amount']) > 0.0:
			inst.stim_roll['base_amount'] = float(s['base_amount'])
		for m in s.get('mods', []):
			inst.stim_mods.append(str(m))
		inst.charges = int(s.get('charges', -1))
		StimRoller.recompute(inst)
	return inst

class_name ShopNPC
extends DialogueNPC

signal stock_changed

@export var shop: ShopDefinition
@export var stock_level: int = 1 
@export var follow_act_level := true
@export var salvage_magic_per := 0.0

var stock: Array[ItemInstance] = []
var _infinite: Array[ItemInstance] = []
var _rerolls := 0
var _stock_rank := 0
var _stock_tier := false

func _ready() -> void:

	_apply_model()
	_use_act_level()

func _use_act_level() -> void:
	if not follow_act_level:
		return
	var n := get_parent()
	while n != null and not (n is TownArea):
		n = n.get_parent()
	if n == null or n.scene_file_path == "":
		return
	var level := Waypoints.top_level(Waypoints.act_of_scene(n.scene_file_path))
	if level > 0:

func _apply_model() -> void:
	if model_scene == null:
		push_warning("ShopNPC: No model scene assigned.")
		return
	
	for child in model_anchor.get_children():
		model_anchor.remove_child(child)
		child.queue_free()
	var model := model_scene.instantiate() as Node3D
	if model == null:
		push_warning("ShopNPC: Model root must inherit Node3D.")
		return
	
	model.scale = model_scale
	model_anchor.add_child(model)

func interact(player: Node3D) -> void:
	if dialogue != null:
		super.interact(player)
		return
	GameEvents.report("talk", npc_id)
	var panel := get_tree().get_first_node_in_group("shop_panel") as ShopPanel
	if panel == null:
		push_warning("ShopNPC: no ShopPanel in the scene (group shop_panel)")
		return
	panel.open_shop(self, player)

func restock() -> void:
	stock.clear()
	_infinite.clear()
	_rerolls = 0
	if shop == null:
		return
	_stock_rank = Ranks.of_player(get_tree())
	_stock_tier = _has_tier()

	for def in shop.fixed_stock:
		if _locked(def, _stock_rank):
			continue
		var item := _make_item(def, false)
		if item == null:
			continue
		stock.append(item)
		if shop.infinite_fixed:
			_infinite.append(item)

	var pool := _build_random_pool()
	_add_random_stock(pool)

	print("[Shop %s] level %d: %d eligible items, %d on sale" % [npc_name, stock_level, pool.size(), stock.size()])
	stock_changed.emit()

func can_reroll() -> bool:
	return shop != null and shop.allow_reroll and shop.random_slots > 0 \
			and (shop.use_catalog or not shop.random_pool.is_empty())


func reroll_cost() -> int:
	if shop == null:
		return 0
	var base := float(shop.reroll_base_cost + shop.reroll_cost_per_level * stock_level)
	return maxi(roundi(base * pow(maxf(shop.reroll_cost_growth, 1.0), _rerolls)), 0)


func try_reroll(wallet: CurrencyComponent) -> String:
	if not can_reroll():
		return "THIS SHOP CAN'T REROLL"
	var pool := _build_random_pool()
	if pool.is_empty():
		return "NOTHING ELSE TO STOCK"
	if wallet == null or not wallet.spend(reroll_cost()):
		return "NOT ENOUGH CREDITS"
	_rerolls += 1
	stock.assign(stock.filter(func(item: ItemInstance) -> bool: return item.definition in shop.fixed_stock))
	_add_random_stock(pool)
	GameEvents.report("shop_reroll", npc_id)
	stock_changed.emit()
	return ""


func _add_random_stock(pool: Array[ItemDefinition]) -> void:
	var p := get_tree().get_first_node_in_group("players") if is_inside_tree() else null
	var lost := YardCrates.shop_slots_lost(p.get("quests") as QuestManager if p != null else null, npc_id)
	for def in _roll_defs(pool, maxi(shop.random_slots - lost, 1)):
		var item := _make_item(def, true)
		if item != null:
			stock.append(item)
	if not _has_tier():
		return
	var copyable := pool.filter(func(d: ItemDefinition) -> bool: return not d.is_gem() and d.stim == null)
	for def in _roll_defs(copyable as Array[ItemDefinition], copy_slots):
		var copy := _make_item(def, true)
		if copy != null:
			copy.copied = true
			stock.append(copy)


func _has_tier() -> bool:
	if tier_quest == "":
		return false
	var p := get_tree().get_first_node_in_group("players") if is_inside_tree() else null
	var qm := p.get("quests") as QuestManager if p != null else null
	return qm != null and qm.is_completed(tier_quest)


func _roll_level() -> int:
	return stock_level + (tier_levels if _has_tier() else 0)


func _build_random_pool() -> Array[ItemDefinition]:
	var found: Array[ItemDefinition] = []
	if shop.use_catalog:
		var folders := PackedStringArray()
		for f in shop.catalog_folders:
			folders.append(f if f.ends_with("/") else f + "/")
		found.append_array(ItemCatalog.for_level(_roll_level(), shop.level_window, shop.include_gems, folders))
	for def in shop.random_pool:
		if def not in found:
			found.append(def)

	var pool: Array[ItemDefinition] = []
	for def in found:
		if def in shop.excluded_items or def in shop.fixed_stock or _locked(def, _stock_rank):
			continue
		pool.append(def)
	return pool

func _roll_defs(pool: Array[ItemDefinition], count: int) -> Array[ItemDefinition]:
	var left: Array[ItemDefinition] = pool.duplicate()
	var out: Array[ItemDefinition] = []
	for i in mini(count, left.size()):
		var total := 0.0
		for d in left:
			total += _weight(d)
		var roll := randf() * total
		var pick: ItemDefinition = left[0]
		for d in left:
			roll -= _weight(d)
			if roll <= 0.0:
				pick = d
				break
		left.erase(pick)
		out.append(pick)
	return out


func _weight(def: ItemDefinition) -> float:
	return pow(maxf(float(def.item_level), 1.0), shop.level_bias)

func buy_price(item: ItemInstance) -> int:
	return maxi(ceili(ItemPricing.value_of(item) * shop.buy_markup), 1)

func sell_price(item: ItemInstance) -> int:
	return maxi(floori(ItemPricing.value_of(item) * shop.sell_ratio), 1)

func try_buy(item: ItemInstance, inv: InventoryComponent, wallet: CurrencyComponent) -> String:
	var idx := stock.find(item)
	if idx < 0:
		return "NO LONGER AVAILABLE"
	if inv.is_full():
		return "BACKPACK FULL"
	if not wallet.spend(buy_price(item)):
		return "NOT ENOUGH CREDITS"

	if item in _infinite:
		var fresh := _make_item(item.definition, false)
		var inf_idx := _infinite.find(item)
		if fresh != null:
			stock[idx] = fresh
			_infinite[inf_idx] = fresh
		else:
			stock.remove_at(idx)
			_infinite.remove_at(inf_idx)
	else:
		stock.remove_at(idx)
	inv.add_item(item)
	stock_changed.emit()
	return ""

func try_sell(item: ItemInstance, inv: InventoryComponent, wallet: CurrencyComponent) -> String:
	if not inv.has_item(item):
		return "ONLY BACKPACK ITEMS CAN BE SOLD"
	var qm: QuestManager = inv.get_parent().get("quests")
	var keep := qm.keep_reason(item) if qm != null else ""
	if keep != "":
		return keep
	wallet.add(sell_price(item))
	inv.remove_item(item)
	return ""

static func _locked(def: ItemDefinition, rank: int) -> bool:
	return def != null and def.support_gem != null and not Ranks.allows(rank, "gem:" + def.support_gem.id)


func engage(player: Node3D) -> void:
	super.engage(player)
	if shop != null and _has_tier() != _stock_tier:
		restock()
	var rank := Ranks.of(player)
	if shop == null or rank <= _stock_rank:
		return
	var before := _stock_rank
	_stock_rank = rank
	for def in shop.fixed_stock:
		if _locked(def, before) and not _locked(def, rank):
			var item := _make_item(def, false)
			if item == null:
				continue
			stock.append(item)
			if shop.infinite_fixed:
				_infinite.append(item)
	for def in _build_random_pool():
		if _locked(def, before):
			var item := _make_item(def, false)
			if item != null:
				stock.append(item)
	stock_changed.emit()


func _make_item(def: ItemDefinition, roll_rarity: bool) -> ItemInstance:
	if def.is_gem():
		return ItemInstance.make_gem(def)
	var rarity := "normal"
	if roll_rarity:
		var salvage := _salvage()
		var rare := minf(shop.rare_chance + salvage_rare_per * salvage, 0.6)
		var magic := minf(shop.magic_chance + salvage_magic_per * salvage, 0.9 - rare)
		var r := randf()
		if r < rare:
			rarity = "rare"
		elif r < rare + magic:
			rarity = "magic"
	return ItemGenerator.generate_item(def, _roll_level(), rarity)


func _salvage() -> int:
	if salvage_magic_per <= 0.0 and salvage_rare_per <= 0.0:
		return 0
	var p := get_tree().get_first_node_in_group("players")
	return SalvageCache.recovered(p.get("quests") as QuestManager) if p != null else 0

func open_shop(shop: ShopNPC, player: Node3D, reach: float = 0.0) -> void:
	if _open or not _built:
		return

	if power != null and power.is_offline():
		QuestItems.toast(get_tree(), "NO POWER\n%s\n%s" % [power.offline_text, power.fix_hint()],
				Color(1.0, 0.35, 0.35))
		return
	_reach = reach
	if _inv == null or _wallet == null:
		push_warning("ShopPanel: not bound. Call bind(inventory, equipment, currency) from the player.")
		return
	_shop = shop
	_player = player
	shop.engage(player)
	_shop.stock_changed.connect(_on_data_changed)
	_inv_panel = get_tree().get_first_node_in_group("inventory_panel") as InventoryPanel
	if _inv_panel:
		_inv_panel.close()
	if "input_locked" in _player:
		_player.input_locked = true

	_title.text = "// %s //" % shop.npc_name.to_upper()
	_greeting.text = shop.shop.greeting if shop.shop else ""
	_selected = null
	_hover = null

	_open = true
	var mode := Input.mouse_mode
	if mode == Input.MOUSE_MODE_CAPTURED or mode == Input.MOUSE_MODE_HIDDEN \
			or mode == Input.MOUSE_MODE_CONFINED_HIDDEN:
		_prev_mouse_mode = mode
		_changed_mouse_mode = true
		Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
	_refresh()
	visible = true
	_fade(1.0)


func close() -> void:
	if not _open:
		return
	_open = false
	if _shop and _shop.stock_changed.is_connected(_on_data_changed):
		_shop.stock_changed.disconnect(_on_data_changed)
	if _player and is_instance_valid(_player) and "input_locked" in _player:
		_player.input_locked = false
	_shop = null
	if _changed_mouse_mode:
		Input.mouse_mode = _prev_mouse_mode
		_changed_mouse_mode = false
	_hover_slot = null
	_update_tooltip()
	_fade(0.0)

	if _open and event is InputEventKey and event.pressed and not event.echo \

func _refresh() -> void:
	if _shop == null:
		return

	_hover_slot = null
	if _selected != null and not (_shop.stock.has(_selected) or _inv.has_item(_selected)):
		_selected = null

	_fill(_bag_grid, _inv.items, "bag", _inv.capacity)

	_update_reroll()
	_sync_selection()
	_update_detail()
	_update_action()


func _fill(grid: GridContainer, items: Array, kind: String, min_cells: int) -> void:
	for child in grid.get_children():
		grid.remove_child(child)
		child.queue_free()
	for i in maxi(items.size(), min_cells):
		var s := _make_slot(kind)
		s.item = items[i] if i < items.size() else null
		grid.add_child(s)

func _update_tooltip() -> void:
	if _tip == null:
		return
	var ctrl := Input.is_key_pressed(KEY_CTRL)
	var slot: InventoryPanel.ItemSlot = null
	if _hover_slot != null and is_instance_valid(_hover_slot) and _hover_slot.item != null:
		slot = _hover_slot
	elif ctrl and _selected != null:
		slot = _slot_showing(_selected)
	if slot == null or not _open or _shop == null:
		_tip.visible = false
		_tip_cmp.visible = false
		_tip_anchor = null
		return
	var item: ItemInstance = slot.item
	_tip_anchor = slot
	_tip.clear()
	if _inv_panel != null:
		_inv_panel.fill_item_detail(_tip.box, item)
	else:
		_tip.box.add_child(_label(item.get_display_name().to_upper(), 20, COL_TEXT))
	_tip.box.add_child(_rule())
	_tip.box.add_child(_label(_price_text(item), 17, _price_color(item)))

	var comparable := _inv_panel != null and _inv_panel.can_compare(item)
	var current: ItemInstance = _inv_panel.equipped_for(item) if comparable else null
	if comparable:
		if ctrl:
			_inv_panel.add_compare_lines(_tip.box, current, item)
		else:
			_tip.box.add_child(_rule())
			var h := _label("HOLD CTRL TO COMPARE", 13, accent)
			h.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
			_tip.box.add_child(h)
	_tip.visible = true

	if comparable and ctrl and current != null:
		_tip_cmp.clear()
		_tip_cmp.box.add_child(_label("CURRENTLY EQUIPPED", 13, accent))
		_inv_panel.fill_item_detail(_tip_cmp.box, current)
		_tip_cmp.visible = true
	else:
		_tip_cmp.visible = false
	_place_tips()


func _slot_showing(item: ItemInstance) -> InventoryPanel.ItemSlot:

		for s: InventoryPanel.ItemSlot in grid.get_children():
			if s.item == item:
				return s
	return null


func _place_tips() -> void:
	if _tip_anchor == null or not is_instance_valid(_tip_anchor) or not _tip.visible:
		return
	var r := _tip_anchor.get_global_rect()
	var screen := get_viewport_rect().size
	var gap := 12.0
	_tip.reset_size()
	var right_side := r.end.x + gap + _tip.size.x <= screen.x - 8.0
	var pos := Vector2(r.end.x + gap if right_side else r.position.x - gap - _tip.size.x, r.position.y)
	pos.x = clampf(pos.x, 8.0, maxf(8.0, screen.x - _tip.size.x - 8.0))
	pos.y = clampf(pos.y, 8.0, maxf(8.0, screen.y - _tip.size.y - 8.0))
	_tip.global_position = pos.round()
	if _tip_cmp.visible:
		_tip_cmp.reset_size()
		var p2 := Vector2(pos.x + _tip.size.x + gap if right_side else pos.x - gap - _tip_cmp.size.x, pos.y)
		if p2.x < 8.0 or p2.x + _tip_cmp.size.x > screen.x - 8.0:
			p2.x = r.position.x - gap - _tip_cmp.size.x if right_side else r.end.x + gap
		p2.x = clampf(p2.x, 8.0, maxf(8.0, screen.x - _tip_cmp.size.x - 8.0))
		p2.y = clampf(p2.y, 8.0, maxf(8.0, screen.y - _tip_cmp.size.y - 8.0))
		_tip_cmp.global_position = p2.round()

	_action_btn.text = "BUY" if (has and _is_shop_item(_selected)) else "SELL"


func _is_shop_item(item: ItemInstance) -> bool:

	_selected = slot.item
	_sync_selection()
	_update_detail()
	_update_action()

		_do_action()


func _do_action() -> void:
	if _selected == null or _shop == null:
		return
	var err := ""
	if _is_shop_item(_selected):
		err = _shop.try_buy(_selected, _inv, _wallet)
	else:
		err = _shop.try_sell(_selected, _inv, _wallet)
	if err != "":
		_notify(err)
		return
	_selected = null
	_hover = null
	_hover_slot = null
	_sync_selection()
	_update_detail()
	_update_action()

	var show := _shop != null and _shop.can_reroll()
	_reroll_btn.visible = show
	if show:
		var cost := _shop.reroll_cost()
		_reroll_btn.text = "REROLL  %d CR" % cost
		_reroll_btn.disabled = _wallet == null or _wallet.amount < cost

	if _shop == null:
		return
	var err := _shop.try_reroll(_wallet)
	if err != "":
		_notify(err)
		return
	if _selected != null and _is_shop_item(_selected):
		_selected = null
	_refresh()

	_msg_label.text = text
	_msg_label.modulate.a = 1.0
	if _msg_tween:
		_msg_tween.kill()
	_msg_tween = create_tween()
	_msg_tween.tween_interval(1.2)
	_msg_tween.tween_property(_msg_label, "modulate:a", 0.0, 0.5)


func _draw() -> void:
	var w := size.x
	var h := size.y
	var poly := PackedVector2Array([
		Vector2(0, 0), Vector2(w - CHAMFER, 0), Vector2(w, CHAMFER),
		Vector2(w, h), Vector2(CHAMFER, h), Vector2(0, h - CHAMFER)])
	draw_colored_polygon(poly, Color(0.02, 0.05, 0.09, bg_opacity))
	var outline := PackedVector2Array(poly)
	outline.append(poly[0])
	draw_polyline(outline, Color(accent, 0.18), 7.0, true)
	draw_polyline(outline, accent, 2.0, true)

	b.add_theme_stylebox_override("disabled", _flat(Color(accent, 0.02), Color(accent, 0.12), 1))
	return b


func _flat(bg: Color, border: Color, border_w: int) -> StyleBoxFlat:
	var sb := StyleBoxFlat.new()
	sb.bg_color = bg
	sb.border_color = border
	sb.set_border_width_all(border_w)
	sb.corner_radius_top_right = 10
	sb.corner_radius_bottom_left = 10
	sb.content_margin_left = 10
	sb.content_margin_right = 10
	sb.content_margin_top = 4
	sb.content_margin_bottom = 4
	return sb


func _rule() -> ColorRect:
	var r := ColorRect.new()
	r.color = Color(accent, 0.5)
	r.custom_minimum_size = Vector2(0, 2)
	r.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return r


func _style_scrollbar(bar: ScrollBar) -> void:
	bar.custom_minimum_size = Vector2(8, 0)
	var track := StyleBoxFlat.new()
	track.bg_color = Color(accent, 0.08)
	bar.add_theme_stylebox_override("scroll", track)
	for entry in [["grabber", 0.55], ["grabber_highlight", 0.85], ["grabber_pressed", 1.0]]:
		var g := StyleBoxFlat.new()
		g.bg_color = Color(accent, entry[1])
		bar.add_theme_stylebox_override(entry[0], g)
