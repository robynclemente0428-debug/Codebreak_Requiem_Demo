class_name AIComponent
extends Node

enum State { IDLE, CHASE, ATTACK }

signal attack_requested(target: Node3D)
signal state_changed(new_state: State)

@export var aggro_range: float = 15.0
@export var attack_range: float = 2.0
@export var eye_height: float = 1.0
@export var lose_target_time: float = 3.0
@export_flags_3d_physics var wall_mask: int = 1
@export var turn_speed: float = 8.0
@export var slot_radius_factor: float = 0.8
@export var separation_radius: float = 1.1
@export var stop_distance: float = 1.2 
@export var basic_attack_enabled: bool = true
@export var debug_stuck := false
@export var preferred_distance: float = 0.0
@export var flee_distance: float = 0.0
@export var kite_delay_min: float = 0.5
@export var kite_delay_max: float = 1.0
var skill_lock: Callable = Callable()
var _kite_timer := -1.0
var is_retreating := false
var _attack_cooldown := 0.0
var busy_time := 0.0

var _stuck_time := 0.0
var _desired := Vector3.ZERO
var _slot_angle: float = 0.0
var _stalled_time := 0.0
var _hold_time := 0.0

const SCAN_INTERVAL := 0.25

static var _crowd_frame := -1
static var _crowd: PackedVector3Array = []
var _sep_frame := -1
var _sep := Vector3.ZERO

var state: State = State.IDLE:
	set(value):
		if value == state:
			return
		state = value
		state_changed.emit(value)
var target: Node3D = null
var objective: Node3D = null

var _body: CharacterBody3D
var _stats: StatComponent
var _nav: NavigationAgent3D
var _scan_timer: float = 0.0
var _time_since_seen: float = 0.0

func setup(body: CharacterBody3D, stats: StatComponent, nav: NavigationAgent3D) -> void:
	_body = body
	_stats = stats
	_nav = nav
	_scan_timer = randf() * SCAN_INTERVAL
	_slot_angle = randf() * TAU
	_body.add_to_group("enemies")
	_nav.velocity_computed.connect(_on_velocity_computed)

func _physics_process(delta: float) -> void:
	if _body == null:
		return
	
	_attack_cooldown = maxf(_attack_cooldown - delta, 0.0)
	busy_time = maxf(busy_time - delta, 0.0)
	
	_scan_timer -= delta
	if _scan_timer <= 0.0:
		_scan_timer = SCAN_INTERVAL
		_update_target()
		if target != null:
			_nav.target_position = target.global_position + _slot_offset()

	var speed := _stats.get_stat(StatTypes.Stat.MOVE_SPEED)
	var move_dir := Vector3.ZERO

	if target == null:
		state = State.IDLE
	else:
		var dist := _body.global_position.distance_to(target.global_position)
		var range_limit := attack_range * (1.25 if state == State.ATTACK else 1.0)
		state = State.ATTACK if dist <= range_limit else State.CHASE

	match state:
		State.CHASE:
			if _hold_time > 0.0:
				_hold_time -= delta
				move_dir = _separation() * 0.15
				_face_direction(target.global_position - _body.global_position, delta)
			else:
				var to_next: Vector3
				if _nav.is_navigation_finished():
					to_next = target.global_position - _body.global_position
				else:
					to_next = _nav.get_next_path_position() - _body.global_position
				to_next.y = 0.0
				move_dir = _steer(to_next.normalized(), 1.2)

				var real := _body.get_real_velocity()
				real.y = 0.0
				if real.length() > 1.0:
					_face_direction(real, delta)
				else:
					_face_direction(target.global_position - _body.global_position, delta)

				if real.length() < 0.5:
					_stalled_time += delta
					if _stalled_time > 0.5:
						_stalled_time = 0.0
						_hold_time = 0.7
				else:
					_stalled_time = 0.0
		State.ATTACK:
			var to_target := target.global_position - _body.global_position
			_face_direction(to_target, delta)
			to_target.y = 0.0
			
			if not basic_attack_enabled and to_target.length() > stop_distance:
				move_dir = (to_target.normalized() + _separation() * 1.2).normalized()
			
			var off := absf(angle_difference(_body.global_rotation.y, atan2(to_target.x, to_target.z)))
			if basic_attack_enabled and off < deg_to_rad(20.0) and _attack_cooldown <= 0.0 and not _is_busy():
				_attack_cooldown = 1.0 / maxf(_stats.get_stat(StatTypes.Stat.ATTACK_SPEED), 0.1)
				attack_requested.emit(target)

	if preferred_distance > 0.0 and target != null:
		var away := _body.global_position - target.global_position
		away.y = 0.0
		var d := away.length()
		_face_direction(-away, delta)

		if is_retreating:
			if d >= preferred_distance:
				is_retreating = false
			else:
				move_dir = (away.normalized() + _separation()).normalized()
		elif d < flee_distance:
			if _kite_timer < 0.0:
				_kite_timer = randf_range(kite_delay_min, kite_delay_max)
			if not _is_busy():
				_kite_timer -= delta
			if _kite_timer <= 0.0:
				is_retreating = true
				_kite_timer = -1.0
			move_dir = _separation() * 0.15
		else:
			_kite_timer = -1.0
			if d > preferred_distance * 1.2:
				var to_next := -away
				if not _nav.is_navigation_finished():
					to_next = _nav.get_next_path_position() - _body.global_position
					to_next.y = 0.0
				move_dir = (to_next.normalized() + _separation()).normalized()
			else:
				move_dir = _separation() * 0.15
	else:
		_kite_timer = -1.0
		is_retreating = false
	if _is_busy():
		move_dir = Vector3.ZERO
	var desired := move_dir * speed
	if debug_stuck:
		_debug_stuck(delta, desired)
	_desired = desired
	if _is_busy():
		_apply_velocity(Vector3.ZERO, delta)
	elif _nav.avoidance_enabled:
		_nav.max_speed = speed
		_nav.velocity = desired
	else:
		_apply_velocity(desired, delta)
	

func halt() -> void:
	set_physics_process(false)
	target = null
	if _nav != null:
		if _nav.velocity_computed.is_connected(_on_velocity_computed):
			_nav.velocity_computed.disconnect(_on_velocity_computed)
		_nav.avoidance_enabled = false
		_nav.velocity = Vector3.ZERO


func _on_velocity_computed(safe_velocity: Vector3) -> void:
	if not is_physics_processing():
		return
	_apply_velocity(safe_velocity, get_physics_process_delta_time())
	
func _apply_velocity(v: Vector3, delta: float) -> void:
	_body.velocity.x = v.x
	_body.velocity.z = v.z
	if not _body.is_on_floor():
		_body.velocity.y -= 20.0 * delta
	_body.move_and_slide()

func _debug_stuck(delta: float, desired: Vector3) -> void:
	if state == State.CHASE and _body.get_real_velocity().length() < 0.5:
		_stuck_time += delta
		if _stuck_time > 1.0 and fmod(_stuck_time, 1.0) < delta:
			print("STUCK %s dist=%.1f want=%.1f finished=%s reachable=%s path=%d" % [
				_body.name, _body.global_position.distance_to(target.global_position),
				desired.length(), _nav.is_navigation_finished(),
				_nav.is_target_reachable(), _nav.get_current_navigation_path().size()])
	else:
		_stuck_time = 0.0

func _slot_offset() -> Vector3:
	return Vector3(sin(_slot_angle), 0.0, cos(_slot_angle)) * attack_range * slot_radius_factor

func _update_target() -> void:
	var best := _find_nearest_visible_player()
	if best == null and objective != null:
		if is_instance_valid(objective) and objective.is_inside_tree():
			target = objective
			_time_since_seen = 0.0
			return
		objective = null
	if best != null:
		target = best
		_time_since_seen = 0.0
	elif target != null:
		_time_since_seen += SCAN_INTERVAL
		if _time_since_seen > lose_target_time or not is_instance_valid(target):
			target = null

func _find_nearest_visible_player() -> Node3D:
	var range_sq := aggro_range * aggro_range
	var best: Node3D = null
	var best_sq := INF
	for p in get_tree().get_nodes_in_group("players"):
		var d_sq := _body.global_position.distance_squared_to(p.global_position)
		if d_sq > range_sq or d_sq >= best_sq:
			continue
		if not _has_line_of_sight(p):
			continue
		best = p
		best_sq = d_sq
	return best

func _has_line_of_sight(p: Node3D) -> bool:
	var from := _body.global_position + Vector3.UP * eye_height
	var to := p.global_position + Vector3.UP * eye_height
	var query := PhysicsRayQueryParameters3D.create(from, to, wall_mask)
	query.exclude = [_body.get_rid()]
	return _body.get_world_3d().direct_space_state.intersect_ray(query).is_empty()

func _separation() -> Vector3:
	var frame := Engine.get_physics_frames()
	if _sep_frame == frame:
		return _sep
	_sep_frame = frame
	if _crowd_frame != frame:
		_crowd_frame = frame
		_crowd.clear()
		for e in get_tree().get_nodes_in_group("enemies"):
			_crowd.append((e as Node3D).global_position)
	var here := _body.global_position
	var r_sq := separation_radius * separation_radius
	var push := Vector3.ZERO
	for pos in _crowd:
		var offset := Vector3(here.x - pos.x, 0.0, here.z - pos.z)
		var d_sq := offset.length_squared()
		if d_sq < r_sq and d_sq > 0.000001:
			var d := sqrt(d_sq)
			push += offset / d * (1.0 - d / separation_radius)
	_sep = push.limit_length(1.0)
	return _sep

func _steer(dir: Vector3, crowd: float) -> Vector3:
	var push := _separation() * crowd
	if not _body.is_on_wall():
		return (dir + push).normalized()
	var n := _body.get_wall_normal()
	n.y = 0.0
	n = n.normalized()
	var want := dir + push * 0.35
	var into := want.dot(n)
	if into < 0.0:
		want -= n * into
	return want.normalized() if want.length_squared() > 0.0001 else dir


func _face_direction(dir: Vector3, delta: float) -> void:
	if _is_busy():
		return
	dir.y = 0.0
	if dir.length_squared() < 0.0001:
		return
	var target_y := atan2(dir.x, dir.z)
	_body.global_rotation.y = lerp_angle(
		_body.global_rotation.y, target_y, 1.0 - pow(0.001, delta * turn_speed))

func _is_busy() -> bool:
	return busy_time > 0.0 or (skill_lock.is_valid() and skill_lock.call())

const TRAIT_CHANCE := {"normal": 0.35, "magic": 0.6, "rare": 1.0, "champion": 1.0}
const SWARM_COUNT := 2


static func for_scene(scene_path: String) -> PackedStringArray:
	var w := Waypoints.for_scene(scene_path)
	if w == null:
		return PackedStringArray()
	return PackedStringArray(w.threats)


static func roll_for(scene_path: String, rng_seed: int) -> PackedStringArray:
	var w := Waypoints.for_scene(scene_path)
	if w == null:
		return PackedStringArray()
	var out := PackedStringArray(w.threats)
	if w.random_threats.y <= 0:
		return out
	var rng := RandomNumberGenerator.new()
	rng.seed = rng_seed + 9241
	var pool: Array = []
	for id in INFO:
		if id != RANDOM and not (id in out) and not INFO[id].get("fixed", false):
			pool.append(id)
	for i in range(pool.size() - 1, 0, -1):
		var j := rng.randi_range(0, i)
		var t = pool[i]
		pool[i] = pool[j]
		pool[j] = t
	var n := mini(rng.randi_range(w.random_threats.x, w.random_threats.y), pool.size())
	for i in n:
		out.append(pool[i])
	return out


static func preview(scene_path: String, level: int, new_instance := false) -> PackedStringArray:
	var w := Waypoints.for_scene(scene_path)
	if w == null:
		return PackedStringArray()
	if not new_instance:
		var inst: DungeonInstance = WorldState.find_open(scene_path, level if level > 0 else w.dungeon_level)
		if inst != null:
			return inst.threats.duplicate()
	var out := PackedStringArray(w.threats)
	if w.random_threats.y > 0:
		out.append(RANDOM)
	return out


static func label(id: String) -> String:
	return INFO.get(id, {}).get("label", id.to_upper())


static func color(id: String) -> Color:
	return INFO.get(id, {}).get("color", Color.WHITE)


static func hint(id: String) -> String:
	return INFO.get(id, {}).get("hint", "")


static func trait_def(id: String) -> EnemyModifierDefinition:
	var path: String = INFO.get(id, {}).get("trait", "")
	return load(path) as EnemyModifierDefinition if path != "" and ResourceLoader.exists(path) else null


static func roll_traits(threats: PackedStringArray, rarity: String, rng: RandomNumberGenerator) -> Array:
	var out: Array = []
	for id in threats:
		if id == "swarm" or INFO.get(id, {}).get("every_enemy", false):
			out.append(trait_def(id))
		elif rng.randf() < float(TRAIT_CHANCE.get(rarity, 0.5)):
			out.append(trait_def(id))
	return out.filter(func(d) -> bool: return d != null)

class_name EnemyBase
extends CharacterBody3D

@onready var model_mount: Node3D = $model
@onready var collision: CollisionShape3D = $CollisionShape3D
@onready var stats: StatComponent = $StatComponent
@onready var health: HealthComponent = $HealthComponent
@onready var ai: AIComponent = $AIComponent
@onready var nav: NavigationAgent3D = $NavigationAgent3D

var modifiers: Array = []
var traits: Array[EnemyTrait] = []

var rarity: String = "normal"
var level := 1
var is_remote := false

@export var hit_delay: float = 0.5

var model: EnemyModel

var display_name: String = ""
var current_health: float = 0.0

func _ready() -> void:
	add_to_group("enemies")

func setup() -> void:
	health.stats = stats              
	health.initialize()
	health.died.connect(_on_died)
	health.health_changed.connect(_on_health_changed)
	stats.stats_changed.connect(health._on_stats_changed)
	
	ai.setup(self, stats, nav)
	ai.attack_requested.connect(_on_attack_requested)

func attach_model(model_scene: PackedScene) -> void:
	model = model_scene.instantiate() as EnemyModel
	model_mount.add_child(model)
	model.bind(ai, health, self)
	EnemySilhouette.cover(model)

func apply_size(radius: float, height: float) -> void:
	var shape := (collision.shape as CylinderShape3D).duplicate() as CylinderShape3D
	shape.radius = radius
	shape.height = height
	collision.shape = shape
	collision.position.y = height * 0.5

func _on_attack_requested(target: Node3D) -> void:
	if model == null:
		return
	var cooldown := 1.0 / maxf(stats.get_stat(StatTypes.Stat.ATTACK_SPEED), 0.1)
	var delay := model.play_attack(cooldown)
	SoundManager.play_skill("attack", self, delay)
	if delay > 0.0:
		await get_tree().create_timer(delay).timeout
		if not is_instance_valid(self) or health.current_health <= 0.0:
			return
		if not is_instance_valid(target):
			return
	if global_position.distance_to(target.global_position) > ai.attack_range * 1.5:
		return
	var dmg := randf_range(
		stats.get_stat(StatTypes.Stat.MIN_ATTACK_DAMAGE),
		stats.get_stat(StatTypes.Stat.MAX_ATTACK_DAMAGE))
	Coop.hit_source = Coop.Hit.ENEMY
	target.health.take_damage(dmg)
	Coop.hit_source = Coop.Hit.NONE

const CORPSE_MAX_TIME := 3.0


func _on_died() -> void:
	remove_from_group("enemies")
	ai.halt()
	velocity = Vector3.ZERO
	collision.set_deferred("disabled", true)
	if model != null:
		EnemySilhouette.uncover(model)
	get_tree().create_timer(CORPSE_MAX_TIME, false).timeout.connect(queue_free)
	var shape := collision.shape as CylinderShape3D
	await GlitchDeath.play(self, model_mount, shape.radius if shape else 0.5, shape.height if shape else 1.8)
	queue_free()

func _on_health_changed(current: float, max: float) -> void:
	pass

func play_skill_animation(key: String, fit_duration: float) -> float:
	if model == null:
		return 0.0
	var hit_in := model.play_skill(key, fit_duration)
	if hit_in > 0.0:
		SoundManager.play_skill(key, self, hit_in)
	return hit_in

class_name EnemySkills
extends Node

@export var check_interval: float = 0.25
@export var skill_cooldown: float = 6.0
@export var cooldown_jitter: float = 2.0
@export var max_cast_distance: float = 12.0
@export var cast_lock_fraction: float = 1.0
var _caster: Node3D
var _stats: StatComponent
var _skills: Array[SkillInstance] = []
var _ready_at: Dictionary = {}
var _busy_until := 0
var _timer := 0.0


func setup(caster: Node3D, stats: StatComponent, gems: Array,
		flag_overrides: Dictionary = {}, supports: Array = [], support_ids: Dictionary = {},
		loops: Dictionary = {}) -> void:
	_caster = caster
	_stats = stats
	for gem in gems:
		if gem == null:
			continue
		var skill := SkillInstance.new()
		skill.gem = gem
		skill.flag_overrides = flag_overrides.get(gem.id, {})
		for sup in supports:
			if sup is SupportGemDefinition and sup not in skill.supports:
				skill.supports.append(sup)
		for id in support_ids.get(gem.id, []):
			var sup := SkillGemRegistry.get_support_gem(str(id))
			if sup == null:
				push_warning("EnemySkills: no support gem '%s' (for %s)" % [id, gem.id])
			elif sup not in skill.supports:
				skill.supports.append(sup)
		if int(loops.get(gem.id, 0)) > 0:
			_give_loop(skill, int(loops[gem.id]))
		skill.rebuild()
		_skills.append(skill)
		_ready_at[skill] = Time.get_ticks_msec() + int(randf_range(1.0, 3.0) * 1000.0)
		print("[EnemySkills] gem=%s overrides=%s supports=%s" % [gem.id, flag_overrides.get(gem.id, "NONE"),
				skill.supports.map(func(s: SupportGemDefinition) -> String: return s.id)])


func copy_skill(gem: SkillGemDefinition) -> void:
	if gem == null or (_skills.size() == 1 and _skills[0].gem == gem):
		return
	var skill := SkillInstance.new()
	skill.gem = gem
	skill.rebuild()
	_skills.clear()
	_ready_at.clear()
	_skills.append(skill)
	_ready_at[skill] = Time.get_ticks_msec() + 1500


func add_copied(skill: SkillInstance) -> void:
	if skill == null or skill in _skills:
		return
	_skills.push_front(skill)
	_ready_at[skill] = Time.get_ticks_msec() + 900


func stun(seconds: float) -> void:
	_busy_until = maxi(_busy_until, Time.get_ticks_msec() + int(seconds * 1000.0))


func _give_loop(skill: SkillInstance, repeats: int) -> void:
	var rule := ProgramRule.new()
	rule.loop = ProgramLoop.make(ProgramLoop.Mode.FOR, repeats)
	var program := SkillProgram.new()
	program.rules.append(rule)
	skill.program = program
	skill.rank = Ranks.LADDER.size() - 1
	skill.bonus_points = 100.0


func _process(delta: float) -> void:
	if _skills.is_empty() or not is_instance_valid(_caster):
		return
	_timer += delta
	if _timer < check_interval:
		return
	_timer = 0.0

	var now := Time.get_ticks_msec()
	if now < _busy_until or _is_dead():
		return
	var target := _pick_target()
	if target == null:
		return

	var dist := _flat_distance(target)
	for skill in _skills:
		if now < int(_ready_at.get(skill, 0)):
			continue
		if dist > minf(_reach(skill), max_cast_distance):
			continue
		_cast(skill, target, now)
		return


func _cast(skill: SkillInstance, target: Node3D, now: int) -> void:
	_face(target)
	var recovery := skill.get_cooldown(_stats)
	_busy_until = now + int(recovery * 1000.0)
	var wait := maxf(recovery, skill_cooldown) + randf_range(0.0, cooldown_jitter)
	_ready_at[skill] = now + int(wait * 1000.0)
	var ai = _caster.get("ai")
	if ai != null and "busy_time" in ai:
		ai.busy_time = recovery * cast_lock_fraction
	Coop.enemy_cast(_caster, _skills.find(skill), String(skill.gem.id))
	skill.cast(_caster, _stats, null)


func remote_cast(index: int, gem_id: String, facing: float) -> void:
	if not is_instance_valid(_caster) or _is_dead():
		return
	var skill := _remote_skill(index, gem_id)
	if skill == null:
		return
	_caster.global_rotation.y = facing
	skill.cast(_caster, _stats, null)


var _borrowed: Dictionary = {}


func _remote_skill(index: int, gem_id: String) -> SkillInstance:
	if index >= 0 and index < _skills.size() and String(_skills[index].gem.id) == gem_id:
		return _skills[index]
	for s in _skills:
		if String(s.gem.id) == gem_id:
			return s
	if _borrowed.has(gem_id):
		return _borrowed[gem_id]
	var gem := SkillGemRegistry.get_skill_gem(gem_id)
	if gem == null:
		return null
	var skill := SkillInstance.new()
	skill.gem = gem
	skill.rebuild()
	_borrowed[gem_id] = skill
	return skill


func _face(target: Node3D) -> void:
	var to := target.global_position - _caster.global_position
	to.y = 0.0
	if to.length_squared() > 0.001:
		_caster.global_rotation.y = atan2(to.x, to.z)


func _reach(skill: SkillInstance) -> float:
	var reach := skill.get_flag("radius") + skill.get_flag("center_offset")
	reach = maxf(reach, skill.get_flag("range"))
	if skill.get_flag("arc_angle") > 0.0:
		reach = maxf(reach, skill.get_melee_range(_stats))
	return reach


func _pick_target() -> Node3D:
	var ai = _caster.get("ai")
	if ai != null:
		var t = ai.target
		return t as Node3D if is_instance_valid(t) else null
	return _nearest_player()


func _nearest_player() -> Node3D:
	var best: Node3D = null
	var best_d := INF
	for p in get_tree().get_nodes_in_group("players"):
		var n := p as Node3D
		if n == null:
			continue
		var d := _flat_distance(n)
		if d < best_d:
			best_d = d
			best = n
	return best


func _flat_distance(target: Node3D) -> float:
	var d := target.global_position - _caster.global_position
	return Vector2(d.x, d.z).length()


func _is_dead() -> bool:
	var h = _caster.get("health")
	return h is HealthComponent and h.current_health <= 0.0

class_name EnemySpawner
extends RefCounted

const RARITY_SCALE := {"normal": 0.8, "magic": 1.3, "rare": 1.5, "champion": 1.8}
const LEVEL_SCALING := {
	StatTypes.Stat.MAX_HEALTH: 0.15,
	StatTypes.Stat.ARMOUR: 0.10,
	StatTypes.Stat.MIN_ATTACK_DAMAGE: 0.07,
	StatTypes.Stat.MAX_ATTACK_DAMAGE: 0.07,
}
const LEVEL_RAMP := {
	StatTypes.Stat.MAX_HEALTH: 0.006,
}
const LEVEL_FLAT := {
	StatTypes.Stat.ARMOUR: 3.0,
}

const ACT_SCALING := 0.5
const ACT_STATS: Array[StatTypes.Stat] = [StatTypes.Stat.MAX_HEALTH, StatTypes.Stat.ARMOUR]


static func act_multiplier_of_scene(scene_path: String) -> float:
	var act := Waypoints.act_of_scene(scene_path)
	return 1.0 + ACT_SCALING * maxi(act.act_number - 1, 0) if act != null else 1.0


static func _act_multiplier(node: Node) -> float:
	while node != null:
		if node.scene_file_path != "" and Waypoints.act_of_scene(node.scene_file_path) != null:
			return act_multiplier_of_scene(node.scene_file_path)
		node = node.get_parent()
	return 1.0


static func spawn_enemy(def: EnemyDefinition, rarity: String, parent: Node,
		world_pos: Variant = null, level: int = 1,
		forced_mods: Variant = null, add_to_random: bool = false) -> Node3D:
	var enemy = def.scene.instantiate()
	parent.add_child(enemy)
	if world_pos != null:
		enemy.global_position = world_pos

	var s: float = RARITY_SCALE.get(rarity, 1.0)
	enemy.apply_size(def.capsule_radius * s, def.capsule_height * s)
	enemy.model_mount.scale = Vector3.ONE * s
	enemy.nav.radius *= s
	if def.model_scene:
		enemy.attach_model(def.model_scene)

	enemy.ai.aggro_range = def.aggro_range
	enemy.ai.attack_range = def.attack_range * s
	enemy.ai.basic_attack_enabled = def.basic_attack_enabled
	enemy.ai.preferred_distance = def.preferred_distance
	enemy.ai.flee_distance = def.flee_distance
	enemy.ai.kite_delay_min = def.kite_delay_min
	enemy.ai.kite_delay_max = def.kite_delay_max
	enemy.stats.base_stats = {
		StatTypes.Stat.MAX_HEALTH: def.max_health,
		StatTypes.Stat.ARMOUR: def.armour,
		StatTypes.Stat.MOVE_SPEED: def.move_speed,
		StatTypes.Stat.MIN_ATTACK_DAMAGE: def.min_attack_damage,
		StatTypes.Stat.MAX_ATTACK_DAMAGE: def.max_attack_damage,
		StatTypes.Stat.ATTACK_SPEED: def.attack_speed,
		StatTypes.Stat.SKILL_DAMAGE_MULT: 1.0,
		StatTypes.Stat.ATTACK_RANGE: def.attack_range * s,
	}
	
	if level > 1:
		for stat in LEVEL_SCALING:
			var scale_mod := Modifier.new()
			scale_mod.stat = stat
			scale_mod.op = Modifier.ModOp.PERCENT_MULT
			scale_mod.value = LEVEL_SCALING[stat] * (level - 1) \
					+ LEVEL_RAMP.get(stat, 0.0) * pow(level - 1, 2)
			scale_mod.source_id = "dungeon_level"
			enemy.stats.add_modifier(scale_mod)
		for stat in LEVEL_FLAT:
			var flat_mod := Modifier.new()
			flat_mod.stat = stat
			flat_mod.op = Modifier.ModOp.FLAT_ADD
			flat_mod.value = LEVEL_FLAT[stat] * (level - 1)
			flat_mod.source_id = "dungeon_level"
			enemy.stats.add_modifier(flat_mod)

	var act_mult := _act_multiplier(parent)
	if act_mult > 1.0:
		for stat in ACT_STATS:
			var act_mod := Modifier.new()
			act_mod.stat = stat
			act_mod.op = Modifier.ModOp.PERCENT_MULT
			act_mod.value = act_mult - 1.0
			act_mod.source_id = "act_scaling"
			enemy.stats.add_modifier(act_mod)

	var mods: Array = []
	if forced_mods == null:
		mods = ModifierRegistry.roll_modifiers(def, rarity)
	else:
		mods = (forced_mods as Array).filter(func(x) -> bool: return x != null)
		if add_to_random:
			for extra in ModifierRegistry.roll_modifiers(def, rarity):
				if not mods.any(func(x) -> bool: return x.id == extra.id):
					mods.append(extra)

	var name_parts: Array[String] = []
	for mod_def in mods:
		for stat_mod in mod_def.stat_modifiers:
			var instance: Modifier = stat_mod.duplicate()
			instance.source_id = "enemymod_%s" % mod_def.id
			enemy.stats.add_modifier(instance)
		name_parts.append(mod_def.display_name_prefix)
		if mod_def.on_spawn_script != null:
			_add_trait(enemy, mod_def.on_spawn_script, mod_def.id)
	for script in def.innate_traits:
		if script != null:
			_add_trait(enemy, script, script.resource_path.get_file().get_basename())

	if "modifiers" in enemy:
		enemy.modifiers = mods
	enemy.rarity = rarity
	enemy.level = level
	enemy.display_name = _with_prefixes(def.display_name, name_parts)
	if not def.skills.is_empty():
		var caster := EnemySkills.new()
		caster.name = "EnemySkills"
		enemy.add_child(caster)
		print("[Spawner] %s skill_flag_overrides=%s  def_path=%s" % [def.id, def.skill_flag_overrides, def.resource_path])
		caster.setup(enemy, enemy.stats, def.skills, def.skill_flag_overrides,
				def.skill_supports, def.skill_support_ids, def.skill_loops)
		caster.skill_cooldown = def.skill_cooldown
		caster.cooldown_jitter = def.cooldown_jitter
		caster.cast_lock_fraction = def.cast_lock_fraction

	enemy.setup()
	Coop.enemy_spawned(enemy, def, rarity, level, mods)
	return enemy


static func empower(enemy: Node3D, health: float, armour: float, damage: float, source := "wave",
		flat_health := 0.0, flat_armour := 0.0) -> void:
	if enemy == null or not "stats" in enemy:
		return
	var given: Array = enemy.get_meta(Coop.NET_BOOST, [])
	given.append([health, armour, damage, source, flat_health, flat_armour])
	enemy.set_meta(Coop.NET_BOOST, given)
	var bonus := {
		StatTypes.Stat.MAX_HEALTH: health,
		StatTypes.Stat.ARMOUR: armour,
		StatTypes.Stat.MIN_ATTACK_DAMAGE: damage,
		StatTypes.Stat.MAX_ATTACK_DAMAGE: damage,
	}
	for stat in bonus:
		if bonus[stat] <= 0.0:
			continue
		var m := Modifier.new()
		m.stat = stat
		m.op = Modifier.ModOp.PERCENT_MULT
		m.value = bonus[stat]
		m.source_id = source
		enemy.stats.add_modifier(m)
	var flat := {StatTypes.Stat.MAX_HEALTH: flat_health, StatTypes.Stat.ARMOUR: flat_armour}
	for stat in flat:
		if flat[stat] <= 0.0:
			continue
		var f := Modifier.new()
		f.stat = stat
		f.op = Modifier.ModOp.FLAT_ADD
		f.value = flat[stat]
		f.source_id = source
		enemy.stats.add_modifier(f)
	var hp = enemy.get("health")
	if hp is HealthComponent:
		hp.revive(1.0)


static func _with_prefixes(base: String, prefixes: Array[String]) -> String:
	var words := " ".join(prefixes)
	if words == "":
		return base
	if base.begins_with("The "):
		return "The %s %s" % [words, base.substr(4)]
	return "%s %s" % [words, base]


static func _add_trait(enemy: Node3D, script: Script, id: String) -> void:
	if not "traits" in enemy:
		return
	var t := Node.new()
	t.set_script(script)
	if t is EnemyTrait:
		t.name = "Trait_%s" % id
		enemy.add_child(t)
		(t as EnemyTrait).setup(enemy)
		enemy.traits.append(t)
	else:
		t.free()
		push_warning("Trait %s: script is not an EnemyTrait" % id)

func _lock_const(key: String, default: Variant) -> Variant:
	var s: Script = get_script()
	if s == null:
		return default
	if not _consts.has(s):
		var all := {}
		var cur := s
		while cur != null:
			var own := cur.get_script_constant_map()
			for k in own:
				if not all.has(k):
					all[k] = own[k]
			cur = cur.get_base_script()
		_consts[s] = all
	return _consts[s].get(key, default)

		return
	var aura := int(_lock_const("LOCK_AURA", 0))
	var model := enemy.get("model") as Node
	if aura <= 0 or model == null:
		return
	enemy.set_meta(&"lock_aura", aura)
	EnemySilhouette.cover_aura(model, aura, lock_color())

const HINT_SECONDS := 2.0


func pop(text: String, col: Color, gap := 0.6) -> void:
	if is_copy() and not _relaying:
		return
	var now := Time.get_ticks_msec() / 1000.0
	if now - _popped_at < gap or not is_instance_valid(enemy):
		return
	_popped_at = now
	_spoke_at = now
	var numbers := get_tree().get_first_node_in_group("damage_numbers") as DamageNumbers
	if numbers != null:
		numbers.pop_text(enemy.global_position + Vector3.UP * 2.2, text, col, 1.0, HINT_SECONDS)
	LockBannerUI.report(self, text, col)
	_tell_copies("trait_say", [0, text, col, 1.0])


func say(text: String, col: Color, size_mult := 1.0) -> void:
	if is_copy() and not _relaying:
		return
	if not is_instance_valid(enemy):
		return
	_spoke_at = Time.get_ticks_msec() / 1000.0
	var numbers := get_tree().get_first_node_in_group("damage_numbers") as DamageNumbers
	if numbers != null:
		numbers.pop_text(enemy.global_position + Vector3.UP * 2.6, text, col, size_mult * 1.2, HINT_SECONDS)
	LockBannerUI.report(self, text, col)
	_tell_copies("trait_say", [1, text, col, size_mult])

func coop_message(what: String, args: Array, from: int) -> void:
	if from != 1 or not is_copy() or args.is_empty() or str(args[0]) != coop_key():
		return
	match what:
		"trait_say":
			if args.size() < 5 or not args[3] is Color:
				return
			_relaying = true
			if int(args[1]) == 0:
				pop(str(args[2]).left(120), args[3], 0.0)
			else:
				say(str(args[2]).left(120), args[3], clampf(float(args[4]), 0.5, 2.0))
			_relaying = false
		"trait_tag":
			_shown_tag = str(args[1]).left(48) if args.size() > 1 else ""
		_:
			_copy_message(what, args.slice(1))

func player_within(reach: float) -> Node3D:
	if not is_instance_valid(enemy) or not enemy.is_inside_tree():
		return null
	var best: Node3D = null
	var best_d := reach
	for p in get_tree().get_nodes_in_group("players"):
		var body := p as Node3D
		var hp := p.get("health") as HealthComponent
		if body == null or (hp != null and hp.current_health <= 0.0):
			continue
		var d := body.global_position.distance_to(enemy.global_position)
		if d <= best_d:
			best = body
			best_d = d
	return best
